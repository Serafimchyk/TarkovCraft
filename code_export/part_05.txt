package com.tarkovcraft.grid;

import com.tarkovcraft.item.BackpackItem;
import com.tarkovcraft.item.TacticalRigItem;
import com.tarkovcraft.item.TarkovCaseItem;
import com.tarkovcraft.search.ContainerSearchMemory;
import net.minecraft.item.ItemStack;
import net.minecraft.nbt.NbtCompound;
import net.minecraft.nbt.NbtElement;
import net.minecraft.nbt.NbtList;
import org.jetbrains.annotations.Nullable;

import java.util.ArrayList;
import java.util.List;
import java.util.UUID;
import java.util.function.Predicate;

/**
 * Helper for reading and writing GridContainers, container UUIDs, and per-player
 * ContainerSearchMemory on wearable container ItemStacks (Backpacks & Tactical Rigs).
 */
public final class ContainerItemHelper {
	private static final String KEY_ROOT = "TarkovContainer";
	private static final String KEY_UUID = "ContainerUuid";
	private static final String KEY_GRID = "Grid";
	private static final String KEY_GRIDS = "Grids";
	private static final String KEY_SEARCH = "SearchMemory";
	private static final String KEY_REQUIRES_SEARCH = "RequiresSearch";

	private ContainerItemHelper() {
	}

	public static boolean isContainerItem(ItemStack stack) {
		return !stack.isEmpty()
			&& (stack.getItem() instanceof BackpackItem
			|| stack.getItem() instanceof TacticalRigItem
			|| stack.getItem() instanceof TarkovCaseItem);
	}

	public static UUID getOrCreateContainerUuid(ItemStack stack) {
		NbtCompound root = stack.getOrCreateSubNbt(KEY_ROOT);
		if (!root.containsUuid(KEY_UUID)) {
			root.putUuid(KEY_UUID, UUID.randomUUID());
		}
		return root.getUuid(KEY_UUID);
	}

	@Nullable
	public static UUID getContainerUuid(ItemStack stack) {
		if (stack == null || stack.isEmpty() || !stack.hasNbt()) {
			return null;
		}
		NbtCompound root = stack.getSubNbt(KEY_ROOT);
		if (root != null && root.containsUuid(KEY_UUID)) {
			return root.getUuid(KEY_UUID);
		}
		return null;
	}

	public static boolean requiresSearch(ItemStack stack) {
		if (!isContainerItem(stack) || !stack.hasNbt()) {
			return false;
		}
		NbtCompound root = stack.getSubNbt(KEY_ROOT);
		return root != null && root.getBoolean(KEY_REQUIRES_SEARCH);
	}

	public static void setRequiresSearch(ItemStack stack, boolean requiresSearch) {
		if (!isContainerItem(stack)) {
			return;
		}
		NbtCompound root = stack.getOrCreateSubNbt(KEY_ROOT);
		root.putBoolean(KEY_REQUIRES_SEARCH, requiresSearch);
	}

	public static ContainerSearchMemory getSearchMemory(ItemStack stack) {
		if (!isContainerItem(stack) || !stack.hasNbt()) {
			return new ContainerSearchMemory();
		}
		NbtCompound root = stack.getSubNbt(KEY_ROOT);
		if (root != null && root.contains(KEY_SEARCH, NbtElement.COMPOUND_TYPE)) {
			return ContainerSearchMemory.fromNbt(root.getCompound(KEY_SEARCH));
		}
		return new ContainerSearchMemory();
	}

	public static void saveSearchMemory(ItemStack stack, ContainerSearchMemory memory) {
		if (!isContainerItem(stack)) {
			return;
		}
		NbtCompound root = stack.getOrCreateSubNbt(KEY_ROOT);
		root.put(KEY_SEARCH, memory.toNbt());
	}

	public static GridContainer readBackpackGrid(ItemStack stack) {
		if (!(stack.getItem() instanceof BackpackItem backpack)) {
			return new GridContainer(1, 1);
		}
		UUID uuid = getOrCreateContainerUuid(stack);
		NbtCompound root = stack.getOrCreateSubNbt(KEY_ROOT);
		if (root.contains(KEY_GRID, NbtElement.COMPOUND_TYPE)) {
			return GridContainer.fromNbt(root.getCompound(KEY_GRID), backpack.getGridWidth(), backpack.getGridHeight());
		}
		GridContainer grid = new GridContainer(backpack.getGridWidth(), backpack.getGridHeight(), uuid);
		root.put(KEY_GRID, grid.toNbt());
		return grid;
	}

	public static void writeBackpackGrid(ItemStack stack, GridContainer grid) {
		if (!(stack.getItem() instanceof BackpackItem)) {
			return;
		}
		NbtCompound root = stack.getOrCreateSubNbt(KEY_ROOT);
		root.putUuid(KEY_UUID, grid.getContainerId());
		root.put(KEY_GRID, grid.toNbt());
	}

	public static List<GridContainer> readRigGrids(ItemStack stack) {
		List<GridContainer> result = new ArrayList<>();
		if (!(stack.getItem() instanceof TacticalRigItem rig)) {
			return result;
		}
		getOrCreateContainerUuid(stack);
		NbtCompound root = stack.getOrCreateSubNbt(KEY_ROOT);
		List<ItemSize> layouts = rig.getCompartmentLayouts();
		NbtList list = root.getList(KEY_GRIDS, NbtElement.COMPOUND_TYPE);
		for (int i = 0; i < layouts.size(); i++) {
			ItemSize sz = layouts.get(i);
			if (i < list.size()) {
				result.add(GridContainer.fromNbt(list.getCompound(i), sz.width(), sz.height()));
			} else {
				result.add(new GridContainer(sz.width(), sz.height()));
			}
		}
		return result;
	}

	public static void writeRigGrids(ItemStack stack, List<GridContainer> grids) {
		if (!(stack.getItem() instanceof TacticalRigItem)) {
			return;
		}
		NbtCompound root = stack.getOrCreateSubNbt(KEY_ROOT);
		NbtList list = new NbtList();
		for (GridContainer g : grids) {
			list.add(g.toNbt());
		}
		root.put(KEY_GRIDS, list);
	}

	public static GridContainer readCaseGrid(ItemStack stack) {
		if (!(stack.getItem() instanceof TarkovCaseItem caseItem)) {
			return new GridContainer(1, 1);
		}
		UUID uuid = getOrCreateContainerUuid(stack);
		NbtCompound root = stack.getOrCreateSubNbt(KEY_ROOT);
		if (root.contains(KEY_GRID, NbtElement.COMPOUND_TYPE)) {
			return GridContainer.fromNbt(root.getCompound(KEY_GRID), caseItem.getGridWidth(), caseItem.getGridHeight());
		}
		GridContainer grid = new GridContainer(caseItem.getGridWidth(), caseItem.getGridHeight(), uuid);
		root.put(KEY_GRID, grid.toNbt());
		return grid;
	}

	public static void writeCaseGrid(ItemStack stack, GridContainer grid) {
		if (!(stack.getItem() instanceof TarkovCaseItem)) {
			return;
		}
		NbtCompound root = stack.getOrCreateSubNbt(KEY_ROOT);
		root.putUuid(KEY_UUID, grid.getContainerId());
		root.put(KEY_GRID, grid.toNbt());
	}

	/**
	 * Validates every saved entry without initializing the case or discarding
	 * entries that no longer fit its grid.
	 */
	public static boolean allCaseContentsMatch(ItemStack caseStack, Predicate<ItemStack> filter) {
		NbtCompound root = caseStack.getSubNbt(KEY_ROOT);
		if (root == null || !root.contains(KEY_GRID)) return true;
		if (!root.contains(KEY_GRID, NbtElement.COMPOUND_TYPE)) return false;
		NbtCompound grid = root.getCompound(KEY_GRID);
		if (!grid.contains("Entries")) return true;
		if (!(grid.get("Entries") instanceof NbtList entries)) return false;
		for (NbtElement entry : entries) {
			if (!(entry instanceof NbtCompound storedEntry)
				|| !filter.test(ItemStack.fromNbt(storedEntry.getCompound("Stack")))) return false;
		}
		return true;
	}

	public static List<GridContainer> readAllGrids(ItemStack stack) {
		if (stack.getItem() instanceof BackpackItem) {
			return List.of(readBackpackGrid(stack));
		}
		if (stack.getItem() instanceof TacticalRigItem) {
			return readRigGrids(stack);
		}
		if (stack.getItem() instanceof TarkovCaseItem) {
			return List.of(readCaseGrid(stack));
		}
		return List.of();
	}

	public static boolean containsContainerId(ItemStack stack, UUID targetId) {
		if (!isContainerItem(stack) || targetId == null) {
			return false;
		}
		UUID selfId = getContainerUuid(stack);
		if (targetId.equals(selfId)) {
			return true;
		}
		for (GridContainer grid : readAllGrids(stack)) {
			if (targetId.equals(grid.getContainerId())) {
				return true;
			}
			for (GridEntry entry : grid.getEntries()) {
				if (containsContainerId(entry.getStack(), targetId)) {
					return true;
				}
			}
		}
		return false;
	}

	public static int getNestingDepth(ItemStack stack) {
		if (!isContainerItem(stack)) {
			return 0;
		}
		int maxChildDepth = 0;
		for (GridContainer grid : readAllGrids(stack)) {
			for (GridEntry entry : grid.getEntries()) {
				maxChildDepth = Math.max(maxChildDepth, getNestingDepth(entry.getStack()));
			}
		}
		return 1 + maxChildDepth;
	}
}

package com.tarkovcraft.grid;

import net.minecraft.item.ItemStack;
import net.minecraft.nbt.NbtCompound;
import net.minecraft.nbt.NbtElement;
import net.minecraft.nbt.NbtList;
import org.jetbrains.annotations.Nullable;

import java.util.ArrayList;
import java.util.Collections;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.UUID;

/**
 * Pure 2D Tetris-style container supporting multi-cell items, rotation, stacking,
 * auto-placement, anti-recursion container nesting checks, and full NBT serialization.
 */
public final class GridContainer {
	public static final int MAX_NESTING_DEPTH = 2;

	private final int width;
	private final int height;
	private final UUID containerId;
	private final Map<UUID, GridEntry> entries = new LinkedHashMap<>();
	private final GridEntry[][] cells;

	public GridContainer(int width, int height) {
		this(width, height, UUID.randomUUID());
	}

	public GridContainer(int width, int height, UUID containerId) {
		if (width < 1 || height < 1) {
			throw new IllegalArgumentException("Grid dimensions must be >= 1, got " + width + "x" + height);
		}
		this.width = width;
		this.height = height;
		this.containerId = containerId;
		this.cells = new GridEntry[width][height];
	}

	public int getWidth() {
		return width;
	}

	public int getHeight() {
		return height;
	}

	public UUID getContainerId() {
		return containerId;
	}

	public List<GridEntry> getEntries() {
		return Collections.unmodifiableList(new ArrayList<>(entries.values()));
	}

	@Nullable
	public GridEntry getEntry(UUID entryId) {
		return entries.get(entryId);
	}

	@Nullable
	public GridEntry getAt(int x, int y) {
		if (x < 0 || y < 0 || x >= width || y >= height) {
			return null;
		}
		return cells[x][y];
	}

	public boolean isEmpty() {
		return entries.isEmpty();
	}

	public boolean canPlace(ItemStack stack, int x, int y, boolean rotated) {
		return canPlace(stack, x, y, rotated, null);
	}

	public boolean canPlace(ItemStack stack, int x, int y, boolean rotated, @Nullable UUID ignoredEntryId) {
		if (stack == null || stack.isEmpty()) {
			return false;
		}
		if (wouldCauseRecursion(stack)) {
			return false;
		}
		ItemSize size = ItemSizeRegistry.getSize(stack).oriented(rotated);
		if (x < 0 || y < 0 || x + size.width() > width || y + size.height() > height) {
			return false;
		}
		for (int dx = 0; dx < size.width(); dx++) {
			for (int dy = 0; dy < size.height(); dy++) {
				GridEntry existing = cells[x + dx][y + dy];
				if (existing != null && !existing.getEntryId().equals(ignoredEntryId)) {
					return false;
				}
			}
		}
		return true;
	}

	/**
	 * Prevents placing a container inside itself or exceeding MAX_NESTING_DEPTH.
	 */
	public boolean wouldCauseRecursion(ItemStack candidate) {
		UUID candidateContainerId = ContainerItemHelper.getContainerUuid(candidate);
		if (candidateContainerId != null && candidateContainerId.equals(this.containerId)) {
			return true;
		}
		if (ContainerItemHelper.containsContainerId(candidate, this.containerId)) {
			return true;
		}
		return ContainerItemHelper.getNestingDepth(candidate) >= MAX_NESTING_DEPTH;
	}

	/**
	 * Attempts to place stack at (x, y, rotated). If the target cell has a matching stackable item,
	 * merges into it first. Returns the created/updated GridEntry or null if placement failed.
	 */
	@Nullable
	public GridEntry place(ItemStack stack, int x, int y, boolean rotated) {
		if (stack == null || stack.isEmpty()) {
			return null;
		}
		// Try stacking onto an existing entry at (x, y)
		GridEntry existing = getAt(x, y);
		if (existing != null && canMerge(existing.getStack(), stack)) {
			int max = existing.getStack().getMaxCount();
			int space = max - existing.getStack().getCount();
			if (space >= stack.getCount()) {
				existing.getStack().increment(stack.getCount());
				return existing;
			}
		}

		if (!canPlace(stack, x, y, rotated, null)) {
			return null;
		}
		GridEntry entry = GridEntry.create(stack.copy(), x, y, rotated);
		insertInternal(entry);
		return entry;
	}

	/**
	 * Places an entry preserving a specific UUID (e.g., when moving within the same container).
	 */
	@Nullable
	public GridEntry placeWithId(UUID entryId, ItemStack stack, int x, int y, boolean rotated) {
		if (!canPlace(stack, x, y, rotated, null)) {
			return null;
		}
		GridEntry entry = new GridEntry(entryId, stack.copy(), x, y, rotated);
		insertInternal(entry);
		return entry;
	}

	/**
	 * Auto-places stack into the first available space (trying stacking first, then normal, then rotated).
	 * Mutates the passed stack's count if partially/fully stacked, returning true if completely inserted.
	 */
	public boolean tryAutoPlace(ItemStack stack) {
		return tryAutoPlace(stack, null);
	}

	/**
	 * Auto-places stack into the first available space (trying stacking first, then normal, then rotated).
	 * Mutates the passed stack's count if partially/fully stacked, returning true if completely inserted.
	 * When {@code canMergeIntoEntry} is provided, only merges into entries whose UUID passes the predicate.
	 */
	public boolean tryAutoPlace(ItemStack stack, @Nullable java.util.function.Predicate<UUID> canMergeIntoEntry) {
		if (stack == null || stack.isEmpty() || wouldCauseRecursion(stack)) {
			return false;
		}
		// 1. Try merging into existing stacks
		if (stack.isStackable()) {
			for (GridEntry entry : entries.values()) {
				if (canMergeIntoEntry != null && !canMergeIntoEntry.test(entry.getEntryId())) {
					continue;
				}
				if (canMerge(entry.getStack(), stack)) {
					int space = entry.getStack().getMaxCount() - entry.getStack().getCount();
					if (space > 0) {
						int transfer = Math.min(space, stack.getCount());
						entry.getStack().increment(transfer);
						stack.decrement(transfer);
						if (stack.isEmpty()) {
							return true;
						}
					}
				}
			}
		}

		// 2. Try free cells (normal orientation, then rotated if non-square)
		ItemSize baseSize = ItemSizeRegistry.getSize(stack);
		boolean[] orientations = baseSize.width() == baseSize.height()
			? new boolean[]{false}
			: new boolean[]{false, true};

		for (boolean rot : orientations) {
			for (int y = 0; y < height; y++) {
				for (int x = 0; x < width; x++) {
					if (canPlace(stack, x, y, rot)) {
						GridEntry entry = GridEntry.create(stack.copy(), x, y, rot);
						insertInternal(entry);
						stack.setCount(0);
						return true;
					}
				}
			}
		}
		return false;
	}

	/**
	 * Places stack into the first completely free cell (normal orientation first, then rotated),
	 * NEVER merging into existing stacks. Returns the created GridEntry or null if full.
	 */
	@Nullable
	public GridEntry tryPlaceInFreeCell(ItemStack stack) {
		if (stack == null || stack.isEmpty() || wouldCauseRecursion(stack)) {
			return null;
		}
		ItemSize baseSize = ItemSizeRegistry.getSize(stack);
		boolean[] orientations = baseSize.width() == baseSize.height()
			? new boolean[]{false}
			: new boolean[]{false, true};

		for (boolean rot : orientations) {
			for (int y = 0; y < height; y++) {
				for (int x = 0; x < width; x++) {
					if (canPlace(stack, x, y, rot)) {
						GridEntry entry = GridEntry.create(stack.copy(), x, y, rot);
						insertInternal(entry);
						return entry;
					}
				}
			}
		}
		return null;
	}

	@Nullable
	public GridEntry remove(UUID entryId) {
		GridEntry entry = entries.remove(entryId);
		if (entry == null) {
			return null;
		}
		clearCells(entry);
		return entry;
	}

	public void clear() {
		entries.clear();
		for (int x = 0; x < width; x++) {
			for (int y = 0; y < height; y++) {
				cells[x][y] = null;
			}
		}
	}

	private void insertInternal(GridEntry entry) {
		entries.put(entry.getEntryId(), entry);
		ItemSize size = entry.getOccupiedSize();
		for (int dx = 0; dx < size.width(); dx++) {
			for (int dy = 0; dy < size.height(); dy++) {
				cells[entry.getX() + dx][entry.getY() + dy] = entry;
			}
		}
	}

	private void clearCells(GridEntry entry) {
		for (int x = 0; x < width; x++) {
			for (int y = 0; y < height; y++) {
				if (cells[x][y] == entry) {
					cells[x][y] = null;
				}
			}
		}
	}

	private static boolean canMerge(ItemStack target, ItemStack incoming) {
		return target.isStackable()
			&& ItemStack.canCombine(target, incoming)
			&& target.getCount() < target.getMaxCount();
	}

	public NbtCompound toNbt() {
		NbtCompound nbt = new NbtCompound();
		nbt.putInt("Width", width);
		nbt.putInt("Height", height);
		nbt.putUuid("ContainerId", containerId);
		NbtList list = new NbtList();
		for (GridEntry entry : entries.values()) {
			if (!entry.getStack().isEmpty()) {
				list.add(entry.toNbt());
			}
		}
		nbt.put("Entries", list);
		return nbt;
	}

	public static GridContainer fromNbt(NbtCompound nbt, int defaultWidth, int defaultHeight) {
		int w = nbt.contains("Width") ? Math.max(1, nbt.getInt("Width")) : defaultWidth;
		int h = nbt.contains("Height") ? Math.max(1, nbt.getInt("Height")) : defaultHeight;
		UUID id = nbt.containsUuid("ContainerId") ? nbt.getUuid("ContainerId") : UUID.randomUUID();
		GridContainer container = new GridContainer(w, h, id);
		NbtList list = nbt.getList("Entries", NbtElement.COMPOUND_TYPE);
		for (int i = 0; i < list.size(); i++) {
			GridEntry entry = GridEntry.fromNbt(list.getCompound(i));
			if (!entry.getStack().isEmpty() && container.canPlace(entry.getStack(), entry.getX(), entry.getY(), entry.isRotated())) {
				container.insertInternal(entry);
			}
		}
		return container;
	}
}

package com.tarkovcraft.grid;

import net.minecraft.item.ItemStack;
import net.minecraft.nbt.NbtCompound;

import java.util.UUID;

public final class GridEntry {
	private final UUID entryId;
	private ItemStack stack;
	private int x;
	private int y;
	private boolean rotated;

	public GridEntry(UUID entryId, ItemStack stack, int x, int y, boolean rotated) {
		this.entryId = entryId;
		this.stack = stack;
		this.x = x;
		this.y = y;
		this.rotated = rotated;
	}

	public static GridEntry create(ItemStack stack, int x, int y, boolean rotated) {
		return new GridEntry(UUID.randomUUID(), stack, x, y, rotated);
	}

	public UUID getEntryId() {
		return entryId;
	}

	public ItemStack getStack() {
		return stack;
	}

	public void setStack(ItemStack stack) {
		this.stack = stack;
	}

	public int getX() {
		return x;
	}

	public void setX(int x) {
		this.x = x;
	}

	public int getY() {
		return y;
	}

	public void setY(int y) {
		this.y = y;
	}

	public boolean isRotated() {
		return rotated;
	}

	public void setRotated(boolean rotated) {
		this.rotated = rotated;
	}

	public ItemSize getOccupiedSize() {
		return ItemSizeRegistry.getSize(stack).oriented(rotated);
	}

	public NbtCompound toNbt() {
		NbtCompound nbt = new NbtCompound();
		nbt.putUuid("EntryId", entryId);
		nbt.put("Stack", stack.writeNbt(new NbtCompound()));
		nbt.putInt("X", x);
		nbt.putInt("Y", y);
		nbt.putBoolean("Rotated", rotated);
		return nbt;
	}

	public static GridEntry fromNbt(NbtCompound nbt) {
		UUID id = nbt.containsUuid("EntryId") ? nbt.getUuid("EntryId") : UUID.randomUUID();
		ItemStack stack = ItemStack.fromNbt(nbt.getCompound("Stack"));
		int x = nbt.getInt("X");
		int y = nbt.getInt("Y");
		boolean rotated = nbt.getBoolean("Rotated");
		return new GridEntry(id, stack, x, y, rotated);
	}
}

package com.tarkovcraft.grid;

public record ItemSize(int width, int height) {
	public static final ItemSize ONE_BY_ONE = new ItemSize(1, 1);

	public ItemSize {
		if (width < 1 || height < 1) {
			throw new IllegalArgumentException("ItemSize dimensions must be >= 1, got " + width + "x" + height);
		}
	}

	public ItemSize rotated() {
		return new ItemSize(height, width);
	}

	public ItemSize oriented(boolean rotated) {
		return rotated ? rotated() : this;
	}
}

/**
 * Resolves Tarkov grid dimensions (width x height) for any ItemStack.
 * Defaults to 1x1 for all unconfigured vanilla or modded items.
 * Includes smart TaCZ gun & magazine size rules:
 * - Pistol magazines: 1x1
 * - Standard rifle/SMG magazines: 1x2
 * - Drum / high-capacity magazines: 2x2
 */
public final class ItemSizeRegistry {
	private static final Logger LOGGER = LoggerFactory.getLogger("TarkovCraft/ItemSize");
	private static final Map<Identifier, ItemSize> ITEM_SIZES = new HashMap<>();
	private static final Map<String, ItemSize> TACZ_GUN_ID_SIZES = new HashMap<>();
	private static final Map<String, ItemSize> TACZ_ATTACHMENT_ID_SIZES = new HashMap<>();

	static {
		registerDefaults();
	}

	private ItemSizeRegistry() {
	}

	private static void registerDefaults() {
		ITEM_SIZES.clear();
		TACZ_GUN_ID_SIZES.clear();
		TACZ_ATTACHMENT_ID_SIZES.clear();

		// TaCZ base items
		ITEM_SIZES.put(new Identifier("tacz", "modern_kinetic_gun"), new ItemSize(4, 2));

	public static void reloadFromResources(ResourceManager manager) {
		registerDefaults();
		Map<Identifier, Resource> resources = manager.findResources("item_sizes", id -> id.getPath().endsWith(".json"));
		for (Map.Entry<Identifier, Resource> entry : resources.entrySet()) {
			try (InputStreamReader reader = new InputStreamReader(entry.getValue().getInputStream(), StandardCharsets.UTF_8)) {
				JsonObject root = JsonParser.parseReader(reader).getAsJsonObject();
				if (root.has("items") && root.get("items").isJsonObject()) {
					for (Map.Entry<String, JsonElement> itemEntry : root.getAsJsonObject("items").entrySet()) {
						JsonObject sz = itemEntry.getValue().getAsJsonObject();
						ITEM_SIZES.put(new Identifier(itemEntry.getKey()), new ItemSize(sz.get("width").getAsInt(), sz.get("height").getAsInt()));
					}
				}
				if (root.has("tacz_guns") && root.get("tacz_guns").isJsonObject()) {
					for (Map.Entry<String, JsonElement> gunEntry : root.getAsJsonObject("tacz_guns").entrySet()) {
						JsonObject sz = gunEntry.getValue().getAsJsonObject();
						TACZ_GUN_ID_SIZES.put(gunEntry.getKey(), new ItemSize(sz.get("width").getAsInt(), sz.get("height").getAsInt()));
					}
				}
				if (root.has("tacz_attachments") && root.get("tacz_attachments").isJsonObject()) {
					for (Map.Entry<String, JsonElement> attEntry : root.getAsJsonObject("tacz_attachments").entrySet()) {
						JsonObject sz = attEntry.getValue().getAsJsonObject();
						TACZ_ATTACHMENT_ID_SIZES.put(attEntry.getKey(), new ItemSize(sz.get("width").getAsInt(), sz.get("height").getAsInt()));
					}
				}
			} catch (Exception e) {
				LOGGER.warn("Failed to load item_sizes from {}: {}", entry.getKey(), e.getMessage());
			}
		}
	}

	public static ItemSize getSize(ItemStack stack) {
		if (stack == null || stack.isEmpty()) {
			return ItemSize.ONE_BY_ONE;
		}
		Item item = stack.getItem();
		if (item instanceof TarkovSizedItem sized) {
			return sized.getStoredItemSize();
		}

		Identifier id = Registries.ITEM.getId(item);
		if ("tacz".equals(id.getNamespace()) && stack.hasNbt()) {
			NbtCompound nbt = stack.getNbt();
			if (nbt != null) {
				if (nbt.contains("GunId")) {
					ItemSize gunSize = TACZ_GUN_ID_SIZES.get(nbt.getString("GunId"));
					if (gunSize != null) return gunSize;
				}
				if (nbt.contains("AttachmentId")) {
					String attId = nbt.getString("AttachmentId");
					ItemSize attSize = TACZ_ATTACHMENT_ID_SIZES.get(attId);
					if (attSize != null) return attSize;
					// Heuristic for custom gunpack magazines
					String lower = attId.toLowerCase();
					if (lower.contains("drum")) return new ItemSize(2, 2);
					if (lower.contains("pistol") && lower.contains("mag")) return new ItemSize(1, 1);
					if (lower.contains("mag")) return new ItemSize(1, 2);
				}
			}
		}

		// Also support magazine-like item IDs from any TaCZ addon / gunpack
		String path = id.getPath().toLowerCase();
		if (path.contains("drum")) return new ItemSize(2, 2);
		if (path.contains("pistol") && path.contains("mag")) return new ItemSize(1, 1);
		if (path.contains("mag") && !path.contains("magma")) return new ItemSize(1, 2);

		return ITEM_SIZES.getOrDefault(id, ItemSize.ONE_BY_ONE);
	}

	public static boolean isPistolLike(ItemStack stack) {
		if (stack == null || stack.isEmpty()) {
			return false;
		}
		ItemSize size = getSize(stack);
		return size.width() <= 2 && size.height() == 1;
	}
}

package com.tarkovcraft.health;

import com.tarkovcraft.loadout.PlayerLoadoutHolder;
import net.minecraft.entity.effect.StatusEffectInstance;
import net.minecraft.entity.effect.StatusEffects;
import net.minecraft.particle.DustParticleEffect;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.network.ServerPlayerEntity;
import net.minecraft.server.world.ServerWorld;
import net.minecraft.sound.SoundCategory;
import net.minecraft.sound.SoundEvents;
import org.joml.Vector3f;

/**
 * Server-side tick orchestrator for bleeding, painkiller timer, broken-leg sprint pain,
 * limb impairment penalties, and blacked-stomach exhaustion.
 */
public final class BodyHealthTicker {
	private static final Vector3f BLOOD_COLOR = new Vector3f(0.78f, 0.05f, 0.05f);

	private BodyHealthTicker() {
	}

	public static void tickServer(MinecraftServer server) {
		for (ServerPlayerEntity player : server.getPlayerManager().getPlayerList()) {
			tickPlayer(player);
		}
	}

	public static void tickPlayer(ServerPlayerEntity player) {
		if (player.isCreative() || player.isSpectator() || !player.isAlive()) {
			return;
		}
		if (!(player instanceof PlayerLoadoutHolder holder)) {
			return;
		}

		BodyPartHealth health = holder.tarkovcraft$getBodyHealth();
		BodyPartHealth.TickResult res = health.tickEffects(player.isSprinting());

		if (res.bleedDamaged()) {
			ServerWorld sw = (ServerWorld) player.getWorld();
			DustParticleEffect blood = new DustParticleEffect(BLOOD_COLOR, res.heavyBleedActive() ? 1.15f : 0.75f);
			sw.spawnParticles(blood, player.getX(), player.getY() + 0.9, player.getZ(),
				res.heavyBleedActive() ? 8 : 3, 0.22, 0.35, 0.22, 0.01);
			if (res.heavyBleedActive()) {
				sw.playSound(null, player.getX(), player.getY(), player.getZ(),
					SoundEvents.ENTITY_PLAYER_HURT, SoundCategory.PLAYERS, 0.35f, 0.75f);
			}
		}

		if (res.sprintPainTriggered()) {
			player.getWorld().playSound(null, player.getX(), player.getY(), player.getZ(),
				SoundEvents.ENTITY_PLAYER_HURT, SoundCategory.PLAYERS, 0.55f, 0.95f);
		}

		if (res.killedByBloodLoss() || health.isDead()) {
			holder.tarkovcraft$syncBodyHealth();
			player.setHealth(0.0f);
			player.onDeath(player.getDamageSources().generic());
			return;
		}

		if (res.impairedLegs() > 0 && !res.painkillerActive()) {
			if (player.isSprinting()) {
				player.setSprinting(false);
			}
			if (player.age % 10 == 0) {
				player.addStatusEffect(new StatusEffectInstance(
					StatusEffects.SLOWNESS, 25, res.impairedLegs() >= 2 ? 2 : 1, true, false, true));
			}
		}

		if (res.impairedArms() > 0 && !res.painkillerActive() && player.age % 10 == 0) {
			int amp = res.impairedArms() >= 2 ? 1 : 0;
			player.addStatusEffect(new StatusEffectInstance(StatusEffects.MINING_FATIGUE, 25, amp, true, false, true));
			player.addStatusEffect(new StatusEffectInstance(StatusEffects.WEAKNESS, 25, amp, true, false, true));
		}

		if (res.exhaustionToAdd() > 0.0f) {
			player.addExhaustion(res.exhaustionToAdd());
		}

		if (res.stateChanged()) {
			player.setHealth(Math.max(1.0f, health.getTotalRatio() * player.getMaxHealth()));
			holder.tarkovcraft$syncBodyHealth();
		}
	}
}

package com.tarkovcraft.health;

/**
 * The 7 Tarkov anatomical health pools (Total = 435 HP).
 * Reducing HEAD or THORAX to 0 HP is immediately fatal.
 */
public enum BodyPart {
	HEAD("head", 35.0f, true),
	THORAX("thorax", 85.0f, true),
	STOMACH("stomach", 70.0f, false),
	LEFT_ARM("left_arm", 60.0f, false),
	RIGHT_ARM("right_arm", 60.0f, false),
	LEFT_LEG("left_leg", 65.0f, false),
	RIGHT_LEG("right_leg", 65.0f, false);

	public static final float TOTAL_MAX_HP = 435.0f;

	private final String id;
	private final float maxHp;
	private final boolean vital;

	BodyPart(String id, float maxHp, boolean vital) {
		this.id = id;
		this.maxHp = maxHp;
		this.vital = vital;
	}

	public String getId() {
		return id;
	}

	public float getMaxHp() {
		return maxHp;
	}

	public boolean isVital() {
		return vital;
	}

	public String getTranslationKey() {
		return "bodypart.tarkovcraft." + id;
	}

	public static BodyPart byId(String id) {
		for (BodyPart part : values()) {
			if (part.id.equalsIgnoreCase(id)) {
				return part;
			}
		}
		return THORAX;
	}
}

package com.tarkovcraft.health;

/**
 * Per-player Tarkov body-part health pool (435 base HP total across 7 parts) with:
 * - Unprotected HEAD_EYES ("Голова — глаза") -> instant fatal oneshot.
 * - Limb hits reduce limb HP to 0 first, then overflow directly to THORAX.
 * - Injuries: LIGHT_BLEED, HEAVY_BLEED, FRACTURE (limbs only) + effectiveMaxHp surgery limits.
 * - Painkiller duration, harsh dual-leg fall damage with weight scaling, and server tick penalties.
 */
public final class BodyPartHealth {
	private static final int MAX_HIT_LOG = 6;
	public static final int MAX_PAINKILLER_TICKS = 6000;

	private final Map<BodyPart, Float> hpMap = new EnumMap<>(BodyPart.class);
	private final Map<BodyPart, Float> effectiveMaxHpMap = new EnumMap<>(BodyPart.class);
	private final Map<BodyPart, EnumSet<InjuryType>> injuries = new EnumMap<>(BodyPart.class);
	private final Deque<DamageReport> recentHits = new ArrayDeque<>();
	@Nullable
	private DamageReport lastFatalReport;

	private int painkillerTicks = 0;
	private int bleedTickCounter = 0;
	private int sprintPainTickCounter = 0;
	private int stomachDrainTickCounter = 0;

	// Active medical/stimulator application state (synced via S2C_HEALTH_SYNC)
	private String activeMedItemId = "";
	private int activeMedElapsedTicks = 0;
	private int activeMedTotalTicks = 0;
	@Nullable
	private BodyPart activeMedTargetPart = null;

	public record TickResult(
		boolean stateChanged,
		boolean killedByBloodLoss,
		boolean bleedDamaged,
		boolean heavyBleedActive,
		boolean sprintPainTriggered,
		int impairedLegs,
		int impairedArms,
		boolean stomachBlacked,
		boolean painkillerActive,
		float exhaustionToAdd
	) {
	}

	public BodyPartHealth() {
		resetFull();
	}

	public void resetFull() {
		for (BodyPart part : BodyPart.values()) {
			effectiveMaxHpMap.put(part, part.getMaxHp());
			hpMap.put(part, part.getMaxHp());
			injuries.put(part, EnumSet.noneOf(InjuryType.class));
		}
		painkillerTicks = 0;
		bleedTickCounter = 0;
		sprintPainTickCounter = 0;
		stomachDrainTickCounter = 0;
		clearActiveMedUse();
		recentHits.clear();
		lastFatalReport = null;
	}

	public void setActiveMedUse(String itemId, int elapsedTicks, int totalTicks, @Nullable BodyPart targetPart) {
		this.activeMedItemId = itemId != null ? itemId : "";
		this.activeMedElapsedTicks = Math.max(0, elapsedTicks);
		this.activeMedTotalTicks = Math.max(0, totalTicks);
		this.activeMedTargetPart = targetPart;
	}

	public void clearActiveMedUse() {
		this.activeMedItemId = "";
		this.activeMedElapsedTicks = 0;
		this.activeMedTotalTicks = 0;
		this.activeMedTargetPart = null;
	}

	public boolean isMedUseActive() {
		return !activeMedItemId.isEmpty() && activeMedTotalTicks > 0;
	}

	public float getActiveMedProgress() {
		if (!isMedUseActive()) return 0.0f;
		return Math.max(0.0f, Math.min(1.0f, activeMedElapsedTicks / (float) activeMedTotalTicks));
	}

	public float getActiveMedRemainingSeconds() {
		if (!isMedUseActive()) return 0.0f;
		return Math.max(0, activeMedTotalTicks - activeMedElapsedTicks) / 20.0f;
	}

	public String getActiveMedItemId() {
		return activeMedItemId;
	}

	@Nullable
	public BodyPart getActiveMedTargetPart() {
		return activeMedTargetPart;
	}

	public float getHp(BodyPart part) {
		return hpMap.getOrDefault(part, getEffectiveMaxHp(part));
	}

	public float getEffectiveMaxHp(BodyPart part) {
		return effectiveMaxHpMap.getOrDefault(part, part.getMaxHp());
	}

	public void setEffectiveMaxHp(BodyPart part, float value) {
		float clamped = Math.max(1.0f, Math.min(part.getMaxHp(), value));
		effectiveMaxHpMap.put(part, clamped);
		if (getHp(part) > clamped) {
			hpMap.put(part, clamped);
		}
	}

	public float getTotalEffectiveMaxHp() {
		float sum = 0.0f;
		for (BodyPart part : BodyPart.values()) {
			sum += getEffectiveMaxHp(part);
		}
		return sum;
	}

	public float getRatio(BodyPart part) {
		float eff = Math.max(1.0f, getEffectiveMaxHp(part));
		return Math.max(0.0f, Math.min(1.0f, getHp(part) / eff));
	}

	public float getBaseRatio(BodyPart part) {
		return Math.max(0.0f, Math.min(1.0f, getHp(part) / part.getMaxHp()));
	}

	public float getEffectiveMaxRatio(BodyPart part) {
		return Math.max(0.0f, Math.min(1.0f, getEffectiveMaxHp(part) / part.getMaxHp()));
	}

	public void setHp(BodyPart part, float value) {
		float effMax = getEffectiveMaxHp(part);
		hpMap.put(part, Math.max(0.0f, Math.min(effMax, value)));
	}

	public float getTotalHp() {
		float sum = 0.0f;
		for (BodyPart part : BodyPart.values()) {
			sum += getHp(part);
		}
		return sum;
	}

	public float getTotalRatio() {
		return Math.max(0.0f, Math.min(1.0f, getTotalHp() / BodyPart.TOTAL_MAX_HP));
	}

	public boolean isDead() {
		return getHp(BodyPart.HEAD) <= 0.001f || getHp(BodyPart.THORAX) <= 0.001f;
	}

	public boolean isBlacked(BodyPart part) {
		return getHp(part) <= 0.001f;
	}

	public static boolean isLimb(BodyPart part) {
		return part == BodyPart.LEFT_ARM
			|| part == BodyPart.RIGHT_ARM
			|| part == BodyPart.LEFT_LEG
			|| part == BodyPart.RIGHT_LEG;
	}

	public boolean hasInjury(BodyPart part, InjuryType type) {
		EnumSet<InjuryType> set = injuries.get(part);
		return set != null && set.contains(type);
	}

	public void addInjury(BodyPart part, InjuryType type) {
		if (part == null || type == null) return;
		if (type == InjuryType.FRACTURE && !isLimb(part)) {
			return;
		}
		EnumSet<InjuryType> set = injuries.computeIfAbsent(part, p -> EnumSet.noneOf(InjuryType.class));
		if (type == InjuryType.HEAVY_BLEED) {
			set.remove(InjuryType.LIGHT_BLEED);
			set.add(InjuryType.HEAVY_BLEED);
		} else if (type == InjuryType.LIGHT_BLEED) {
			if (!set.contains(InjuryType.HEAVY_BLEED)) {
				set.add(InjuryType.LIGHT_BLEED);
			}
		} else {
			set.add(type);
		}
	}

	public boolean removeInjury(BodyPart part, InjuryType type) {
		EnumSet<InjuryType> set = injuries.get(part);
		return set != null && set.remove(type);
	}

	public EnumSet<InjuryType> getInjuries(BodyPart part) {
		EnumSet<InjuryType> set = injuries.get(part);
		return (set == null || set.isEmpty()) ? EnumSet.noneOf(InjuryType.class) : EnumSet.copyOf(set);
	}

		return new ArrayList<>(recentHits);
	}

		return lastFatalReport;
	}

	private void recordReport(DamageReport report, boolean fatal) {
		recentHits.addFirst(report);
		while (recentHits.size() > MAX_HIT_LOG) {
			recentHits.removeLast();
		}
		if (fatal) {
			lastFatalReport = report;
		}
	}

	public boolean applyHit(

		HitZone zone,
		float rawDamage,

		float distanceMeters
	) {
		if (rawDamage <= 0.0f) {
			return isDead();
		}

		ZonalArmorHelper.ArmorResult armorRes = ZonalArmorHelper.evaluateHit(
			loadout, zone, rawDamage, penetrationPower,
			armorWearMultiplier, bypassesArmor, canRicochet, ricochetRoll
		);
		BodyPart targetPart = zone.getBodyPart();
		float dmg = armorRes.remainingDamage();
		boolean oneshot = false;
		boolean overflowed = false;

		// Rule 1: Unprotected HEAD_EYES ("Голова — глаза") -> instant oneshot (only for non-bypassing hits)
		if (zone.isUnprotectedOneshot() && !bypassesArmor && !armorRes.protectedZone()) {
			setHp(BodyPart.HEAD, 0.0f);
			dmg = Math.max(dmg, BodyPart.HEAD.getMaxHp());
			oneshot = true;
		} else if (!targetPart.isVital()) {
			float currentLimbHp = getHp(targetPart);
			if (currentLimbHp >= dmg) {
				setHp(targetPart, currentLimbHp - dmg);
			} else {
				float excess = dmg - currentLimbHp;
				setHp(targetPart, 0.0f);
				if (excess > 0.0f) {
					setHp(BodyPart.THORAX, getHp(BodyPart.THORAX) - excess);
					overflowed = true;
				}
			}
		} else {
			setHp(targetPart, getHp(targetPart) - dmg);
		}

		boolean killed = isDead();

		// Rule 3: Combat injuries (only for penetrating/unblocked non-bypassing alive hits)
		if (!killed && !bypassesArmor && !armorRes.ricocheted() && !armorRes.stoppedByArmor() && dmg > 0.0f) {
			float heavyChance = dmg >= 25.0f ? Math.min(0.45f, 0.12f + (dmg - 25.0f) * 0.008f) : 0.0f;
			if (dmg >= 25.0f && bleedRoll < heavyChance) {
				addInjury(targetPart, InjuryType.HEAVY_BLEED);
			} else if (dmg >= 10.0f) {
				float lightChance = Math.min(0.50f, 0.18f + (dmg - 10.0f) * 0.010f);
				float threshold = heavyChance + lightChance;
				if (bleedRoll < threshold) {
					addInjury(targetPart, InjuryType.LIGHT_BLEED);
				}
			}

			if (isLimb(targetPart)) {
				float fractureChance = 0.0f;
				if (isBlacked(targetPart) && dmg >= 20.0f) {
					fractureChance = 0.70f;
				} else if (dmg >= 18.0f) {
					fractureChance = Math.min(0.45f, 0.15f + (dmg - 18.0f) * 0.008f);
				}
				if (fractureChance > 0.0f && fractureRoll < fractureChance) {
					addInjury(targetPart, InjuryType.FRACTURE);
				}
			}
		}

		DamageReport report = new DamageReport(
			zone, targetPart, rawDamage, dmg, overflowed,
			armorRes.stoppedByArmor(), armorRes.ricocheted(), oneshot,
			killerName, weaponName, ammoName, distanceMeters
		);
		recordReport(report, killed);
		return killed;
	}

	/**
	 * Harsh EFT fall impact across BOTH legs with encumbrance weight multiplier and leg-to-thorax overflow.
	 */
	public boolean applyFallDamage(float vanillaAmount, float weightKg, float randomRoll) {
		if (vanillaAmount <= 0.0f) {
			return isDead();
		}

		double excessKg = Math.max(0.0, weightKg - 20.0);
		double weightMult = 1.0 + Math.min(0.80, excessKg * 0.02);
		float effectiveFall = (float) (vanillaAmount * weightMult);
		float perLegDamage = effectiveFall * 11.0f;

		float leftBefore = getHp(BodyPart.LEFT_LEG);
		float rightBefore = getHp(BodyPart.RIGHT_LEG);
		float leftExcess = Math.max(0.0f, perLegDamage - leftBefore);
		float rightExcess = Math.max(0.0f, perLegDamage - rightBefore);

		setHp(BodyPart.LEFT_LEG, leftBefore - perLegDamage);
		setHp(BodyPart.RIGHT_LEG, rightBefore - perLegDamage);

		float thoraxOverflow = leftExcess + rightExcess;
		if (thoraxOverflow > 0.0f) {
			setHp(BodyPart.THORAX, getHp(BodyPart.THORAX) - thoraxOverflow);
		}

		if (effectiveFall >= 6.0f) {
			addInjury(BodyPart.LEFT_LEG, InjuryType.FRACTURE);
			addInjury(BodyPart.RIGHT_LEG, InjuryType.FRACTURE);
			if (effectiveFall >= 8.0f) {
				addInjury(BodyPart.LEFT_LEG, InjuryType.HEAVY_BLEED);
				addInjury(BodyPart.RIGHT_LEG, InjuryType.LIGHT_BLEED);
			} else {
				addInjury(BodyPart.LEFT_LEG, InjuryType.LIGHT_BLEED);
				if (randomRoll < 0.50f) {
					addInjury(BodyPart.RIGHT_LEG, InjuryType.LIGHT_BLEED);
				}
			}
		} else if (effectiveFall >= 3.0f) {
			addInjury(BodyPart.LEFT_LEG, InjuryType.FRACTURE);
			if (effectiveFall >= 4.0f || randomRoll < 0.50f) {
				addInjury(BodyPart.RIGHT_LEG, InjuryType.FRACTURE);
			}
		}

		boolean killed = isDead();
		DamageReport report = new DamageReport(
			HitZone.LEFT_LEG,
			BodyPart.LEFT_LEG,
			perLegDamage * 2.0f,
			perLegDamage * 2.0f,
			thoraxOverflow > 0.0f,
			false,
			false,
			false,
			"Fall Impact",
			"Gravity",
			String.format(Locale.ROOT, "%.1f kg", weightKg),
			vanillaAmount + 3.0f
		);
		recordReport(report, killed);
		return killed;
	}

	/**
	 * Advances bleeding, painkiller timer, broken-leg sprint pain, and blacked-stomach exhaustion by 1 server tick.
	 */
	public TickResult tickEffects(boolean isSprinting) {
		boolean stateChanged = false;
		boolean killedByBloodLoss = false;
		boolean bleedDamaged = false;
		boolean heavyBleedActive = countInjuries(InjuryType.HEAVY_BLEED) > 0;
		boolean sprintPainTriggered = false;
		float exhaustionToAdd = 0.0f;

		if (painkillerTicks > 0) {
			painkillerTicks--;
			if (painkillerTicks == 0 || painkillerTicks % 20 == 0) {
				stateChanged = true;
			}
		}

		// 1. Bleeding ticks (every 40 ticks = 2.0s)
		if (hasAnyBleed()) {
			bleedTickCounter++;
			if (bleedTickCounter >= 40) {
				bleedTickCounter = 0;
				bleedDamaged = true;
				stateChanged = true;

				float totalBleedDmg = 0.0f;
				boolean overflowed = false;
				boolean anyHeavy = false;
				for (BodyPart part : BodyPart.values()) {
					float drain = 0.0f;
					if (hasHeavyBleed(part)) {
						drain = 4.0f;
						anyHeavy = true;
					} else if (hasLightBleed(part)) {
						drain = 1.5f;
					}
					if (drain > 0.0f) {
						totalBleedDmg += drain;
						if (applyInternalDamage(part, drain)) {
							overflowed = true;
						}
					}
				}

				if (isDead()) {
					killedByBloodLoss = true;
					recordReport(new DamageReport(
						HitZone.THORAX, BodyPart.THORAX, totalBleedDmg, totalBleedDmg,
						overflowed, false, false, false,
						"Blood Loss", anyHeavy ? "Heavy Bleeding" : "Light Bleeding", "", 0.0f
					), true);
				}
			}
		} else {
			bleedTickCounter = 0;
		}

		// 2. Sprinting on impaired legs while painkiller is active (1.5 HP / sec)
		int impairedLegs = getImpairedLegCount();
		if (impairedLegs > 0 && hasPainkiller() && isSprinting && !isDead()) {
			sprintPainTickCounter++;
			if (sprintPainTickCounter >= 20) {
				sprintPainTickCounter = 0;
				sprintPainTriggered = true;
				stateChanged = true;

				boolean leftImpaired = isBlacked(BodyPart.LEFT_LEG) || hasFracture(BodyPart.LEFT_LEG);
				boolean rightImpaired = isBlacked(BodyPart.RIGHT_LEG) || hasFracture(BodyPart.RIGHT_LEG);
				float perLeg = 1.5f / Math.max(1, (leftImpaired ? 1 : 0) + (rightImpaired ? 1 : 0));
				boolean overflowed = false;
				if (leftImpaired && applyInternalDamage(BodyPart.LEFT_LEG, perLeg)) {
					overflowed = true;
				}
				if (rightImpaired && applyInternalDamage(BodyPart.RIGHT_LEG, perLeg)) {
					overflowed = true;
				}
				if (isDead()) {
					killedByBloodLoss = true;
					recordReport(new DamageReport(
						HitZone.LEFT_LEG, BodyPart.LEFT_LEG, 1.5f, 1.5f,
						overflowed, false, false, false,
						"Pain Shock", "Broken Legs Sprint", "", 0.0f
					), true);
				}
			}
		} else {
			sprintPainTickCounter = 0;
		}

		// 3. Blacked stomach hunger/hydration drain (every 40 ticks)
		boolean stomachBlacked = isStomachBlacked();
		if (stomachBlacked && !isDead()) {
			stomachDrainTickCounter++;
			if (stomachDrainTickCounter >= 40) {
				stomachDrainTickCounter = 0;
				exhaustionToAdd = 0.45f;
			}
		} else {
			stomachDrainTickCounter = 0;
		}

		return new TickResult(
			stateChanged,
			killedByBloodLoss,
			bleedDamaged,
			heavyBleedActive,
			sprintPainTriggered,
			impairedLegs,
			getImpairedArmCount(),
			stomachBlacked,
			hasPainkiller(),
			exhaustionToAdd
		);
	}

	private boolean applyInternalDamage(BodyPart part, float amount) {
		if (!part.isVital()) {
			float cur = getHp(part);
			if (cur >= amount) {
				setHp(part, cur - amount);
				return false;
			}
			float excess = amount - cur;
			setHp(part, 0.0f);
			if (excess > 0.0f) {
				setHp(BodyPart.THORAX, getHp(BodyPart.THORAX) - excess);
				return true;
			}
			return false;
		}
		setHp(part, getHp(part) - amount);
		return false;
	}

	/**
	 * Gentle natural regeneration / food healing helper (only heals non-blacked parts up to effectiveMaxHp).
	 */
	public void healEvenly(float amount) {
		if (amount <= 0.0f || isDead()) return;
		for (BodyPart part : BodyPart.values()) {
			float cur = getHp(part);
			float effMax = getEffectiveMaxHp(part);
			if (cur > 0.001f && cur < effMax) {
				setHp(part, cur + amount);
			}
		}
	}

	public NbtCompound writeNbt() {
		NbtCompound root = new NbtCompound();

		NbtCompound effNbt = new NbtCompound();
		NbtCompound injNbt = new NbtCompound();
		for (BodyPart part : BodyPart.values()) {
			root.putFloat(part.getId(), getHp(part));
			effNbt.putFloat(part.getId(), getEffectiveMaxHp(part));
			int mask = 0;
			for (InjuryType t : getInjuries(part)) {
				mask |= t.getBitMask();
			}
			if (mask != 0) {
				injNbt.putInt(part.getId(), mask);
			}
		}
		root.put("EffectiveMaxHp", effNbt);
		if (!injNbt.isEmpty()) {
			root.put("Injuries", injNbt);
		}

		root.putInt("PainkillerTicks", painkillerTicks);
		root.putInt("BleedTick", bleedTickCounter);
		root.putInt("SprintPainTick", sprintPainTickCounter);
		root.putInt("StomachDrainTick", stomachDrainTickCounter);

		if (isMedUseActive()) {
			root.putString("ActiveMedItem", activeMedItemId);
			root.putInt("ActiveMedElapsed", activeMedElapsedTicks);
			root.putInt("ActiveMedTotal", activeMedTotalTicks);
			if (activeMedTargetPart != null) {
				root.putString("ActiveMedPart", activeMedTargetPart.getId());
			}
		}

		NbtList hits = new NbtList();
		for (DamageReport r : recentHits) {
			hits.add(r.toNbt());
		}
		root.put("RecentHits", hits);
		if (lastFatalReport != null) {
			root.put("Fatal", lastFatalReport.toNbt());
		}
		return root;
	}

	public void readNbt(NbtCompound root) {
		if (root == null) {
			resetFull();
			return;
		}

		// 1. Read EffectiveMaxHp BEFORE part HP so setHp clamps accurately
		NbtCompound effNbt = root.contains("EffectiveMaxHp", NbtElement.COMPOUND_TYPE)
			? root.getCompound("EffectiveMaxHp")
			: null;
		for (BodyPart part : BodyPart.values()) {
			float eff = (effNbt != null && effNbt.contains(part.getId()))
				? effNbt.getFloat(part.getId())
				: part.getMaxHp();
			effectiveMaxHpMap.put(part, Math.max(1.0f, Math.min(part.getMaxHp(), eff)));
		}

		// 2. Read HP per part
		for (BodyPart part : BodyPart.values()) {
			if (root.contains(part.getId())) {
				setHp(part, root.getFloat(part.getId()));
			} else {
				setHp(part, getEffectiveMaxHp(part));
			}
		}

		// 3. Read Injuries bitmask per part
		NbtCompound injNbt = root.contains("Injuries", NbtElement.COMPOUND_TYPE)
			? root.getCompound("Injuries")
			: null;
		for (BodyPart part : BodyPart.values()) {
			EnumSet<InjuryType> set = EnumSet.noneOf(InjuryType.class);
			if (injNbt != null && injNbt.contains(part.getId())) {
				int mask = injNbt.getInt(part.getId());
				for (InjuryType t : InjuryType.values()) {
					if ((mask & t.getBitMask()) != 0) {
						if (t != InjuryType.FRACTURE || isLimb(part)) {
							set.add(t);
						}
					}
				}
				if (set.contains(InjuryType.HEAVY_BLEED)) {
					set.remove(InjuryType.LIGHT_BLEED);
				}
			}
			injuries.put(part, set);
		}

		painkillerTicks = Math.max(0, Math.min(MAX_PAINKILLER_TICKS, root.getInt("PainkillerTicks")));
		bleedTickCounter = Math.max(0, root.getInt("BleedTick"));
		sprintPainTickCounter = Math.max(0, root.getInt("SprintPainTick"));
		stomachDrainTickCounter = Math.max(0, root.getInt("StomachDrainTick"));

		if (root.contains("ActiveMedItem", NbtElement.STRING_TYPE) && root.getInt("ActiveMedTotal") > 0) {
			this.activeMedItemId = root.getString("ActiveMedItem");
			this.activeMedElapsedTicks = Math.max(0, root.getInt("ActiveMedElapsed"));
			this.activeMedTotalTicks = Math.max(1, root.getInt("ActiveMedTotal"));
			this.activeMedTargetPart = root.contains("ActiveMedPart", NbtElement.STRING_TYPE)
				? BodyPart.byId(root.getString("ActiveMedPart"))
				: null;
		} else {
			clearActiveMedUse();
		}

		recentHits.clear();
		if (root.contains("RecentHits", NbtElement.LIST_TYPE)) {
			NbtList hits = root.getList("RecentHits", NbtElement.COMPOUND_TYPE);
			for (int i = 0; i < hits.size(); i++) {
				recentHits.addLast(DamageReport.fromNbt(hits.getCompound(i)));
			}
		}
		lastFatalReport = root.contains("Fatal", NbtElement.COMPOUND_TYPE)
			? DamageReport.fromNbt(root.getCompound("Fatal"))
			: null;
	}
}

package com.tarkovcraft.health;

/**
 * Granular Tarkov hit zones mapped onto BodyPart health pools.
 * Armor protection is evaluated strictly against the exact HitZone struck —
 * e.g. a helmet covering only HEAD_TOP does NOT protect HEAD_EARS or HEAD_EYES.
 */
public enum HitZone {
	HEAD_TOP("head_top", BodyPart.HEAD, false),
	HEAD_EYES("head_eyes", BodyPart.HEAD, true),
	HEAD_JAWS("head_jaws", BodyPart.HEAD, false),
	HEAD_EARS("head_ears", BodyPart.HEAD, false),
	NECK("neck", BodyPart.THORAX, false),
	THORAX("thorax", BodyPart.THORAX, false),
	ARMPIT_LEFT("armpit_left", BodyPart.THORAX, false),
	ARMPIT_RIGHT("armpit_right", BodyPart.THORAX, false),
	STOMACH("stomach", BodyPart.STOMACH, false),
	LEFT_ARM("left_arm", BodyPart.LEFT_ARM, false),
	RIGHT_ARM("right_arm", BodyPart.RIGHT_ARM, false),
	LEFT_LEG("left_leg", BodyPart.LEFT_LEG, false),
	RIGHT_LEG("right_leg", BodyPart.RIGHT_LEG, false);

	private final String id;
	private final BodyPart bodyPart;
	private final boolean unprotectedOneshot;

	HitZone(String id, BodyPart bodyPart, boolean unprotectedOneshot) {
		this.id = id;
		this.bodyPart = bodyPart;
		this.unprotectedOneshot = unprotectedOneshot;
	}

	public String getId() {
		return id;
	}

	public BodyPart getBodyPart() {
		return bodyPart;
	}

	/**
	 * True for HEAD_EYES ("Голова — глаза"): an unprotected hit is an instant lethal oneshot.
	 */
	public boolean isUnprotectedOneshot() {
		return unprotectedOneshot;
	}

	/**
	 * True for helmet dome/ear zones eligible for ballistic ricochet.
	 */
	public boolean isHelmetRicochetZone() {
		return this == HEAD_TOP || this == HEAD_EARS;
	}

	public String getTranslationKey() {
		return "hitzone.tarkovcraft." + id;
	}

	public static HitZone byId(String id) {
		for (HitZone zone : values()) {
			if (zone.id.equalsIgnoreCase(id)) {
				return zone;
			}
		}
		return THORAX;
	}
}

// ========== .\src\main\java\com\tarkovcraft\health\HitZoneBoxes.java ==========
package com.tarkovcraft.health;

import net.minecraft.entity.LivingEntity;
import net.minecraft.util.math.Vec3d;
import org.jetbrains.annotations.Nullable;

import java.util.List;

/**
 * Single source of truth for 3D oriented bounding boxes (OBB) of all 13 HitZones.
 * Used both by server-side HitZoneResolver raycasting and client-side HitZoneDebugRenderer.
 */
public final class HitZoneBoxes {
	public static final int[][] EDGE_INDICES = {
		{0, 1}, {1, 2}, {2, 3}, {3, 0},
		{4, 5}, {5, 6}, {6, 7}, {7, 4},
		{0, 4}, {1, 5}, {2, 6}, {3, 7}
	};

	private HitZoneBoxes() {
	}

	public record ZoneBox(
		HitZone zone,
		Vec3d origin,
		float yawDeg,
		double minRight, double maxRight,
		double minY, double maxY,
		double minFwd, double maxFwd
	) {
		/**
		 * Kay-Kajiya 3-axis OBB slab intersection. Returns entry t >= 0 along normalized rayDir, or Double.NaN on miss.
		 */
		public double intersectRay(Vec3d rayOrigin, Vec3d rayDir, double maxDist) {
			double yawRad = Math.toRadians(yawDeg);
			double fwdX = -Math.sin(yawRad);
			double fwdZ = Math.cos(yawRad);
			double rightX = -fwdZ;
			double rightZ = fwdX;

			double dx = rayOrigin.x - origin.x;
			double dy = rayOrigin.y - origin.y;
			double dz = rayOrigin.z - origin.z;

			double ox = dx * rightX + dz * rightZ;
			double oy = dy;
			double oz = dx * fwdX + dz * fwdZ;

			double rdx = rayDir.x * rightX + rayDir.z * rightZ;
			double rdy = rayDir.y;
			double rdz = rayDir.x * fwdX + rayDir.z * fwdZ;

			double tMin = 0.0;
			double tMax = maxDist;

			// Right axis
			if (Math.abs(rdx) < 1e-9) {
				if (ox < minRight || ox > maxRight) return Double.NaN;
			} else {
				double t1 = (minRight - ox) / rdx;
				double t2 = (maxRight - ox) / rdx;
				if (t1 > t2) { double tmp = t1; t1 = t2; t2 = tmp; }
				tMin = Math.max(tMin, t1);
				tMax = Math.min(tMax, t2);
				if (tMin > tMax) return Double.NaN;
			}

			// Y axis
			if (Math.abs(rdy) < 1e-9) {
				if (oy < minY || oy > maxY) return Double.NaN;
			} else {
				double t1 = (minY - oy) / rdy;
				double t2 = (maxY - oy) / rdy;
				if (t1 > t2) { double tmp = t1; t1 = t2; t2 = tmp; }
				tMin = Math.max(tMin, t1);
				tMax = Math.min(tMax, t2);
				if (tMin > tMax) return Double.NaN;
			}

			// Forward axis
			if (Math.abs(rdz) < 1e-9) {
				if (oz < minFwd || oz > maxFwd) return Double.NaN;
			} else {
				double t1 = (minFwd - oz) / rdz;
				double t2 = (maxFwd - oz) / rdz;
				if (t1 > t2) { double tmp = t1; t1 = t2; t2 = tmp; }
				tMin = Math.max(tMin, t1);
				tMax = Math.min(tMax, t2);
				if (tMin > tMax) return Double.NaN;
			}

			return (tMin <= tMax && tMin >= 0.0 && tMin <= maxDist) ? tMin : Double.NaN;
		}

		/**
		 * Returns the 8 world-space vertices of this OBB in standard box order (0..3 bottom, 4..7 top).
		 */
		public Vec3d[] getWorldCorners() {
			double yawRad = Math.toRadians(yawDeg);
			double fwdX = -Math.sin(yawRad);
			double fwdZ = Math.cos(yawRad);
			double rightX = -fwdZ;
			double rightZ = fwdX;

			return new Vec3d[]{
				corner(rightX, rightZ, fwdX, fwdZ, minRight, minY, minFwd),
				corner(rightX, rightZ, fwdX, fwdZ, maxRight, minY, minFwd),
				corner(rightX, rightZ, fwdX, fwdZ, maxRight, minY, maxFwd),
				corner(rightX, rightZ, fwdX, fwdZ, minRight, minY, maxFwd),
				corner(rightX, rightZ, fwdX, fwdZ, minRight, maxY, minFwd),
				corner(rightX, rightZ, fwdX, fwdZ, maxRight, maxY, minFwd),
				corner(rightX, rightZ, fwdX, fwdZ, maxRight, maxY, maxFwd),
				corner(rightX, rightZ, fwdX, fwdZ, minRight, maxY, maxFwd)
			};
		}

		private Vec3d corner(double rx, double rz, double fx, double fz, double r, double y, double f) {
			return new Vec3d(
				origin.x + rx * r + fx * f,
				origin.y + y,
				origin.z + rz * r + fz * f
			);
		}
	}

	public record RayHit(HitZone zone, double distance, Vec3d hitPos) {
	}

	public static List<ZoneBox> computeForEntity(LivingEntity entity) {
		return compute(entity.getPos(), entity.getHeight(), entity.getWidth(), entity.getBodyYaw(), entity.getHeadYaw());
	}

	public static List<ZoneBox> compute(Vec3d feetPos, float height, float width, float bodyYawDeg, float headYawDeg) {
		double h = Math.max(0.4, height);
		double ws = Math.max(0.5, width / 0.60);

		return List.of(
			// Head zones (use headYawDeg)
			box(HitZone.HEAD_TOP, feetPos, headYawDeg, -0.23 * ws, 0.23 * ws, 0.92 * h, 1.02 * h, -0.23 * ws, 0.23 * ws),
			box(HitZone.HEAD_TOP, feetPos, headYawDeg, -0.22 * ws, 0.22 * ws, 0.78 * h, 0.92 * h, -0.23 * ws, 0.02 * ws),
			box(HitZone.HEAD_EYES, feetPos, headYawDeg, -0.19 * ws, 0.19 * ws, 0.85 * h, 0.92 * h, 0.02 * ws, 0.24 * ws),
			box(HitZone.HEAD_JAWS, feetPos, headYawDeg, -0.19 * ws, 0.19 * ws, 0.78 * h, 0.85 * h, 0.02 * ws, 0.24 * ws),
			box(HitZone.HEAD_EARS, feetPos, headYawDeg, -0.25 * ws, -0.19 * ws, 0.81 * h, 0.92 * h, -0.12 * ws, 0.12 * ws),
			box(HitZone.HEAD_EARS, feetPos, headYawDeg, 0.19 * ws, 0.25 * ws, 0.81 * h, 0.92 * h, -0.12 * ws, 0.12 * ws),
			// Body & limbs (use bodyYawDeg)
			box(HitZone.NECK, feetPos, bodyYawDeg, -0.13 * ws, 0.13 * ws, 0.74 * h, 0.78 * h, -0.13 * ws, 0.13 * ws),
			box(HitZone.THORAX, feetPos, bodyYawDeg, -0.20 * ws, 0.20 * ws, 0.54 * h, 0.74 * h, -0.15 * ws, 0.15 * ws),
			box(HitZone.STOMACH, feetPos, bodyYawDeg, -0.20 * ws, 0.20 * ws, 0.40 * h, 0.54 * h, -0.15 * ws, 0.15 * ws),
			box(HitZone.ARMPIT_LEFT, feetPos, bodyYawDeg, -0.23 * ws, -0.17 * ws, 0.60 * h, 0.72 * h, -0.14 * ws, 0.14 * ws),
			box(HitZone.ARMPIT_RIGHT, feetPos, bodyYawDeg, 0.17 * ws, 0.23 * ws, 0.60 * h, 0.72 * h, -0.14 * ws, 0.14 * ws),
			box(HitZone.LEFT_ARM, feetPos, bodyYawDeg, -0.42 * ws, -0.23 * ws, 0.40 * h, 0.74 * h, -0.11 * ws, 0.11 * ws),
			box(HitZone.RIGHT_ARM, feetPos, bodyYawDeg, 0.23 * ws, 0.42 * ws, 0.40 * h, 0.74 * h, -0.11 * ws, 0.11 * ws),
			box(HitZone.LEFT_LEG, feetPos, bodyYawDeg, -0.20 * ws, -0.005 * ws, 0.00 * h, 0.40 * h, -0.14 * ws, 0.14 * ws),
			box(HitZone.RIGHT_LEG, feetPos, bodyYawDeg, 0.005 * ws, 0.20 * ws, 0.00 * h, 0.40 * h, -0.14 * ws, 0.14 * ws)
		);
	}

	private static ZoneBox box(
		HitZone zone, Vec3d origin, float yawDeg,
		double minR, double maxR, double minY, double maxY, double minF, double maxF
	) {
		return new ZoneBox(zone, origin, yawDeg, minR, maxR, minY, maxY, minF, maxF);
	}

	@Nullable
	public static RayHit raycastZones(List<ZoneBox> boxes, Vec3d rayOrigin, Vec3d rayDir, double maxDist) {
		double lenSq = rayDir.lengthSquared();
		if (lenSq < 1e-9 || maxDist <= 0.0) {
			return null;
		}
		Vec3d normDir = rayDir.multiply(1.0 / Math.sqrt(lenSq));

		HitZone bestZone = null;
		double bestT = Double.MAX_VALUE;
		for (ZoneBox box : boxes) {
			double t = box.intersectRay(rayOrigin, normDir, maxDist);
			if (!Double.isNaN(t) && t >= 0.0 && t < bestT) {
				bestT = t;
				bestZone = box.zone();
			}
		}

		if (bestZone == null) {
			return null;
		}
		return new RayHit(bestZone, bestT, rayOrigin.add(normDir.multiply(bestT)));
	}
}

package com.tarkovcraft.health;

import net.minecraft.entity.Entity;
import net.minecraft.entity.LivingEntity;
import net.minecraft.entity.damage.DamageSource;
import net.minecraft.registry.tag.DamageTypeTags;
import net.minecraft.util.math.Vec3d;
import org.jetbrains.annotations.Nullable;

import java.util.List;
import java.util.Optional;

/**
 * Pure 3D geometric hit-zone classifier relative to victim's position, height, and body yaw.
 * Separates Head into HEAD_EYES, HEAD_JAWS, HEAD_EARS, HEAD_TOP; includes abstract NECK strip
 * and ARMPIT_LEFT / ARMPIT_RIGHT flank gaps under the arms.
 */
public final class HitZoneResolver {
	private HitZoneResolver() {
	}

	/**
	 * Resolves exact HitZone from a live Minecraft DamageSource hitting a LivingEntity.
	 */
	public static HitZone resolveFromDamage(LivingEntity victim, DamageSource source) {
		String msgId = source.getName();
		if (source.isIn(DamageTypeTags.IS_FALL) || "fall".equals(msgId) || "stalagmite".equals(msgId)) {
			return victim.getRandom().nextBoolean() ? HitZone.LEFT_LEG : HitZone.RIGHT_LEG;
		}
		if (source.isIn(DamageTypeTags.IS_DROWNING)
			|| source.isIn(DamageTypeTags.IS_FIRE)
			|| source.isIn(DamageTypeTags.IS_FREEZING)
			|| "drown".equals(msgId)
			|| "inWall".equals(msgId)
			|| "starve".equals(msgId)
			|| "magic".equals(msgId)
			|| "wither".equals(msgId)) {
			return HitZone.THORAX;
		}

		List<HitZoneBoxes.ZoneBox> boxes = HitZoneBoxes.computeForEntity(victim);
		Entity direct = source.getSource();
		Entity attacker = source.getAttacker();

		// 1. Projectile / Bullet trajectory raycast against 3D OBB HitZoneBoxes
		if (direct != null && direct != attacker) {
			Vec3d directPos = direct.getPos();
			Vec3d vel = direct.getVelocity();
			Vec3d dir = null;

			if (vel.lengthSquared() > 1e-6) {
				dir = vel.normalize();
				Vec3d origin = directPos.subtract(dir.multiply(0.5));
				double maxDist = Math.max(16.0, vel.length() * 2.0 + 6.0);
				HitZoneBoxes.RayHit hit = HitZoneBoxes.raycastZones(boxes, origin, dir, maxDist);
				if (hit != null) {
					return hit.zone();
				}
			}

			Vec3d prevPos = new Vec3d(direct.prevX, direct.prevY, direct.prevZ);
			Vec3d delta = directPos.subtract(prevPos);
			if (delta.lengthSquared() > 1e-8) {
				Vec3d prevDir = delta.normalize();
				HitZoneBoxes.RayHit hit = HitZoneBoxes.raycastZones(boxes, prevPos, prevDir, delta.length() + 6.0);
				if (hit != null) {
					return hit.zone();
				}
				if (dir == null) {
					dir = prevDir;
				}
			}

			if (attacker != null) {
				Vec3d aEye = attacker.getEyePos();
				Vec3d aLook = attacker.getRotationVec(1.0f);
				HitZoneBoxes.RayHit hit = HitZoneBoxes.raycastZones(boxes, aEye, aLook, attacker.distanceTo(victim) + 6.0);
				if (hit != null) {
					return hit.zone();
				}
			}

			// Geometric fallback along projectile line (never raw source.getPosition() feet-height trap)
			Vec3d fallbackPos;
			if (dir != null) {
				Vec3d center = victim.getPos().add(0, victim.getHeight() * 0.5, 0);
				double projT = center.subtract(directPos).dotProduct(dir);
				fallbackPos = directPos.add(dir.multiply(projT));
			} else {
				Vec3d srcPos = source.getPosition();
				fallbackPos = srcPos != null ? srcPos : directPos;
			}
			return resolveGeometric(
				victim.getPos(), victim.getHeight(),
				victim.getBodyYaw(), victim.getHeadYaw(),
				fallbackPos, dir
			);
		}

		// 2. Melee / direct attacker raycast against 3D OBB HitZoneBoxes
		if (attacker != null) {
			Vec3d aEye = attacker.getEyePos();
			Vec3d aLook = attacker.getRotationVec(1.0f);
			HitZoneBoxes.RayHit hit = HitZoneBoxes.raycastZones(boxes, aEye, aLook, 8.0);
			if (hit != null) {
				return hit.zone();
			}
			Vec3d hitPos = raycastMeleeHitPos(victim, attacker);
			return resolveGeometric(
				victim.getPos(), victim.getHeight(),
				victim.getBodyYaw(), victim.getHeadYaw(),
				hitPos, aLook
			);
		}

		Vec3d hitPos = source.getPosition();
		if (hitPos == null) {
			return HitZone.THORAX;
		}
		return resolveGeometric(
			victim.getPos(), victim.getHeight(),
			victim.getBodyYaw(), victim.getHeadYaw(),
			hitPos, null
		);
	}

	private static Vec3d raycastMeleeHitPos(LivingEntity victim, Entity attacker) {
		Vec3d aEye = attacker.getEyePos();
		Vec3d aLook = attacker.getRotationVec(1.0f);
		Optional<Vec3d> boxHit = victim.getBoundingBox().expand(0.15).raycast(aEye, aEye.add(aLook.multiply(6.0)));
		if (boxHit.isPresent()) {
			return boxHit.get();
		}
		double dist = aEye.distanceTo(victim.getPos().add(0, victim.getHeight() * 0.6, 0));
		return aEye.add(aLook.multiply(Math.max(0.5, dist)));
	}

	/**
	 * Pure deterministic geometric classifier — fully unit-testable without Minecraft world state.
	 */
	public static HitZone resolveGeometric(
		Vec3d victimFeetPos,
		float victimHeight,
		float bodyYawDeg,
		Vec3d hitPos,
		@Nullable Vec3d incomingDir
	) {
		return resolveGeometric(victimFeetPos, victimHeight, bodyYawDeg, bodyYawDeg, hitPos, incomingDir);
	}

	/**
	 * Pure deterministic geometric classifier with independent body and head yaw.
	 *
	 * @param victimFeetPos feet world position
	 * @param victimHeight  entity bounding box height (1.80m standing)
	 * @param bodyYawDeg    Minecraft body yaw in degrees (0 = +Z South, 90 = -X West, 180 = -Z North, -90 = +X East)
	 * @param headYawDeg    Minecraft head yaw in degrees (used for head zones relY >= 0.78)
	 * @param hitPos        world impact point
	 * @param incomingDir   normalized projectile travel vector (attacker -> victim), nullable
	 */
	public static HitZone resolveGeometric(
		Vec3d victimFeetPos,
		float victimHeight,
		float bodyYawDeg,
		float headYawDeg,
		Vec3d hitPos,
		@Nullable Vec3d incomingDir
	) {
		double h = Math.max(0.4, victimHeight);
		double relY = (hitPos.y - victimFeetPos.y) / h;

		float activeYawDeg = relY >= 0.78 ? headYawDeg : bodyYawDeg;

		// Victim forward & right horizontal unit vectors (Minecraft yaw convention)
		double yawRad = Math.toRadians(activeYawDeg);
		double fwdX = -Math.sin(yawRad);
		double fwdZ = Math.cos(yawRad);
		// Right vector = cross(up, forward) -> (-fwdZ, fwdX)
		// Check: yaw=0 -> fwd=(0, 1) South -> player's RIGHT is West (-1, 0) -> (-fwdZ, fwdX) = (-1, 0)!
		double rightX = -fwdZ;
		double rightZ = fwdX;

		double dx = hitPos.x - victimFeetPos.x;
		double dz = hitPos.z - victimFeetPos.z;

		// Local signed coordinates: forward > 0 means front of player, right > 0 means player's right side
		double localFwd = dx * fwdX + dz * fwdZ;
		double localRight = dx * rightX + dz * rightZ;

		// Incoming shot direction relative to victim facing (direction FROM victim TO shooter = -incomingDir)
		double shotFromFrontDot;
		double shotFromSideAbs;
		if (incomingDir != null && (incomingDir.x * incomingDir.x + incomingDir.z * incomingDir.z) > 1e-6) {
			Vec3d horiz = new Vec3d(incomingDir.x, 0, incomingDir.z).normalize();
			// If bullet travels in -forward direction, it hit the FRONT face
			shotFromFrontDot = -(horiz.x * fwdX + horiz.z * fwdZ);
			shotFromSideAbs = Math.abs(horiz.x * rightX + horiz.z * rightZ);
		} else {
			double len = Math.hypot(localFwd, localRight);
			shotFromFrontDot = len > 1e-5 ? (localFwd / len) : 1.0;
			shotFromSideAbs = len > 1e-5 ? Math.abs(localRight / len) : 0.0;
		}

		// 1. HEAD ZONE (relY >= 0.78)
		if (relY >= 0.78) {
			// Crown / top of helmet
			if (relY >= 0.93) {
				return HitZone.HEAD_TOP;
			}
			// Side hit in ear height band -> HEAD_EARS (exposed unless helmet covers ears!)
			if ((shotFromSideAbs >= 0.62 || Math.abs(localRight) > Math.abs(localFwd) * 1.15)
				&& relY >= 0.81 && relY <= 0.92) {
				return HitZone.HEAD_EARS;
			}
			// Front face -> EYES vs JAWS
			if (shotFromFrontDot > 0.25) {
				if (relY >= 0.85) {
					return HitZone.HEAD_EYES; // "Голова — глаза"
				}
				return HitZone.HEAD_JAWS;
			}
			// Back of head -> HEAD_TOP / nape
			return HitZone.HEAD_TOP;
		}

		// 2. ABSTRACT NECK STRIP (thin gap between head & thorax: relY in [0.74, 0.78))
		if (relY >= 0.74) {
			return HitZone.NECK;
		}

		// 3. UPPER TORSO + ARMPITS + ARMS (relY in [0.54, 0.74))
		if (relY >= 0.54) {
			// Side-angle shot into upper flank gap -> ARMPIT (bypasses front/back chest plates!)
			if (relY >= 0.59 && relY <= 0.72 && shotFromSideAbs >= 0.72 && Math.abs(localRight) >= 0.12 && Math.abs(localRight) <= 0.24) {
				return localRight >= 0 ? HitZone.ARMPIT_RIGHT : HitZone.ARMPIT_LEFT;
			}
			if (Math.abs(localRight) > 0.19) {
				return localRight >= 0 ? HitZone.RIGHT_ARM : HitZone.LEFT_ARM;
			}
			return HitZone.THORAX;
		}

		// 4. LOWER TORSO / STOMACH + LOWER ARMS (relY in [0.40, 0.54))
		if (relY >= 0.40) {
			if (Math.abs(localRight) > 0.21) {
				return localRight >= 0 ? HitZone.RIGHT_ARM : HitZone.LEFT_ARM;
			}
			return HitZone.STOMACH;
		}

		// 5. LEGS (relY < 0.40)
		return localRight >= 0 ? HitZone.RIGHT_LEG : HitZone.LEFT_LEG;
	}
}

package com.tarkovcraft.health;

import com.tarkovcraft.item.MedicalType;
import com.tarkovcraft.item.TarkovMedicalItem;
import net.minecraft.item.ItemStack;
import net.minecraft.server.network.ServerPlayerEntity;
import net.minecraft.sound.SoundCategory;
import net.minecraft.sound.SoundEvents;
import org.jetbrains.annotations.Nullable;

/**
 * Pure common-side medical application & target priority logic.
 */
public final class MedicalHelper {
	private static final BodyPart[] BLEED_PRIORITY_ORDER = {
		BodyPart.THORAX, BodyPart.HEAD, BodyPart.STOMACH,
		BodyPart.LEFT_LEG, BodyPart.RIGHT_LEG, BodyPart.LEFT_ARM, BodyPart.RIGHT_ARM
	};
	private static final BodyPart[] LIMB_PRIORITY_ORDER = {
		BodyPart.LEFT_LEG, BodyPart.RIGHT_LEG, BodyPart.LEFT_ARM, BodyPart.RIGHT_ARM
	};
	private static final BodyPart[] SURGERY_PRIORITY_ORDER = {
		BodyPart.STOMACH, BodyPart.LEFT_LEG, BodyPart.RIGHT_LEG, BodyPart.LEFT_ARM, BodyPart.RIGHT_ARM
	};

	private MedicalHelper() {
	}

	public record ApplyResult(
		boolean applied,
		@Nullable BodyPart targetPart,
		float exhaustionToAdd,
		boolean depleted
	) {
	}

	public static boolean canApplyToPart(BodyPartHealth health, ItemStack stack, BodyPart part) {
		if (health == null || stack == null || stack.isEmpty() || part == null) {
			return false;
		}
		if (!(stack.getItem() instanceof TarkovMedicalItem med)) {
			return false;
		}
		int remaining = TarkovMedicalItem.getRemaining(stack);
		if (remaining <= 0) {
			return false;
		}
		MedicalType type = med.getMedicalType();
		if (type.isPainkillerOnly()) {
			return health.getPainkillerTicks() < type.getPainkillerTicks() - 20;
		}
		if (type.isCuresHeavyBleed() && health.hasHeavyBleed(part) && remaining >= type.getHeavyBleedCost()) {
			return true;
		}
		if (type.isCuresLightBleed() && health.hasLightBleed(part) && remaining >= type.getLightBleedCost()) {
			return true;
		}
		if (type.isCuresFracture() && health.hasFracture(part) && remaining >= type.getFractureCost()) {
			return true;
		}
		if (type.isSurgicalKit() && !part.isVital() && health.isBlacked(part)) {
			return true;
		}
		return type.getMaxHealPerUse() > 0
			&& !health.isBlacked(part)
			&& health.getHp(part) + 0.01f < health.getEffectiveMaxHp(part);
	}

	@Nullable
	public static BodyPart selectAutoTarget(BodyPartHealth health, ItemStack stack) {
		if (health == null || stack == null || stack.isEmpty() || !(stack.getItem() instanceof TarkovMedicalItem med)) {
			return null;
		}
		int remaining = TarkovMedicalItem.getRemaining(stack);
		if (remaining <= 0) {
			return null;
		}
		MedicalType type = med.getMedicalType();

		if (type.isPainkillerOnly()) {
			return health.getPainkillerTicks() < type.getPainkillerTicks() - 20 ? BodyPart.HEAD : null;
		}

		if (type.isCuresHeavyBleed() && remaining >= type.getHeavyBleedCost()) {
			for (BodyPart part : BLEED_PRIORITY_ORDER) {
				if (health.hasHeavyBleed(part)) {
					return part;
				}
			}
		}

		if (type.isCuresLightBleed() && remaining >= type.getLightBleedCost()) {
			for (BodyPart part : BLEED_PRIORITY_ORDER) {
				if (health.hasLightBleed(part)) {
					return part;
				}
			}
		}

		if (type.isSurgicalKit()) {
			for (BodyPart part : SURGERY_PRIORITY_ORDER) {
				if (health.isBlacked(part)) {
					return part;
				}
			}
		}

		if (type.isCuresFracture() && remaining >= type.getFractureCost()) {
			for (BodyPart part : LIMB_PRIORITY_ORDER) {
				if (health.hasFracture(part)) {
					return part;
				}
			}
		}

		if (type.getMaxHealPerUse() <= 0) {
			return null;
		}

		// HP healing: prioritize vital parts (HEAD, THORAX) by lowest ratio, then non-vital
		BodyPart bestVital = null;
		float bestVitalRatio = 2.0f;
		BodyPart bestOther = null;
		float bestOtherRatio = 2.0f;

		for (BodyPart part : BodyPart.values()) {
			if (!health.isBlacked(part) && health.getHp(part) + 0.01f < health.getEffectiveMaxHp(part)) {
				float ratio = health.getRatio(part);
				if (part.isVital()) {
					if (ratio < bestVitalRatio) {
						bestVitalRatio = ratio;
						bestVital = part;
					}
				} else {
					if (ratio < bestOtherRatio) {
						bestOtherRatio = ratio;
						bestOther = part;
					}
				}
			}
		}
		return bestVital != null ? bestVital : bestOther;
	}

	public static ApplyResult applyMedical(
		BodyPartHealth health,
		ItemStack stack,
		@Nullable BodyPart requestedPart,
		boolean infiniteCreative
	) {
		if (health == null || stack == null || stack.isEmpty() || !(stack.getItem() instanceof TarkovMedicalItem med)) {
			return new ApplyResult(false, null, 0.0f, false);
		}
		BodyPart target = requestedPart != null ? requestedPart : selectAutoTarget(health, stack);
		if (target == null || !canApplyToPart(health, stack, target)) {
			return new ApplyResult(false, null, 0.0f, false);
		}

		MedicalType type = med.getMedicalType();
		int remaining = TarkovMedicalItem.getRemaining(stack);
		int cost = 1;

		if (type.isPainkillerOnly()) {
			health.setPainkillerTicks(Math.max(health.getPainkillerTicks(), type.getPainkillerTicks()));
		} else if (type.isCuresHeavyBleed() && health.hasHeavyBleed(target) && remaining >= type.getHeavyBleedCost()) {
			health.removeInjury(target, InjuryType.HEAVY_BLEED);
			cost = type.getHeavyBleedCost();
		} else if (type.isCuresLightBleed() && health.hasLightBleed(target) && remaining >= type.getLightBleedCost()) {
			health.removeInjury(target, InjuryType.LIGHT_BLEED);
			cost = type.getLightBleedCost();
		} else if (type.isSurgicalKit() && !target.isVital() && health.isBlacked(target)) {
			float newEff = Math.min(
				health.getEffectiveMaxHp(target),
				Math.max(1.0f, Math.round(target.getMaxHp() * type.getSurgeryRetention()))
			);
			health.setEffectiveMaxHp(target, newEff);
			health.setHp(target, 1.0f);
			if (type.isCuresFracture()) {
				health.removeInjury(target, InjuryType.FRACTURE);
			}
		} else if (type.isCuresFracture() && health.hasFracture(target) && remaining >= type.getFractureCost()) {
			health.removeInjury(target, InjuryType.FRACTURE);
			cost = type.getFractureCost();
		} else if (type.getMaxHealPerUse() > 0 && !health.isBlacked(target)
			&& health.getHp(target) + 0.01f < health.getEffectiveMaxHp(target)) {
			float missing = health.getEffectiveMaxHp(target) - health.getHp(target);
			float heal = Math.min(missing, Math.min(type.getMaxHealPerUse(), remaining));
			health.setHp(target, health.getHp(target) + heal);
			cost = Math.min(remaining, Math.max(1, Math.round(heal)));
		} else {
			return new ApplyResult(false, null, 0.0f, false);
		}

		if (type.getPainkillerTicks() > 0 && !type.isPainkillerOnly()) {
			health.setPainkillerTicks(Math.max(health.getPainkillerTicks(), type.getPainkillerTicks()));
		}

		if (!infiniteCreative) {
			TarkovMedicalItem.setRemaining(stack, remaining - cost);
		}
		return new ApplyResult(true, target, type.getExhaustionCost(), stack.isEmpty());
	}

	public static void playMedicalSound(ServerPlayerEntity player, MedicalType type) {
		if (type.isSurgicalKit()) {
			player.getWorld().playSound(null, player.getX(), player.getY(), player.getZ(),
				SoundEvents.ENTITY_SHEEP_SHEAR, SoundCategory.PLAYERS, 0.7f, 0.95f);
		} else if (type.getPainkillerTicks() > 0) {
			player.getWorld().playSound(null, player.getX(), player.getY(), player.getZ(),
				SoundEvents.ENTITY_GENERIC_DRINK, SoundCategory.PLAYERS, 0.7f, 1.3f);
		} else if (type.getCapacityMode() == MedicalType.CapacityMode.HP_POOL) {
			player.getWorld().playSound(null, player.getX(), player.getY(), player.getZ(),
				SoundEvents.ITEM_HONEYCOMB_WAX_ON, SoundCategory.PLAYERS, 0.75f, 1.15f);
		} else if (type.isCuresFracture()) {
			player.getWorld().playSound(null, player.getX(), player.getY(), player.getZ(),
				SoundEvents.ITEM_ARMOR_EQUIP_LEATHER, SoundCategory.PLAYERS, 0.8f, 1.2f);
		} else {
			player.getWorld().playSound(null, player.getX(), player.getY(), player.getZ(),
				SoundEvents.BLOCK_WOOL_PLACE, SoundCategory.PLAYERS, 0.8f, 1.1f);
		}
	}
}

package com.tarkovcraft.health;

import com.tarkovcraft.grid.ContainerItemHelper;
import com.tarkovcraft.grid.GridContainer;
import com.tarkovcraft.grid.GridEntry;
import com.tarkovcraft.item.BackpackItem;
import com.tarkovcraft.item.MedicalType;
import com.tarkovcraft.item.TacticalRigItem;
import com.tarkovcraft.item.TarkovMedicalItem;
import com.tarkovcraft.item.TarkovStimulatorItem;
import com.tarkovcraft.loadout.EquipmentSlotType;
import com.tarkovcraft.loadout.PlayerLoadout;
import com.tarkovcraft.loadout.PlayerLoadoutHolder;
import com.tarkovcraft.network.TarkovNetworking;
import com.tarkovcraft.screen.TarkovScreenHandler;
import net.minecraft.item.Item;
import net.minecraft.item.ItemStack;
import net.minecraft.registry.Registries;
import net.minecraft.server.network.ServerPlayerEntity;
import net.minecraft.sound.SoundCategory;
import net.minecraft.sound.SoundEvents;
import net.minecraft.text.Text;
import net.minecraft.util.Hand;
import org.jetbrains.annotations.Nullable;

import java.util.List;
import java.util.Map;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;

/**
 * Server-authoritative timer for applying EFT medical items and stimulators.
 * Prevents instant healing and double-use, verifies source presence each tick,
 * consumes charges only on completion, and syncs progress via BodyPartHealth.
 */
public final class MedicalUseManager {
	private static final Map<UUID, ActiveMedUse> ACTIVE = new ConcurrentHashMap<>();

	private static final class ActiveMedUse {
		final boolean fromHand;
		@Nullable
		final Hand hand;
		final byte srcLoc;
		final int srcIdx;
		@Nullable
		final UUID srcEntryId;
		final Item expectedItem;
		@Nullable
		final BodyPart targetPart;
		final boolean isStimulator;
		final int totalTicks;
		int elapsedTicks = 0;

		ActiveMedUse(
			boolean fromHand,
			@Nullable Hand hand,
			byte srcLoc,
			int srcIdx,
			@Nullable UUID srcEntryId,
			Item expectedItem,
			@Nullable BodyPart targetPart,
			boolean isStimulator,
			int totalTicks
		) {
			this.fromHand = fromHand;
			this.hand = hand;
			this.srcLoc = srcLoc;
			this.srcIdx = srcIdx;
			this.srcEntryId = srcEntryId;
			this.expectedItem = expectedItem;
			this.targetPart = targetPart;
			this.isStimulator = isStimulator;
			this.totalTicks = Math.max(1, totalTicks);
		}
	}

	private MedicalUseManager() {
	}

	public static boolean isUsing(UUID playerId) {
		return playerId != null && ACTIVE.containsKey(playerId);
	}

	public static void interruptForCombat(ServerPlayerEntity player) {
		if (isUsing(player.getUuid()) && player instanceof PlayerLoadoutHolder holder) {
			cancelInternal(player, holder.tarkovcraft$getBodyHealth(), holder, true);
		}
	}

	public static void removePlayer(UUID playerId) {
		if (playerId != null) {
			ACTIVE.remove(playerId);
		}
	}

	public static boolean startFromHand(ServerPlayerEntity player, Hand hand, ItemStack stack) {
		if (player == null || stack == null || stack.isEmpty()) {
			return false;
		}
		return startInternal(player, true, hand, (byte) -1, -1, null, stack, null);
	}

	public static boolean startFromInventory(
		ServerPlayerEntity player,
		byte srcLoc,
		int srcIdx,
		@Nullable UUID srcEntryId,
		@Nullable BodyPart targetPart
	) {
		if (player == null || !(player.currentScreenHandler instanceof TarkovScreenHandler tsh)) {
			return false;
		}
		ItemStack stack = tsh.peekSourceForMed(player, srcLoc, srcIdx, srcEntryId);
		if (stack.isEmpty()) {
			return false;
		}
		return startInternal(player, false, null, srcLoc, srcIdx, srcEntryId, stack, targetPart);
	}

	private static boolean startInternal(
		ServerPlayerEntity player,
		boolean fromHand,
		@Nullable Hand hand,
		byte srcLoc,
		int srcIdx,
		@Nullable UUID srcEntryId,
		ItemStack stack,
		@Nullable BodyPart targetPart
	) {
		if (player.isSpectator() || !player.isAlive() || !(player instanceof PlayerLoadoutHolder holder)) {
			return false;
		}
		UUID playerId = player.getUuid();
		if (holder.tarkovcraft$getBodyHealth().isMedUseActive()) {
			player.sendMessage(Text.translatable("message.tarkovcraft.med_busy"), true);
			return false;
		}

		Item item = stack.getItem();
		if (player.getItemCooldownManager().isCoolingDown(item)) {
			return false;
		}

		BodyPartHealth health = holder.tarkovcraft$getBodyHealth();
		boolean isStim = item instanceof TarkovStimulatorItem;
		int durationTicks;
		if (item instanceof TarkovMedicalItem med) {
			MedicalType type = med.getMedicalType();
			boolean validTarget = targetPart != null
				? MedicalHelper.canApplyToPart(health, stack, targetPart)
				: MedicalHelper.selectAutoTarget(health, stack) != null;
			if (!validTarget) {
				player.sendMessage(Text.translatable(
					targetPart != null ? "message.tarkovcraft.med_invalid_target" : "message.tarkovcraft.med_not_needed"
				), true);
				return false;
			}
			durationTicks = type.getUseDurationTicks();
		} else if (isStim) {
			durationTicks = TarkovStimulatorItem.USE_DURATION_TICKS;
		} else {
			return false;
		}

		ActiveMedUse use = new ActiveMedUse(
			fromHand, hand, srcLoc, srcIdx, srcEntryId,
			item, targetPart, isStim, durationTicks
		);
		ACTIVE.put(playerId, use);
		health.setActiveMedUse(Registries.ITEM.getId(item).toString(), 0, durationTicks, targetPart);
		holder.tarkovcraft$syncBodyHealth();

		player.getWorld().playSound(null, player.getX(), player.getY(), player.getZ(),
			SoundEvents.ITEM_ARMOR_EQUIP_LEATHER, SoundCategory.PLAYERS, 0.45f, 1.2f);
		return true;
	}

	public static void tickPlayer(ServerPlayerEntity player) {
		if (player == null) {
			return;
		}
		UUID playerId = player.getUuid();
		ActiveMedUse active = ACTIVE.get(playerId);
		if (active == null) {
			return;
		}
		if (!(player instanceof PlayerLoadoutHolder holder)) {
			ACTIVE.remove(playerId);
			return;
		}
		BodyPartHealth health = holder.tarkovcraft$getBodyHealth();
		if (!player.isAlive() || player.isSpectator()) {
			cancelInternal(player, health, holder, false);
			return;
		}
		if (player.isSprinting()) {
			cancelInternal(player, health, holder, true);
			return;
		}

		ItemStack current = resolveCurrentStack(player, holder.tarkovcraft$getLoadout(), active);
		if (current.isEmpty() || current.getItem() != active.expectedItem) {
			cancelInternal(player, health, holder, true);
			return;
		}

		active.elapsedTicks++;
		health.setActiveMedUse(
			Registries.ITEM.getId(active.expectedItem).toString(),
			active.elapsedTicks,
			active.totalTicks,
			active.targetPart
		);

		if (active.elapsedTicks >= active.totalTicks) {
			completeInternal(player, holder, health, active);
			return;
		}

		if (active.elapsedTicks % 2 == 0) {
			holder.tarkovcraft$syncBodyHealth();
		}
	}

	private static void cancelInternal(
		ServerPlayerEntity player,
		BodyPartHealth health,
		PlayerLoadoutHolder holder,
		boolean notify
	) {
		ACTIVE.remove(player.getUuid());
		health.clearActiveMedUse();
		holder.tarkovcraft$syncBodyHealth();
		if (notify) {
			player.sendMessage(Text.translatable("message.tarkovcraft.med_cancelled"), true);
		}
	}

	private static void completeInternal(
		ServerPlayerEntity player,
		PlayerLoadoutHolder holder,
		BodyPartHealth health,
		ActiveMedUse active
	) {
		ACTIVE.remove(player.getUuid());
		health.clearActiveMedUse();

		if (active.fromHand && active.hand != null) {
			ItemStack handStack = player.getStackInHand(active.hand);
			if (!handStack.isEmpty() && handStack.getItem() == active.expectedItem) {
				if (active.isStimulator) {
					TarkovStimulatorItem.applyStimulator(player, handStack);
					if (handStack.isEmpty() && !player.isCreative()) {
						player.setStackInHand(active.hand, ItemStack.EMPTY);
					}
				} else if (handStack.getItem() instanceof TarkovMedicalItem med) {
					MedicalHelper.ApplyResult res = MedicalHelper.applyMedical(health, handStack, active.targetPart, player.isCreative());
					if (res.applied()) {
						if (res.exhaustionToAdd() > 0.0f) {
							player.addExhaustion(res.exhaustionToAdd());
						}
						player.getItemCooldownManager().set(med, med.getMedicalType().getCooldownTicks());
						if (handStack.isEmpty() && !player.isCreative()) {
							player.setStackInHand(active.hand, ItemStack.EMPTY);
						}
						player.setHealth(Math.max(1.0f, health.getTotalRatio() * player.getMaxHealth()));
						MedicalHelper.playMedicalSound(player, med.getMedicalType());
					}
				}
			}
			holder.tarkovcraft$syncBodyHealth();
			holder.tarkovcraft$syncLoadout();
			if (player.currentScreenHandler instanceof TarkovScreenHandler tsh) {
				tsh.syncToClient(player);
			}
			return;
		}

		// Inventory/container source
		if (player.currentScreenHandler instanceof TarkovScreenHandler tsh) {
			tsh.completeTimedMedUse(player, active.srcLoc, active.srcIdx, active.srcEntryId, active.targetPart, active.isStimulator);
			return;
		}

		// Player closed inventory mid-heal; own storage (Pocket/Rig/Backpack) still completes safely
		completeInOwnLoadout(player, holder, health, active);
		holder.tarkovcraft$syncBodyHealth();
		holder.tarkovcraft$syncLoadout();
	}

	private static void completeInOwnLoadout(
		ServerPlayerEntity player,
		PlayerLoadoutHolder holder,
		BodyPartHealth health,
		ActiveMedUse active
	) {
		PlayerLoadout loadout = holder.tarkovcraft$getLoadout();
		UUID playerId = player.getUuid();
		boolean creative = player.isCreative();

		if (active.srcLoc == TarkovNetworking.LOC_POCKET) {
			ItemStack st = loadout.getPocket(active.srcIdx);
			if (st.isEmpty() || st.getItem() != active.expectedItem) return;
			applyOnStack(player, health, st, active, creative);
			if (st.isEmpty()) loadout.setPocket(active.srcIdx, ItemStack.EMPTY);
			loadout.invalidateExposedCache();
			return;
		}
		if (active.srcLoc == TarkovNetworking.LOC_SECURE_CONTAINER && active.srcEntryId != null) {
			GridContainer grid = loadout.getSecureContainer();
			GridEntry entry = grid.getEntry(active.srcEntryId);
			if (entry == null || entry.getStack().getItem() != active.expectedItem) return;
			if (applyOnStack(player, health, entry.getStack(), active, creative)) {
				if (entry.getStack().isEmpty()) grid.remove(active.srcEntryId);
				loadout.invalidateExposedCache();
			}
			return;
		}
		if (active.srcLoc == TarkovNetworking.LOC_RIG_GRID && active.srcEntryId != null) {
			ItemStack rig = loadout.getEquipment(EquipmentSlotType.RIG);
			if (rig.isEmpty() || !(rig.getItem() instanceof TacticalRigItem)) return;
			List<GridContainer> grids = ContainerItemHelper.readRigGrids(rig);
			if (active.srcIdx < 0 || active.srcIdx >= grids.size()) return;
			GridEntry entry = grids.get(active.srcIdx).getEntry(active.srcEntryId);
			if (entry == null || entry.getStack().getItem() != active.expectedItem) return;
			if (applyOnStack(player, health, entry.getStack(), active, creative)) {
				if (entry.getStack().isEmpty()) {
					grids.get(active.srcIdx).remove(active.srcEntryId);
				}
				ContainerItemHelper.writeRigGrids(rig, grids);
				loadout.invalidateExposedCache();
			}
			return;
		}
		if (active.srcLoc == TarkovNetworking.LOC_BACKPACK_GRID && active.srcEntryId != null) {
			ItemStack bp = loadout.getEquipment(EquipmentSlotType.BACKPACK);
			if (bp.isEmpty() || !(bp.getItem() instanceof BackpackItem)) return;
			GridContainer grid = ContainerItemHelper.readBackpackGrid(bp);
			GridEntry entry = grid.getEntry(active.srcEntryId);
			if (entry == null || entry.getStack().getItem() != active.expectedItem) return;
			if (applyOnStack(player, health, entry.getStack(), active, creative)) {
				if (entry.getStack().isEmpty()) {
					grid.remove(active.srcEntryId);
				}
				ContainerItemHelper.writeBackpackGrid(bp, grid);
				loadout.invalidateExposedCache();
			}
		}
	}

	private static boolean applyOnStack(
		ServerPlayerEntity player,
		BodyPartHealth health,
		ItemStack st,
		ActiveMedUse active,
		boolean creative
	) {
		if (active.isStimulator) {
			return TarkovStimulatorItem.applyStimulator(player, st);
		}
		if (st.getItem() instanceof TarkovMedicalItem med) {
			MedicalHelper.ApplyResult res = MedicalHelper.applyMedical(health, st, active.targetPart, creative);
			if (res.applied()) {
				if (res.exhaustionToAdd() > 0.0f) {
					player.addExhaustion(res.exhaustionToAdd());
				}
				player.getItemCooldownManager().set(med, med.getMedicalType().getCooldownTicks());
				player.setHealth(Math.max(1.0f, health.getTotalRatio() * player.getMaxHealth()));
				MedicalHelper.playMedicalSound(player, med.getMedicalType());
				return true;
			}
		}
		return false;
	}

	private static ItemStack resolveCurrentStack(
		ServerPlayerEntity player,
		PlayerLoadout loadout,
		ActiveMedUse active
	) {
		if (active.fromHand && active.hand != null) {
			return player.getStackInHand(active.hand);
		}
		if (player.currentScreenHandler instanceof TarkovScreenHandler tsh) {
			return tsh.peekSourceForMed(player, active.srcLoc, active.srcIdx, active.srcEntryId);
		}
		return switch (active.srcLoc) {
			case TarkovNetworking.LOC_POCKET -> loadout.getPocket(active.srcIdx);
			case TarkovNetworking.LOC_SECURE_CONTAINER -> {
				GridEntry entry = active.srcEntryId == null ? null : loadout.getSecureContainer().getEntry(active.srcEntryId);
				yield entry == null ? ItemStack.EMPTY : entry.getStack();
			}
			case TarkovNetworking.LOC_RIG_GRID -> {
				if (active.srcEntryId == null) yield ItemStack.EMPTY;
				ItemStack rig = loadout.getEquipment(EquipmentSlotType.RIG);
				if (rig.isEmpty() || !(rig.getItem() instanceof TacticalRigItem)) yield ItemStack.EMPTY;
				List<GridContainer> grids = ContainerItemHelper.readRigGrids(rig);
				if (active.srcIdx < 0 || active.srcIdx >= grids.size()) yield ItemStack.EMPTY;
				GridEntry entry = grids.get(active.srcIdx).getEntry(active.srcEntryId);
				yield entry != null ? entry.getStack() : ItemStack.EMPTY;
			}
			case TarkovNetworking.LOC_BACKPACK_GRID -> {
				if (active.srcEntryId == null) yield ItemStack.EMPTY;
				ItemStack bp = loadout.getEquipment(EquipmentSlotType.BACKPACK);
				if (bp.isEmpty() || !(bp.getItem() instanceof BackpackItem)) yield ItemStack.EMPTY;
				GridEntry entry = ContainerItemHelper.readBackpackGrid(bp).getEntry(active.srcEntryId);
				yield entry != null ? entry.getStack() : ItemStack.EMPTY;
			}
			default -> ItemStack.EMPTY;
		};
	}
}

package com.tarkovcraft.health;

import com.tarkovcraft.ammo.AmmoTier;
import net.minecraft.entity.Entity;
import net.minecraft.entity.LivingEntity;
import net.minecraft.entity.damage.DamageSource;
import net.minecraft.entity.projectile.ProjectileEntity;
import net.minecraft.item.ItemStack;
import net.minecraft.server.network.ServerPlayerEntity;
import net.minecraft.util.Identifier;

import java.lang.reflect.Method;
import java.util.Locale;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import org.jetbrains.annotations.Nullable;

/**
 * Zero-hard-dependency bridge to TaCZ Refabricated's EntityKineticBullet.
 * Extracts gun ID, ammo ID, and the AmmoTier (HP / FMJ / AP / IGN) captured when the projectile spawns,
 * computing effective penetration power and flesh damage multiplier.
 */
public final class TaCZBulletHelper {
	private record BulletGetters(@Nullable Method getGunId, @Nullable Method getAmmoId) {
	}

	private static final ConcurrentHashMap<Class<?>, BulletGetters> GETTER_CACHE = new ConcurrentHashMap<>();
	private static final Map<Integer, AmmoTier> PROJECTILE_TIERS = new ConcurrentHashMap<>();

	private TaCZBulletHelper() {
	}

	public record BulletInfo(
		boolean isTaczBullet,
		String weaponName,
		String ammoName,
		AmmoTier tier,
		int penetrationPower,
		float damageMultiplier,
		float armorWearMultiplier
	) {
	}

	public static void onEntityLoad(Entity entity) {
		if (!(entity instanceof ProjectileEntity projectile) || !isTaczProjectile(entity)
				|| !(projectile.getOwner() instanceof LivingEntity shooter)) {
			return;
		}
		ItemStack gun = shooter.getMainHandStack();
		if (AmmoTier.isTaczGunStack(gun) && gun.getNbt() != null
				&& gun.getNbt().contains("GunId") && gun.getNbt().getString("GunId").equals(invokeCachedGetter(entity, true))) {
			PROJECTILE_TIERS.put(entity.getId(), AmmoTier.getLoadedTierFromGun(gun));
		}
		if (shooter instanceof ServerPlayerEntity player) {
			MedicalUseManager.interruptForCombat(player);
		}
	}

	public static void onEntityUnload(Entity entity) {
		PROJECTILE_TIERS.remove(entity.getId());
	}

	public static void clearProjectiles() {
		PROJECTILE_TIERS.clear();
	}

	private static boolean isTaczProjectile(@Nullable Entity entity) {
		return entity != null && entity.getClass().getName().contains("EntityKineticBullet");
	}

	public static BulletInfo inspectDamageSource(DamageSource source) {
		Entity direct = source.getSource();
		Entity attacker = source.getAttacker();

		boolean isTacz = isTaczProjectile(direct)
			|| (source.getName() != null && source.getName().toLowerCase(Locale.ROOT).contains("tacz.bullet"));
		if (isTacz) {
			AmmoTier loadedTier = direct != null ? PROJECTILE_TIERS.getOrDefault(direct.getId(), AmmoTier.FMJ) : AmmoTier.FMJ;
			String gunStr = direct != null ? invokeCachedGetter(direct, true) : "";
			String ammoStr = direct != null ? invokeCachedGetter(direct, false) : "";
			String cleanGun = prettifyId(gunStr, "Firearm");
			String cleanAmmo = prettifyId(ammoStr, "Ballistic") + " [" + loadedTier.getBadge() + "]";
			int pen = computePenetration(ammoStr, loadedTier);
			return new BulletInfo(true, cleanGun, cleanAmmo, loadedTier, pen,
				loadedTier.getDamageMultiplier(), loadedTier.getArmorWearMultiplier());
		}

		if (attacker instanceof LivingEntity living) {
			ItemStack main = living.getMainHandStack();
			String wep = !main.isEmpty() ? main.getName().getString() : source.getName();
			String ammo = (direct != null && direct != attacker) ? direct.getName().getString() : "Melee";
			return new BulletInfo(false, wep, ammo, AmmoTier.FMJ, 15, 1.0f, 1.0f);
		}

		return new BulletInfo(false, source.getName(), "", AmmoTier.FMJ, 0, 1.0f, 0.0f);
	}

	private static BulletGetters resolveGetters(Class<?> bulletClass) {
		return GETTER_CACHE.computeIfAbsent(bulletClass, cls -> {
			Method gun = null;
			Method ammo = null;
			try {
				gun = cls.getMethod("getGunId");
			} catch (NoSuchMethodException ignored) {
			}
			try {
				ammo = cls.getMethod("getAmmoId");
			} catch (NoSuchMethodException ignored) {
			}
			return new BulletGetters(gun, ammo);
		});
	}

	private static String invokeCachedGetter(Entity bullet, boolean isGun) {
		try {
			BulletGetters getters = resolveGetters(bullet.getClass());
			Method m = isGun ? getters.getGunId() : getters.getAmmoId();
			if (m == null) {
				return "";
			}
			Object res = m.invoke(bullet);
			if (res instanceof Identifier id) {
				return id.toString();
			}
			return res != null ? res.toString() : "";
		} catch (Exception ignored) {
			return "";
		}
	}

	private static String prettifyId(String id, String fallback) {
		if (id == null || id.isBlank()) return fallback;
		int colon = id.indexOf(':');
		String path = colon >= 0 ? id.substring(colon + 1) : id;
		return path.replace('_', ' ').toUpperCase();
	}

	/**
	 * Computes final penetration power (0..75 scale against Armor Class 1..6 thresholds 10..60)
	 * by combining base caliber penetration with the AmmoTier delta.
	 */
	public static int computePenetration(String ammoId, AmmoTier tier) {
		int base = estimateBaseCaliberPenetration(ammoId);
		return Math.max(2, base + tier.getPenetrationDelta());
	}

	public static int estimateBaseCaliberPenetration(String ammoId) {
		if (ammoId == null) return 22;
		String lower = ammoId.toLowerCase();
		if (lower.contains("50bmg")) return 54;
		if (lower.contains("338")) return 48;
		if (lower.contains("762x54")) return 39;
		if (lower.contains("308") || lower.contains("30_06") || lower.contains("68x51") || lower.contains("792x57")) return 38;
		if (lower.contains("45_70")) return 34;
		if (lower.contains("762x39")) return 31;
		if (lower.contains("58x42") || lower.contains("300blk")) return 30;

// ========== .\src\main\java\com\tarkovcraft\health\ZonalArmorHelper.java ==========
package com.tarkovcraft.health;

import com.tarkovcraft.item.TarkovArmorItem;
import com.tarkovcraft.loadout.EquipmentSlotType;
import com.tarkovcraft.loadout.PlayerLoadout;
import net.minecraft.item.ItemStack;

import java.util.ArrayList;
import java.util.List;

/**
 * Evaluates strictly zonal, layered armor protection (outside-in: HELMET -> FACE_MASK -> BODY_ARMOR).
 * Only intact armor pieces whose coveredZones include the exact HitZone can ricochet, mitigate, or stop a hit.
 * Open zones and broken (0 durability) armor receive ZERO protection.
 */
public final class ZonalArmorHelper {
	private static final EquipmentSlotType[] ARMOR_LAYER_ORDER = {
		EquipmentSlotType.HELMET,
		EquipmentSlotType.FACE_MASK,
		EquipmentSlotType.BODY_ARMOR
	};

	private ZonalArmorHelper() {
	}

	public record ArmorResult(
		float remainingDamage,
		boolean protectedZone,
		boolean stoppedByArmor,
		boolean ricocheted,
		int absorbedArmorClass,
		int durabilityDamageApplied
	) {
	}

	private record ArmorLayer(ItemStack stack, TarkovArmorItem armor, int remainingDurability) {
	}

	public static ArmorResult evaluateHit(PlayerLoadout loadout, HitZone zone, float rawDamage, int penetrationPower) {
		return evaluateHit(loadout, zone, rawDamage, penetrationPower, 1.0f, false, false, 1.0f);
	}

	public static ArmorResult evaluateHit(
		PlayerLoadout loadout,
		HitZone zone,
		float rawDamage,
		int penetrationPower,
		float armorWearMultiplier,
		boolean bypassesArmor,
		boolean canRicochet,
		float ricochetRoll
	) {
		if (bypassesArmor || loadout == null || zone == null) {
			return new ArmorResult(rawDamage, false, false, false, 0, 0);
		}

		List<ArmorLayer> layers = new ArrayList<>(3);
		for (EquipmentSlotType slot : ARMOR_LAYER_ORDER) {
			ItemStack stack = loadout.getEquipment(slot);
			if (stack.isEmpty() || !(stack.getItem() instanceof TarkovArmorItem armor)) {
				continue;
			}
			int remainingDur = TarkovArmorItem.getRemainingDurability(stack);
			if (remainingDur > 0 && armor.coversZone(zone)) {
				layers.add(new ArmorLayer(stack, armor, remainingDur));
			}
		}

		if (layers.isEmpty()) {
			return new ArmorResult(rawDamage, false, false, false, 0, 0);
		}

		float currentDmg = rawDamage;
		int currentPen = Math.max(0, penetrationPower);
		float safeWearMult = Math.max(0.0f, armorWearMultiplier);
		int totalWear = 0;
		int highestClass = 0;

		for (ArmorLayer layer : layers) {
			ItemStack stack = layer.stack();
			TarkovArmorItem armor = layer.armor();
			int maxDur = Math.max(1, stack.getMaxDamage());
			float durRatio = layer.remainingDurability() / (float) maxDur;
			highestClass = Math.max(highestClass, armor.getArmorClass());

			if (canRicochet && zone.isHelmetRicochetZone() && armor.getRicochetChance() > 0.0f) {
				float effRicochet = Math.min(0.65f, armor.getRicochetChance() * (0.5f + 0.5f * durRatio));
				if (ricochetRoll < effRicochet) {
					int wear = Math.max(1, Math.round(currentDmg * 0.08f * armor.getMaterial().getDestructibility() * safeWearMult));
					applyWear(stack, maxDur, wear);
					totalWear += wear;
					return new ArmorResult(Math.max(1.0f, currentDmg * 0.05f), true, true, true, armor.getArmorClass(), totalWear);
				}
			}

			float effectiveThreshold = armor.getArmorClass() * 10.0f * (0.45f + 0.55f * durRatio);
			float baseWear = currentDmg * 0.25f * armor.getMaterial().getDestructibility() * safeWearMult;

			if (currentPen < effectiveThreshold) {
				int wear = Math.max(1, Math.round(baseWear));
				applyWear(stack, maxDur, wear);
				totalWear += wear;
				float blunt = Math.max(1.0f, currentDmg * armor.getMaterial().getBluntThroughput());
				return new ArmorResult(blunt, true, true, false, armor.getArmorClass(), totalWear);
			}

			int wear = Math.max(1, Math.round(baseWear * 1.20f));
			applyWear(stack, maxDur, wear);
			totalWear += wear;

			float overPen = currentPen - effectiveThreshold;
			float reduction = Math.max(0.10f, 0.30f - overPen * 0.01f);
			currentDmg = Math.max(1.0f, currentDmg * (1.0f - reduction));
			currentPen = Math.max(0, Math.round(currentPen - effectiveThreshold * 0.35f));
		}

		return new ArmorResult(currentDmg, true, false, false, highestClass, totalWear);
	}

	private static void applyWear(ItemStack stack, int maxDur, int wear) {
		stack.setDamage(Math.min(maxDur, stack.getDamage() + wear));
	}

	/**
	 * Returns the highest intact armor class covering the given zone, or 0 if exposed / broken.
	 */
	public static int getHighestProtectingClass(PlayerLoadout loadout, HitZone zone) {
		if (loadout == null || zone == null) {
			return 0;
		}
		int best = 0;
		for (EquipmentSlotType slot : ARMOR_LAYER_ORDER) {
			ItemStack stack = loadout.getEquipment(slot);
			if (stack.isEmpty() || !(stack.getItem() instanceof TarkovArmorItem armor)) {
				continue;
			}
			if (TarkovArmorItem.getRemainingDurability(stack) > 0 && armor.coversZone(zone)) {
				best = Math.max(best, armor.getArmorClass());
			}
		}
		return best;
	}
}

	private HeldItemInteraction() {
	}

	public static boolean takesPriority(PlayerEntity player, Hand hand) {
		if (player.isSpectator() || player.isCreative()) {
			return false;
		}
		var stack = player.getStackInHand(hand);
		if (hand == Hand.MAIN_HAND && stack.isEmpty()) {
			stack = player.getOffHandStack();
		}
		return stack.getItem() instanceof TarkovGrenadeItem
			|| stack.getItem() instanceof TarkovArmorRepairItem
			|| stack.getItem() instanceof TarkovMedicalItem
			|| stack.getItem() instanceof TarkovStimulatorItem
			|| AmmoTier.isTaczGunStack(stack);
	}

		this.gasProtection = gasProtection;

	public static int getRemainingDurability(ItemStack stack) {
		if (stack == null || stack.isEmpty() || !(stack.getItem() instanceof TarkovArmorItem armor)) {
			return 0;
		}
		if (armor.getArmorClass() <= 0 || !stack.isDamageable()) {
			return 0;
		}
		return Math.min(getEffectiveMaxDurability(stack), Math.max(0, stack.getMaxDamage() - stack.getDamage()));
	}

	public static int getEffectiveMaxDurability(ItemStack stack) {
		int base = stack.getMaxDamage();
		if (stack.getNbt() == null || !stack.getNbt().contains(MAX_DURABILITY_KEY, net.minecraft.nbt.NbtElement.NUMBER_TYPE)) {
			return base;
		}
		return Math.min(base, Math.max(1, stack.getNbt().getInt(MAX_DURABILITY_KEY)));
	}

	public static int repair(ItemStack stack, int points, float lossRatio) {
		int maximum = getEffectiveMaxDurability(stack);
		int current = getRemainingDurability(stack);
		int gap = maximum - current;
		int restored = Math.min(points, Math.round(gap / (1.0f + lossRatio)));
		int loss = Math.min(maximum - 1, Math.round(restored * lossRatio));
		if (restored + loss > gap) {
			restored--;
			loss = Math.min(maximum - 1, Math.round(restored * lossRatio));
		}
		if (restored <= 0) return 0;
		stack.getOrCreateNbt().putInt(MAX_DURABILITY_KEY, maximum - loss);
		stack.setDamage(stack.getMaxDamage() - current - restored);
		return restored;
	}

	public static String formatDurability(ItemStack stack) {
		return getRemainingDurability(stack) + "/" + getEffectiveMaxDurability(stack) + " (" + stack.getMaxDamage() + ")";
	}

	public static boolean isBroken(ItemStack stack) {
		if (stack == null || stack.isEmpty() || !(stack.getItem() instanceof TarkovArmorItem armor)) {
			return false;
		}
		return armor.getArmorClass() > 0 && getRemainingDurability(stack) <= 0;
	}

	@Override
	public void appendTooltip(ItemStack stack, @Nullable World world, List<Text> tooltip, TooltipContext context) {
		if (armorClass > 0) {
			tooltip.add(Text.translatable("tooltip.tarkovcraft.armor_class", armorClass).formatted(Formatting.AQUA));
			tooltip.add(Text.translatable("tooltip.tarkovcraft.armor_material",
				Text.translatable(material.getTranslationKey())).formatted(Formatting.GRAY));

			int curDur = getRemainingDurability(stack);
			float ratio = curDur / (float) Math.max(1, getEffectiveMaxDurability(stack));
			Formatting durColor = ratio < 0.25f ? Formatting.RED
				: (ratio < 0.50f ? Formatting.YELLOW : Formatting.GRAY);
			tooltip.add(Text.translatable("tooltip.tarkovcraft.armor_durability", formatDurability(stack)).formatted(durColor));

			if (ricochetChance > 0.0f) {
				tooltip.add(Text.translatable("tooltip.tarkovcraft.armor_ricochet", Math.round(ricochetChance * 100.0f))
					.formatted(Formatting.DARK_AQUA));
			}

			if (!coveredZones.isEmpty()) {
				String zonesStr = coveredZones.stream()
					.map(z -> Text.translatable(z.getTranslationKey()).getString())
					.collect(Collectors.joining(", "));
				tooltip.add(Text.translatable("tooltip.tarkovcraft.armor_zones", zonesStr).formatted(Formatting.DARK_GREEN));
			}

			List<HitZone> slotZones = switch (targetSlot) {
				case HELMET -> HELMET_VULNERABLE_ZONES;
				case BODY_ARMOR -> BODY_ARMOR_VULNERABLE_ZONES;
				default -> List.of();
			};
			if (!slotZones.isEmpty()) {
				String exposedStr = slotZones.stream()
					.filter(z -> !coveredZones.contains(z))
					.map(z -> Text.translatable(z.getTranslationKey()).getString())
					.collect(Collectors.joining(", "));
				if (!exposedStr.isEmpty()) {
					tooltip.add(Text.translatable("tooltip.tarkovcraft.armor_exposed", exposedStr).formatted(Formatting.RED));
				}
			}
		} else {
			tooltip.add(Text.translatable("tooltip.tarkovcraft.armor_cosmetic").formatted(Formatting.DARK_GRAY));
		}

		tooltip.add(Text.translatable("tooltip.tarkovcraft.armor_weight", String.format(Locale.ROOT, "%.1f", weightKg))
			.formatted(Formatting.DARK_GRAY));
	}
}

public final class TarkovArmorRepairItem extends Item implements TarkovSizedItem {
	public enum Tier {
		T1(25, 2, 0.30f, 60),
		T2(45, 4, 0.22f, 80),
		T3(65, 5, 0.15f, 90),
		T4(90, 6, 0.10f, 100);

		private final int repairPoints;
		private final int maxArmorClass;
		private final float maxDurabilityLossRatio;
		private final int durationTicks;

		Tier(int repairPoints, int maxArmorClass, float maxDurabilityLossRatio, int durationTicks) {
			this.repairPoints = repairPoints;
			this.maxArmorClass = maxArmorClass;
			this.maxDurabilityLossRatio = maxDurabilityLossRatio;
			this.durationTicks = durationTicks;
		}
	}

	private record Pending(ServerPlayerEntity player, int startTick, int duration, ItemStack expectedKit,
			EquipmentSlotType slot, ItemStack target, Supplier<ItemStack> source, Consumer<ItemStack> replace) {}

	private static final EquipmentSlotType[] REPAIR_PRIORITY = {
		EquipmentSlotType.BODY_ARMOR, EquipmentSlotType.HELMET, EquipmentSlotType.FACE_MASK
	};
	private static final String REPAIR_POINTS_KEY = "TarkovRepairPoints";
	private static final Map<UUID, Pending> PENDING = new HashMap<>();

	private final ItemSize storedItemSize;
	private final ItemRarityTier rarityTier;
	private final float weightKg;
	private final Tier tier;

	public TarkovArmorRepairItem(Settings settings, Tier tier, ItemSize storedItemSize, ItemRarityTier rarityTier, float weightKg) {
		super(settings.maxCount(1));
		this.tier = tier;
		this.storedItemSize = storedItemSize;
		this.rarityTier = rarityTier;
		this.weightKg = Math.max(0.0f, weightKg);
	}

	public int getRepairAmount() {
		return tier.repairPoints;
	}

	public int getMaxArmorClass() {
		return tier.maxArmorClass;
	}

	private int getRemainingPoints(ItemStack stack) {
		if (stack.getNbt() == null || !stack.getNbt().contains(REPAIR_POINTS_KEY)) return tier.repairPoints;
		return Math.max(0, Math.min(tier.repairPoints, stack.getNbt().getInt(REPAIR_POINTS_KEY)));
	}

	@Override
	public ItemSize getStoredItemSize() {
		return storedItemSize;
	}

	@Override
	public ItemRarityTier getRarityTier() {
		return rarityTier;
	}

	@Override
	public float getWeightKg() {
		return weightKg;
	}

	@Override
	public TypedActionResult<ItemStack> use(World world, PlayerEntity user, Hand hand) {
		ItemStack stack = user.getStackInHand(hand);
		if (world.isClient) return TypedActionResult.success(stack, true);
		if (user instanceof ServerPlayerEntity player
			&& start(player, () -> player.getStackInHand(hand), result -> player.setStackInHand(hand, result))) {
			return TypedActionResult.success(stack, false);
		}
		return TypedActionResult.fail(stack);
	}

	public static boolean start(ServerPlayerEntity player, Supplier<ItemStack> source, Consumer<ItemStack> replace) {
		if (!player.isAlive() || player.isSpectator() || PENDING.containsKey(player.getUuid())
			|| !(player instanceof PlayerLoadoutHolder holder) || holder.tarkovcraft$getBodyHealth().isMedUseActive()) return false;
		ItemStack kit = source.get();
		if (!(kit.getItem() instanceof TarkovArmorRepairItem repair) || repair.getRemainingPoints(kit) == 0) return false;
		PlayerLoadout loadout = holder.tarkovcraft$getLoadout();
		for (EquipmentSlotType slot : REPAIR_PRIORITY) {
			ItemStack armor = loadout.getEquipment(slot);
			if (!(armor.getItem() instanceof TarkovArmorItem piece) || !armor.isDamageable()
				|| piece.getArmorClass() > repair.tier.maxArmorClass
				|| TarkovArmorItem.getRemainingDurability(armor) >= TarkovArmorItem.getEffectiveMaxDurability(armor)) continue;
			PENDING.put(player.getUuid(), new Pending(player, player.getServer().getTicks(), repair.tier.durationTicks,
				kit.copy(), slot, armor, source, replace));
			holder.tarkovcraft$getBodyHealth().setActiveMedUse(Registries.ITEM.getId(kit.getItem()).toString(),
				0, repair.tier.durationTicks, null);
			holder.tarkovcraft$syncBodyHealth();
			return true;
		}
		player.sendMessage(Text.translatable("message.tarkovcraft.repair_not_needed"), true);
		return false;
	}

	public static void tickPlayer(ServerPlayerEntity player) {
		Pending pending = PENDING.get(player.getUuid());
		if (pending == null || !(player instanceof PlayerLoadoutHolder holder)) return;
		ItemStack kit = player == pending.player() && player.isAlive() ? pending.source().get() : ItemStack.EMPTY;
		ItemStack armor = holder.tarkovcraft$getLoadout().getEquipment(pending.slot());
		if (player != pending.player() || !player.isAlive() || player.isSpectator() || player.isSprinting()
			|| armor != pending.target() || !ItemStack.areEqual(kit, pending.expectedKit())) {
			finishTimer(player, holder);
			player.sendMessage(Text.translatable("message.tarkovcraft.repair_cancelled"), true);
			return;
		}
		int elapsed = player.getServer().getTicks() - pending.startTick();
		if (elapsed < pending.duration()) {
			holder.tarkovcraft$getBodyHealth().setActiveMedUse(Registries.ITEM.getId(kit.getItem()).toString(),
				elapsed, pending.duration(), null);
			if (elapsed % 2 == 0) holder.tarkovcraft$syncBodyHealth();
			return;
		}
		finishTimer(player, holder);
		TarkovArmorRepairItem repair = (TarkovArmorRepairItem) kit.getItem();
		int previous = TarkovArmorItem.getRemainingDurability(armor);
		int available = repair.getRemainingPoints(kit);
		int spent = TarkovArmorItem.repair(armor, available, repair.tier.maxDurabilityLossRatio);
		if (spent == 0) return;
		if (!player.isCreative()) {
			ItemStack updated = spent == available ? ItemStack.EMPTY : kit.copy();
			if (!updated.isEmpty()) updated.getOrCreateNbt().putInt(REPAIR_POINTS_KEY, available - spent);
			pending.replace().accept(updated);
		}
		if (player.currentScreenHandler instanceof TarkovScreenHandler handler) {
			handler.syncToClient(player);
		} else {
			holder.tarkovcraft$syncLoadout();
		}
		player.getWorld().playSound(null, player.getX(), player.getY(), player.getZ(),
			SoundEvents.BLOCK_ANVIL_USE, SoundCategory.PLAYERS, 0.55f, 1.3f);
		player.sendMessage(Text.translatable("message.tarkovcraft.repair_done",
			TarkovArmorItem.getRemainingDurability(armor) - previous), true);
	}

	private static void finishTimer(ServerPlayerEntity player, PlayerLoadoutHolder holder) {
		PENDING.remove(player.getUuid());
		holder.tarkovcraft$getBodyHealth().clearActiveMedUse();
		holder.tarkovcraft$syncBodyHealth();
	}

	public static void removePlayer(UUID playerId) {
		PENDING.remove(playerId);
	}

	@Override
	public void appendTooltip(ItemStack stack, @Nullable World world, List<Text> tooltip, TooltipContext context) {
		tooltip.add(Text.translatable("tooltip.tarkovcraft.repair_points", getRemainingPoints(stack), tier.repairPoints,
			tier.maxArmorClass, Math.round(tier.maxDurabilityLossRatio * 100)).formatted(net.minecraft.util.Formatting.AQUA));
		tooltip.add(Text.translatable("tooltip.tarkovcraft.item_size", storedItemSize.width(), storedItemSize.height()));
	}
}

/**
 * Openable TarkovCraft storage case (Wallet, Keytool, Docs, SICC, Medcase, THICC, etc.).
 * Persists an internal GridContainer in NBT via ContainerItemHelper.
 */
public class TarkovCaseItem extends Item implements TarkovSizedItem {
	public enum CaseRole {
		GENERAL("general"),
		MEDICAL("medical"),
		INJECTOR("injector"),
		AMMO("ammo"),
		GRENADE("grenade"),
		FOOD("food"),
		WEAPON("weapon"),
		ARMOR_PLATE("armor_plate"),
		MONEY("money");

		private final String translationSuffix;

		CaseRole(String translationSuffix) {
			this.translationSuffix = translationSuffix;
		}

		public String getTranslationKey() {
			return "tooltip.tarkovcraft.case_role." + translationSuffix;
		}

		public static CaseRole fromString(String raw) {
			if (raw == null || raw.isBlank()) {
				return GENERAL;
			}
			for (CaseRole role : values()) {
				if (role.name().equalsIgnoreCase(raw)) {
					return role;
				}
			}
			return GENERAL;
		}
	}

	private final int gridWidth;
	private final int gridHeight;
	private final ItemSize storedItemSize;
	private final ItemRarityTier rarityTier;
	private final float weightKg;
	private final CaseRole caseRole;

	public TarkovCaseItem(
		Settings settings,
		int gridWidth,
		int gridHeight,
		ItemSize storedItemSize,
		ItemRarityTier rarityTier,
		float weightKg
	) {
		this(settings, gridWidth, gridHeight, storedItemSize, rarityTier, weightKg, CaseRole.GENERAL);
	}

	public TarkovCaseItem(
		Settings settings,
		int gridWidth,
		int gridHeight,
		ItemSize storedItemSize,
		ItemRarityTier rarityTier,
		float weightKg,
		CaseRole caseRole
	) {
		super(settings.maxCount(1));
		this.gridWidth = Math.max(1, gridWidth);
		this.gridHeight = Math.max(1, gridHeight);
		this.storedItemSize = storedItemSize != null ? storedItemSize : ItemSize.ONE_BY_ONE;
		this.rarityTier = rarityTier != null ? rarityTier : ItemRarityTier.RARE;
		this.weightKg = Math.max(0.0f, weightKg);
		this.caseRole = caseRole != null ? caseRole : CaseRole.GENERAL;
	}

	public CaseRole getCaseRole() {
		return caseRole;
	}

	public boolean isMedicalCase() {
		return caseRole == CaseRole.MEDICAL || caseRole == CaseRole.INJECTOR;
	}

	/**
	 * Returns true if the given ItemStack is allowed inside this storage case.
	 */
	public boolean canAcceptItem(ItemStack stack) {
		if (stack == null || stack.isEmpty() || stack.getItem() instanceof TarkovCaseItem) {
			return false;
		}
		Item item = stack.getItem();
		return switch (caseRole) {
			case GENERAL -> true;
			case MEDICAL -> item instanceof TarkovMedicalItem || item instanceof TarkovStimulatorItem;
			case INJECTOR -> item instanceof TarkovStimulatorItem;
			case AMMO -> isTaczAmmoOrAttachment(stack);
			case GRENADE -> item instanceof TarkovGrenadeItem;
			case FOOD -> stack.isFood() || item instanceof TarkovFoodItem
				|| item instanceof PotionItem || item instanceof MilkBucketItem;
			case WEAPON -> AmmoTier.isTaczGunStack(stack) || isTaczAmmoOrAttachment(stack)
				|| item instanceof TarkovMeleeItem || item instanceof TarkovGrenadeItem
				|| item instanceof SwordItem || item instanceof BowItem
				|| item instanceof CrossbowItem || item instanceof TridentItem;
			case ARMOR_PLATE -> item instanceof TarkovArmorRepairItem || item instanceof TarkovArmorItem;
			case MONEY -> item instanceof TarkovFunctionalItem functional && functional.isValuable()
				|| item == Items.EMERALD || item == Items.GOLD_INGOT || item == Items.GOLD_NUGGET
				|| item == Items.DIAMOND || item == Items.NETHERITE_INGOT || item == Items.NETHERITE_SCRAP
				|| item == Items.AMETHYST_SHARD || item == Items.PAPER;
		};
	}

	private static boolean isTaczAmmoOrAttachment(ItemStack stack) {
		if (AmmoTier.isTaczAmmoStack(stack)) {
			return true;
		}
		Identifier id = Registries.ITEM.getId(stack.getItem());
		if (!"tacz".equals(id.getNamespace())) {
			return false;
		}
		String path = id.getPath();
		return path.contains("ammo") || path.contains("attachment") || path.contains("mag");
	}

			return TypedActionResult.pass(stack);

		tooltip.add(Text.translatable("tooltip.tarkovcraft.med_weight",
