/**
 * Renders the Tarkov-style body-part health silhouette, injury badges, painkiller timer,
 * effectiveMaxHp surgery limits, armor coverage, and interactive HEALTH tab hitboxes.
 */
public final class BodyHealthHudHelper {
	private static final HitZone[] COVERAGE_ZONES_LEFT = {
		HitZone.HEAD_TOP, HitZone.HEAD_EARS, HitZone.HEAD_EYES, HitZone.HEAD_JAWS
	};
	private static final HitZone[] COVERAGE_ZONES_RIGHT = {
		HitZone.NECK, HitZone.THORAX, HitZone.ARMPIT_LEFT, HitZone.STOMACH
	};

	public record PartHitbox(BodyPart part, int x, int y, int w, int h) {
		public boolean contains(double mx, double my) {
			return mx >= x && mx < x + w && my >= y && my < y + h;
		}
	}

	public record QuickHealMed(byte loc, int idx, @Nullable UUID entryId, ItemStack stack) {
	}

	public record QuickHealHitbox(BodyPart part, QuickHealMed med, int x, int y, int w, int h) {
		public boolean contains(double mx, double my) {
			return mx >= x && mx < x + w && my >= y && my < y + h;
		}
	}

	private BodyHealthHudHelper() {
	}

	public static void drawCompactTopRight(DrawContext context, TextRenderer textRenderer, PlayerEntity player, int screenW) {
		MinecraftClient mc = MinecraftClient.getInstance();
		if (mc != null && (mc.currentScreen instanceof TarkovInventoryScreen || mc.currentScreen instanceof TarkovDeathScreen)) {
			return;
		}
		BodyPartHealth health = ClientLoadoutCache.getBodyHealth();

		int boxW = 48;
		int boxH = 66;
		int x = screenW - boxW - 8;
		int y = 8;

		context.fill(x, y, x + boxW, y + boxH, 0x88101316);
		context.drawBorder(x, y, boxW, boxH, 0x99384048);

		int cx = x + boxW / 2;
		int top = y + 5;

		drawBodyPartRect(context, cx - 4, top, 8, 8, health.getRatio(BodyPart.HEAD));
		drawBodyPartRect(context, cx - 6, top + 10, 12, 11, health.getRatio(BodyPart.THORAX));
		drawBodyPartRect(context, cx - 5, top + 22, 10, 7, health.getRatio(BodyPart.STOMACH));
		drawBodyPartRect(context, cx - 12, top + 10, 5, 17, health.getRatio(BodyPart.LEFT_ARM));
		drawBodyPartRect(context, cx + 7, top + 10, 5, 17, health.getRatio(BodyPart.RIGHT_ARM));
		drawBodyPartRect(context, cx - 6, top + 30, 5, 19, health.getRatio(BodyPart.LEFT_LEG));
		drawBodyPartRect(context, cx + 1, top + 30, 5, 19, health.getRatio(BodyPart.RIGHT_LEG));

		int curTotal = Math.round(health.getTotalHp());
		int effMaxTotal = Math.round(health.getTotalEffectiveMaxHp());
		String label = curTotal + "/" + effMaxTotal;
		context.getMatrices().push();
		context.getMatrices().scale(0.72f, 0.72f, 1.0f);
		int tx = (int) (cx / 0.72f);
		int ty = (int) ((y + boxH - 9) / 0.72f);
		context.drawCenteredTextWithShadow(textRenderer, label, tx, ty, hpToColor(health.getTotalRatio()));
		context.getMatrices().pop();

		// Active trauma & painkiller badges directly below the silhouette
		int py = y + boxH + 3;
		int heavyCount = health.countInjuries(InjuryType.HEAVY_BLEED);
		if (heavyCount > 0) {
			py = drawHudBadge(context, textRenderer, x + boxW, py, "HB x" + heavyCount, 0xDD7A1414, 0xFFFF4444, 0xFFFFCCCC);
		}
		int lightCount = health.countInjuries(InjuryType.LIGHT_BLEED);
		if (lightCount > 0) {
			py = drawHudBadge(context, textRenderer, x + boxW, py, "LB x" + lightCount, 0xDD6E2A16, 0xFFFF8855, 0xFFFFDDCC);
		}
		int fracCount = health.countInjuries(InjuryType.FRACTURE);
		if (fracCount > 0) {
			py = drawHudBadge(context, textRenderer, x + boxW, py, "FR x" + fracCount, 0xDD4D3C12, 0xFFE0B23B, 0xFFFFEEAA);
		}
		if (health.hasPainkiller()) {
			int secs = (health.getPainkillerTicks() + 19) / 20;
			py = drawHudBadge(context, textRenderer, x + boxW, py, "PK " + secs + "s", 0xDD143B34, 0xFF4FD1C5, 0xFFB2F5EA);
		}
		if (player != null) {
			int energy = Math.max(0, Math.min(100, player.getHungerManager().getFoodLevel() * 5));
			int eBg = energy <= 15 ? 0xDD6A1414 : (energy <= 30 ? 0xDD4D3C12 : 0xDD14222D);
			int eBorder = energy <= 15 ? 0xFFFF4444 : (energy <= 30 ? 0xFFE0B23B : 0xFF3E6888);
			int eText = energy <= 15 ? 0xFFFFCCCC : (energy <= 30 ? 0xFFFFEEAA : 0xFF90CDF4);
			drawHudBadge(context, textRenderer, x + boxW, py, "E:" + energy, eBg, eBorder, eText);
		}
	}

	private static int drawHudBadge(
		DrawContext context, TextRenderer tr, int rightX, int y,
		String text, int bgCol, int borderCol, int textCol
	) {
		int bw = Math.max(30, Math.round(tr.getWidth(text) * 0.72f) + 8);
		int bx = rightX - bw;
		context.fill(bx, y, rightX, y + 10, bgCol);
		context.drawBorder(bx, y, bw, 10, borderCol);
		context.getMatrices().push();
		context.getMatrices().scale(0.72f, 0.72f, 1.0f);
		context.drawCenteredTextWithShadow(tr, text, (int) ((bx + bw / 2.0f) / 0.72f), (int) ((y + 2) / 0.72f), textCol);
		context.getMatrices().pop();
		return y + 12;
	}

	public static void drawDetailedHealthTab(DrawContext context, TextRenderer textRenderer, PlayerEntity player, int x, int y, int w, int h) {
		drawDetailedHealthTab(context, textRenderer, player, x, y, w, h, null, null, -1, -1);

	public static void drawDetailedHealthTab(
		DrawContext context,
		TextRenderer textRenderer,
		PlayerEntity player,
		int x,
		int y,
		int w,
		int h,
		@Nullable ItemStack selectedMedStack,
		@Nullable List<PartHitbox> outHitboxes,
		int mouseX,
		int mouseY
	) {

	}

	public static void drawDetailedHealthTab(
		DrawContext context,
		TextRenderer textRenderer,
		PlayerEntity player,
		int x,
		int y,
		int w,
		int h,
		@Nullable ItemStack selectedMedStack,
		@Nullable List<PartHitbox> outHitboxes,
		@Nullable List<QuickHealMed> availableMeds,
		@Nullable List<QuickHealHitbox> outQuickHeals,
		int mouseX,
		int mouseY
	) {
		BodyPartHealth health = ClientLoadoutCache.getBodyHealth();
		PlayerLoadout loadout = ClientLoadoutCache.getLoadout();

		int silTop = y + 8;
		drawSilhouetteWithZoneBars(context, textRenderer, health, x + w / 2, silTop,
			selectedMedStack, outHitboxes, availableMeds, outQuickHeals, mouseX, mouseY);

		List<DamageReport> hits = health.getRecentHits();
		int panelW = Math.min(w - 16, 420);
		int panelX = x + (w - panelW) / 2;

		// 1. Penalties & Painkiller status strip under silhouette
		int statusY = silTop + 123;
		drawStatusPenaltiesBar(context, textRenderer, health, panelX, statusY, panelW);

		// 2. Armor Coverage panel
		int covY = statusY + 15;
		int covH = 50;
		context.fill(panelX, covY, panelX + panelW, covY + covH, 0xCC0E1114);
		context.drawBorder(panelX, covY, panelW, covH, 0xFF2B3238);
		drawTiny(context, textRenderer, Text.translatable("armor.tarkovcraft.coverage_header").getString(),
			panelX + 5, covY + 3, 0xFFD4B46A);

		int col0X = panelX + 6;
		int col1X = panelX + panelW / 2 + 4;
		int colW = panelW / 2 - 12;
		drawCoverageColumn(context, textRenderer, loadout, COVERAGE_ZONES_LEFT, col0X, covY + 13, colW);
		drawCoverageColumn(context, textRenderer, loadout, COVERAGE_ZONES_RIGHT, col1X, covY + 13, colW);

		// 3. Recent Hit Log
		if (!hits.isEmpty()) {
			int logY = covY + covH + 3;
			int logH = Math.min(42, Math.max(24, y + h - logY - 2));
			if (logH >= 24) {
				context.fill(panelX, logY, panelX + panelW, logY + logH, 0xCC0E1114);
				context.drawBorder(panelX, logY, panelW, logH, 0xFF2B3238);
				drawTiny(context, textRenderer, Text.translatable("health.tarkovcraft.recent_hits").getString(),
					panelX + 5, logY + 3, 0xFFD4B46A);
				int rowY = logY + 12;
				int maxRows = Math.min(3, Math.max(1, (logH - 12) / 10));
				int shown = Math.min(maxRows, hits.size());
				for (int i = 0; i < shown; i++) {
					DamageReport r = hits.get(i);
					String zoneTxt = Text.translatable(r.hitZone().getTranslationKey()).getString();
					String armorTag = r.ricocheted() ? " [RICOCHET]" : (r.armorBlocked() ? " [ARMOR]" : "");
					String line = String.format("%s  -%.0f HP%s%s",
						zoneTxt,
						r.finalDamage(),
						armorTag,
						r.overflowedToThorax() ? " -> THORAX" : ""
					);
					drawTiny(context, textRenderer, line, panelX + 5, rowY, r.headEyesOneshot() ? 0xFFFF5555 : 0xFFC8D0D8);
					rowY += 10;
				}
			}
		}
	}

	private static void drawStatusPenaltiesBar(DrawContext context, TextRenderer tr, BodyPartHealth health, int x, int y, int w) {
		context.fill(x, y, x + w, y + 12, 0xCC0E1114);
		context.drawBorder(x, y, w, 12, 0xFF2B3238);

		List<String> tags = new ArrayList<>();
		List<Integer> cols = new ArrayList<>();

		if (health.hasPainkiller()) {
			int secs = (health.getPainkillerTicks() + 19) / 20;
			tags.add(Text.translatable("health.tarkovcraft.status_painkiller", secs).getString());
			cols.add(0xFF4FD1C5);
		}
		if (health.getImpairedLegCount() > 0) {
			tags.add(Text.translatable("health.tarkovcraft.penalty_limp").getString());
			cols.add(0xFFE0B23B);
		}
		if (health.getImpairedArmCount() > 0) {
			tags.add(Text.translatable("health.tarkovcraft.penalty_arms").getString());
			cols.add(0xFFE08A3B);
		}
		if (health.isStomachBlacked()) {
			tags.add(Text.translatable("health.tarkovcraft.penalty_stomach").getString());
			cols.add(0xFFFF6B6B);
		}

		if (tags.isEmpty() && !health.hasAnyBleed()) {
			drawTiny(context, tr, Text.translatable("health.tarkovcraft.status_healthy").getString(), x + 5, y + 3, 0xFF77B85A);
			return;
		}
		if (tags.isEmpty()) {
			tags.add(Text.translatable("health.tarkovcraft.tag_light_bleed").getString());
			cols.add(0xFFFF8855);
		}

		int cx = x + 5;
		for (int i = 0; i < tags.size(); i++) {
			String txt = (i > 0 ? " | " : "") + tags.get(i);
			drawTiny(context, tr, txt, cx, y + 3, cols.get(i));
			cx += Math.round(tr.getWidth(txt) * 0.72f) + 2;
			if (cx > x + w - 20) break;
		}
	}

	private static void drawCoverageColumn(
		DrawContext context, TextRenderer textRenderer,
		PlayerLoadout loadout, HitZone[] zones, int x, int startY, int colW
	) {
		String openTxt = Text.translatable("armor.tarkovcraft.zone_open").getString();
		int rowY = startY;
		for (HitZone zone : zones) {
			int armorClass = ZonalArmorHelper.getHighestProtectingClass(loadout, zone);
			String zoneTxt = Text.translatable(zone.getTranslationKey()).getString();
			String statusTxt = armorClass > 0 ? ("CL." + armorClass) : openTxt;
			int statusCol = armorClass > 0 ? 0xFF63D2FF : 0xFFFF5555;

			drawTiny(context, textRenderer, zoneTxt, x, rowY, 0xFFC8D0D8);
			int sw = Math.round(textRenderer.getWidth(statusTxt) * 0.72f);
			drawTiny(context, textRenderer, statusTxt, x + colW - sw, rowY, statusCol);
			rowY += 9;
		}
	}

		DrawContext context,
		TextRenderer textRenderer,

	public static void drawSilhouetteWithZoneBars(
		DrawContext context,
		TextRenderer textRenderer,
		BodyPartHealth health,
		int cx,
		int top,
		@Nullable ItemStack selectedMedStack,
		@Nullable List<PartHitbox> outHitboxes,
		@Nullable List<QuickHealMed> availableMeds,
		@Nullable List<QuickHealHitbox> outQuickHeals,
		int mouseX,
		int mouseY
	) {
		drawInteractivePartRect(context, BodyPart.HEAD, cx - 10, top, 20, 20, health, selectedMedStack, outHitboxes, mouseX, mouseY);
		drawInteractivePartRect(context, BodyPart.THORAX, cx - 15, top + 24, 30, 28, health, selectedMedStack, outHitboxes, mouseX, mouseY);
		drawInteractivePartRect(context, BodyPart.STOMACH, cx - 13, top + 54, 26, 18, health, selectedMedStack, outHitboxes, mouseX, mouseY);
		drawInteractivePartRect(context, BodyPart.LEFT_ARM, cx - 28, top + 24, 11, 42, health, selectedMedStack, outHitboxes, mouseX, mouseY);
		drawInteractivePartRect(context, BodyPart.RIGHT_ARM, cx + 17, top + 24, 11, 42, health, selectedMedStack, outHitboxes, mouseX, mouseY);
		drawInteractivePartRect(context, BodyPart.LEFT_LEG, cx - 14, top + 74, 12, 46, health, selectedMedStack, outHitboxes, mouseX, mouseY);
		drawInteractivePartRect(context, BodyPart.RIGHT_LEG, cx + 2, top + 74, 12, 46, health, selectedMedStack, outHitboxes, mouseX, mouseY);

		drawZoneBar(context, textRenderer, cx + 38, top, BodyPart.HEAD, health, selectedMedStack, outHitboxes, availableMeds, outQuickHeals, true, mouseX, mouseY);
		drawZoneBar(context, textRenderer, cx + 38, top + 24, BodyPart.THORAX, health, selectedMedStack, outHitboxes, availableMeds, outQuickHeals, true, mouseX, mouseY);
		drawZoneBar(context, textRenderer, cx + 38, top + 48, BodyPart.STOMACH, health, selectedMedStack, outHitboxes, availableMeds, outQuickHeals, true, mouseX, mouseY);
		drawZoneBar(context, textRenderer, cx + 38, top + 72, BodyPart.RIGHT_ARM, health, selectedMedStack, outHitboxes, availableMeds, outQuickHeals, true, mouseX, mouseY);
		drawZoneBar(context, textRenderer, cx + 38, top + 96, BodyPart.RIGHT_LEG, health, selectedMedStack, outHitboxes, availableMeds, outQuickHeals, true, mouseX, mouseY);

		drawZoneBar(context, textRenderer, cx - 130, top + 34, BodyPart.LEFT_ARM, health, selectedMedStack, outHitboxes, availableMeds, outQuickHeals, false, mouseX, mouseY);
		drawZoneBar(context, textRenderer, cx - 130, top + 76, BodyPart.LEFT_LEG, health, selectedMedStack, outHitboxes, availableMeds, outQuickHeals, false, mouseX, mouseY);
	}

	private static void drawInteractivePartRect(
		DrawContext context, BodyPart part, int x, int y, int w, int h,
		BodyPartHealth health, @Nullable ItemStack selectedMedStack,
		@Nullable List<PartHitbox> outHitboxes, int mouseX, int mouseY
	) {
		drawBodyPartRect(context, x, y, w, h, health.getRatio(part));
		if (selectedMedStack != null && MedicalHelper.canApplyToPart(health, selectedMedStack, part)) {
			boolean hov = mouseX >= x && mouseX < x + w && mouseY >= y && mouseY < y + h;
			context.drawBorder(x, y, w, h, hov ? 0xFFD4B46A : 0xFF4CAF50);
		}
		if (outHitboxes != null) {
			outHitboxes.add(new PartHitbox(part, x, y, w, h));
		}
	}

	private static void drawZoneBar(
		DrawContext context,
		TextRenderer textRenderer,
		int x,
		int y,
		BodyPart part,
		BodyPartHealth health,
		@Nullable ItemStack selectedMedStack,
		@Nullable List<PartHitbox> outHitboxes,
		@Nullable List<QuickHealMed> availableMeds,
		@Nullable List<QuickHealHitbox> outQuickHeals,
		boolean iconsOnRight,
		int mouseX,
		int mouseY
	) {
		int bw = 92;
		int bh = 22;
		float ratio = health.getRatio(part);
		int cur = Math.round(health.getHp(part));
		int effMax = Math.round(health.getEffectiveMaxHp(part));
		int baseMax = (int) part.getMaxHp();
		int barColor = hpToColor(ratio);

		boolean canHeal = selectedMedStack != null && MedicalHelper.canApplyToPart(health, selectedMedStack, part);
		boolean hovered = mouseX >= x && mouseX < x + bw && mouseY >= y && mouseY < y + bh;
		int border = canHeal
			? (hovered ? 0xFFD4B46A : 0xFF4CAF50)
			: (ratio <= 0.001f ? 0xFFC93838 : 0xFF3E464E);

		context.fill(x, y, x + bw, y + bh, (canHeal && hovered) ? 0x334CAF50 : 0xCC161A1E);
		context.drawBorder(x, y, bw, bh, border);

		context.getMatrices().push();
		context.getMatrices().scale(0.72f, 0.72f, 1.0f);
		String name = Text.translatable(part.getTranslationKey()).getString();
		context.drawTextWithShadow(textRenderer, name, (int) ((x + 3) / 0.72f), (int) ((y + 2) / 0.72f), 0xFFD4D8DC);

		String curStr = String.valueOf(cur);
		String maxStr = "/" + effMax;
		int curW = textRenderer.getWidth(curStr);
		int maxW = textRenderer.getWidth(maxStr);
		int rightScaledX = (int) ((x + bw - 3) / 0.72f);
		int textScaledY = (int) ((y + 2) / 0.72f);
		context.drawTextWithShadow(textRenderer, curStr, rightScaledX - maxW - curW, textScaledY,
			ratio <= 0.001f ? 0xFFFF5555 : barColor);
		context.drawTextWithShadow(textRenderer, maxStr, rightScaledX - maxW, textScaledY,
			effMax < baseMax ? 0xFF9AA0A8 : 0xFFB8C0C8);
		context.getMatrices().pop();

		int innerW = bw - 6;
		int barY1 = y + 10;
		int barY2 = y + 14;
		context.fill(x + 3, barY1, x + 3 + innerW, barY2, 0xFF0E1114);
		if (effMax < baseMax) {
			int effW = Math.round(innerW * health.getEffectiveMaxRatio(part));
			if (effW < innerW) {
				context.fill(x + 3 + effW, barY1, x + 3 + innerW, barY2, 0xFF34383E);
			}
		}
		int fillW = Math.round(innerW * health.getBaseRatio(part));
		if (fillW > 0) {
			context.fill(x + 3, barY1, x + 3 + fillW, barY2, barColor);
		}

		// Injury & blacked-out tags on bottom line (y + 15)
		int tagX = x + 3;
		if (health.isBlacked(part)) {
			tagX = drawMicroTag(context, textRenderer, tagX, y + 15,
				Text.translatable("health.tarkovcraft.tag_blacked").getString(), 0xFFE53935);
		}
		if (health.hasHeavyBleed(part)) {
			tagX = drawMicroTag(context, textRenderer, tagX, y + 15,
				Text.translatable("health.tarkovcraft.tag_heavy_bleed").getString(), 0xFFFF4444);
		} else if (health.hasLightBleed(part)) {
			tagX = drawMicroTag(context, textRenderer, tagX, y + 15,
				Text.translatable("health.tarkovcraft.tag_light_bleed").getString(), 0xFFFF8855);
		}
		if (health.hasFracture(part)) {
			drawMicroTag(context, textRenderer, tagX, y + 15,
				Text.translatable("health.tarkovcraft.tag_fracture").getString(), 0xFFE0B23B);
		}

		if (outHitboxes != null) {
			outHitboxes.add(new PartHitbox(part, x, y, bw, bh));
		}

		// Quick-Heal 14x14 mini-icons next to injured parts (no durability text)
		if (availableMeds != null && outQuickHeals != null) {
			boolean needsHelp = health.getHp(part) < health.getEffectiveMaxHp(part) - 0.5f
				|| health.hasLightBleed(part)
				|| health.hasHeavyBleed(part)
				|| health.hasFracture(part)
				|| health.isBlacked(part);
			if (needsHelp) {
				List<QuickHealMed> suitable = collectQuickHealsForPart(health, part, availableMeds);
				int btnSize = 14;
				int btnY = y + (bh - btnSize) / 2;
				for (int i = 0; i < suitable.size(); i++) {
					QuickHealMed qm = suitable.get(i);
					int btnX = iconsOnRight
						? (x + bw + 4 + i * (btnSize + 2))
						: (x - 4 - btnSize - i * (btnSize + 2));
					boolean qHov = mouseX >= btnX && mouseX < btnX + btnSize && mouseY >= btnY && mouseY < btnY + btnSize;
					context.fill(btnX, btnY, btnX + btnSize, btnY + btnSize, qHov ? 0xDD272B30 : 0xCC121619);
					context.drawBorder(btnX, btnY, btnSize, btnSize, qHov ? TarkovUiTheme.ACCENT_SAND : 0xFF3A424A);
					TarkovUiTheme.drawScaledItem(context, qm.stack(), btnX + 1, btnY + 1, 12, 12);
					outQuickHeals.add(new QuickHealHitbox(part, qm, btnX, btnY, btnSize, btnSize));
				}
			}
		}
	}

	private static List<QuickHealMed> collectQuickHealsForPart(
		BodyPartHealth health, BodyPart part, List<QuickHealMed> allMeds
	) {
		Map<Item, QuickHealMed> unique = new LinkedHashMap<>();
		for (QuickHealMed m : allMeds) {
			if (!unique.containsKey(m.stack().getItem()) && MedicalHelper.canApplyToPart(health, m.stack(), part)) {
				unique.put(m.stack().getItem(), m);
			}
		}
		List<QuickHealMed> sorted = new ArrayList<>(unique.values());
		sorted.sort(Comparator.comparingInt(m -> quickHealPriority(health, part, m.stack())));
		return sorted.size() > 4 ? sorted.subList(0, 4) : sorted;
	}

	private static int quickHealPriority(BodyPartHealth health, BodyPart part, ItemStack stack) {
		if (!(stack.getItem() instanceof TarkovMedicalItem med)) return 99;
		MedicalType t = med.getMedicalType();
		if (health.hasHeavyBleed(part) && t.isCuresHeavyBleed()) return 0;
		if (health.isBlacked(part) && t.isSurgicalKit()) return 1;
		if (health.hasFracture(part) && t.isCuresFracture()) return 2;
		if (health.hasLightBleed(part) && t.isCuresLightBleed()) return 3;
		if (t.getMaxHealPerUse() > 0) return 4;
		return 5;
	}

	private static int drawMicroTag(DrawContext context, TextRenderer tr, int x, int y, String text, int color) {
		context.getMatrices().push();
		context.getMatrices().scale(0.62f, 0.62f, 1.0f);
		context.drawText(tr, text, Math.round(x / 0.62f), Math.round(y / 0.62f), color, false);
		context.getMatrices().pop();
		return x + Math.round(tr.getWidth(text) * 0.62f) + 4;
	}

	public static void drawBodyPartRect(DrawContext context, int x, int y, int w, int h, float ratio) {
		int fill = hpToColor(ratio);
		int outline;
		if (ratio <= 0.001f) {
			outline = 0xFFE53935;
		} else if (ratio < 0.25f) {
			outline = 0xFF8E2A22;
		} else {
			outline = 0xCC0D1013;
		}
		context.fill(x, y, x + w, y + h, fill);
		context.drawBorder(x, y, w, h, outline);
	}

		context.drawTextWithShadow(tr, text, (int) (x / 0.72f), (int) (y / 0.72f), col);
		context.getMatrices().pop();
	}

	private static final float[] GRADIENT_STOPS = {0.0f, 0.25f, 0.50f, 0.75f, 1.0f};
	private static final int[] GRADIENT_COLORS = {
		0xC62828,
		0xE0662F,
		0xE0B23B,
		0xA9BE4A,
		0x77B85A
	};

	public static int hpToColor(float ratio) {
		if (ratio <= 0.001f) return 0xFF141414;
		float r = Math.max(0.0f, Math.min(1.0f, ratio));
		for (int i = 0; i < GRADIENT_STOPS.length - 1; i++) {
			float a = GRADIENT_STOPS[i];
			float b = GRADIENT_STOPS[i + 1];
			if (r <= b) {
				float t = (r - a) / (b - a);
				return 0xFF000000 | lerpRgb(GRADIENT_COLORS[i], GRADIENT_COLORS[i + 1], t);
			}
		}
		return 0xFF000000 | GRADIENT_COLORS[GRADIENT_COLORS.length - 1];
	}

	private static int lerpRgb(int c1, int c2, float t) {
		int r = Math.round(((c1 >> 16) & 0xFF) + (((c2 >> 16) & 0xFF) - ((c1 >> 16) & 0xFF)) * t);
		int g = Math.round(((c1 >> 8) & 0xFF) + (((c2 >> 8) & 0xFF) - ((c1 >> 8) & 0xFF)) * t);
		int b = Math.round((c1 & 0xFF) + ((c2 & 0xFF) - (c1 & 0xFF)) * t);
		return (r << 16) | (g << 8) | b;
	}

	private static final float DROPPED_SCALE = 0.85F;
	private static final float PIXELS_PER_BLOCK = 16.0F;

	private final BakedPart[] parts;
	private final Identifier baseTexture;
	private final boolean translucent;
	private final boolean groundModel;
	private final double centerX;
	private final double centerDepth;
	private final double bottom;

	GearModel(int textureWidth, int textureHeight, boolean translucent, Part... geometry) {
		this(textureWidth, textureHeight, translucent, null, geometry);
	}

	GearModel(int textureWidth, int textureHeight, boolean translucent, String textureName, Part... geometry) {
		this.groundModel = geometry[0].anchor() == Anchor.GROUND;
		this.translucent = translucent;
		String folder = groundModel ? "ground" : "wearable";
		String file = textureName == null ? name().toLowerCase(Locale.ROOT) : textureName;
		this.baseTexture = new Identifier("tarkovcraft", "textures/equipment/" + folder + "/" + file + ".png");

		ModelData data = new ModelData();
		float minX = Float.POSITIVE_INFINITY;
		float maxX = Float.NEGATIVE_INFINITY;
		float minDepth = Float.POSITIVE_INFINITY;
		float maxDepth = Float.NEGATIVE_INFINITY;
		float lowestPoint = Float.POSITIVE_INFINITY;
		for (Part part : geometry) {
			ModelPartBuilder builder = ModelPartBuilder.create();
			Anchor anchor = part.anchor();
			for (Cuboid cube : part.cuboids()) {
				cube.addTo(builder);
				minX = Math.min(minX, anchor.x + cube.x() - cube.dilation());
				maxX = Math.max(maxX, anchor.x + cube.x() + cube.width() + cube.dilation());
				if (groundModel) {
					minDepth = Math.min(minDepth, cube.z() - cube.dilation());
					maxDepth = Math.max(maxDepth, cube.z() + cube.depth() + cube.dilation());
					lowestPoint = Math.min(lowestPoint, -(cube.y() + cube.height() + cube.dilation()));
				} else {
					// When laid face up, model Y becomes horizontal depth and -Z becomes height.
					minDepth = Math.min(minDepth, -(anchor.y + cube.y() + cube.height() + cube.dilation()));
					maxDepth = Math.max(maxDepth, -(anchor.y + cube.y() - cube.dilation()));
					lowestPoint = Math.min(lowestPoint, -(cube.z() + cube.depth() + cube.dilation()));
				}
			}
			data.getRoot().addChild(anchor.name(), builder, ModelTransform.NONE);
		}
		ModelPart model = TexturedModelData.of(data, textureWidth, textureHeight).createModel();
		this.parts = new BakedPart[geometry.length];
		for (int index = 0; index < geometry.length; index++) {
			Anchor anchor = geometry[index].anchor();
			parts[index] = new BakedPart(anchor, model.getChild(anchor.name()));
		}
		this.centerX = (minX + maxX) / (2.0 * PIXELS_PER_BLOCK);
		this.centerDepth = (minDepth + maxDepth) / (2.0 * PIXELS_PER_BLOCK);
		this.bottom = lowestPoint / PIXELS_PER_BLOCK;
	}

	public void renderWorn(MatrixStack matrices, VertexConsumerProvider consumers, int light, BipedEntityModel<?> playerModel, Item item) {
		if (groundModel) {
			return;
		}
		VertexConsumer vertices = consumers.getBuffer(renderLayer(item));
		for (BakedPart part : parts) {
			ModelPart anchor = part.anchor().fromPlayer(playerModel);
			if (!anchor.visible || anchor.hidden) {
				continue;
			}
			matrices.push();
			anchor.rotate(matrices);
			part.model().render(matrices, vertices, light, OverlayTexture.DEFAULT_UV);
			matrices.pop();
		}
	}

	public void renderOnGround(MatrixStack matrices, VertexConsumerProvider consumers, int light, Item item) {
		VertexConsumer vertices = consumers.getBuffer(renderLayer(item));
		matrices.push();
		matrices.scale(DROPPED_SCALE, DROPPED_SCALE, DROPPED_SCALE);
		matrices.translate(centerX, -bottom, -centerDepth);
		if (!groundModel) {
			matrices.multiply(RotationAxis.POSITIVE_X.rotationDegrees(90.0F));
		}
		matrices.scale(-1.0F, -1.0F, 1.0F);
		for (BakedPart part : parts) {
			Anchor anchor = part.anchor();
			matrices.push();
			matrices.translate(anchor.x / PIXELS_PER_BLOCK, anchor.y / PIXELS_PER_BLOCK, 0.0);
			part.model().render(matrices, vertices, light, OverlayTexture.DEFAULT_UV);
			matrices.pop();
		}
		matrices.pop();
	}

	private RenderLayer renderLayer(Item item) {
		Identifier texture = this == GROUND_CASE_MEDKIT || this == VISOR ? baseTexture : GearPalette.textureFor(item, baseTexture);
		return translucent ? RenderLayer.getEntityTranslucent(texture) : RenderLayer.getEntityCutoutNoCull(texture);
	}

	private static Part part(Anchor anchor, Cuboid... cuboids) {
		return new Part(anchor, cuboids);
	}

	private static Cuboid cube(int u, int v, float x, float y, float z, float width, float height, float depth) {
		return cube(u, v, x, y, z, width, height, depth, 0.0F);
	}

	private static Cuboid cube(int u, int v, float x, float y, float z, float width, float height, float depth, float dilation) {
		return new Cuboid(u, v, x, y, z, width, height, depth, dilation);
	}

	private record Cuboid(int u, int v, float x, float y, float z, float width, float height, float depth, float dilation) {
		void addTo(ModelPartBuilder builder) {
			builder.uv(u, v).cuboid(x, y, z, width, height, depth, new Dilation(dilation));
		}
	}

	private record Part(Anchor anchor, Cuboid[] cuboids) {}
	private record BakedPart(Anchor anchor, ModelPart model) {}

	private enum Anchor {
		HEAD(0, 0),
		BODY(0, 0),
		RIGHT_ARM(-5, 2),
		LEFT_ARM(5, 2),
		GROUND(0, 0);

		private final float x;
		private final float y;

		Anchor(float x, float y) {
			this.x = x;
			this.y = y;
		}

		ModelPart fromPlayer(BipedEntityModel<?> playerModel) {
			return switch (this) {
				case HEAD -> playerModel.head;
				case BODY -> playerModel.body;
				case RIGHT_ARM -> playerModel.rightArm;
				case LEFT_ARM -> playerModel.leftArm;
				case GROUND -> throw new IllegalStateException("Ground geometry has no player model part");
			};
		}
	}
}

/** Recolors the pack's materials from the inventory icon for each equipment item. */
public final class GearPalette {
	private static final Map<Identifier, Optional<Color>> ICON_COLORS = new HashMap<>();
	private static final Map<TextureKey, Optional<Identifier>> RECOLORED_TEXTURES = new HashMap<>();

	private GearPalette() {
	}

	public static void register() {
		ResourceManagerHelper.get(ResourceType.CLIENT_RESOURCES).registerReloadListener(new SimpleSynchronousResourceReloadListener() {
			@Override
			public Identifier getFabricId() {
				return new Identifier("tarkovcraft", "gear_palette");
			}

			@Override
			public void reload(ResourceManager manager) {
				TextureManager textures = MinecraftClient.getInstance().getTextureManager();
				for (Optional<Identifier> texture : RECOLORED_TEXTURES.values()) {
					texture.ifPresent(textures::destroyTexture);
				}
				RECOLORED_TEXTURES.clear();
				ICON_COLORS.clear();
			}
		});
	}

	static Identifier textureFor(Item item, Identifier baseTexture) {
		Identifier itemId = Registries.ITEM.getId(item);
		TextureKey key = new TextureKey(itemId, baseTexture);
		return RECOLORED_TEXTURES.computeIfAbsent(key, GearPalette::createTexture).orElse(baseTexture);
	}

	private static Optional<Identifier> createTexture(TextureKey key) {
		MinecraftClient client = MinecraftClient.getInstance();
		ResourceManager resources = client.getResourceManager();
		Optional<Color> icon = ICON_COLORS.computeIfAbsent(key.item(), item -> readIconColor(resources, item));
		if (icon.isEmpty()) {
			return Optional.empty();
		}

		NativeImage image;
		try {
			image = readImage(resources, key.baseTexture());
		} catch (IOException | IllegalArgumentException e) {
			return Optional.empty();
		}
		NativeImageBackedTexture texture = null;
		boolean registered = false;
		try {
			Color material = materialColor(image, 255);
			if (material == null) {
				return Optional.empty();
			}
			recolor(image, material, icon.get());
			texture = new NativeImageBackedTexture(image);
			// Stable IDs let RenderLayer reuse its cached layer across reloads.
			Identifier id = new Identifier("tarkovcraft", "gear_palette/"
				+ key.baseTexture().getPath() + "/" + key.item().getNamespace() + "/" + key.item().getPath());
			client.getTextureManager().registerTexture(id, texture);
			registered = true;
			return Optional.of(id);
		} finally {
			if (!registered) {
				if (texture == null) image.close();
				else texture.close();
			}
		}
	}

	private static Optional<Color> readIconColor(ResourceManager resources, Identifier item) {
		// TarkovCraft's equipment item models use a matching textures/item/<id>.png layer.
		Identifier texture = new Identifier(item.getNamespace(), "textures/item/" + item.getPath() + ".png");
		try (NativeImage icon = readImage(resources, texture)) {
			return Optional.ofNullable(materialColor(icon, 128));
		} catch (IOException | IllegalArgumentException e) {
			return Optional.empty();
		}
	}

	private static NativeImage readImage(ResourceManager resources, Identifier texture) throws IOException {
		try (InputStream stream = resources.open(texture)) {
			return NativeImage.read(NativeImage.Format.RGBA, stream);
		}
	}

	@Nullable
	private static Color materialColor(NativeImage image, int minAlpha) {
		int[] brightnessCounts = new int[256];
		int count = 0;
		for (int y = 0; y < image.getHeight(); y++) {
			for (int x = 0; x < image.getWidth(); x++) {
				int pixel = image.getColor(x, y);
				if ((pixel >>> 24) >= minAlpha) {
					brightnessCounts[Math.round(Color.fromPixel(pixel).luminance())]++;
					count++;
				}
			}
		}
		if (count == 0) {
			return null;
		}
		int low = percentile(brightnessCounts, count * 15 / 100);
		int high = percentile(brightnessCounts, count * 85 / 100);
		int red = 0;
		int green = 0;
		int blue = 0;
		int included = 0;
		for (int y = 0; y < image.getHeight(); y++) {
			for (int x = 0; x < image.getWidth(); x++) {
				int pixel = image.getColor(x, y);
				if ((pixel >>> 24) < minAlpha) continue;
				Color color = Color.fromPixel(pixel);
				int shade = Math.round(color.luminance());
				if (shade < low || shade > high) continue;
				red += color.red();
				green += color.green();
				blue += color.blue();
				included++;
			}
		}
		return new Color(red / included, green / included, blue / included);
	}

	private static int percentile(int[] counts, int index) {
		for (int shade = 0; shade < counts.length; shade++) {
			index -= counts[shade];
			if (index < 0) return shade;
		}
		return 255;
	}

	private static void recolor(NativeImage image, Color material, Color target) {
		float baseShade = material.luminance();
		for (int y = 0; y < image.getHeight(); y++) {
			for (int x = 0; x < image.getWidth(); x++) {
				int pixel = image.getColor(x, y);
				// Preserve transparent UV padding, openings and translucent lenses exactly.
				if ((pixel >>> 24) != 255) continue;
				Color original = Color.fromPixel(pixel);
				float shade = original.luminance();
				float chromaDistance = original.chromaDistance(material, shade, baseShade);
				float materialWeight = 1.0F - MathHelper.clamp((chromaDistance - 0.18F) / 0.30F, 0.0F, 1.0F);
				float lightRatio = MathHelper.clamp(shade / Math.max(baseShade, 1.0F), 0.0F, 2.0F);
				int red = shadeChannel(original.red(), target.red(), lightRatio, materialWeight);
				int green = shadeChannel(original.green(), target.green(), lightRatio, materialWeight);
				int blue = shadeChannel(original.blue(), target.blue(), lightRatio, materialWeight);
				image.setColor(x, y, 0xFF000000 | blue << 16 | green << 8 | red);
			}
		}
	}

	private static int shadeChannel(int original, int target, float ratio, float weight) {
		return MathHelper.clamp(Math.round(MathHelper.lerp(weight, original, target * ratio)), 0, 255);
	}

	private record TextureKey(Identifier item, Identifier baseTexture) {}

	private record Color(int red, int green, int blue) {
		static Color fromPixel(int abgr) {
			return new Color(abgr & 255, abgr >>> 8 & 255, abgr >>> 16 & 255);
		}

		float luminance() {
			return (3 * red + 6 * green + blue) / 10.0F;
		}

		float chromaDistance(Color other, float shade, float otherShade) {
			float scale = Math.max(shade, 24.0F);
			float otherScale = Math.max(otherShade, 24.0F);
			float redGreen = (red - green) / scale - (other.red - other.green) / otherScale;
			float blueGreen = (blue - green) / scale - (other.blue - other.green) / otherScale;
			return Math.max(Math.abs(redGreen), Math.abs(blueGreen));
		}
	}
}

public class AirdropEntityRenderer extends EntityRenderer<AirdropEntity> {
	private final BlockRenderManager blockRenderManager;

	public AirdropEntityRenderer(EntityRendererFactory.Context ctx) {
		super(ctx);
		this.blockRenderManager = ctx.getBlockRenderManager();
		this.shadowRadius = 0.6f;
	}

	@Override
	public void render(AirdropEntity entity, float yaw, float tickDelta, MatrixStack matrices,
			VertexConsumerProvider vertexConsumers, int light) {
		super.render(entity, yaw, tickDelta, matrices, vertexConsumers, light);

		float sway = (float) Math.sin((entity.age + tickDelta) * 0.08f) * 3.5f;

		matrices.push();
		matrices.multiply(RotationAxis.POSITIVE_Z.rotationDegrees(sway));
		matrices.multiply(RotationAxis.POSITIVE_X.rotationDegrees(sway * 0.5f));

		// Crate body centered around origin
		matrices.push();
		matrices.translate(-0.5, 0.0, -0.5);
		blockRenderManager.renderBlockAsEntity(
			ModBlocks.AIRDROP_CRATE.getDefaultState(),
			matrices,
			vertexConsumers,
			light,
			OverlayTexture.DEFAULT_UV
		);
		matrices.pop();

		// 4 shroud cords connecting crate corners to canopy
		BlockState cordState = Blocks.GRAY_CONCRETE.getDefaultState();
		float[][] corners = {
			{-0.42f, -0.42f, 12f, -12f},
			{0.42f, -0.42f, 12f, 12f},
			{-0.42f, 0.42f, -12f, -12f},
			{0.42f, 0.42f, -12f, 12f}
		};
		for (float[] c : corners) {
			matrices.push();
			matrices.translate(c[0], 0.92f, c[1]);
			matrices.multiply(RotationAxis.POSITIVE_X.rotationDegrees(c[2]));
			matrices.multiply(RotationAxis.POSITIVE_Z.rotationDegrees(c[3]));
			matrices.scale(0.05f, 1.55f, 0.05f);
			matrices.translate(-0.5, 0.0, -0.5);
			blockRenderManager.renderBlockAsEntity(cordState, matrices, vertexConsumers, light, OverlayTexture.DEFAULT_UV);
			matrices.pop();
		}

		// Parachute canopy: 3 striped panels (red / white / red)
		renderCanopyPanel(matrices, vertexConsumers, light, Blocks.RED_WOOL.getDefaultState(), -0.82f, 2.38f, 0.82f, 0.22f, 2.4f, -10f);
		renderCanopyPanel(matrices, vertexConsumers, light, Blocks.WHITE_WOOL.getDefaultState(), 0.0f, 2.48f, 0.84f, 0.24f, 2.45f, 0f);
		renderCanopyPanel(matrices, vertexConsumers, light, Blocks.RED_WOOL.getDefaultState(), 0.82f, 2.38f, 0.82f, 0.22f, 2.4f, 10f);

		matrices.pop();
	}

	private void renderCanopyPanel(MatrixStack matrices, VertexConsumerProvider vertexConsumers, int light,
			BlockState state, float xOffset, float yOffset, float width, float height, float depth, float tiltDeg) {
		matrices.push();
		matrices.translate(xOffset, yOffset, 0.0f);
		matrices.multiply(RotationAxis.POSITIVE_Z.rotationDegrees(-tiltDeg));
		matrices.scale(width, height, depth);
		matrices.translate(-0.5, 0.0, -0.5);
		blockRenderManager.renderBlockAsEntity(state, matrices, vertexConsumers, light, OverlayTexture.DEFAULT_UV);
		matrices.pop();
	}

	}

		model.hat.copyTransform(model.head);
		model.body.setAngles(0.0f, 0.0f, 0.0f);
		model.jacket.copyTransform(model.body);
		model.leftArm.setAngles(-0.15f, 0.0f, -0.35f);
		model.leftSleeve.copyTransform(model.leftArm);
		model.rightArm.setAngles(0.10f, 0.0f, 0.42f);

package com.tarkovcraft.client.renderer;

import com.tarkovcraft.client.model.GearModel;
import com.tarkovcraft.health.HitZone;

import org.jetbrains.annotations.Nullable;

/** Selects a model from the pack for each supported equipment family. */
public final class GearAppearance {
	private GearAppearance() {
	}

	@Nullable
	public static GearModel worn(Item item) {
		if (item instanceof BackpackItem backpack) {
			return backpack.getGridWidth() * backpack.getGridHeight() >= 24
				? GearModel.BACKPACK_LARGE : GearModel.BACKPACK_SMALL;
		}

		if (item instanceof TacticalRigItem) {
			String path = Registries.ITEM.getId(item).getPath();
			if (path.startsWith("remenno_plechevaya_") || path.equals("bank_robber_rig")) {
				return GearModel.RIG_BELT_RPS;
			}
			if (path.startsWith("chekhol_dlya_broneplit_")) {
				return GearModel.RIG_ARMORED;
			}
			return path.equals("scav_vest") || path.startsWith("zhilet_okhrannika_") || path.startsWith("okhotnichiy_zhilet_")
				? GearModel.RIG_SCAV_VEST : GearModel.RIG_CHEST_HEAVY;
		}
		if (item instanceof TarkovArmorItem armor) {
			String path = Registries.ITEM.getId(item).getPath();
			return switch (armor.getTargetSlot()) {
				case HELMET -> helmet(path, armor);
				case FACE_MASK -> faceMask(path, armor);
				case BODY_ARMOR -> bodyArmor(path, armor);
				default -> null;
			};
		}
		if (item instanceof TarkovFunctionalItem functional && functional.isHeadset()) {
			return GearModel.HEADSET_ACTIVE;
		}
		return null;
	}

	private static GearModel helmet(String path, TarkovArmorItem armor) {
		if (path.contains("tagilly")) return GearModel.MASK_TAGILLA;
		if (path.contains("ronin")) return GearModel.HELMET_RONIN;
		if (path.contains("ushanka") || path.contains("shapka")) return GearModel.USHANKA;
		if (path.contains("kepka") || path.startsWith("panama_") || path.startsWith("beret_")) {
			return GearModel.TACTICAL_CAP;
		}
		// Hit zones describe protection. They do not equip a visor on an open shell.
		return armor.coversZone(HitZone.HEAD_EARS) || path.equals("ssh68_helmet") || path.startsWith("broneshlem_altyn_")
			? GearModel.HELMET_LOWCUT : GearModel.HELMET_HIGHCUT;
	}

	private static GearModel faceMask(String path, TarkovArmorItem armor) {
		if (path.startsWith("vizor_")) return GearModel.VISOR;
		if (path.contains("ochki")) return GearModel.TACTICAL_GOGGLES;
		if (armor.isGasProtection() || path.startsWith("respirator")) return GearModel.GAS_MASK;
		if (path.contains("tagilly") || path.contains("ten_smer") || path.equals("death_shadow_mask")) {
			return GearModel.MASK_TAGILLA;
		}
		return GearModel.BALACLAVA;
	}

	private static GearModel bodyArmor(String path, TarkovArmorItem armor) {
		if (path.contains("slick") || path.contains("paca") || path.contains("hexgrid")) {
			return GearModel.ARMOR_SLICK;
		}
		return armor.getCoveredZones().contains(HitZone.NECK) || path.equals("zabralo_armor")
			? GearModel.ARMOR_HEAVY : GearModel.ARMOR_MEDIUM;
	}

	@Nullable
	public static GearModel ground(ItemStack stack) {
		if (stack.isEmpty()) return null;
		Item item = stack.getItem();
		if ((item instanceof TarkovMedicalItem medical && medical.getMedicalType().getCapacityMode() == MedicalType.CapacityMode.HP_POOL)
			|| (item instanceof TarkovCaseItem storage && storage.getCaseRole() == TarkovCaseItem.CaseRole.MEDICAL)) {
			return GearModel.GROUND_CASE_MEDKIT;
		}
		GearModel worn = worn(item);
		if (worn == null) return null;
		return switch (worn) {
			case BACKPACK_SMALL -> GearModel.GROUND_BACKPACK_SMALL;
			case BACKPACK_LARGE -> GearModel.GROUND_BACKPACK_LARGE;
			case ARMOR_HEAVY, ARMOR_MEDIUM, ARMOR_SLICK -> GearModel.GROUND_ARMOR;
			case RIG_ARMORED, RIG_BELT_RPS, RIG_CHEST_HEAVY, RIG_SCAV_VEST -> GearModel.GROUND_RIG;
			case HELMET_HIGHCUT, HELMET_LOWCUT -> GearModel.GROUND_HELMET_OPEN;
			case HELMET_RONIN, HELMET_VISOR -> GearModel.GROUND_HELMET_VISOR;
			case MASK_TAGILLA -> GearModel.GROUND_MASK_TAGILLA;
			default -> worn;
		};
	}
}

package com.tarkovcraft.client.renderer;

import com.tarkovcraft.health.HitZone;
import com.tarkovcraft.health.HitZoneBoxes;
import net.fabricmc.api.EnvType;
import net.fabricmc.api.Environment;
import net.fabricmc.fabric.api.client.rendering.v1.WorldRenderContext;
import net.minecraft.client.MinecraftClient;
import net.minecraft.client.render.RenderLayer;

import net.minecraft.client.util.math.MatrixStack;
import net.minecraft.entity.Entity;
import net.minecraft.entity.LivingEntity;
import net.minecraft.entity.decoration.ArmorStandEntity;
import net.minecraft.entity.mob.MobEntity;
import net.minecraft.entity.player.PlayerEntity;
import net.minecraft.util.math.MathHelper;
import net.minecraft.util.math.Vec3d;
import org.joml.Matrix3f;
import org.joml.Matrix4f;

import java.util.List;

/**
 * Renders live 3D OBB hitboxes from HitZoneBoxes when F3+B OR F8 is active.
 */
@Environment(EnvType.CLIENT)
public final class HitZoneDebugRenderer {
	private static final double MAX_RENDER_DIST_SQ = 64.0 * 64.0;
	private static boolean userToggled = false;

	private HitZoneDebugRenderer() {
	}

	public static boolean toggle() {
		userToggled = !userToggled;
		return userToggled;
	}

	public static boolean isUserToggled() {
		return userToggled;
	}

	public static void render(WorldRenderContext ctx) {
		MinecraftClient client = MinecraftClient.getInstance();
		if (client == null || ctx.world() == null || client.player == null) {
			return;
		}
		boolean enabled = userToggled || client.getEntityRenderDispatcher().shouldRenderHitboxes();
		if (!enabled) {
			return;
		}
		VertexConsumerProvider consumers = ctx.consumers();
		if (consumers == null) {
			return;
		}

		MatrixStack matrices = ctx.matrixStack();
		Vec3d camPos = ctx.camera().getPos();
		float tickDelta = ctx.tickDelta();
		boolean firstPerson = client.options.getPerspective().isFirstPerson();

		VertexConsumer lines = consumers.getBuffer(RenderLayer.getLines());
		matrices.push();
		matrices.translate(-camPos.x, -camPos.y, -camPos.z);
		Matrix4f posMat = matrices.peek().getPositionMatrix();
		Matrix3f normMat = matrices.peek().getNormalMatrix();

		for (Entity e : ctx.world().getEntities()) {
			if (!(e instanceof LivingEntity living) || !living.isAlive()) {
				continue;
			}
			if (!shouldRenderFor(living, client.player, firstPerson)) {
				continue;
			}
			if (living.squaredDistanceTo(camPos.x, camPos.y, camPos.z) > MAX_RENDER_DIST_SQ) {
				continue;
			}

			Vec3d lerpedPos = living.getLerpedPos(tickDelta);
			float bodyYaw = MathHelper.lerpAngleDegrees(tickDelta, living.prevBodyYaw, living.bodyYaw);
			float headYaw = MathHelper.lerpAngleDegrees(tickDelta, living.prevHeadYaw, living.headYaw);

			List<HitZoneBoxes.ZoneBox> boxes = HitZoneBoxes.compute(
				lerpedPos, living.getHeight(), living.getWidth(), bodyYaw, headYaw
			);
			for (HitZoneBoxes.ZoneBox box : boxes) {
				float[] rgba = colorForZone(box.zone());
				Vec3d[] c = box.getWorldCorners();
				for (int[] edge : HitZoneBoxes.EDGE_INDICES) {
					drawEdge(lines, posMat, normMat, c[edge[0]], c[edge[1]], rgba[0], rgba[1], rgba[2], rgba[3]);
				}
			}
		}

		matrices.pop();
	}

	private static boolean shouldRenderFor(LivingEntity living, PlayerEntity localPlayer, boolean firstPerson) {
		if (living instanceof PlayerEntity) {
			return !(living == localPlayer && firstPerson);
		}
		if (living instanceof ArmorStandEntity) {
			return true;
		}
		if (living instanceof MobEntity) {
			return living.getHeight() >= 1.35f && living.getWidth() <= 0.85f;
		}
		return false;
	}

	private static void drawEdge(
		VertexConsumer vc, Matrix4f posMat, Matrix3f normMat,
		Vec3d p0, Vec3d p1, float r, float g, float b, float a
	) {
		float dx = (float) (p1.x - p0.x);
		float dy = (float) (p1.y - p0.y);
		float dz = (float) (p1.z - p0.z);
		float len = (float) Math.sqrt(dx * dx + dy * dy + dz * dz);
		float nx = len > 1e-6f ? dx / len : 0.0f;
		float ny = len > 1e-6f ? dy / len : 1.0f;
		float nz = len > 1e-6f ? dz / len : 0.0f;

		vc.vertex(posMat, (float) p0.x, (float) p0.y, (float) p0.z)
			.color(r, g, b, a)
			.normal(normMat, nx, ny, nz)
			.next();
		vc.vertex(posMat, (float) p1.x, (float) p1.y, (float) p1.z)
			.color(r, g, b, a)
			.normal(normMat, nx, ny, nz)
			.next();
	}

	private static float[] colorForZone(HitZone zone) {
		return switch (zone) {
			case HEAD_EYES -> new float[]{1.00f, 0.12f, 0.12f, 1.0f};
			case HEAD_EARS -> new float[]{0.95f, 0.15f, 0.90f, 1.0f};
			case HEAD_JAWS -> new float[]{1.00f, 0.55f, 0.10f, 1.0f};
			case HEAD_TOP -> new float[]{0.25f, 0.78f, 1.00f, 1.0f};
			case NECK -> new float[]{1.00f, 0.92f, 0.15f, 1.0f};
			case THORAX -> new float[]{0.15f, 0.90f, 0.30f, 1.0f};
			case STOMACH -> new float[]{0.62f, 0.96f, 0.22f, 1.0f};
			case ARMPIT_LEFT, ARMPIT_RIGHT -> new float[]{0.65f, 0.25f, 1.00f, 1.0f};
			case LEFT_ARM, RIGHT_ARM -> new float[]{0.20f, 0.48f, 1.00f, 1.0f};
			case LEFT_LEG, RIGHT_LEG -> new float[]{0.12f, 0.88f, 0.82f, 1.0f};
		};
	}
}

import net.fabricmc.fabric.api.client.networking.v1.ClientPlayNetworking;

import net.minecraft.client.MinecraftClient;
import net.minecraft.client.network.AbstractClientPlayerEntity;

import net.minecraft.client.render.entity.feature.FeatureRenderer;

import java.util.HashMap;
import java.util.Map;
import java.util.UUID;

public final class PlayerGearFeature extends FeatureRenderer<AbstractClientPlayerEntity, PlayerEntityModel<AbstractClientPlayerEntity>> {
	private static final Map<UUID, EquipmentAppearance> REMOTE_APPEARANCES = new HashMap<>();

	private PlayerGearFeature(PlayerEntityRenderer renderer) {
		super(renderer);
	}

	public static void register() {
		LivingEntityFeatureRendererRegistrationCallback.EVENT.register((type, renderer, helper, context) -> {
			if (renderer instanceof PlayerEntityRenderer playerRenderer) {
				helper.register(new PlayerGearFeature(playerRenderer));
			}
		});
		ClientPlayNetworking.registerGlobalReceiver(EquipmentAppearance.CHANNEL, (client, handler, buf, sender) -> {
			UUID playerId = buf.readUuid();
			EquipmentAppearance appearance = EquipmentAppearance.read(buf);
			client.execute(() -> {
				if (client.getNetworkHandler() != handler) return;
				if (appearance.isEmpty()) {
					REMOTE_APPEARANCES.remove(playerId);
				} else {
					REMOTE_APPEARANCES.put(playerId, appearance);
				}
			});
		});
		ClientPlayConnectionEvents.DISCONNECT.register((handler, client) -> REMOTE_APPEARANCES.clear());
	}

	@Override
	public void render(MatrixStack matrices, VertexConsumerProvider vertices, int light, AbstractClientPlayerEntity player,
			float limbAngle, float limbDistance, float tickDelta, float animationProgress, float headYaw, float headPitch) {
		if (player.isInvisible() || player.isSpectator()) return;
		EquipmentAppearance appearance = REMOTE_APPEARANCES.get(player.getUuid());
		if (player == MinecraftClient.getInstance().player && player instanceof PlayerLoadoutHolder holder) {
			appearance = EquipmentAppearance.capture(holder.tarkovcraft$getLoadout());
		}
		if (appearance == null) return;

		for (Item item : appearance.items()) {
			GearModel model = GearAppearance.worn(item);
			if (model != null) {
				model.renderWorn(matrices, vertices, light, getContextModel(), item);
			}
		}
	}
}

package com.tarkovcraft.client.screen;

import com.tarkovcraft.ammo.AmmoTier;
import com.tarkovcraft.ammo.TaCZCaliberCatalog;
import com.tarkovcraft.network.AmmoSelectionNetworking;
import net.fabricmc.fabric.api.client.networking.v1.ClientPlayNetworking;

import org.lwjgl.glfw.GLFW;

import java.util.Map;

/** Selects which available ammo tier TaCZ should draw from first on the next reload. */
public final class AmmoRadialScreen extends Screen {
	private static final int CARD_WIDTH = 60;
	private static final int CARD_HEIGHT = 28;
	private static final int RADIUS = 70;
	private static final AmmoTier[] TIERS = AmmoTier.values();

	private final String gunId;
	private final TaCZCaliberCatalog.CaliberSpec caliber;
	private final Map<AmmoTier, Integer> ammoCounts;

	public AmmoRadialScreen(String gunId, String ammoId, Map<AmmoTier, Integer> ammoCounts) {
		super(Text.translatable("screen.tarkovcraft.ammo_radial"));
		this.gunId = gunId;
		this.caliber = TaCZCaliberCatalog.getSpecOrDynamic(ammoId);
		this.ammoCounts = Map.copyOf(ammoCounts);
	}

	@Override
	public void render(DrawContext context, int mouseX, int mouseY, float delta) {
		int centerX = width / 2;
		int centerY = height / 2;
		context.fillGradient(0, 0, width, height, 0x79131517, 0xB1121315);
		context.drawCenteredTextWithShadow(textRenderer, title, centerX, centerY - 12, 0xE5CE9F);
		context.drawCenteredTextWithShadow(textRenderer, caliber.displayName(), centerX, centerY + 2, 0xB7B4AD);
		context.drawCenteredTextWithShadow(textRenderer, Text.translatable("screen.tarkovcraft.ammo_radial_hint"),
			centerX, centerY + radiusY() + 18, 0xB7B4AD);

		for (AmmoTier tier : TIERS) {
			int x = cardX(tier);
			int y = cardY(tier);
			boolean selectable = isSelectable(tier);
			boolean hovered = selectable && isOverCard(tier, mouseX, mouseY);
			context.fill(x, y, x + CARD_WIDTH, y + CARD_HEIGHT, hovered ? 0xE7393630 : 0xE7222427);
			context.drawBorder(x, y, CARD_WIDTH, CARD_HEIGHT, hovered ? tier.getArgbColor() : 0xFF585750);
			context.drawCenteredTextWithShadow(textRenderer, tier.getBadge(), x + CARD_WIDTH / 2, y + 4,
				selectable ? tier.getArgbColor() : 0x808080);
			context.drawCenteredTextWithShadow(textRenderer, Integer.toString(ammoCounts.getOrDefault(tier, 0)),
				x + CARD_WIDTH / 2, y + 16, selectable ? 0xE7E3DB : 0x808080);
		}
	}

	@Override
	public boolean mouseClicked(double mouseX, double mouseY, int button) {
		if (button == GLFW.GLFW_MOUSE_BUTTON_LEFT) {
			for (AmmoTier tier : TIERS) {
				if (isSelectable(tier) && isOverCard(tier, mouseX, mouseY)) {
					if (ClientPlayNetworking.canSend(AmmoSelectionNetworking.SELECT)) {
						PacketByteBuf buf = PacketByteBufs.create();
						buf.writeByte(tier.ordinal());
						buf.writeString(gunId, AmmoSelectionNetworking.MAX_GUN_ID_LENGTH);
						ClientPlayNetworking.send(AmmoSelectionNetworking.SELECT, buf);
					}
					close();
					return true;
				}
			}
		}
		return super.mouseClicked(mouseX, mouseY, button);
	}

	@Override
	public boolean shouldPause() {
		return false;
	}

	private boolean isSelectable(AmmoTier tier) {
		return caliber.isTierAllowed(tier) && ammoCounts.getOrDefault(tier, 0) > 0;
	}

	private boolean isOverCard(AmmoTier tier, double mouseX, double mouseY) {
		int x = cardX(tier);
		int y = cardY(tier);
		return mouseX >= x && mouseX < x + CARD_WIDTH && mouseY >= y && mouseY < y + CARD_HEIGHT;
	}

	private int cardX(AmmoTier tier) {
		int radius = Math.min(RADIUS, width / 2 - CARD_WIDTH / 2 - 4);
		int offset = switch (tier) {
			case FMJ -> radius;
			case IGN -> -radius;
			default -> 0;
		};
		return width / 2 + offset - CARD_WIDTH / 2;
	}

	private int cardY(AmmoTier tier) {
		int offset = switch (tier) {
			case HP -> -radiusY();
			case AP -> radiusY();
			default -> 0;
		};
		return height / 2 + offset - CARD_HEIGHT / 2;
	}

	private int radiusY() {
		return Math.min(RADIUS, height / 2 - 30);
	}
}

package com.tarkovcraft.client.screen;

import com.tarkovcraft.client.ClientLoadoutCache;

import java.util.List;

/**
 * Escape-from-Tarkov-style KIA Death Screen:
 * Displays killer, weapon/ammo, exact fatal HitZone (e.g. "Голова — глаза (Ваншот!)"),
 * shot distance, body-part silhouette showing final damage state, and recent hit log.
 */
public class TarkovDeathScreen extends Screen {
	private int ticksSinceDeath = 0;
	private ButtonWidget respawnButton;
	private ButtonWidget titleButton;

	public TarkovDeathScreen() {
		super(Text.translatable("screen.tarkovcraft.kia_title"));
	}

	@Override
	public boolean shouldCloseOnEsc() {
		return false;
	}

	@Override
	protected void init() {
		super.init();
		int btnY = Math.min(this.height - 32, (this.height + 210) / 2);
		respawnButton = this.addDrawableChild(ButtonWidget.builder(
			Text.translatable("screen.tarkovcraft.respawn"),
			btn -> {
				if (this.client != null && this.client.player != null) {
					this.client.player.requestRespawn();
					this.client.setScreen(null);
				}
			}
		).dimensions(this.width / 2 - 125, btnY, 120, 20).build());

		titleButton = this.addDrawableChild(ButtonWidget.builder(
			Text.translatable("deathScreen.titleScreen"),
			btn -> {
				if (this.client != null && this.client.world != null) {
					this.client.world.disconnect();
					this.client.disconnect();
					this.client.setScreen(new TitleScreen());
				}
			}
		).dimensions(this.width / 2 + 5, btnY, 120, 20).build());

		respawnButton.active = false;
	}

	@Override
	public void tick() {
		super.tick();
		ticksSinceDeath++;
		if (ticksSinceDeath >= 15 && respawnButton != null) {
			respawnButton.active = true;
		}
	}

	@Override
	public void render(DrawContext context, int mouseX, int mouseY, float delta) {
		// Deep crimson-charcoal Tarkov KIA vignette
		context.fillGradient(0, 0, this.width, this.height, 0xE6140909, 0xF20A0C0E);

		BodyPartHealth health = ClientLoadoutCache.getBodyHealth();
		DamageReport fatal = health.getLastFatalReport();

		int panelW = Math.min(this.width - 24, 430);
		int panelH = 216;
		int px = (this.width - panelW) / 2;
		int py = (this.height - panelH) / 2 - 12;

		context.fill(px, py, px + panelW, py + panelH, 0xE6111417);
		context.drawBorder(px, py, panelW, panelH, 0xFF7A2424);

		// Crimson KIA Header Bar
		context.fill(px + 1, py + 1, px + panelW - 1, py + 24, 0xFF3A1212);
		context.fill(px + 1, py + 24, px + panelW - 1, py + 25, 0xFF9E2B2B);
		context.drawCenteredTextWithShadow(textRenderer,
			Text.translatable("screen.tarkovcraft.kia_header"),
			px + panelW / 2, py + 8, 0xFFFF5E5E);

		// Left Panel: Final Body Silhouette
		int leftW = 148;
		int leftX = px + 8;
		int contentY = py + 32;
		int contentH = panelH - 40;
		context.fill(leftX, contentY, leftX + leftW, contentY + contentH, 0xCC161A1E);
		context.drawBorder(leftX, contentY, leftW, contentH, 0xFF2D333A);

		int silCx = leftX + leftW / 2;
		int silTop = contentY + 12;
		BodyHealthHudHelper.drawBodyPartRect(context, silCx - 10, silTop, 20, 20, health.getRatio(BodyPart.HEAD));
		BodyHealthHudHelper.drawBodyPartRect(context, silCx - 15, silTop + 24, 30, 28, health.getRatio(BodyPart.THORAX));
		BodyHealthHudHelper.drawBodyPartRect(context, silCx - 13, silTop + 54, 26, 18, health.getRatio(BodyPart.STOMACH));
		BodyHealthHudHelper.drawBodyPartRect(context, silCx - 28, silTop + 24, 11, 42, health.getRatio(BodyPart.LEFT_ARM));
		BodyHealthHudHelper.drawBodyPartRect(context, silCx + 17, silTop + 24, 11, 42, health.getRatio(BodyPart.RIGHT_ARM));
		BodyHealthHudHelper.drawBodyPartRect(context, silCx - 14, silTop + 74, 12, 46, health.getRatio(BodyPart.LEFT_LEG));
		BodyHealthHudHelper.drawBodyPartRect(context, silCx + 2, silTop + 74, 12, 46, health.getRatio(BodyPart.RIGHT_LEG));

		String totalStr = Math.round(health.getTotalHp()) + " / " + (int) BodyPart.TOTAL_MAX_HP + " HP";
		context.drawCenteredTextWithShadow(textRenderer, totalStr, silCx, contentY + contentH - 16, 0xFFCF5555);

		// Right Panel: Cause of Death + Hit Log
		int rightX = leftX + leftW + 8;
		int rightW = panelW - leftW - 24;
		context.fill(rightX, contentY, rightX + rightW, contentY + contentH, 0xCC161A1E);
		context.drawBorder(rightX, contentY, rightW, contentH, 0xFF2D333A);

		int ry = contentY + 8;
		if (fatal != null) {
			String zoneName = Text.translatable(fatal.hitZone().getTranslationKey()).getString();
			if (fatal.headEyesOneshot()) {
				zoneName += " (" + Text.translatable("screen.tarkovcraft.oneshot").getString() + ")";
			} else if (fatal.overflowedToThorax()) {
				zoneName += " -> " + Text.translatable(BodyPart.THORAX.getTranslationKey()).getString();
			}

			drawInfoRow(context, rightX + 8, ry, Text.translatable("screen.tarkovcraft.kia_killer").getString(),
				fatal.killerName().isEmpty() ? "Unknown" : fatal.killerName(), 0xFFEFEFEF);
			ry += 18;

			drawInfoRow(context, rightX + 8, ry, Text.translatable("screen.tarkovcraft.kia_zone").getString(),
				zoneName, 0xFFFF5E5E);
			ry += 18;

			String weaponLine = fatal.weaponName().isEmpty() ? "-" : fatal.weaponName();
			if (!fatal.ammoName().isEmpty()) {
				weaponLine += " // " + fatal.ammoName();
			}
			drawInfoRow(context, rightX + 8, ry, Text.translatable("screen.tarkovcraft.kia_weapon").getString(),
				weaponLine, 0xFFD4B46A);
			ry += 18;

			String distLine = fatal.distanceMeters() > 0.1f ? String.format("%.1f m", fatal.distanceMeters()) : "-";
			drawInfoRow(context, rightX + 8, ry, Text.translatable("screen.tarkovcraft.kia_distance").getString(),
				distLine, 0xFFB8C0C8);
			ry += 22;
		} else {
			context.drawTextWithShadow(textRenderer, Text.translatable("screen.tarkovcraft.kia_unknown"), rightX + 8, ry, 0xFFB8C0C8);
			ry += 20;
		}

		// Recent Hits Log
		context.fill(rightX + 6, ry, rightX + rightW - 6, ry + 1, 0xFF2D333A);
		ry += 6;
		drawTiny(context, Text.translatable("health.tarkovcraft.recent_hits").getString(), rightX + 8, ry, 0xFFD4B46A);
		ry += 10;

		List<DamageReport> hits = health.getRecentHits();
		int shown = Math.min(5, hits.size());
		for (int i = 0; i < shown; i++) {
			DamageReport r = hits.get(i);
			String zt = Text.translatable(r.hitZone().getTranslationKey()).getString();
			String armorTag = r.ricocheted() ? " [RICOCHET]" : (r.armorBlocked() ? " [ARMOR]" : "");
			String line = String.format("#%d  %s  -%.0f HP%s%s",
				i + 1, zt, r.finalDamage(),
				armorTag,
				r.overflowedToThorax() ? " -> THORAX" : ""
			);
			drawTiny(context, line, rightX + 8, ry, i == 0 ? 0xFFFF7A7A : 0xFFB8C0C8);
			ry += 10;
		}

		super.render(context, mouseX, mouseY, delta);
	}

	private void drawInfoRow(DrawContext context, int x, int y, String label, String value, int valueColor) {
		drawTiny(context, label, x, y, 0xFF88929C);
		context.drawTextWithShadow(textRenderer, value, x, y + 7, valueColor);
	}

	private void drawTiny(DrawContext context, String text, int x, int y, int col) {
		context.getMatrices().push();
		context.getMatrices().scale(0.72f, 0.72f, 1.0f);
		context.drawTextWithShadow(textRenderer, text, (int) (x / 0.72f), (int) (y / 0.72f), col);
		context.getMatrices().pop();
	}
}

/**
 * Draggable EFT-style item inspection window rendered above the inventory on z = 420.
 */
@Environment(EnvType.CLIENT)
public final class TarkovInspectWindow {
	private static final int WIN_W = 256;
	private static final int WIN_H = 224;
	private static final int HEADER_H = 20;

	private final ItemStack stack;
	private int x;
	private int y;
	private int scrollY = 0;
	private int maxScrollY = 0;
	private boolean draggingHeader = false;

	public TarkovInspectWindow(ItemStack stack, int screenW, int screenH) {
		this(stack, screenW, screenH, 0);
	}

	public TarkovInspectWindow(ItemStack stack, int screenW, int screenH, int cascadeIndex) {
		this.stack = stack.copy();
		int offset = cascadeIndex * 16;
		this.x = Math.max(8, (screenW - WIN_W) / 2 + offset);
		this.y = Math.max(8, (screenH - WIN_H) / 2 + offset);
		clampToScreen(screenW, screenH);
	}

	public void render(DrawContext ctx, TextRenderer tr, int mouseX, int mouseY, int screenW, int screenH) {
		clampToScreen(screenW, screenH);
		TarkovUiTheme.clearDepthBuffer(ctx);

		ctx.getMatrices().push();
		ctx.getMatrices().translate(0.0f, 0.0f, TarkovUiTheme.Z_MODAL);

		ctx.fill(x, y, x + WIN_W, y + WIN_H, TarkovUiTheme.BG_MODAL);
		ctx.drawBorder(x, y, WIN_W, WIN_H, TarkovUiTheme.BORDER_POPUP);

		// Header
		ctx.fill(x + 1, y + 1, x + WIN_W - 1, y + HEADER_H, TarkovUiTheme.BG_MODAL_HEADER);
		ctx.fill(x + 1, y + HEADER_H, x + WIN_W - 1, y + HEADER_H + 1, TarkovUiTheme.BORDER_PANEL);
		String title = tr.trimToWidth(stack.getName().getString(), WIN_W - 32);
		ctx.drawText(tr, title, x + 6, y + 6, TarkovUiTheme.TEXT_PRIMARY, false);

		// Close [X] button
		int closeX = x + WIN_W - 18;
		int closeY = y + 3;
		boolean closeHov = mouseX >= closeX && mouseX < closeX + 14 && mouseY >= closeY && mouseY < closeY + 14;
		ctx.fill(closeX, closeY, closeX + 14, closeY + 14, closeHov ? TarkovUiTheme.HOVER_MENU_BG : 0xFF1D2025);
		TarkovUiTheme.drawCenteredFlatText(ctx, tr, "X", closeX + 7, closeY + 4,
			closeHov ? TarkovUiTheme.HOVER_MENU_TEXT : TarkovUiTheme.TEXT_PRIMARY, TarkovUiTheme.TEXT_SCALE_SMALL);

		// Left 70x70 large item preview
		int previewX = x + 8;
		int previewY = y + HEADER_H + 8;
		ctx.fill(previewX, previewY, previewX + 70, previewY + 70, TarkovUiTheme.itemCategoryTint(stack));
		ctx.drawBorder(previewX, previewY, 70, 70, TarkovUiTheme.BORDER_OCCUPIED);

		ctx.getMatrices().push();
		ctx.getMatrices().translate(previewX + 35.0f, previewY + 35.0f, 10.0f);
		ctx.getMatrices().scale(3.0f, 3.0f, 1.0f);
		ctx.drawItem(stack, -8, -8);
		ctx.getMatrices().pop();

		// Clear OpenGL depth buffer after 3D item draw so labels/lines/subsequent windows never get clipped
		TarkovUiTheme.clearDepthBuffer(ctx);

		ItemSize size = ItemSizeRegistry.getSize(stack);
		TarkovUiTheme.drawRightAlignedFlatText(ctx, tr, size.width() + "x" + size.height(),
			previewX + 67, previewY + 60, TarkovUiTheme.TEXT_SECONDARY, TarkovUiTheme.TEXT_SCALE_SMALL);

		// Right metadata column
		int metaX = previewX + 78;
		int metaY = previewY + 2;
		float weight = (stack.getItem() instanceof TarkovArmorItem armor)
			? armor.getWeightKg()
			: PlayerLoadout.estimateItemWeight(stack);
		ItemRarityTier rarity = ItemRarityRegistry.getTier(stack);

		TarkovUiTheme.drawFlatText(ctx, tr,
			Text.translatable("ui.tarkovcraft.inspect_weight", String.format(Locale.ROOT, "%.2f", weight)).getString(),
			metaX, metaY, TarkovUiTheme.TEXT_PRIMARY, TarkovUiTheme.TEXT_SCALE_SMALL);
		metaY += 12;
		TarkovUiTheme.drawFlatText(ctx, tr,
			Text.translatable("ui.tarkovcraft.inspect_count", stack.getCount() + "/" + stack.getMaxCount()).getString(),
			metaX, metaY, TarkovUiTheme.TEXT_SECONDARY, TarkovUiTheme.TEXT_SCALE_SMALL);
		metaY += 12;
		TarkovUiTheme.drawFlatText(ctx, tr,
			Text.translatable("ui.tarkovcraft.inspect_size", size.width(), size.height()).getString(),
			metaX, metaY, TarkovUiTheme.TEXT_SECONDARY, TarkovUiTheme.TEXT_SCALE_SMALL);
		metaY += 12;
		TarkovUiTheme.drawFlatText(ctx, tr,
			Text.translatable("ui.tarkovcraft.inspect_rarity", rarity.name()).getString(),
			metaX, metaY, TarkovUiTheme.ACCENT_SAND, TarkovUiTheme.TEXT_SCALE_SMALL);
		metaY += 12;

		FoodComponent fc = stack.getItem().getFoodComponent();
		if (fc != null) {
			int energy = fc.getHunger() * 5;
			int sat = Math.round(fc.getHunger() * fc.getSaturationModifier() * 10.0f);
			TarkovUiTheme.drawFlatText(ctx, tr,
				Text.translatable("ui.tarkovcraft.inspect_food", energy, sat).getString(),
				metaX, metaY, 0xFF77B85A, TarkovUiTheme.TEXT_SCALE_SMALL);
		}

		// Divider & scrollable colored tooltip lines
		int detY = previewY + 76;
		ctx.fill(x + 8, detY, x + WIN_W - 8, detY + 1, TarkovUiTheme.BORDER_PANEL);
		detY += 5;

		int boxBottom = y + WIN_H - 6;
		int viewH = Math.max(12, boxBottom - detY);
		List<Text> details = collectDetailLines();
		int lineH = 10;
		maxScrollY = Math.max(0, details.size() * lineH - viewH);
		scrollY = Math.max(0, Math.min(scrollY, maxScrollY));

		ctx.enableScissor(x + 6, detY, x + WIN_W - 6, boxBottom);
		float s = TarkovUiTheme.TEXT_SCALE_SMALL;
		int maxUnscaledW = Math.round((WIN_W - 22) / s);
		for (int i = 0; i < details.size(); i++) {
			int ly = detY + i * lineH - scrollY;
			if (ly + lineH < detY || ly > boxBottom) {
				continue;
			}
			Text line = details.get(i);
			String plain = line.getString();
			if (tr.getWidth(plain) > maxUnscaledW) {
				String trimmed = tr.trimToWidth(plain, maxUnscaledW - tr.getWidth("...")) + "...";
				line = Text.literal(trimmed).setStyle(line.getStyle());
			}
			ctx.getMatrices().push();
			ctx.getMatrices().scale(s, s, 1.0f);
			ctx.drawText(tr, line, Math.round((x + 8) / s), Math.round(ly / s), TarkovUiTheme.TEXT_SECONDARY, false);
			ctx.getMatrices().pop();
		}
		ctx.disableScissor();

		if (maxScrollY > 0) {
			int sbX = x + WIN_W - 6;
			ctx.fill(sbX, detY, sbX + 2, boxBottom, 0x6620242A);
			int thumbH = Math.max(10, Math.round(viewH * (viewH / (float) (viewH + maxScrollY))));
			int thumbY = detY + Math.round((viewH - thumbH) * (scrollY / (float) maxScrollY));
			ctx.fill(sbX, thumbY, sbX + 2, thumbY + thumbH, TarkovUiTheme.ACCENT_SAND);
		}

		ctx.getMatrices().pop();
	}

	private List<Text> collectDetailLines() {
		List<Text> out = new ArrayList<>();
		MinecraftClient mc = MinecraftClient.getInstance();
		boolean advanced = mc != null && mc.options.advancedItemTooltips;
		TooltipContext ctx = advanced ? TooltipContext.ADVANCED : TooltipContext.BASIC;
		List<Text> tooltip = stack.getTooltip(mc != null ? mc.player : null, ctx);
		for (int i = 1; i < tooltip.size(); i++) {
			Text t = tooltip.get(i);
			String s = t.getString().trim();
			if (s.isEmpty()) {
				continue;
			}
			if (!advanced && isTechnicalDebugLine(s)) {
				continue;
			}
			out.add(t);
		}
		return out;
	}

	private static boolean isTechnicalDebugLine(String s) {
		String lower = s.toLowerCase(Locale.ROOT);
		if (lower.startsWith("nbt:") || lower.startsWith("gunid:") || lower.equals("tacz:modern_kinetic_gun")) {
			return true;
		}
		return lower.matches("^[a-z0-9_]+:[a-z0-9_./-]+$");
	}

	public boolean handleMouseScrolled(double mx, double my, double amount) {
		if (!contains(mx, my) || maxScrollY <= 0) {
			return false;
		}
		scrollY = Math.max(0, Math.min(maxScrollY, scrollY - (int) Math.round(amount * 14)));
		return true;
	}

	public boolean contains(double mx, double my) {
		return mx >= x && mx < x + WIN_W && my >= y && my < y + WIN_H;
	}

	/**
	 * Returns true if the window was closed by clicking [X].
	 */
	public boolean handleMouseClicked(double mx, double my, int button) {
		if (button != 0 || !contains(mx, my)) {
			return false;
		}
		int closeX = x + WIN_W - 18;
		int closeY = y + 3;
		if (mx >= closeX && mx < closeX + 14 && my >= closeY && my < closeY + 14) {
			return true;
		}
		if (my >= y && my < y + HEADER_H) {
			draggingHeader = true;
		}
		return false;
	}

	public boolean handleMouseDragged(double dx, double dy, int screenW, int screenH) {
		if (!draggingHeader) {
			return false;
		}
		x += Math.round(dx);
		y += Math.round(dy);
		clampToScreen(screenW, screenH);
		return true;
	}

	public void handleMouseReleased() {
		draggingHeader = false;
	}

	private void clampToScreen(int screenW, int screenH) {
		x = Math.max(4, Math.min(screenW - WIN_W - 4, x));
		y = Math.max(4, Math.min(screenH - WIN_H - 4, y));
	}
