package com.tarkovcraft.workbench;

import com.tarkovcraft.ammo.AmmoTier;
import com.tarkovcraft.ammo.TaCZCaliberCatalog;
import com.tarkovcraft.grid.ContainerItemHelper;
import com.tarkovcraft.grid.GridContainer;
import com.tarkovcraft.grid.GridEntry;
import com.tarkovcraft.item.BackpackItem;
import com.tarkovcraft.item.TacticalRigItem;
import com.tarkovcraft.loadout.EquipmentSlotType;
import com.tarkovcraft.loadout.PlayerLoadout;
import com.tarkovcraft.loadout.PlayerLoadoutHolder;
import net.minecraft.item.Item;
import net.minecraft.item.ItemStack;
import net.minecraft.server.network.ServerPlayerEntity;
import net.minecraft.sound.SoundCategory;
import net.minecraft.sound.SoundEvents;
import net.minecraft.text.Text;
import net.minecraft.util.Formatting;
import org.jetbrains.annotations.Nullable;

import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;
import java.util.UUID;

/**
 * Server-side inventory scanning, atomic resource deduction (Pockets -> Rig -> Backpack),
 * and crafting execution for the Tarkov Workbench.
 */
public final class WorkbenchInventoryHelper {
	public record SyncPayload(
		Map<Item, Integer> itemCounts,
		Map<String, Integer> fmjAmmoCounts,
		List<String> extraCalibers
	) {
	}

	private WorkbenchInventoryHelper() {
	}

	public static SyncPayload collectSyncPayload(ServerPlayerEntity player) {
		if (player == null || !(player instanceof PlayerLoadoutHolder holder)) {
			return new SyncPayload(Map.of(), Map.of(), List.of());
		}
		if (player.isCreative()) {
			// Scan creative vanilla inventory for any extra custom gunpack calibers
			Set<String> extras = new LinkedHashSet<>();
			for (ItemStack st : player.getInventory().main) {
				collectExtraCaliber(st, extras);
			}
			for (ItemStack st : player.getInventory().offHand) {
				collectExtraCaliber(st, extras);
			}
			return new SyncPayload(Map.of(), Map.of(), List.copyOf(extras));
		}
		return collectSyncPayload(holder.tarkovcraft$getLoadout(), player.getUuid());
	}

	public static SyncPayload collectSyncPayload(PlayerLoadout loadout, @Nullable UUID playerId) {
		Map<Item, Integer> items = new LinkedHashMap<>();
		Map<String, Integer> fmj = new LinkedHashMap<>();
		Set<String> extras = new LinkedHashSet<>();

		for (int i = 0; i < PlayerLoadout.POCKET_COUNT; i++) {
			tallyStack(loadout.getPocket(i), items, fmj, extras);
		}

		ItemStack rig = loadout.getEquipment(EquipmentSlotType.RIG);
		if (!rig.isEmpty() && rig.getItem() instanceof TacticalRigItem && isContainerAccessible(rig, playerId)) {
			for (GridContainer g : ContainerItemHelper.readRigGrids(rig)) {
				for (GridEntry entry : g.getEntries()) {
					tallyStack(entry.getStack(), items, fmj, extras);
				}
			}
		}

		ItemStack bp = loadout.getEquipment(EquipmentSlotType.BACKPACK);
		if (!bp.isEmpty() && bp.getItem() instanceof BackpackItem && isContainerAccessible(bp, playerId)) {
			GridContainer bpGrid = ContainerItemHelper.readBackpackGrid(bp);
			for (GridEntry entry : bpGrid.getEntries()) {
				tallyStack(entry.getStack(), items, fmj, extras);
			}
		}

		return new SyncPayload(items, fmj, List.copyOf(extras));
	}

	private static void tallyStack(
		ItemStack stack,
		Map<Item, Integer> items,
		Map<String, Integer> fmj,
		Set<String> extras
	) {
		if (stack == null || stack.isEmpty()) {
			return;
		}
		if (AmmoTier.isTaczAmmoStack(stack)) {
			String ammoId = AmmoTier.getAmmoId(stack);
			if (TaCZCaliberCatalog.isBallisticAmmoId(ammoId)) {
				collectExtraCaliber(stack, extras);
				if (AmmoTier.getFromAmmoStack(stack) == AmmoTier.FMJ) {
					fmj.merge(ammoId, stack.getCount(), Integer::sum);
				}
			}
			return;
		}
		items.merge(stack.getItem(), stack.getCount(), Integer::sum);
	}

	private static void collectExtraCaliber(ItemStack stack, Set<String> extras) {
		if (stack == null || stack.isEmpty() || !AmmoTier.isTaczAmmoStack(stack)) {
			return;
		}
		String ammoId = AmmoTier.getAmmoId(stack);
		if (TaCZCaliberCatalog.isBallisticAmmoId(ammoId)) {
			boolean isDefault = TaCZCaliberCatalog.getDefaultCalibers().stream()
				.anyMatch(c -> c.ammoId().equals(ammoId));
			if (!isDefault) {
				extras.add(ammoId);
			}
		}
	}

	public static boolean hasMaterials(
		PlayerLoadout loadout,
		@Nullable UUID playerId,
		List<TarkovWorkbenchRecipes.Ingredient> materials,
		@Nullable String requiredFmjAmmoId,
		int requiredFmjCount
	) {
		SyncPayload payload = collectSyncPayload(loadout, playerId);
		for (TarkovWorkbenchRecipes.Ingredient ing : materials) {
			if (payload.itemCounts().getOrDefault(ing.item(), 0) < ing.count()) {
				return false;
			}
		}
		if (requiredFmjCount > 0 && requiredFmjAmmoId != null) {
			if (payload.fmjAmmoCounts().getOrDefault(requiredFmjAmmoId, 0) < requiredFmjCount) {
				return false;
			}
		}
		return true;
	}

	public static boolean consumeMaterials(
		PlayerLoadout loadout,
		@Nullable UUID playerId,
		List<TarkovWorkbenchRecipes.Ingredient> materials,
		@Nullable String requiredFmjAmmoId,
		int requiredFmjCount
	) {
		if (!hasMaterials(loadout, playerId, materials, requiredFmjAmmoId, requiredFmjCount)) {
			return false;
		}

		for (TarkovWorkbenchRecipes.Ingredient ing : materials) {
			consumeFromLoadout(loadout, playerId, ing.item(), null, ing.count());
		}
		if (requiredFmjCount > 0 && requiredFmjAmmoId != null) {
			consumeFromLoadout(loadout, playerId, null, requiredFmjAmmoId, requiredFmjCount);
		}
		return true;
	}

	private static void consumeFromLoadout(
		PlayerLoadout loadout,
		@Nullable UUID playerId,
		@Nullable Item targetItem,
		@Nullable String targetFmjAmmoId,
		int amount
	) {
		int remaining = amount;

		// 1. Pockets
		for (int i = 0; i < PlayerLoadout.POCKET_COUNT && remaining > 0; i++) {
			ItemStack p = loadout.getPocket(i);
			if (matchesConsumptionTarget(p, targetItem, targetFmjAmmoId)) {
				int take = Math.min(remaining, p.getCount());
				p.decrement(take);
				remaining -= take;
				if (p.isEmpty()) {
					loadout.setPocket(i, ItemStack.EMPTY);
				}
			}
		}

		// 2. Equipped Rig
		if (remaining > 0) {
			ItemStack rig = loadout.getEquipment(EquipmentSlotType.RIG);
			if (!rig.isEmpty() && rig.getItem() instanceof TacticalRigItem && isContainerAccessible(rig, playerId)) {
				List<GridContainer> grids = ContainerItemHelper.readRigGrids(rig);
				boolean modified = false;
				for (GridContainer g : grids) {
					List<GridEntry> entries = new ArrayList<>(g.getEntries());
					for (GridEntry entry : entries) {
						if (remaining <= 0) break;
						ItemStack st = entry.getStack();
						if (matchesConsumptionTarget(st, targetItem, targetFmjAmmoId)) {
							int take = Math.min(remaining, st.getCount());
							st.decrement(take);
							remaining -= take;
							if (st.isEmpty()) {
								g.remove(entry.getEntryId());
							}
							modified = true;
						}
					}
					if (remaining <= 0) break;
				}
				if (modified) {
					ContainerItemHelper.writeRigGrids(rig, grids);
				}
			}
		}

		// 3. Equipped Backpack
		if (remaining > 0) {
			ItemStack bp = loadout.getEquipment(EquipmentSlotType.BACKPACK);
			if (!bp.isEmpty() && bp.getItem() instanceof BackpackItem && isContainerAccessible(bp, playerId)) {
				GridContainer bpGrid = ContainerItemHelper.readBackpackGrid(bp);
				boolean modified = false;
				List<GridEntry> entries = new ArrayList<>(bpGrid.getEntries());
				for (GridEntry entry : entries) {
					if (remaining <= 0) break;
					ItemStack st = entry.getStack();
					if (matchesConsumptionTarget(st, targetItem, targetFmjAmmoId)) {
						int take = Math.min(remaining, st.getCount());
						st.decrement(take);
						remaining -= take;
						if (st.isEmpty()) {
							bpGrid.remove(entry.getEntryId());
						}
						modified = true;
					}
				}
				if (modified) {
					ContainerItemHelper.writeBackpackGrid(bp, bpGrid);
				}
			}
		}
	}

	private static boolean matchesConsumptionTarget(
		ItemStack stack,
		@Nullable Item targetItem,
		@Nullable String targetFmjAmmoId
	) {
		if (stack == null || stack.isEmpty()) {
			return false;
		}
		if (targetFmjAmmoId != null) {
			return AmmoTier.isTaczAmmoStack(stack)
				&& targetFmjAmmoId.equals(AmmoTier.getAmmoId(stack))
				&& AmmoTier.getFromAmmoStack(stack) == AmmoTier.FMJ;
		}
		return targetItem != null && !AmmoTier.isTaczAmmoStack(stack) && stack.getItem() == targetItem;
	}

	private static boolean isContainerAccessible(ItemStack container, @Nullable UUID playerId) {
		return !ContainerItemHelper.requiresSearch(container)
			|| (playerId != null && ContainerItemHelper.getSearchMemory(container).isGridRevealed(playerId));
	}

	public static boolean craftAmmo(ServerPlayerEntity player, String ammoId, AmmoTier tier, boolean upgradeFromFmj) {
		if (!TaCZCaliberCatalog.isBallisticAmmoId(ammoId) || tier == null) {
			return false;
		}
		TarkovWorkbenchRecipes.AmmoRecipe recipe = TarkovWorkbenchRecipes.getAmmoRecipe(ammoId, tier, upgradeFromFmj);
		ItemStack result = AmmoTier.createTaczAmmoStack(recipe.ammoId(), recipe.tier(), recipe.batchSize());

		if (!executeCraftTransaction(player, recipe.materials(), recipe.upgradeFromFmj() ? recipe.ammoId() : null, recipe.requiredFmjCount(), result)) {
			return false;
		}

		player.getWorld().playSound(null, player.getX(), player.getY(), player.getZ(),
			SoundEvents.BLOCK_SMITHING_TABLE_USE, SoundCategory.PLAYERS, 0.7f, 1.15f);
		return true;
	}

	public static boolean craftGear(ServerPlayerEntity player, Item gearItem) {
		if (!TarkovWorkbenchRecipes.isCraftableGear(gearItem)) {
			return false;
		}
		List<TarkovWorkbenchRecipes.Ingredient> mats = TarkovWorkbenchRecipes.computeGearIngredients(gearItem);
		ItemStack result = new ItemStack(gearItem);

		if (!executeCraftTransaction(player, mats, null, 0, result)) {
			return false;
		}

		player.getWorld().playSound(null, player.getX(), player.getY(), player.getZ(),
			SoundEvents.BLOCK_ANVIL_USE, SoundCategory.PLAYERS, 0.6f, 1.2f);
		return true;
	}

	public static boolean giveOrDrop(ServerPlayerEntity player, PlayerLoadout loadout, ItemStack crafted) {
		ItemStack toInsert = crafted.copy();
		loadout.tryInsertItem(toInsert, player.getUuid());
		boolean droppedAny = !toInsert.isEmpty();
		if (droppedAny) {
			player.dropItem(toInsert, false);
		}
		return droppedAny;
	}

	private static boolean executeCraftTransaction(
		ServerPlayerEntity player,
		List<TarkovWorkbenchRecipes.Ingredient> materials,
		@Nullable String requiredFmjAmmoId,
		int requiredFmjCount,
		ItemStack result
	) {
		if (player.isCreative()) {
			ItemStack creativeCopy = result.copy();
			boolean inserted = player.getInventory().insertStack(creativeCopy) && creativeCopy.isEmpty();
			if (!inserted && !creativeCopy.isEmpty()) {
				player.dropItem(creativeCopy, false);
			}
			if (inserted) {
				player.sendMessage(Text.translatable("message.tarkovcraft.workbench_crafted", result.getName()), true);
			} else {
				Text warn = Text.translatable("message.tarkovcraft.wb_crafted_dropped", result.getName())
					.formatted(Formatting.YELLOW);
				player.sendMessage(warn, false);
				player.sendMessage(warn, true);
			}
			return true;
		}

		if (!(player instanceof PlayerLoadoutHolder holder)) {
			return false;
		}
		PlayerLoadout loadout = holder.tarkovcraft$getLoadout();
		if (!consumeMaterials(loadout, player.getUuid(), materials, requiredFmjAmmoId, requiredFmjCount)) {
			player.sendMessage(Text.translatable("message.tarkovcraft.workbench_no_materials"), true);
			return false;
		}

		boolean droppedAny = giveOrDrop(player, loadout, result);
		holder.tarkovcraft$syncLoadout();

		if (!droppedAny) {
			player.sendMessage(Text.translatable("message.tarkovcraft.workbench_crafted", result.getName()), true);
		} else {
			Text warn = Text.translatable("message.tarkovcraft.wb_crafted_dropped", result.getName())
				.formatted(Formatting.YELLOW);
			player.sendMessage(warn, false);
			player.sendMessage(warn, true);
		}
		return true;
	}
}

package com.tarkovcraft.worldgen;

import com.tarkovcraft.TarkovCraftMod;
import net.fabricmc.fabric.api.biome.v1.BiomeModifications;
import net.fabricmc.fabric.api.biome.v1.BiomeSelectors;
import net.minecraft.registry.Registries;
import net.minecraft.registry.Registry;
import net.minecraft.registry.RegistryKey;
import net.minecraft.registry.RegistryKeys;
import net.minecraft.registry.tag.BiomeTags;
import net.minecraft.util.Identifier;
import net.minecraft.world.gen.GenerationStep;
import net.minecraft.world.gen.feature.DefaultFeatureConfig;
import net.minecraft.world.gen.feature.Feature;

/**
 * Registers Overworld surface loot structures (roadblock, med_tent, military_camp, forest_stash).
 */
public final class ModWorldGen {
	public static final Feature<DefaultFeatureConfig> ROADBLOCK = registerFeature("roadblock",
		new TarkovStructureFeature(DefaultFeatureConfig.CODEC, TarkovStructureFeature.StructureKind.ROADBLOCK));
	public static final Feature<DefaultFeatureConfig> MED_TENT = registerFeature("med_tent",
		new TarkovStructureFeature(DefaultFeatureConfig.CODEC, TarkovStructureFeature.StructureKind.MED_TENT));
	public static final Feature<DefaultFeatureConfig> MILITARY_CAMP = registerFeature("military_camp",
		new TarkovStructureFeature(DefaultFeatureConfig.CODEC, TarkovStructureFeature.StructureKind.MILITARY_CAMP));
	public static final Feature<DefaultFeatureConfig> FOREST_STASH = registerFeature("forest_stash",
		new TarkovStructureFeature(DefaultFeatureConfig.CODEC, TarkovStructureFeature.StructureKind.FOREST_STASH));

	private static boolean initialized = false;

	private ModWorldGen() {
	}

	private static Feature<DefaultFeatureConfig> registerFeature(String name, Feature<DefaultFeatureConfig> feature) {
		return Registry.register(Registries.FEATURE, new Identifier(TarkovCraftMod.MOD_ID, name), feature);
	}

	public static synchronized void init() {
		if (initialized) {
			return;
		}
		initialized = true;
		var nonWaterOverworld = BiomeSelectors.foundInOverworld()
			.and(BiomeSelectors.tag(BiomeTags.IS_OCEAN).negate())
			.and(BiomeSelectors.tag(BiomeTags.IS_DEEP_OCEAN).negate())
			.and(BiomeSelectors.tag(BiomeTags.IS_RIVER).negate())
			.and(BiomeSelectors.tag(BiomeTags.IS_BEACH).negate());

		addStructure("roadblock", nonWaterOverworld);
		addStructure("med_tent", nonWaterOverworld);
		addStructure("military_camp", nonWaterOverworld);
		addStructure("forest_stash", BiomeSelectors.tag(BiomeTags.IS_FOREST).or(BiomeSelectors.tag(BiomeTags.IS_TAIGA)));
	}

	private static void addStructure(String name, java.util.function.Predicate<net.fabricmc.fabric.api.biome.v1.BiomeSelectionContext> selector) {
		BiomeModifications.addFeature(
			selector,
			GenerationStep.Feature.SURFACE_STRUCTURES,
			RegistryKey.of(RegistryKeys.PLACED_FEATURE, new Identifier(TarkovCraftMod.MOD_ID, name))
		);
	}
}

package com.tarkovcraft.worldgen;

import com.mojang.serialization.Codec;
import com.tarkovcraft.block.SearchableLootBlock;
import com.tarkovcraft.block.SearchableLootBlockEntity;
import com.tarkovcraft.config.TarkovConfig;
import com.tarkovcraft.loot.TarkovLootPool;
import com.tarkovcraft.registry.ModBlocks;
import net.minecraft.block.Block;
import net.minecraft.block.BlockState;
import net.minecraft.block.Blocks;
import net.minecraft.registry.tag.BlockTags;
import net.minecraft.util.math.BlockPos;
import net.minecraft.util.math.random.Random;
import net.minecraft.world.Heightmap;
import net.minecraft.world.StructureWorldAccess;
import net.minecraft.world.gen.feature.DefaultFeatureConfig;
import net.minecraft.world.gen.feature.Feature;
import net.minecraft.world.gen.feature.util.FeatureContext;

/**
 * Overworld surface structures for TarkovCraft:
 * ROADBLOCK, MED_TENT, MILITARY_CAMP, FOREST_STASH.
 */
public class TarkovStructureFeature extends Feature<DefaultFeatureConfig> {
	public enum StructureKind {
		ROADBLOCK(2, 2),
		MED_TENT(2, 2),
		MILITARY_CAMP(3, 2),
		FOREST_STASH(1, 1);

		private final int halfW;
		private final int halfD;

		StructureKind(int halfW, int halfD) {
			this.halfW = halfW;
			this.halfD = halfD;
		}
	}

	private final StructureKind kind;

	public TarkovStructureFeature(Codec<DefaultFeatureConfig> codec, StructureKind kind) {
		super(codec);
		this.kind = kind;
	}

	@Override
	public boolean generate(FeatureContext<DefaultFeatureConfig> context) {
		if (!TarkovConfig.get().worldgenStructuresEnabled) {
			return false;
		}
		StructureWorldAccess world = context.getWorld();
		Random random = context.getRandom();
		BlockPos origin = context.getOrigin();

		if (!isFootprintSafe(world, origin, kind.halfW, kind.halfD)) {
			return false;
		}

		switch (kind) {
			case ROADBLOCK -> buildRoadblock(world, origin, random);
			case MED_TENT -> buildMedTent(world, origin, random);
			case MILITARY_CAMP -> buildMilitaryCamp(world, origin, random);
			case FOREST_STASH -> buildForestStash(world, origin, random);
		}
		try {
			net.minecraft.server.world.ServerWorld sw = world.toServerWorld();
			if (sw != null) {
				TarkovStructureRegistry.get(sw).record(kind, origin);
			}
		} catch (Exception ignored) {
		}
		return true;
	}

	private static boolean isFootprintSafe(StructureWorldAccess world, BlockPos origin, int halfW, int halfD) {
		int ox = origin.getX();
		int oy = origin.getY();
		int oz = origin.getZ();

		for (int dx = -halfW; dx <= halfW; dx++) {
			for (int dz = -halfD; dz <= halfD; dz++) {
				int x = ox + dx;
				int z = oz + dz;
				int topY = world.getTopY(Heightmap.Type.MOTION_BLOCKING_NO_LEAVES, x, z);
				if (Math.abs(topY - oy) > 2) {
					return false;
				}
				BlockPos groundPos = new BlockPos(x, topY - 1, z);
				BlockState ground = world.getBlockState(groundPos);
				if (!ground.isOpaqueFullCube(world, groundPos)
					|| !ground.getFluidState().isEmpty()
					|| ground.isIn(BlockTags.LOGS)
					|| ground.isIn(BlockTags.LEAVES)
					|| ground.isOf(Blocks.ICE)
					|| ground.isOf(Blocks.PACKED_ICE)
					|| ground.isOf(Blocks.BLUE_ICE)) {
					return false;
				}
				for (int dy = 0; dy <= 3; dy++) {
					BlockPos above = new BlockPos(x, topY + dy, z);
					if (!world.getBlockState(above).isReplaceable()) {
						return false;
					}
				}
			}
		}
		return true;
	}

	private static void buildRoadblock(StructureWorldAccess world, BlockPos origin, Random random) {
		int y = origin.getY();
		int ox = origin.getX();
		int oz = origin.getZ();

		for (int dx = -2; dx <= 2; dx++) {
			for (int dz = -2; dz <= 2; dz++) {
				setFoundation(world, ox + dx, y - 1, oz + dz, Blocks.COBBLESTONE.getDefaultState());
			}
		}

		// Sandbag / cobblestone barricade on north and south edges
		for (int dx = -2; dx <= 2; dx++) {
			if (dx != 0) {
				BlockState wall = (dx % 2 == 0 ? Blocks.COBBLESTONE_WALL : Blocks.MOSSY_COBBLESTONE_WALL).getDefaultState();
				setBlock(world, ox + dx, y, oz - 2, wall);
				setBlock(world, ox + dx, y, oz + 2, Blocks.SANDSTONE.getDefaultState());
			}
		}
		setBlock(world, ox - 2, y, oz - 1, Blocks.HAY_BLOCK.getDefaultState());
		setBlock(world, ox - 2, y, oz, Blocks.IRON_BARS.getDefaultState());
		setBlock(world, ox + 2, y, oz, Blocks.IRON_BARS.getDefaultState());
		setBlock(world, ox + 2, y, oz - 1, Blocks.OAK_FENCE.getDefaultState());
		setBlock(world, ox + 2, y + 1, oz - 1, Blocks.LANTERN.getDefaultState());

		placeCrate(world, new BlockPos(ox - 1, y, oz), ModBlocks.WEAPON_CRATE, TarkovLootPool.MILITARY, random);
		if (random.nextBoolean()) {
			placeCrate(world, new BlockPos(ox + 1, y, oz + 1), ModBlocks.WEAPON_CRATE, TarkovLootPool.MILITARY, random);
		}
	}

	private static void buildMedTent(StructureWorldAccess world, BlockPos origin, Random random) {
		int y = origin.getY();
		int ox = origin.getX();
		int oz = origin.getZ();

		for (int dx = -2; dx <= 2; dx++) {
			for (int dz = -2; dz <= 2; dz++) {
				setFoundation(world, ox + dx, y - 1, oz + dz, Blocks.COARSE_DIRT.getDefaultState());
			}
		}

		// Corner fence posts (height 3)
		int[][] corners = {{-2, -2}, {2, -2}, {-2, 2}, {2, 2}};
		for (int[] c : corners) {
			for (int dy = 0; dy < 3; dy++) {
				setBlock(world, ox + c[0], y + dy, oz + c[1], Blocks.OAK_FENCE.getDefaultState());
			}
		}

		// White wool roof with red cross in center
		for (int dx = -2; dx <= 2; dx++) {
			for (int dz = -2; dz <= 2; dz++) {
				boolean isCross = (dx == 0 && Math.abs(dz) <= 1) || (dz == 0 && Math.abs(dx) <= 1);
				Block roofBlock = isCross ? Blocks.RED_WOOL : Blocks.WHITE_WOOL;
				setBlock(world, ox + dx, y + 3, oz + dz, roofBlock.getDefaultState());
			}
		}

		setBlock(world, ox, y + 2, oz, Blocks.LANTERN.getDefaultState());
		placeCrate(world, new BlockPos(ox - 1, y, oz), ModBlocks.MEDBAG, TarkovLootPool.MEDICAL, random);
		if (random.nextFloat() < 0.65f) {
			placeCrate(world, new BlockPos(ox + 1, y, oz + 1), ModBlocks.MEDBAG, TarkovLootPool.MEDICAL, random);
		}
	}

	private static void buildMilitaryCamp(StructureWorldAccess world, BlockPos origin, Random random) {
		int y = origin.getY();
		int ox = origin.getX();
		int oz = origin.getZ();

		for (int dx = -3; dx <= 3; dx++) {
			for (int dz = -2; dz <= 2; dz++) {
				setFoundation(world, ox + dx, y - 1, oz + dz, Blocks.COARSE_DIRT.getDefaultState());
			}
		}

		// Campfire in center
		setBlock(world, ox, y, oz, Blocks.CAMPFIRE.getDefaultState());
		setBlock(world, ox - 1, y, oz + 1, Blocks.SPRUCE_STAIRS.getDefaultState());

		// Small green wool tent on left side
		for (int dy = 0; dy < 2; dy++) {
			setBlock(world, ox - 3, y + dy, oz - 2, Blocks.SPRUCE_FENCE.getDefaultState());
			setBlock(world, ox - 3, y + dy, oz + 2, Blocks.SPRUCE_FENCE.getDefaultState());
			setBlock(world, ox - 1, y + dy, oz - 2, Blocks.SPRUCE_FENCE.getDefaultState());
		}
		for (int dx = -3; dx <= -1; dx++) {
			for (int dz = -2; dz <= 2; dz++) {
				setBlock(world, ox + dx, y + 2, oz + dz, Blocks.GREEN_WOOL.getDefaultState());
			}
		}

		setBlock(world, ox + 2, y, oz - 1, Blocks.BARREL.getDefaultState());
		setBlock(world, ox + 3, y, oz + 1, Blocks.BARREL.getDefaultState());

		placeCrate(world, new BlockPos(ox - 2, y, oz), ModBlocks.WEAPON_CRATE, TarkovLootPool.MILITARY, random);
		placeCrate(world, new BlockPos(ox + 2, y, oz + 1), ModBlocks.TECHNICAL_CRATE, TarkovLootPool.CIVILIAN, random);
	}

	private static void buildForestStash(StructureWorldAccess world, BlockPos origin, Random random) {
		int y = origin.getY();
		int ox = origin.getX();
		int oz = origin.getZ();

		for (int dx = -1; dx <= 1; dx++) {
			for (int dz = -1; dz <= 1; dz++) {
				setFoundation(world, ox + dx, y - 1, oz + dz, Blocks.MOSSY_COBBLESTONE.getDefaultState());
			}
		}

		setBlock(world, ox - 1, y, oz - 1, Blocks.OAK_LOG.getDefaultState());
		setBlock(world, ox + 1, y, oz - 1, Blocks.MOSSY_COBBLESTONE.getDefaultState());
		setBlock(world, ox - 1, y, oz + 1, Blocks.FERN.getDefaultState());

		boolean isMilitary = random.nextFloat() < 0.30f;
		SearchableLootBlock crate = isMilitary ? ModBlocks.WEAPON_CRATE : ModBlocks.TECHNICAL_CRATE;
		TarkovLootPool pool = isMilitary ? TarkovLootPool.MILITARY : TarkovLootPool.CIVILIAN;
		placeCrate(world, new BlockPos(ox, y, oz), crate, pool, random);
	}

	private static void setFoundation(StructureWorldAccess world, int x, int startY, int z, BlockState state) {
		for (int dy = 0; dy < 3; dy++) {
			BlockPos pos = new BlockPos(x, startY - dy, z);
			if (dy == 0 || world.getBlockState(pos).isReplaceable()) {
				world.setBlockState(pos, state, Block.NOTIFY_LISTENERS);
			} else {
				break;
			}
		}
	}

	private static void setBlock(StructureWorldAccess world, int x, int y, int z, BlockState state) {
		world.setBlockState(new BlockPos(x, y, z), state, Block.NOTIFY_LISTENERS);
	}

	private static void placeCrate(
		StructureWorldAccess world,
		BlockPos cratePos,
		SearchableLootBlock crateBlock,
		TarkovLootPool pool,
		Random random
	) {
		world.setBlockState(cratePos, crateBlock.getDefaultState(), Block.NOTIFY_LISTENERS);
		if (world.getBlockEntity(cratePos) instanceof SearchableLootBlockEntity be) {
			be.populateFromPool(pool, random);
		}
	}
}

package com.tarkovcraft.worldgen;

import net.minecraft.nbt.NbtCompound;
import net.minecraft.nbt.NbtElement;
import net.minecraft.nbt.NbtList;
import net.minecraft.server.world.ServerWorld;
import net.minecraft.util.math.BlockPos;
import net.minecraft.world.PersistentState;
import org.jetbrains.annotations.Nullable;

import java.util.ArrayList;
import java.util.EnumMap;
import java.util.List;
import java.util.Map;

/**
 * Persistent world registry of generated TarkovCraft structures so /tarkov locate can find them.
 */
public final class TarkovStructureRegistry extends PersistentState {
	private static final String STATE_KEY = "tarkovcraft_structures";
	private static final int MAX_PER_KIND = 512;

	public record FoundStructure(TarkovStructureFeature.StructureKind kind, BlockPos pos) {
	}

	private final Map<TarkovStructureFeature.StructureKind, List<BlockPos>> positions =
		new EnumMap<>(TarkovStructureFeature.StructureKind.class);

	public TarkovStructureRegistry() {
		for (TarkovStructureFeature.StructureKind kind : TarkovStructureFeature.StructureKind.values()) {
			positions.put(kind, new ArrayList<>());
		}
	}

	public static TarkovStructureRegistry get(ServerWorld world) {
		return world.getPersistentStateManager().getOrCreate(
			TarkovStructureRegistry::fromNbt,
			TarkovStructureRegistry::new,
			STATE_KEY
		);
	}

	public synchronized void record(TarkovStructureFeature.StructureKind kind, BlockPos pos) {
		if (kind == null || pos == null) {
			return;
		}
		List<BlockPos> list = positions.get(kind);
		BlockPos imm = pos.toImmutable();
		if (list.contains(imm)) {
			return;
		}
		if (list.size() >= MAX_PER_KIND) {
			list.remove(0);
		}
		list.add(imm);
		markDirty();
	}

	@Nullable
	public synchronized FoundStructure findNearest(
		@Nullable TarkovStructureFeature.StructureKind filterKind,
		BlockPos origin
	) {
		FoundStructure best = null;
		double bestDistSq = Double.MAX_VALUE;
		for (TarkovStructureFeature.StructureKind kind : TarkovStructureFeature.StructureKind.values()) {
			if (filterKind != null && kind != filterKind) {
				continue;
			}
			for (BlockPos pos : positions.get(kind)) {
				double distSq = pos.getSquaredDistance(origin);
				if (distSq < bestDistSq) {
					bestDistSq = distSq;
					best = new FoundStructure(kind, pos);
				}
			}
		}
		return best;
	}

	@Override
	public synchronized NbtCompound writeNbt(NbtCompound nbt) {
		for (TarkovStructureFeature.StructureKind kind : TarkovStructureFeature.StructureKind.values()) {
			NbtList list = new NbtList();
			for (BlockPos pos : positions.get(kind)) {
				NbtCompound entry = new NbtCompound();
				entry.putInt("x", pos.getX());
				entry.putInt("y", pos.getY());
				entry.putInt("z", pos.getZ());
				list.add(entry);
			}
			nbt.put(kind.name(), list);
		}
		return nbt;
	}

	public static TarkovStructureRegistry fromNbt(NbtCompound nbt) {
		TarkovStructureRegistry reg = new TarkovStructureRegistry();
		for (TarkovStructureFeature.StructureKind kind : TarkovStructureFeature.StructureKind.values()) {
			if (!nbt.contains(kind.name(), NbtElement.LIST_TYPE)) {
				continue;
			}
			NbtList list = nbt.getList(kind.name(), NbtElement.COMPOUND_TYPE);
			List<BlockPos> target = reg.positions.get(kind);
			for (int i = 0; i < list.size() && target.size() < MAX_PER_KIND; i++) {
				NbtCompound c = list.getCompound(i);
				target.add(new BlockPos(c.getInt("x"), c.getInt("y"), c.getInt("z")));
			}
		}
		return reg;
	}
}
