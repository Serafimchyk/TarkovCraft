@Mixin(PlayerEntity.class)
public abstract class PlayerEntityMixin implements PlayerLoadoutHolder {
	@Unique
	private static final float TARKOV_DAMAGE_SCALE = 4.5f;

	@Unique
	private final PlayerLoadout tarkovcraft$loadout = new PlayerLoadout();

	@Unique
	private final BodyPartHealth tarkovcraft$bodyHealth = new BodyPartHealth();

	@Override
	public PlayerLoadout tarkovcraft$getLoadout() {
		return tarkovcraft$loadout;
	}

	@Override
	public void tarkovcraft$syncLoadout() {
		PlayerEntity self = (PlayerEntity) (Object) this;
		if (self instanceof ServerPlayerEntity serverPlayer) {
			TarkovNetworking.sendLoadoutSync(serverPlayer, tarkovcraft$loadout);
		}
	}

	@Override
	public BodyPartHealth tarkovcraft$getBodyHealth() {
		return tarkovcraft$bodyHealth;
	}

	@Override
	public void tarkovcraft$syncBodyHealth() {
		PlayerEntity self = (PlayerEntity) (Object) this;
		if (self instanceof ServerPlayerEntity serverPlayer) {
			TarkovNetworking.sendHealthSync(serverPlayer, tarkovcraft$bodyHealth);
		}
	}

	@Inject(method = "writeCustomDataToNbt", at = @At("TAIL"))
	private void tarkovcraft$writeNbt(NbtCompound nbt, CallbackInfo ci) {
		nbt.put("TarkovCraftLoadout", tarkovcraft$loadout.writeNbt());
		nbt.put("TarkovCraftHealth", tarkovcraft$bodyHealth.writeNbt());
	}

	@Inject(method = "readCustomDataFromNbt", at = @At("TAIL"))
	private void tarkovcraft$readNbt(NbtCompound nbt, CallbackInfo ci) {
		if (nbt.contains("TarkovCraftLoadout", NbtElement.COMPOUND_TYPE)) {
			tarkovcraft$loadout.readNbt(nbt.getCompound("TarkovCraftLoadout"));
		}
		if (nbt.contains("TarkovCraftHealth", NbtElement.COMPOUND_TYPE)) {
			tarkovcraft$bodyHealth.readNbt(nbt.getCompound("TarkovCraftHealth"));
			tarkovcraft$bodyHealth.clearActiveMedUse();
		}
	}

	/**
	 * Disables vanilla armor slots in Survival/Adventure mode (dispensers, equip calls, attribute reads).
	 */
	@Inject(method = "canEquip(Lnet/minecraft/item/ItemStack;)Z", at = @At("HEAD"), cancellable = true)
	private void tarkovcraft$blockVanillaCanEquip(ItemStack stack, CallbackInfoReturnable<Boolean> cir) {
		PlayerEntity self = (PlayerEntity) (Object) this;
		if (!self.isCreative() && !self.isSpectator()
			&& MobEntity.getPreferredEquipmentSlot(stack).getType() == EquipmentSlot.Type.ARMOR) {
			cir.setReturnValue(false);
		}
	}

	@Inject(method = "equipStack(Lnet/minecraft/entity/EquipmentSlot;Lnet/minecraft/item/ItemStack;)V", at = @At("HEAD"), cancellable = true)
	private void tarkovcraft$blockVanillaEquipStack(EquipmentSlot slot, ItemStack stack, CallbackInfo ci) {
		PlayerEntity self = (PlayerEntity) (Object) this;
		if (!self.isCreative() && !self.isSpectator()
			&& slot.getType() == EquipmentSlot.Type.ARMOR && !stack.isEmpty()) {
			ci.cancel();
		}
	}

	@Inject(method = "getEquippedStack(Lnet/minecraft/entity/EquipmentSlot;)Lnet/minecraft/item/ItemStack;", at = @At("HEAD"), cancellable = true)
	private void tarkovcraft$hideVanillaArmorStack(EquipmentSlot slot, CallbackInfoReturnable<ItemStack> cir) {
		PlayerEntity self = (PlayerEntity) (Object) this;
		if (!self.isCreative() && !self.isSpectator() && slot.getType() == EquipmentSlot.Type.ARMOR) {
			cir.setReturnValue(ItemStack.EMPTY);
		}
	}

	/**
	 * Routes all Survival/Adventure damage through TarkovCraft's 3D HitZoneResolver,
	 * zonal armor check, unprotected head-eyes oneshot, and limb-to-thorax overflow!
	 */
	@Inject(method = "applyDamage", at = @At("HEAD"), cancellable = true)
	private void tarkovcraft$applyZonalBodyDamage(DamageSource source, float amount, CallbackInfo ci) {
		PlayerEntity self = (PlayerEntity) (Object) this;
		if (self.getWorld().isClient || self.isCreative() || self.isSpectator() || self.isInvulnerableTo(source)) {
			return;
		}
		if (amount <= 0.0f) {
			return;
		}

		String msgId = source.getName();
		// Allow /kill and void death to kill immediately while still recording a clean KIA report
		if ("outOfWorld".equals(msgId) || "genericKill".equals(msgId)) {
			tarkovcraft$bodyHealth.applyHit(tarkovcraft$loadout, HitZone.THORAX, 9999.0f, 100, 0.0f, true, false, 1.0f, msgId, "", "", 0.0f);
			tarkovcraft$syncBodyHealth();
			return;
		}

		ci.cancel();

		StatusEffectInstance res = self.getStatusEffect(StatusEffects.RESISTANCE);
		if (res != null && !source.isIn(DamageTypeTags.BYPASSES_RESISTANCE)) {
			int reduction = Math.min(25, (res.getAmplifier() + 1) * 5);
			float mult = Math.max(0.0f, (25 - reduction) / 25.0f);
			amount *= mult;
			if (amount <= 0.001f) {
				return;
			}
		}

		if (source.isIn(DamageTypeTags.IS_FALL)) {
			float weightKg = tarkovcraft$loadout.calculateWeightKg();
			boolean killed = tarkovcraft$bodyHealth.applyFallDamage(amount, weightKg, self.getRandom().nextFloat());
			if (tarkovcraft$bodyHealth.hasFracture(BodyPart.LEFT_LEG) || tarkovcraft$bodyHealth.hasFracture(BodyPart.RIGHT_LEG)) {
				self.getWorld().playSound(null, self.getX(), self.getY(), self.getZ(),
					SoundEvents.BLOCK_BONE_BLOCK_BREAK, SoundCategory.PLAYERS, 0.9f, 0.85f);
			}
			self.getWorld().playSound(null, self.getX(), self.getY(), self.getZ(),
				SoundEvents.ENTITY_PLAYER_HURT, SoundCategory.PLAYERS, 0.85f, 0.95f);
			tarkovcraft$syncBodyHealth();
			if (killed) {
				self.setHealth(0.0f);
				self.onDeath(source);
			} else {
				self.setHealth(Math.max(1.0f, tarkovcraft$bodyHealth.getTotalRatio() * self.getMaxHealth()));
			}
			return;
		}

		HitZone zone = HitZoneResolver.resolveFromDamage(self, source);
		TaCZBulletHelper.BulletInfo bulletInfo = TaCZBulletHelper.inspectDamageSource(source);

		boolean envBypass = source.isIn(DamageTypeTags.IS_FALL)
			|| source.isIn(DamageTypeTags.IS_DROWNING)
			|| source.isIn(DamageTypeTags.IS_FIRE)
			|| source.isIn(DamageTypeTags.IS_FREEZING);
		// TaCZ BULLET_IGNORE_ARMOR sets BYPASSES_ARMOR; force ZonalArmorHelper to evaluate ALL TaCZ bullets!
		boolean bypassesArmor = envBypass || (!bulletInfo.isTaczBullet() && source.isIn(DamageTypeTags.BYPASSES_ARMOR));
		boolean canRicochet = !bypassesArmor && (bulletInfo.isTaczBullet() || source.isIn(DamageTypeTags.IS_PROJECTILE));

		Entity attacker = source.getAttacker();
		String killerName = attacker != null ? attacker.getDisplayName().getString() : msgId;
		float distance = attacker != null ? self.distanceTo(attacker) : 0.0f;

		float tarkovDamage = amount * TARKOV_DAMAGE_SCALE * bulletInfo.damageMultiplier();
		float absorptionVanilla = self.getAbsorptionAmount();
		if (absorptionVanilla > 0.0f && tarkovDamage > 0.0f && !source.isIn(DamageTypeTags.BYPASSES_INVULNERABILITY)) {
			float absorptionTarkov = absorptionVanilla * TARKOV_DAMAGE_SCALE;
			float absorbedTarkov = Math.min(tarkovDamage, absorptionTarkov);
			tarkovDamage -= absorbedTarkov;
			self.setAbsorptionAmount(Math.max(0.0f, absorptionVanilla - absorbedTarkov / TARKOV_DAMAGE_SCALE));
			if (tarkovDamage <= 0.001f) {
				self.getWorld().playSound(null, self.getX(), self.getY(), self.getZ(),
					SoundEvents.ITEM_SHIELD_BLOCK, SoundCategory.PLAYERS, 0.5f, 1.35f);
				return;
			}
		}

		boolean killed = tarkovcraft$bodyHealth.applyHit(
			tarkovcraft$loadout,
			zone,
			tarkovDamage,
			bulletInfo.penetrationPower(),
			bulletInfo.armorWearMultiplier(),
			bypassesArmor,
			canRicochet,
			self.getRandom().nextFloat(),
			self.getRandom().nextFloat(),
			self.getRandom().nextFloat(),
			killerName,
			bulletInfo.weaponName(),
			bulletInfo.ammoName(),
			distance
		);

		List<DamageReport> hits = tarkovcraft$bodyHealth.getRecentHits();
		if (!hits.isEmpty()) {
			DamageReport latest = hits.get(0);
			if (latest.ricocheted()) {
				self.getWorld().playSound(null, self.getX(), self.getY(), self.getZ(),
					SoundEvents.BLOCK_ANVIL_LAND, SoundCategory.PLAYERS, 0.35f, 1.9f);
			} else if (latest.armorBlocked()) {
				self.getWorld().playSound(null, self.getX(), self.getY(), self.getZ(),
					SoundEvents.ITEM_SHIELD_BLOCK, SoundCategory.PLAYERS, 0.65f, 1.15f);
			}
		}

		tarkovcraft$syncBodyHealth();
		tarkovcraft$syncLoadout(); // Sync armor durability changes

		if (killed) {
			self.setHealth(0.0f);
			self.onDeath(source);
		} else {
			// Keep vanilla health proportional (>= 1.0f so vanilla never prematurely kills on limb black-out)
			float syncedVanillaHp = Math.max(1.0f, tarkovcraft$bodyHealth.getTotalRatio() * self.getMaxHealth());
			self.setHealth(syncedVanillaHp);
		}
	}

	@Inject(method = "dropInventory", at = @At("HEAD"), cancellable = true)
	private void tarkovcraft$spawnCorpseInsteadOfDropping(CallbackInfo ci) {
		PlayerEntity self = (PlayerEntity) (Object) this;
		if (self.getWorld().isClient) {
			return;
		}
		if (self.getWorld().getGameRules().getBoolean(GameRules.KEEP_INVENTORY)) {
			ci.cancel();
			return;
		}

		List<ItemStack> extras = new ArrayList<>();
		for (ItemStack st : self.getInventory().offHand) {
			if (!st.isEmpty()) {
				extras.add(st.copy());
			}
		}
		for (ItemStack st : self.getInventory().main) {
			if (!st.isEmpty()) {
				extras.add(st.copy());
			}
		}

		if (!tarkovcraft$loadout.isLootEmpty() || !extras.isEmpty()) {
			List<ItemStack> overflow = new ArrayList<>();
			CorpseEntity corpse = CorpseEntity.fromPlayer(self.getWorld(), self, tarkovcraft$loadout, extras, overflow);
			self.getWorld().spawnEntity(corpse);
			for (ItemStack st : overflow) {
				if (!st.isEmpty()) {
					self.dropItem(st, true, false);
				}
			}
			tarkovcraft$loadout.clearLoot();
			self.getInventory().clear();
			tarkovcraft$syncLoadout();
			ci.cancel();
		}
	}
}

package com.tarkovcraft.mixin;

import com.tarkovcraft.ammo.AmmoTier;
import com.tarkovcraft.client.WorldInteractState;
import com.tarkovcraft.loadout.EquipmentSlotType;
import com.tarkovcraft.loadout.PlayerLoadout;
import com.tarkovcraft.loadout.PlayerLoadoutHolder;
import com.tarkovcraft.screen.TarkovScreenHandler;
import net.minecraft.entity.player.PlayerEntity;
import net.minecraft.entity.player.PlayerInventory;
import net.minecraft.item.ItemStack;
import org.spongepowered.asm.mixin.Final;
import org.spongepowered.asm.mixin.Mixin;
import org.spongepowered.asm.mixin.Shadow;
import org.spongepowered.asm.mixin.Unique;
import org.spongepowered.asm.mixin.injection.At;
import org.spongepowered.asm.mixin.injection.Inject;
import org.spongepowered.asm.mixin.injection.callback.CallbackInfo;
import org.spongepowered.asm.mixin.injection.callback.CallbackInfoReturnable;

/**
 * Bridges PlayerInventory (slots 0..35) to TarkovCraft's Loadout in Survival/Adventure mode:
 * - Slots 0..2: Primary 1, Primary 2, Pistol
 * - Slots 3..8: 6x (1x1) Pockets
 * - Slots 9..35:
 *     * In combat (no external workbench open): ONLY Tactical Rig items (NEVER Backpack!) ->
 *       TaCZ fast-reload pulls ammo/magazines strictly from Pockets and Tactical Rig!
 *     * While a workbench screen (e.g. TaCZ Gunsmith Table) is open: also exposes Backpack
 *       materials so crafting can use ingots/gunpowder stored in the Backpack!
 * - Whenever TaCZ consumes an ammo stack during reload, stamps its AmmoTier onto the held gun!
 */
@Mixin(PlayerInventory.class)
public abstract class PlayerInventoryMixin {
	@Shadow
	@Final
	public PlayerEntity player;

	@Shadow
	public int selectedSlot;

	@Unique
	private boolean isExternalWorkbenchOpen() {
		return player.currentScreenHandler != null
			&& player.currentScreenHandler != player.playerScreenHandler
			&& !(player.currentScreenHandler instanceof TarkovScreenHandler);
	}

	@Inject(method = "getMainHandStack", at = @At("HEAD"), cancellable = true)
	private void tarkovcraft$getMainHandStack(CallbackInfoReturnable<ItemStack> cir) {
		if (!player.isCreative() && player instanceof PlayerLoadoutHolder holder) {
			cir.setReturnValue(holder.tarkovcraft$getLoadout().getActiveHandStack(selectedSlot));
		}
	}

	@Inject(method = "getStack", at = @At("HEAD"), cancellable = true)
	private void tarkovcraft$getStack(int slot, CallbackInfoReturnable<ItemStack> cir) {
		if (!player.isCreative() && player instanceof PlayerLoadoutHolder holder && slot >= 0 && slot < 36) {
			PlayerLoadout loadout = holder.tarkovcraft$getLoadout();
			if (slot < 9) {
				cir.setReturnValue(loadout.getHotbarDisplayStack(slot));
			} else {
				long worldTick = player.getWorld() != null ? player.getWorld().getTime() : 0L;
				cir.setReturnValue(loadout.getExposedStorageStackCached(slot - 9, player.getUuid(), isExternalWorkbenchOpen(), worldTick));
			}
		}
	}

	@Inject(method = "removeStack(II)Lnet/minecraft/item/ItemStack;", at = @At("HEAD"), cancellable = true)
	private void tarkovcraft$removeStackAmount(int slot, int amount, CallbackInfoReturnable<ItemStack> cir) {
		if (!player.isCreative() && player instanceof PlayerLoadoutHolder holder && slot >= 0 && slot < 36) {
			PlayerLoadout loadout = holder.tarkovcraft$getLoadout();
			ItemStack extracted = ItemStack.EMPTY;
			if (slot == 0) {
				extracted = splitEquipment(loadout, EquipmentSlotType.PRIMARY_1, amount);
			} else if (slot == 1) {
				extracted = splitEquipment(loadout, EquipmentSlotType.PRIMARY_2, amount);
			} else if (slot == 2) {
				extracted = splitEquipment(loadout, EquipmentSlotType.PISTOL, amount);
			} else if (slot < 9) {
				ItemStack p = loadout.getPocket(slot - 3);
				if (!p.isEmpty()) {
					extracted = p.split(amount);
					if (p.isEmpty()) loadout.setPocket(slot - 3, ItemStack.EMPTY);
				}
			} else {
				extracted = loadout.removeExposedStorageStack(slot - 9, amount, player.getUuid(), isExternalWorkbenchOpen());
			}

			// If TaCZ just consumed ammo during a reload, stamp the AmmoTier onto the held gun!
			if (!extracted.isEmpty() && AmmoTier.isTaczAmmoStack(extracted)) {
				ItemStack activeGun = loadout.getActiveHandStack(selectedSlot);
				if (!activeGun.isEmpty() && AmmoTier.isTaczGunStack(activeGun)) {
					AmmoTier.stampLoadedTierOnGun(activeGun, AmmoTier.getFromAmmoStack(extracted));
				}
			}

			if (!extracted.isEmpty() && !player.getWorld().isClient) {
				holder.tarkovcraft$syncLoadout();
			}
			cir.setReturnValue(extracted);
		}
	}

	@Inject(method = "removeStack(I)Lnet/minecraft/item/ItemStack;", at = @At("HEAD"), cancellable = true)
	private void tarkovcraft$removeStackFull(int slot, CallbackInfoReturnable<ItemStack> cir) {
		if (!player.isCreative() && player instanceof PlayerLoadoutHolder holder && slot >= 0 && slot < 36) {
			PlayerLoadout loadout = holder.tarkovcraft$getLoadout();
			ItemStack extracted = ItemStack.EMPTY;
			if (slot == 0) {
				extracted = splitEquipment(loadout, EquipmentSlotType.PRIMARY_1, Integer.MAX_VALUE);
			} else if (slot == 1) {
				extracted = splitEquipment(loadout, EquipmentSlotType.PRIMARY_2, Integer.MAX_VALUE);
			} else if (slot == 2) {
				extracted = splitEquipment(loadout, EquipmentSlotType.PISTOL, Integer.MAX_VALUE);
			} else if (slot < 9) {
				ItemStack p = loadout.getPocket(slot - 3);
				if (!p.isEmpty()) {
					extracted = p.copy();
					loadout.setPocket(slot - 3, ItemStack.EMPTY);
				}
			} else {
				extracted = loadout.removeExposedStorageStack(slot - 9, Integer.MAX_VALUE, player.getUuid(), isExternalWorkbenchOpen());
			}

			if (!extracted.isEmpty() && AmmoTier.isTaczAmmoStack(extracted)) {
				ItemStack activeGun = loadout.getActiveHandStack(selectedSlot);
				if (!activeGun.isEmpty() && AmmoTier.isTaczGunStack(activeGun)) {
					AmmoTier.stampLoadedTierOnGun(activeGun, AmmoTier.getFromAmmoStack(extracted));
				}
			}

			if (!extracted.isEmpty() && !player.getWorld().isClient) {
				holder.tarkovcraft$syncLoadout();
			}
			cir.setReturnValue(extracted);
		}
	}

	@Inject(method = "setStack", at = @At("HEAD"), cancellable = true)
	private void tarkovcraft$setStack(int slot, ItemStack stack, CallbackInfo ci) {
		if (!player.isCreative() && player instanceof PlayerLoadoutHolder holder && slot >= 0 && slot < 36) {
			PlayerLoadout loadout = holder.tarkovcraft$getLoadout();
			ItemStack safe = stack == null ? ItemStack.EMPTY : stack;
			if (slot == 0) {
				loadout.setEquipment(EquipmentSlotType.PRIMARY_1, safe);
			} else if (slot == 1) {
				loadout.setEquipment(EquipmentSlotType.PRIMARY_2, safe);
			} else if (slot == 2) {
				loadout.setEquipment(EquipmentSlotType.PISTOL, safe);
			} else if (slot < 9) {
				loadout.setPocket(slot - 3, safe);
			} else {
				loadout.setExposedStorageStack(slot - 9, safe, player.getUuid(), isExternalWorkbenchOpen());
			}
			if (!player.getWorld().isClient) {
				holder.tarkovcraft$syncLoadout();
			}
			ci.cancel();
		}
	}

	@Inject(method = "insertStack(Lnet/minecraft/item/ItemStack;)Z", at = @At("HEAD"), cancellable = true)
	private void tarkovcraft$insertStack(ItemStack stack, CallbackInfoReturnable<Boolean> cir) {
		if (!player.isCreative() && player instanceof PlayerLoadoutHolder holder) {
			PlayerLoadout loadout = holder.tarkovcraft$getLoadout();
			boolean inserted = loadout.trySmartPickup(stack, player.getUuid());
			if (inserted) {
				holder.tarkovcraft$syncLoadout();
			}
			cir.setReturnValue(inserted);
		}
	}

	@Inject(method = "insertStack(ILnet/minecraft/item/ItemStack;)Z", at = @At("HEAD"), cancellable = true)
	private void tarkovcraft$insertStackAtSlot(int slot, ItemStack stack, CallbackInfoReturnable<Boolean> cir) {
		if (!player.isCreative() && player instanceof PlayerLoadoutHolder holder) {
			if (stack == null || stack.isEmpty()) {
				cir.setReturnValue(false);
				return;
			}
			PlayerLoadout loadout = holder.tarkovcraft$getLoadout();
			boolean inserted = false;
			if (slot >= 3 && slot < 9) {
				int pocketIdx = slot - 3;
				ItemStack existing = loadout.getPocket(pocketIdx);
				if (existing.isEmpty() && loadout.canPlaceInPocket(stack)) {
					loadout.setPocket(pocketIdx, stack.copy());
					stack.setCount(0);
					inserted = true;
				} else if (!existing.isEmpty() && existing.isStackable()
					&& ItemStack.canCombine(existing, stack) && existing.getCount() < existing.getMaxCount()) {
					int transfer = Math.min(existing.getMaxCount() - existing.getCount(), stack.getCount());
					existing.increment(transfer);
					stack.decrement(transfer);
					inserted = stack.isEmpty();
				}
			}
			if (!inserted && !stack.isEmpty()) {
				inserted = loadout.trySmartPickup(stack, player.getUuid());
			}
			if (inserted && !player.getWorld().isClient) {
				holder.tarkovcraft$syncLoadout();
			}
			cir.setReturnValue(inserted);
		}
	}

	@Inject(method = "scrollInHotbar", at = @At("HEAD"), cancellable = true)
	private void tarkovcraft$interceptScrollForWorldMenuOrClearMelee(double scrollAmount, CallbackInfo ci) {
		if (player.getWorld().isClient && WorldInteractState.hasMultiActionMenu()) {
			WorldInteractState.scrollAction(scrollAmount > 0 ? -1 : 1);
			ci.cancel();
			return;
		}
		if (!player.isCreative() && player instanceof PlayerLoadoutHolder holder) {
			PlayerLoadout loadout = holder.tarkovcraft$getLoadout();
			if (loadout.isMeleeSelected()) {
				loadout.setMeleeSelected(false);
			}
		}
	}

	private static ItemStack splitEquipment(PlayerLoadout loadout, EquipmentSlotType slot, int amount) {
		ItemStack cur = loadout.getEquipment(slot);
		if (cur.isEmpty()) return ItemStack.EMPTY;
		ItemStack out = cur.split(amount);
		if (cur.isEmpty()) {
			loadout.setEquipment(slot, ItemStack.EMPTY);
		}
		return out;
	}
}

package com.tarkovcraft.mixin;

@Mixin(PlayerManager.class)
abstract class PlayerManagerSoundMixin {
	@Inject(method = "sendToAround", at = @At("TAIL"))
	private void tarkovcraft$extendTacticalSounds(PlayerEntity source, double x, double y, double z, double distance,
			RegistryKey<World> world, Packet<?> packet, CallbackInfo callback) {
		boolean tactical = packet instanceof PlaySoundS2CPacket positioned
			&& HearingProfile.isQuietTactical(positioned.getSound().value().getId(), positioned.getCategory())
			|| packet instanceof PlaySoundFromEntityS2CPacket attached
			&& HearingProfile.isQuietTactical(attached.getSound().value().getId(), attached.getCategory());
		if (!tactical || distance <= 0) return;

		double baseSq = distance * distance;
		double boostedSq = baseSq * HearingProfile.ACTIVE_RANGE * HearingProfile.ACTIVE_RANGE;
		for (var player : ((PlayerManager) (Object) this).getPlayerList()) {
			if (player == source || player.getWorld().getRegistryKey() != world
				|| !(player instanceof PlayerLoadoutHolder holder)) continue;
			double distanceSq = player.squaredDistanceTo(x, y, z);
			if (distanceSq >= baseSq && distanceSq < boostedSq
				&& HearingProfile.forLoadout(holder.tarkovcraft$getLoadout()).range() == HearingProfile.ACTIVE_RANGE) {
				player.networkHandler.sendPacket(packet);
			}
		}
	}
}

@Mixin(SoundSystem.class)
abstract class SoundSystemMixin {
	@ModifyVariable(method = "play(Lnet/minecraft/client/sound/SoundInstance;)V", at = @At("STORE"), ordinal = 1)
	private float tarkovcraft$attenuation(float range, SoundInstance sound) {
		HearingProfile profile = tarkovcraft$profile(sound);
		return profile.muffled() || tarkovcraft$isTactical(sound) && !tarkovcraft$isLoud(sound)
			? range * profile.range() : range;
	}

	@ModifyVariable(method = "play(Lnet/minecraft/client/sound/SoundInstance;)V", at = @At("STORE"), ordinal = 2)
	private float tarkovcraft$initialVolume(float volume, SoundInstance sound) {
		return tarkovcraft$volume(volume, sound);
	}

	@Inject(method = "getAdjustedVolume(Lnet/minecraft/client/sound/SoundInstance;)F", at = @At("RETURN"), cancellable = true)
	private void tarkovcraft$tickingVolume(SoundInstance sound, CallbackInfoReturnable<Float> callback) {
		callback.setReturnValue(tarkovcraft$volume(callback.getReturnValueF(), sound));
	}

	@Inject(method = "getAdjustedPitch", at = @At("RETURN"), cancellable = true)
	private void tarkovcraft$pitch(SoundInstance sound, CallbackInfoReturnable<Float> callback) {
		callback.setReturnValue(callback.getReturnValueF() * tarkovcraft$profile(sound).pitch());
	}

	@Unique
	private static float tarkovcraft$volume(float volume, SoundInstance sound) {
		HearingProfile profile = tarkovcraft$profile(sound);
		if (profile == HearingProfile.DEFAULT) return volume;
		if (!profile.muffled()) {
			if (tarkovcraft$isLoud(sound)) return Math.min(volume, profile.loudClamp());
			if (!tarkovcraft$isTactical(sound)) return volume;
		}
		return Math.min(1.0f, volume * profile.ambientGain());
	}

	@Unique
	private static HearingProfile tarkovcraft$profile(SoundInstance sound) {
		if (MinecraftClient.getInstance().player == null || sound.getCategory() == SoundCategory.MUSIC
			|| sound.getCategory() == SoundCategory.RECORDS || sound.getCategory() == SoundCategory.VOICE
			|| sound.getCategory() == SoundCategory.MASTER && sound.isRelative()) return HearingProfile.DEFAULT;
		return HearingProfile.forLoadout(ClientLoadoutCache.getLoadout());
	}

	@Unique
	private static boolean tarkovcraft$isTactical(SoundInstance sound) {
		return HearingProfile.isQuietTactical(sound.getId(), sound.getCategory())
			|| sound.getSound() != null
				&& HearingProfile.isQuietTactical(sound.getSound().getIdentifier(), sound.getCategory());
	}

	@Unique
	private static boolean tarkovcraft$isLoud(SoundInstance sound) {
		return HearingProfile.isLoud(sound.getId().getPath())
			|| sound.getSound() != null && HearingProfile.isLoud(sound.getSound().getIdentifier().getPath());
	}
}

public final class AmmoSelectionNetworking {
	public static final Identifier SELECT = new Identifier(TarkovCraftMod.MOD_ID, "ammo_tier_select");
	public static final int MAX_GUN_ID_LENGTH = 128;

	private AmmoSelectionNetworking() {
	}

	public static void register() {
		ServerPlayNetworking.registerGlobalReceiver(SELECT, (server, player, handler, buf, sender) -> {
			int ordinal;
			String gunId;
			try {
				ordinal = buf.readUnsignedByte();
				gunId = buf.readString(MAX_GUN_ID_LENGTH);
			} catch (DecoderException | IndexOutOfBoundsException ignored) {
				return;
			}
			AmmoTier[] tiers = AmmoTier.values();
			if (ordinal >= tiers.length || buf.isReadable() || gunId.isEmpty() || Identifier.tryParse(gunId) == null) {
				return;
			}
			server.execute(() -> selectTier(player, gunId, tiers[ordinal]));
		});
	}

	private static void selectTier(ServerPlayerEntity player, String expectedGunId, AmmoTier tier) {
		if (!player.isAlive() || player.isSpectator() || player.currentScreenHandler != player.playerScreenHandler) {
			return;
		}
		ItemStack gun = player.getMainHandStack();
		if (!AmmoTier.isTaczGunStack(gun) || gun.getNbt() == null
				|| !expectedGunId.equals(gun.getNbt().getString("GunId"))) {
			return;
		}
		if (AmmoReloadTracker.prioritizeAmmoTier(player, AmmoReloadTracker.getGunAmmoId(gun), tier)) {
			player.sendMessage(Text.translatable("message.tarkovcraft.ammo_priority", Text.translatable(tier.getTranslationKey())), true);
		}
	}
}

import net.fabricmc.fabric.api.networking.v1.ServerPlayNetworking;
import net.minecraft.network.PacketByteBuf;
import net.minecraft.server.network.ServerPlayerEntity;

import java.util.IdentityHashMap;
import java.util.Map;

public final class EquipmentAppearanceSync {
	private static final Map<ServerPlayerEntity, EquipmentAppearance> LAST_SENT = new IdentityHashMap<>();

	private EquipmentAppearanceSync() {
	}

	public static void register() {
		// Players may already be tracked by the time the client's channel is registered.
		S2CPlayChannelEvents.REGISTER.register((handler, sender, server, channels) -> {
			if (!channels.contains(EquipmentAppearance.CHANNEL)) return;
			ServerPlayerEntity observer = handler.player;
			for (ServerPlayerEntity player : PlayerLookup.world(observer.getServerWorld())) {
				if (player != observer && PlayerLookup.tracking(player).contains(observer)) {
					sendCurrent(observer, player);
				}
			}
		});
		EntityTrackingEvents.START_TRACKING.register((entity, observer) -> {
			if (entity instanceof ServerPlayerEntity player) {
				sendCurrent(observer, player);
			}
		});
		EntityTrackingEvents.STOP_TRACKING.register((entity, observer) -> {
			if (entity instanceof ServerPlayerEntity player) {
				send(observer, player, EquipmentAppearance.EMPTY);
			}
		});
		ServerPlayConnectionEvents.DISCONNECT.register((handler, server) -> LAST_SENT.remove(handler.player));
		ServerPlayerEvents.AFTER_RESPAWN.register((oldPlayer, newPlayer, alive) -> LAST_SENT.remove(oldPlayer));
	}

	private static void sendCurrent(ServerPlayerEntity observer, ServerPlayerEntity player) {
		if (player instanceof PlayerLoadoutHolder holder) {
			send(observer, player, EquipmentAppearance.capture(holder.tarkovcraft$getLoadout()));
		}
	}

	static void update(ServerPlayerEntity player, PlayerLoadout loadout) {
		EquipmentAppearance appearance = EquipmentAppearance.capture(loadout);
		if (!appearance.equals(LAST_SENT.put(player, appearance))) {
			for (ServerPlayerEntity observer : PlayerLookup.tracking(player)) {
				send(observer, player, appearance);
			}
		}
	}

	private static void send(ServerPlayerEntity observer, ServerPlayerEntity player, EquipmentAppearance appearance) {
		if (observer == player || !ServerPlayNetworking.canSend(observer, EquipmentAppearance.CHANNEL)) return;
		PacketByteBuf buf = PacketByteBufs.create();
		buf.writeUuid(player.getUuid());
		appearance.write(buf);
		ServerPlayNetworking.send(observer, EquipmentAppearance.CHANNEL, buf);
	}
}

			GridContainer container,

			@Nullable ContainerSearchMemory memory,
			UUID playerId,
			boolean requiresSearch
		) {
			boolean revealed = !requiresSearch || memory == null || memory.isGridRevealed(playerId);
			List<ViewEntry> list = new ArrayList<>();
			if (revealed) {
				for (GridEntry e : container.getEntries()) {
					boolean disc = !requiresSearch || memory == null || memory.isEntryDiscovered(playerId, e.getEntryId());
					ItemSize sz = e.getOccupiedSize();

						e.getEntryId(),
						e.getX(),
						e.getY(),
						sz.width(),
						sz.height(),
						e.isRotated(),
						disc,

			writeGridList(buf, corpseRigViews);
			buf.writeBoolean(corpseBackpackView != null);
			if (corpseBackpackView != null) corpseBackpackView.write(buf);
			buf.writeByte(searchStage);
			buf.writeBoolean(searchTargetId != null);
			if (searchTargetId != null) buf.writeUuid(searchTargetId);
			buf.writeBoolean(searchEntryId != null);
			if (searchEntryId != null) buf.writeUuid(searchEntryId);
			buf.writeFloat(searchProgress);
			List<OpenContainerView> open = openContainers == null ? List.of() : openContainers;
			buf.writeVarInt(open.size());
			for (OpenContainerView v : open) {
				v.write(buf);
			}
		}

		public static ScreenStateSnapshot read(PacketByteBuf buf) {
			byte mode = buf.readByte();
			String title = buf.readString();
			List<GridView> ownRig = readGridList(buf);
			GridView bpView = buf.readBoolean() ? GridView.read(buf) : null;
			GridView extView = buf.readBoolean() ? GridView.read(buf) : null;
			PlayerLoadout corpse = null;
			if (buf.readBoolean()) {
				corpse = new PlayerLoadout();
				corpse.readNbt(buf.readNbt());
			}
			List<GridView> corpseRig = readGridList(buf);
			GridView corpseBp = buf.readBoolean() ? GridView.read(buf) : null;
			byte stage = buf.readByte();
			UUID targetId = buf.readBoolean() ? buf.readUuid() : null;
			UUID entryId = buf.readBoolean() ? buf.readUuid() : null;
			float progress = buf.readFloat();
			int openCount = buf.readVarInt();
			List<OpenContainerView> open = new ArrayList<>(openCount);
			for (int i = 0; i < openCount; i++) {
				open.add(OpenContainerView.read(buf));
			}
			return new ScreenStateSnapshot(mode, title, ownRig, bpView, extView, corpse, corpseRig, corpseBp, stage, targetId, entryId, progress, open);
		}

		static void writeGridList(PacketByteBuf buf, List<GridView> list) {
			buf.writeVarInt(list.size());
			for (GridView v : list) {
				v.write(buf);
			}
		}

		static List<GridView> readGridList(PacketByteBuf buf) {
			int n = buf.readVarInt();
			List<GridView> out = new ArrayList<>(n);
			for (int i = 0; i < n; i++) {
				out.add(GridView.read(buf));
			}
			return out;
		}
	}

	private static void handleInteract(ServerPlayerEntity player, byte actionIndex) {
		if (player.isSpectator()) return;
		double reach = 4.2;
		Vec3d eye = player.getCameraPosVec(1.0f);
		Vec3d look = player.getRotationVec(1.0f);
		Vec3d end = eye.add(look.multiply(reach));

		BlockHitResult blockHit = player.getWorld().raycast(new RaycastContext(
			eye, end, RaycastContext.ShapeType.OUTLINE, RaycastContext.FluidHandling.NONE, player
		));
		double maxDistSq = blockHit.getType() == HitResult.Type.BLOCK
			? eye.squaredDistanceTo(blockHit.getPos())
			: reach * reach;

		Box searchBox = player.getBoundingBox().stretch(look.multiply(reach)).expand(1.2);
		EntityHitResult entityHit = ProjectileUtil.raycast(
			player, eye, end, searchBox,
			e -> (e instanceof CorpseEntity || e instanceof ItemEntity) && !e.isRemoved(),
			maxDistSq
		);

		Entity targetEntity = entityHit != null ? entityHit.getEntity() : findClosestLookedItem(player, eye, look, Math.sqrt(maxDistSq));

		if (targetEntity instanceof CorpseEntity corpse) {
			corpse.openForPlayer(player);
			return;
		}

		if (targetEntity instanceof ItemEntity itemEntity && player instanceof PlayerLoadoutHolder holder) {
			ItemStack stack = itemEntity.getStack();
			if (stack.isEmpty()) return;

			// Ground Backpack / Tactical Rig / Case supports scrollable action: 0 = Search on ground, 1 = Equip/Take!
			boolean isGroundContainer = stack.getItem() instanceof BackpackItem
				|| stack.getItem() instanceof TacticalRigItem
				|| stack.getItem() instanceof TarkovCaseItem;
			if (isGroundContainer && actionIndex == WorldInteractState.ACTION_SEARCH) {
				TarkovScreenHandler.openGroundContainer(player, itemEntity);
				return;
			}

			PlayerLoadout loadout = holder.tarkovcraft$getLoadout();
			int beforeCount = stack.getCount();
			boolean fullyPicked = loadout.trySmartPickup(stack, player.getUuid());
			if (fullyPicked || stack.getCount() < beforeCount) {
				if (stack.isEmpty()) {
					itemEntity.discard();
				} else {
					itemEntity.setStack(stack);
				}
				player.getWorld().playSound(null, player.getX(), player.getY(), player.getZ(),
					SoundEvents.ENTITY_ITEM_PICKUP, SoundCategory.PLAYERS, 0.35f, 1.2f);
				holder.tarkovcraft$syncLoadout();
			} else {
				player.sendMessage(Text.translatable("message.tarkovcraft.no_space"), true);
			}
			return;
		}

		if (blockHit.getType() == HitResult.Type.BLOCK) {
			if (player.getWorld().getBlockState(blockHit.getBlockPos()).getBlock() instanceof TarkovWorkbenchBlock) {
				TarkovWorkbenchScreenHandler.openForPlayer(player, blockHit.getBlockPos());
				return;
			}
			BlockEntity be = player.getWorld().getBlockEntity(blockHit.getBlockPos());
			if (be instanceof SearchableLootBlockEntity lootBe) {
				lootBe.openForPlayer(player);
				return;
			}
			com.tarkovcraft.container.VanillaContainerBridge.tryOpenForPlayer(player, blockHit.getBlockPos());
		}
	}

	@Nullable
	public static ItemEntity findClosestLookedItem(Entity viewer, Vec3d eye, Vec3d look, double maxDist) {
		Box box = viewer.getBoundingBox().expand(maxDist);
		List<ItemEntity> items = viewer.getWorld().getEntitiesByClass(ItemEntity.class, box, e -> !e.isRemoved() && !e.getStack().isEmpty());
		ItemEntity best = null;
		double bestScore = Double.MAX_VALUE;
		for (ItemEntity item : items) {
			Vec3d center = item.getPos().add(0, 0.18, 0);
			Vec3d toItem = center.subtract(eye);
			double dist = toItem.length();
			if (dist > maxDist || dist < 0.05) continue;
			double dot = toItem.normalize().dotProduct(look);
			if (dot > 0.93) {
				double score = dist * (1.05 - dot);
				if (score < bestScore) {
					bestScore = score;
					best = item;
				}
			}
		}
		return best;
	}

	public static void sendLoadoutSync(ServerPlayerEntity player, PlayerLoadout loadout) {
		ItemStack headset = loadout.getEquipment(com.tarkovcraft.loadout.EquipmentSlotType.HEADSET);
		if (!headset.isEmpty() && loadout.blocksHeadset()) {
			loadout.setEquipment(com.tarkovcraft.loadout.EquipmentSlotType.HEADSET, ItemStack.EMPTY);
			if (!loadout.tryInsertItem(headset, player.getUuid()) && !headset.isEmpty()) {
				player.dropItem(headset, false);
			}
			player.sendMessage(Text.translatable("message.tarkovcraft.headset_removed"), true);
		}
		PacketByteBuf buf = PacketByteBufs.create();
		buf.writeNbt(loadout.writeNbt());
		ServerPlayNetworking.send(player, S2C_LOADOUT_SYNC, buf);
		EquipmentAppearanceSync.update(player, loadout);
			com.tarkovcraft.compat.voicechat.VoicechatAdapter.updateServerProfile(player.getUuid(),
			com.tarkovcraft.audio.HearingProfile.forLoadout(loadout));
	}

/**
 * Data-driven item catalog loader reading /data/tarkovcraft/catalog/items_catalog.json
 * and registering all items into Registries.ITEM and 6 category-specific creative tab lists.
 */
public final class TarkovItemCatalog {
	public static final String CATALOG_RESOURCE = "/data/tarkovcraft/catalog/items_catalog.json";
	private static final Logger LOGGER = LoggerFactory.getLogger("TarkovCraft/Catalog");

	private static final List<Item> ARMOR_ITEMS = new ArrayList<>();
	private static final List<Item> RIG_ITEMS = new ArrayList<>();
	private static final List<Item> BACKPACK_ITEMS = new ArrayList<>();
	private static final List<Item> MEDICAL_ITEMS = new ArrayList<>();
	private static final List<Item> FOOD_ITEMS = new ArrayList<>();
	private static final List<Item> TACTICAL_ITEMS = new ArrayList<>();
	private static boolean registered = false;

	private TarkovItemCatalog() {
	}

	public static synchronized void registerAll() {
		if (registered) {
			return;
		}
		registered = true;

		try (InputStream stream = TarkovItemCatalog.class.getResourceAsStream(CATALOG_RESOURCE)) {
			if (stream == null) {
				LOGGER.warn("Catalog resource {} not found on classpath; skipping dynamic item registration.", CATALOG_RESOURCE);
				return;
			}
			try (InputStreamReader reader = new InputStreamReader(stream, StandardCharsets.UTF_8)) {
				JsonObject root = JsonParser.parseReader(reader).getAsJsonObject();
				if (!root.has("items") || !root.get("items").isJsonArray()) {
					LOGGER.warn("Catalog {} missing 'items' array.", CATALOG_RESOURCE);
					return;
				}
				JsonArray items = root.getAsJsonArray("items");
				int count = 0;
				for (JsonElement elem : items) {
					if (!elem.isJsonObject()) {
						continue;
					}
					if (registerEntry(elem.getAsJsonObject())) {
						count++;
					}
				}
				LOGGER.info("Registered {} dynamic items from {}.", count, CATALOG_RESOURCE);
			}
		} catch (Exception e) {
			throw new IllegalStateException("Failed to load TarkovCraft item catalog from " + CATALOG_RESOURCE, e);
		}
	}

	private static boolean registerEntry(JsonObject obj) {
		String id = getString(obj, "id", "");
		if (id.isEmpty()) {
			return false;
		}
		Identifier identifier = new Identifier(TarkovCraftMod.MOD_ID, id);
		if (Registries.ITEM.containsId(identifier)) {
			return false;
		}

		String category = getString(obj, "category", "").toLowerCase(Locale.ROOT);
		int width = Math.max(1, getInt(obj, "width", 1));
		int height = Math.max(1, getInt(obj, "height", 1));
		ItemSize size = new ItemSize(width, height);
		ItemRarityTier rarity = parseRarity(getString(obj, "rarity", "COMMON"));
		float weight = Math.max(0.0f, getFloat(obj, "weight", 0.2f));

		Item item = createItemForCategory(obj, category, size, rarity, weight);
		if (item == null) {
			LOGGER.warn("Skipping catalog item '{}' with unsupported category '{}'", id, category);
			return false;
		}

		Registry.register(Registries.ITEM, identifier, item);
		addToTabBucket(category, item);
		return true;
	}

	private static Item createItemForCategory(
		JsonObject obj,
		String category,
		ItemSize size,
		ItemRarityTier rarity,
		float weight
	) {
		FabricItemSettings settings = new FabricItemSettings();
		return switch (category) {
			case "medical" -> {
				String presetStr = getString(obj, "medPreset", "");
				MedicalType medType = MedicalType.fromPreset(presetStr);
				if (medType == null) {
					throw new IllegalArgumentException("Unknown medPreset '" + presetStr + "' for item " + getString(obj, "id", "?"));
				}
				yield new TarkovMedicalItem(settings, medType, size, rarity, weight);
			}
			case "stimulator" -> {
				String stimStr = getString(obj, "stimPreset", "MORPHINE");
				TarkovStimulatorItem.StimPreset preset = TarkovStimulatorItem.StimPreset.fromString(stimStr);
				yield new TarkovStimulatorItem(settings, preset, size, rarity, weight);
			}
			case "food" -> {
				int hunger = getInt(obj, "hunger", 4);
				float saturation = getFloat(obj, "saturation", 0.5f);
				boolean drink = getBoolean(obj, "drink", false);
				yield new TarkovFoodItem(settings, hunger, saturation, drink, size, rarity, weight);
			}
			case "helmet" -> createArmor(obj, settings, EquipmentSlotType.HELMET, size, rarity, weight);
			case "body_armor" -> createArmor(obj, settings, EquipmentSlotType.BODY_ARMOR, size, rarity, weight);
			case "face_mask" -> createArmor(obj, settings, EquipmentSlotType.FACE_MASK, size, rarity, weight);
			case "rig" -> {
				List<ItemSize> compartments = parseCompartments(obj);
				yield new TacticalRigItem(settings, compartments, size, rarity, weight);
			}
			case "backpack" -> {
				int gridW = Math.max(1, getInt(obj, "gridWidth", 2));
				int gridH = Math.max(1, getInt(obj, "gridHeight", 2));
				yield new BackpackItem(settings, gridW, gridH, size, rarity, weight);
			}
			case "repair_kit" -> new TarkovArmorRepairItem(settings,
				TarkovArmorRepairItem.Tier.valueOf(getString(obj, "repairTier", "T1")), size, rarity, weight);
			case "melee" -> {
				float dmg = Math.max(1.0f, getFloat(obj, "meleeDamage", 6.0f));
				yield new TarkovMeleeItem(settings, dmg, size, rarity, weight);
			}
			case "case" -> {
				int gridW = Math.max(1, getInt(obj, "gridWidth", 4));
				int gridH = Math.max(1, getInt(obj, "gridHeight", 4));
				TarkovCaseItem.CaseRole role = TarkovCaseItem.CaseRole.fromString(getString(obj, "caseRole", "GENERAL"));
				yield new TarkovCaseItem(settings, gridW, gridH, size, rarity, weight, role);
			}
			case "grenade" -> {
				String gTypeStr = getString(obj, "grenadeType", "FRAG_F1");
				TarkovGrenadeItem.GrenadeType gType = TarkovGrenadeItem.GrenadeType.fromString(gTypeStr);
				yield new TarkovGrenadeItem(settings, gType, size, rarity, weight);
			}
			case "headset" -> new TarkovFunctionalItem(settings, category, size, rarity, weight);
			case "barter", "valuable" -> new TarkovFunctionalItem(settings, category, size, rarity, weight,
				obj.has("basePrice") ? obj.get("basePrice").getAsLong() : 0L);
			default -> null;
		};
	}

	private static TarkovArmorItem createArmor(
		JsonObject obj,
		FabricItemSettings settings,
		EquipmentSlotType slot,
		ItemSize size,
		ItemRarityTier rarity,
		float weight
	) {
		int armorClass = Math.max(0, getInt(obj, "armorClass", 0));
		int durability = Math.max(0, getInt(obj, "durability", 0));
		ArmorMaterialType material = parseMaterial(getString(obj, "material", "COSMETIC"));
		float ricochet = getFloat(obj, "ricochet", 0.0f);
		Set<HitZone> zones = parseZones(obj);
		boolean gasProtection = getBoolean(obj, "gasProtection", false);
		return new TarkovArmorItem(settings, slot, size, armorClass, durability, material, ricochet, weight,
			zones, rarity, gasProtection, getBoolean(obj, "blocksHeadset", false), getFloat(obj, "hearingMultiplier", 1.0f));
	}

	private static void addToTabBucket(String category, Item item) {
		switch (category) {
			case "helmet", "body_armor", "face_mask", "repair_kit" -> ARMOR_ITEMS.add(item);
			case "rig" -> RIG_ITEMS.add(item);
			case "backpack" -> BACKPACK_ITEMS.add(item);
			case "medical", "stimulator" -> MEDICAL_ITEMS.add(item);
			case "food" -> FOOD_ITEMS.add(item);
			default -> TACTICAL_ITEMS.add(item);
		}
	}

	private static List<ItemSize> parseCompartments(JsonObject obj) {
		List<ItemSize> out = new ArrayList<>();
		if (obj.has("compartments") && obj.get("compartments").isJsonArray()) {
			for (JsonElement elem : obj.getAsJsonArray("compartments")) {
				if (elem.isJsonArray()) {
					JsonArray pair = elem.getAsJsonArray();
					if (pair.size() >= 2) {
						out.add(new ItemSize(Math.max(1, pair.get(0).getAsInt()), Math.max(1, pair.get(1).getAsInt())));
					}
				}
			}
		}
		if (out.isEmpty()) {
			out.add(new ItemSize(1, 2));
		}
		return out;
	}

	private static Set<HitZone> parseZones(JsonObject obj) {
		EnumSet<HitZone> out = EnumSet.noneOf(HitZone.class);
		if (obj.has("zones") && obj.get("zones").isJsonArray()) {
			for (JsonElement elem : obj.getAsJsonArray("zones")) {
				if (elem.isJsonPrimitive()) {
					String name = elem.getAsString().trim();
					if (!name.isEmpty()) {
						out.add(HitZone.byId(name));
					}
				}
			}
		}
		return out;
	}

	public static ItemRarityTier parseRarity(String raw) {
		if (raw == null || raw.isBlank()) {
			return ItemRarityTier.COMMON;
		}
		for (ItemRarityTier tier : ItemRarityTier.values()) {
			if (tier.name().equalsIgnoreCase(raw.trim())) {
				return tier;
			}
		}
		return ItemRarityTier.COMMON;
	}

	public static ArmorMaterialType parseMaterial(String raw) {
		if (raw == null || raw.isBlank()) {
			return ArmorMaterialType.COSMETIC;
		}
		for (ArmorMaterialType mat : ArmorMaterialType.values()) {
			if (mat.name().equalsIgnoreCase(raw.trim()) || mat.getId().equalsIgnoreCase(raw.trim())) {
				return mat;
			}
		}
		return ArmorMaterialType.COSMETIC;
	}

	private static String getString(JsonObject obj, String key, String fallback) {
		return obj.has(key) && !obj.get(key).isJsonNull() ? obj.get(key).getAsString() : fallback;
	}

	private static int getInt(JsonObject obj, String key, int fallback) {
		return obj.has(key) && !obj.get(key).isJsonNull() ? obj.get(key).getAsInt() : fallback;
	}

	private static float getFloat(JsonObject obj, String key, float fallback) {
		return obj.has(key) && !obj.get(key).isJsonNull() ? obj.get(key).getAsFloat() : fallback;
	}

	private static boolean getBoolean(JsonObject obj, String key, boolean fallback) {
		return obj.has(key) && !obj.get(key).isJsonNull() ? obj.get(key).getAsBoolean() : fallback;
	}
