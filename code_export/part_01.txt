package com.tarkovcraft.airdrop;

import com.tarkovcraft.config.TarkovConfig;
import com.tarkovcraft.arena.ArenaEventsState;
import com.tarkovcraft.entity.AirdropEntity;
import com.tarkovcraft.registry.ModEntities;
import net.minecraft.block.BlockState;
import net.minecraft.registry.tag.BlockTags;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.network.ServerPlayerEntity;
import net.minecraft.server.world.ServerWorld;
import net.minecraft.sound.SoundCategory;
import net.minecraft.sound.SoundEvents;
import net.minecraft.text.Text;
import net.minecraft.util.math.BlockPos;
import net.minecraft.world.Heightmap;

import java.util.List;

public final class AirdropManager {
	private static int tickCounter = 0;

	private AirdropManager() {
	}

	public static void resetTimer() {
		tickCounter = 0;
	}

	public static int getTickCounter() {
		return tickCounter;
	}

	public static void onConfigReloaded() {
		int maxTicks = TarkovConfig.get().getAirdropIntervalTicks();
		if (tickCounter >= maxTicks) {
			tickCounter = 0;
		}
	}

	public static void tickServer(MinecraftServer server) {
		TarkovConfig cfg = TarkovConfig.get();
		if (!cfg.airdropEnabled) {
			return;
		}
		ServerWorld overworld = server.getOverworld();
		if (overworld == null) {
			return;
		}
		if (ArenaEventsState.get(overworld).arenas().stream().anyMatch(ArenaEventsState.Arena::ready)) {
			return;
		}
		List<ServerPlayerEntity> activePlayers = overworld.getPlayers(p -> p.isAlive() && !p.isSpectator());
		if (activePlayers.isEmpty()) {
			return;
		}
		tickCounter++;
		if (tickCounter >= cfg.getAirdropIntervalTicks()) {
			tickCounter = 0;
			if (activePlayers.size() >= 2) {
				spawnAirdropBetweenPlayers(overworld, activePlayers);
			} else {
				ServerPlayerEntity target = activePlayers.get(overworld.random.nextInt(activePlayers.size()));
				spawnAirdropNear(target);
			}
		}
	}

	public static boolean isSafeArenaPoint(ServerWorld world, ArenaEventsState.Arena arena, BlockPos pos) {
		if (!world.isInBuildLimit(pos) || pos.getY() > arena.maxGroundY()) return false;
		for (int dx = -2; dx <= 2; dx++) {
			for (int dz = -2; dz <= 2; dz++) {
				int x = pos.getX() + dx;
				int z = pos.getZ() + dz;
				if (!arena.contains(x, z) || !world.isChunkLoaded(x >> 4, z >> 4)) return false;
				int y = world.getTopY(Heightmap.Type.MOTION_BLOCKING, x, z);
				if (y != pos.getY()) return false;
				BlockPos surface = new BlockPos(x, y, z);
				BlockPos floor = surface.down();
				BlockState support = world.getBlockState(floor);
				if (!support.isSolidBlock(world, floor) || support.isIn(BlockTags.LEAVES) || support.isIn(BlockTags.LOGS)
					|| !support.getFluidState().isEmpty() || !world.getBlockState(surface).isAir()
					|| !world.isSkyVisible(surface)) return false;
			}
		}
		return true;
	}

	public static AirdropEntity spawnForArena(ServerWorld world, ArenaEventsState.Arena arena) {
		if (!arena.ready() || !arena.airdropsEnabled()) return null;
		if (!arena.airdropPoints().isEmpty()) {
			List<BlockPos> safe = arena.airdropPoints().stream().filter(pos -> isSafeArenaPoint(world, arena, pos)).toList();
			if (safe.isEmpty()) return null;
			return spawnArenaDrop(world, safe.get(world.random.nextInt(safe.size())));
		}
		int minX = arena.corners().stream().mapToInt(BlockPos::getX).min().orElseThrow();
		int maxX = arena.corners().stream().mapToInt(BlockPos::getX).max().orElseThrow();
		int minZ = arena.corners().stream().mapToInt(BlockPos::getZ).min().orElseThrow();
		int maxZ = arena.corners().stream().mapToInt(BlockPos::getZ).max().orElseThrow();
		for (int i = 0; i < 96; i++) {
			int x = i == 0 ? arena.center().getX() : minX + world.random.nextInt(maxX - minX + 1);
			int z = i == 0 ? arena.center().getZ() : minZ + world.random.nextInt(maxZ - minZ + 1);
			if (!arena.contains(x, z) || !world.isChunkLoaded(x >> 4, z >> 4)) continue;
			BlockPos pos = new BlockPos(x, world.getTopY(Heightmap.Type.MOTION_BLOCKING, x, z), z);
			if (isSafeArenaPoint(world, arena, pos)) return spawnArenaDrop(world, pos);
		}
		return null;
	}

	private static AirdropEntity spawnArenaDrop(ServerWorld world, BlockPos landing) {
		return spawnAirdropAt(world, landing.getX(), landing.getZ(), true, landing);
	}

	public static AirdropEntity spawnFromCommand(ServerPlayerEntity player, boolean here) {
		ServerWorld world = player.getServerWorld();
		BlockPos pos = player.getBlockPos();
		var arenas = ArenaEventsState.get(world).arenas().stream().filter(ArenaEventsState.Arena::ready).toList();
		if (arenas.isEmpty()) {
			return here ? spawnAirdropAt(world, pos.getX(), pos.getZ(), true) : spawnAirdropNear(player);
		}
		var arena = arenas.stream().filter(a -> a.contains(pos.getX(), pos.getZ())).findFirst().orElse(null);
		if (arena == null || !arena.airdropsEnabled()) return null;
		if (!here) return spawnForArena(world, arena);
		if (!arena.airdropPoints().isEmpty() && !arena.airdropPoints().contains(pos)) return null;
		return isSafeArenaPoint(world, arena, pos) ? spawnArenaDrop(world, pos) : null;
	}

	public static AirdropEntity spawnAirdropBetweenPlayers(ServerWorld world, List<ServerPlayerEntity> players) {
		if (players == null || players.isEmpty()) {
			return null;
		}
		if (players.size() == 1) {
			return spawnAirdropNear(players.get(0));
		}

		// Collect pairs within 1200 blocks XZ
		List<int[]> validPairs = new java.util.ArrayList<>();
		for (int i = 0; i < players.size(); i++) {
			for (int j = i + 1; j < players.size(); j++) {
				ServerPlayerEntity a = players.get(i);
				ServerPlayerEntity b = players.get(j);
				double dx = a.getX() - b.getX();
				double dz = a.getZ() - b.getZ();
				if (dx * dx + dz * dz <= 1200.0 * 1200.0) {
					validPairs.add(new int[]{i, j});
				}
			}
		}
		if (validPairs.isEmpty()) {
			return spawnAirdropNear(players.get(world.random.nextInt(players.size())));
		}

		int[] pair = validPairs.get(world.random.nextInt(validPairs.size()));
		ServerPlayerEntity a = players.get(pair[0]);
		ServerPlayerEntity b = players.get(pair[1]);

		double[] tSteps = {0.50, 0.42, 0.58, 0.34, 0.66, 0.25, 0.75, 0.15, 0.85};
		for (double t : tSteps) {
			double baseX = a.getX() + (b.getX() - a.getX()) * t;
			double baseZ = a.getZ() + (b.getZ() - a.getZ()) * t;
			for (int s = 0; s < 6; s++) {
				double offsetAngle = world.random.nextDouble() * Math.PI * 2.0;
				double offsetDist = s == 0 ? 0.0 : (12.0 + world.random.nextDouble() * 24.0);
				int cx = (int) Math.round(baseX + Math.cos(offsetAngle) * offsetDist);
				int cz = (int) Math.round(baseZ + Math.sin(offsetAngle) * offsetDist);
				if (!world.isChunkLoaded(cx >> 4, cz >> 4)) {
					continue;
				}
				int surfaceY = world.getTopY(Heightmap.Type.MOTION_BLOCKING_NO_LEAVES, cx, cz);
				net.minecraft.util.math.BlockPos surfacePos = new net.minecraft.util.math.BlockPos(cx, surfaceY - 1, cz);
				if (world.getFluidState(surfacePos).isEmpty() && world.getFluidState(surfacePos.up()).isEmpty()) {
					return spawnAirdropAt(world, cx, cz, true);
				}
			}
		}

		return spawnAirdropNear(players.get(world.random.nextInt(players.size())));
	}

	public static AirdropEntity spawnAirdropNear(ServerPlayerEntity player) {
		ServerWorld world = player.getServerWorld();
		TarkovConfig cfg = TarkovConfig.get();
		int minR = cfg.airdropMinRadius;
		int maxR = Math.max(minR, cfg.airdropMaxRadius);

		int chosenX = player.getBlockX();
		int chosenZ = player.getBlockZ();
		boolean found = false;
		for (int attempt = 0; attempt < 16; attempt++) {
			double angle = world.random.nextDouble() * Math.PI * 2.0;
			double radius = minR + world.random.nextDouble() * (maxR - minR);
			int cx = (int) Math.round(player.getX() + Math.cos(angle) * radius);
			int cz = (int) Math.round(player.getZ() + Math.sin(angle) * radius);
			if (!world.isChunkLoaded(cx >> 4, cz >> 4)) {
				continue;
			}
			int surfaceY = world.getTopY(Heightmap.Type.MOTION_BLOCKING_NO_LEAVES, cx, cz);
			net.minecraft.util.math.BlockPos surfacePos = new net.minecraft.util.math.BlockPos(cx, surfaceY - 1, cz);
			if (world.getFluidState(surfacePos).isEmpty() && world.getFluidState(surfacePos.up()).isEmpty()) {
				chosenX = cx;
				chosenZ = cz;
				found = true;
				break;
			}
		}
		if (!found) {
			chosenX = player.getBlockX();
			chosenZ = player.getBlockZ();
		}
		return spawnAirdropAt(world, chosenX, chosenZ, true);
	}

	@org.jetbrains.annotations.Nullable
	public static AirdropEntity spawnAirdropAt(ServerWorld world, int x, int z, boolean announce) {
		return spawnAirdropAt(world, x, z, announce, null);
	}

	private static AirdropEntity spawnAirdropAt(ServerWorld world, int x, int z, boolean announce, BlockPos landing) {
		if (!world.isChunkLoaded(x >> 4, z >> 4)) {
			return null;
		}
		int surfaceY = world.getTopY(Heightmap.Type.MOTION_BLOCKING_NO_LEAVES, x, z);
		int spawnY = Math.min(world.getTopY() - 8, Math.max(surfaceY + 120, world.getSeaLevel() + 110));

		AirdropEntity airdrop = new AirdropEntity(ModEntities.AIRDROP, world);
		if (landing != null) airdrop.setLandingPoint(landing);
		airdrop.refreshPositionAndAngles(x + 0.5, spawnY, z + 0.5, 0.0f, 0.0f);
		if (spawnY <= surfaceY || !world.spawnEntity(airdrop)) return null;

		for (ServerPlayerEntity p : world.getPlayers()) {
			double dx = p.getX() - x;
			double dz = p.getZ() - z;
			if (dx * dx + dz * dz <= 400.0 * 400.0) {
				world.playSound(null, p.getX(), p.getY(), p.getZ(),
					SoundEvents.BLOCK_BEACON_ACTIVATE, SoundCategory.AMBIENT, 0.9f, 0.85f);
			}
		}

		if (announce) {
			Text msg = TarkovConfig.get().airdropAnnounceCoords
				? Text.translatable("message.tarkovcraft.airdrop_incoming_coords", x, z)
				: Text.translatable("message.tarkovcraft.airdrop_incoming");
			MinecraftServer server = world.getServer();
			if (server != null) {
				server.getPlayerManager().broadcast(msg, false);
			}
		}
		return airdrop;
	}
}

package com.tarkovcraft.ammo;

import com.tarkovcraft.grid.ContainerItemHelper;
import com.tarkovcraft.grid.GridContainer;
import com.tarkovcraft.grid.GridEntry;
import com.tarkovcraft.grid.ItemSize;
import com.tarkovcraft.grid.ItemSizeRegistry;
import com.tarkovcraft.item.TacticalRigItem;
import com.tarkovcraft.loadout.EquipmentSlotType;
import com.tarkovcraft.loadout.PlayerLoadout;
import com.tarkovcraft.loadout.PlayerLoadoutHolder;
import com.tarkovcraft.search.ContainerSearchMemory;
import net.minecraft.entity.player.PlayerEntity;
import net.minecraft.item.ItemStack;
import net.minecraft.nbt.NbtCompound;
import net.minecraft.server.network.ServerPlayerEntity;
import net.minecraft.util.Identifier;
import org.jetbrains.annotations.Nullable;

import java.lang.reflect.Method;
import java.util.ArrayList;
import java.util.EnumMap;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Optional;
import java.util.UUID;
import java.util.function.Consumer;

/**
 * Server-side fallback tracker for TaCZ firearm reloads:
 * - In Creative mode (where TaCZ does not call PlayerInventory.removeStack), detects GunCurrentAmmoCount
 *   increases and stamps the matching AmmoTier from the player's inventory/loadout onto the held gun.
 * - In Survival mode, ensures guns without TarkovLoadedTier receive FMJ when ammo count increases.
 */
public final class AmmoReloadTracker {
	private static final Map<UUID, Integer> LAST_AMMO_COUNTS = new HashMap<>();

	private record AmmoSlot(ItemStack stack, Consumer<ItemStack> replace) {}

	private AmmoReloadTracker() {
	}

	public static String getGunAmmoId(ItemStack gun) {
		if (!AmmoTier.isTaczGunStack(gun) || gun.getNbt() == null) {
			return "";
		}
		Identifier gunId = Identifier.tryParse(gun.getNbt().getString("GunId"));
		if (gunId != null && GunLookup.GET_GUN_INDEX != null) {
			try {
				Object result = GunLookup.GET_GUN_INDEX.invoke(null, gunId);
				if (result instanceof Optional<?> index && index.isPresent()) {
					Object gunIndex = index.get();
					Object data = gunIndex.getClass().getMethod("getGunData").invoke(gunIndex);
					Object ammoId = data.getClass().getMethod("getAmmoId").invoke(data);
					if (ammoId instanceof Identifier id) return id.toString();
				}
			} catch (ReflectiveOperationException | LinkageError ignored) {
				// Optional bridge. A gun may instead expose its caliber directly in the stack NBT.
			}
		}
		return gun.getNbt().getString("AmmoId");
	}

	private static final class GunLookup {
		@Nullable
		private static final Method GET_GUN_INDEX = findGunIndexMethod();

		@Nullable
		private static Method findGunIndexMethod() {
			try {
				return Class.forName("com.tacz.guns.api.TimelessAPI").getMethod("getCommonGunIndex", Identifier.class);
			} catch (ReflectiveOperationException | LinkageError ignored) {
				return null;
			}
		}
	}

	public static Map<AmmoTier, Integer> countAvailableAmmo(PlayerEntity player, String ammoId) {
		if (player.isCreative()) {
			return countAvailableAmmo(creativeAmmoSlots(player), ammoId);
		}
		if (player instanceof PlayerLoadoutHolder holder) {
			return countAvailableAmmo(holder.tarkovcraft$getLoadout(), player.getUuid(), ammoId);
		}
		return Map.of();
	}

	static Map<AmmoTier, Integer> countAvailableAmmo(PlayerLoadout loadout, UUID playerId, String ammoId) {
		ItemStack rig = loadout.getEquipment(EquipmentSlotType.RIG);
		return countAvailableAmmo(loadoutAmmoSlots(loadout, playerId, rig, readRevealedRig(rig, playerId)), ammoId);
	}

	private static Map<AmmoTier, Integer> countAvailableAmmo(List<AmmoSlot> slots, String ammoId) {
		Map<AmmoTier, Integer> counts = new EnumMap<>(AmmoTier.class);
		for (AmmoSlot slot : slots) {
			ItemStack stack = slot.stack();
			if (isMatchingAmmo(stack, ammoId)) {
				counts.merge(AmmoTier.getFromAmmoStack(stack), stack.getCount(), Integer::sum);
			}
		}
		return counts;
	}

	public static boolean prioritizeAmmoTier(ServerPlayerEntity player, String ammoId, AmmoTier tier) {
		if (!TaCZCaliberCatalog.isBallisticAmmoId(ammoId) || !TaCZCaliberCatalog.isTierAllowed(ammoId, tier)) {
			return false;
		}
		if (player.isCreative()) {
			if (!prioritizeAmmoTier(creativeAmmoSlots(player), ammoId, tier)) {
				return false;
			}
			player.getInventory().markDirty();
			player.currentScreenHandler.sendContentUpdates();
			return true;
		}
		if (player instanceof PlayerLoadoutHolder holder
				&& prioritizeAmmoTier(holder.tarkovcraft$getLoadout(), player.getUuid(), ammoId, tier)) {
			holder.tarkovcraft$syncLoadout();
			return true;
		}
		return false;
	}

	static boolean prioritizeAmmoTier(PlayerLoadout loadout, UUID playerId, String ammoId, AmmoTier tier) {
		ItemStack rig = loadout.getEquipment(EquipmentSlotType.RIG);
		List<GridContainer> grids = readRevealedRig(rig, playerId);
		if (!prioritizeAmmoTier(loadoutAmmoSlots(loadout, playerId, rig, grids), ammoId, tier)) {
			return false;
		}
		if (!grids.isEmpty()) {
			ContainerItemHelper.writeRigGrids(rig, grids);
		}
		loadout.invalidateExposedCache();
		return true;
	}

	private static boolean prioritizeAmmoTier(List<AmmoSlot> slots, String ammoId, AmmoTier tier) {
		Map<ItemSize, List<AmmoSlot>> slotsBySize = new HashMap<>();
		boolean hasPreferred = false;
		for (AmmoSlot slot : slots) {
			if (isMatchingAmmo(slot.stack(), ammoId)) {
				slotsBySize.computeIfAbsent(ItemSizeRegistry.getSize(slot.stack()), size -> new ArrayList<>()).add(slot);
				hasPreferred |= AmmoTier.getFromAmmoStack(slot.stack()) == tier;
			}
		}
		if (!hasPreferred) return false;

		// TaCZ scans inventory slots in order. Reorder equal-footprint ammo for this caliber only.
		for (List<AmmoSlot> sameSize : slotsBySize.values()) {
			List<ItemStack> prioritized = new ArrayList<>(sameSize.size());
			for (AmmoSlot slot : sameSize) {
				if (AmmoTier.getFromAmmoStack(slot.stack()) == tier) prioritized.add(slot.stack());
			}
			for (AmmoSlot slot : sameSize) {
				if (AmmoTier.getFromAmmoStack(slot.stack()) != tier) prioritized.add(slot.stack());
			}
			for (int i = 0; i < sameSize.size(); i++) {
				AmmoSlot slot = sameSize.get(i);
				ItemStack next = prioritized.get(i);
				if (slot.stack() != next) slot.replace().accept(next);
			}
		}
		return true;
	}

	private static List<AmmoSlot> creativeAmmoSlots(PlayerEntity player) {
		List<AmmoSlot> slots = new ArrayList<>();
		addCreativeAmmoSlots(slots, player.getInventory().main);
		addCreativeAmmoSlots(slots, player.getInventory().offHand);
		return slots;
	}

	private static void addCreativeAmmoSlots(List<AmmoSlot> slots, List<ItemStack> inventory) {
		for (int i = 0; i < inventory.size(); i++) {
			int index = i;
			ItemStack stack = inventory.get(index);
			if (AmmoTier.isTaczAmmoStack(stack)) slots.add(new AmmoSlot(stack, next -> inventory.set(index, next)));
		}
	}

	private static List<AmmoSlot> loadoutAmmoSlots(PlayerLoadout loadout, UUID playerId, ItemStack rig,
			List<GridContainer> grids) {
		List<AmmoSlot> slots = new ArrayList<>();
		for (int i = 0; i < PlayerLoadout.POCKET_COUNT; i++) {
			int index = i;
			ItemStack stack = loadout.getPocket(index);
			if (AmmoTier.isTaczAmmoStack(stack)) slots.add(new AmmoSlot(stack, next -> loadout.setPocket(index, next)));
		}

		ContainerSearchMemory search = ContainerItemHelper.requiresSearch(rig) ? ContainerItemHelper.getSearchMemory(rig) : null;
		int exposed = 0;
		for (GridEntry entry : loadout.getSecureContainer().getEntries()) {
			if (!AmmoTier.isTaczAmmoStack(entry.getStack())) continue;
			exposed++;
			slots.add(new AmmoSlot(entry.getStack(), entry::setStack));
		}
		for (GridContainer grid : grids) {
			for (GridEntry entry : grid.getEntries()) {
				if (search != null && !search.isEntryDiscovered(playerId, entry.getEntryId())) continue;
				if (exposed++ == 36 - PlayerLoadout.HOTBAR_SIZE) return slots;
				if (AmmoTier.isTaczAmmoStack(entry.getStack())) slots.add(new AmmoSlot(entry.getStack(), entry::setStack));
			}
		}
		return slots;
	}

	private static List<GridContainer> readRevealedRig(ItemStack rig, UUID playerId) {
		if (!(rig.getItem() instanceof TacticalRigItem)
				|| (ContainerItemHelper.requiresSearch(rig)
					&& !ContainerItemHelper.getSearchMemory(rig).isGridRevealed(playerId))) {
			return List.of();
		}
		return ContainerItemHelper.readRigGrids(rig);
	}

	private static boolean isMatchingAmmo(ItemStack stack, String ammoId) {
		return !ammoId.isBlank() && AmmoTier.isTaczAmmoStack(stack) && ammoId.equals(AmmoTier.getAmmoId(stack));
	}

	public static void removePlayer(UUID playerId) {
		LAST_AMMO_COUNTS.remove(playerId);
	}

	public static void tickPlayer(ServerPlayerEntity player) {
		if (player == null || player.isSpectator()) {
			return;
		}
		ItemStack held = player.getMainHandStack();
		if (!AmmoTier.isTaczGunStack(held) || !held.hasNbt()) {
			LAST_AMMO_COUNTS.remove(player.getUuid());
			return;
		}

		NbtCompound nbt = held.getNbt();
		if (nbt == null || !nbt.contains("GunCurrentAmmoCount")) {
			return;
		}

		int curCount = nbt.getInt("GunCurrentAmmoCount");
		Integer prevCount = LAST_AMMO_COUNTS.put(player.getUuid(), curCount);
		if (prevCount == null || curCount <= prevCount) {
			return;
		}

		if (player.isCreative()) {
			AmmoTier found = findAmmoTierForCreativeReload(player);
			if (found != null) {
				AmmoTier.stampLoadedTierOnGun(held, found);
			} else if (!nbt.contains(AmmoTier.NBT_LOADED_TIER)) {
				AmmoTier.stampLoadedTierOnGun(held, AmmoTier.FMJ);
			}
		} else if (!nbt.contains(AmmoTier.NBT_LOADED_TIER)) {
			AmmoTier.stampLoadedTierOnGun(held, AmmoTier.FMJ);
		}
	}

	@Nullable
	private static AmmoTier findAmmoTierForCreativeReload(ServerPlayerEntity player) {
		String ammoId = getGunAmmoId(player.getMainHandStack());
		if (!ammoId.isBlank()) {
			for (AmmoSlot slot : creativeAmmoSlots(player)) {
				if (isMatchingAmmo(slot.stack(), ammoId)) return AmmoTier.getFromAmmoStack(slot.stack());
			}
			return null;
		}

		AmmoTier fallbackFmj = null;
		for (int i = 0; i < player.getInventory().main.size(); i++) {
			ItemStack st = player.getInventory().main.get(i);
			if (AmmoTier.isTaczAmmoStack(st)) {
				AmmoTier t = AmmoTier.getFromAmmoStack(st);
				if (t != AmmoTier.FMJ) return t;
				fallbackFmj = t;
			}
		}
		for (int i = 0; i < player.getInventory().offHand.size(); i++) {
			ItemStack st = player.getInventory().offHand.get(i);
			if (AmmoTier.isTaczAmmoStack(st)) {
				AmmoTier t = AmmoTier.getFromAmmoStack(st);
				if (t != AmmoTier.FMJ) return t;
				fallbackFmj = t;
			}
		}
		if (player instanceof PlayerLoadoutHolder holder) {
			PlayerLoadout loadout = holder.tarkovcraft$getLoadout();
			for (int i = 0; i < PlayerLoadout.POCKET_COUNT; i++) {
				ItemStack st = loadout.getPocket(i);
				if (AmmoTier.isTaczAmmoStack(st)) {
					AmmoTier t = AmmoTier.getFromAmmoStack(st);
					if (t != AmmoTier.FMJ) return t;
					fallbackFmj = t;
				}
			}
			ItemStack rig = loadout.getEquipment(EquipmentSlotType.RIG);
			if (!rig.isEmpty() && rig.getItem() instanceof TacticalRigItem) {
				ContainerSearchMemory search = ContainerItemHelper.requiresSearch(rig) ? ContainerItemHelper.getSearchMemory(rig) : null;
				for (GridContainer g : readRevealedRig(rig, player.getUuid())) {
					for (GridEntry entry : g.getEntries()) {
						if (search != null && !search.isEntryDiscovered(player.getUuid(), entry.getEntryId())) continue;
						ItemStack st = entry.getStack();
						if (AmmoTier.isTaczAmmoStack(st)) {
							AmmoTier t = AmmoTier.getFromAmmoStack(st);
							if (t != AmmoTier.FMJ) return t;
							fallbackFmj = t;
						}
					}
				}
			}
		}
		return fallbackFmj;
	}
}

package com.tarkovcraft.ammo;

import net.minecraft.item.Item;

/**
 * Tarkov ballistic ammo tiers for TaCZ ammunition:
 * - HP  (Hollow Point / Экспансивные): +30% flesh damage, -12 penetration (stops on Class 2-3, shreds limbs/face)
 * - FMJ (Standard / Стандартные PS/FMJ): baseline damage & penetration (Class 2-3)
 * - AP  (Armor-Piercing / Бронебойные BP/M855A1/M80): -8% flesh damage, +16 penetration (Class 4-5)
 * - IGN (Ultra-AP / Сверхбронебойные BS/Igolnik/M995/M61): -15% flesh damage, +28 penetration (Class 5-6)
 */
public enum AmmoTier {
	HP("hp", "HP", 1.30f, -12, 0.70f, Formatting.GREEN, 0xFF55FF55),
	FMJ("fmj", "PS/FMJ", 1.00f, 0, 1.00f, Formatting.GRAY, 0xFFB0B8C0),
	AP("ap", "BP/AP", 0.92f, 16, 1.25f, Formatting.GOLD, 0xFFFFAA00),
	IGN("ign", "BS/M995", 0.85f, 28, 1.50f, Formatting.RED, 0xFFFF5555);

	public static final String NBT_AMMO_TIER = "TarkovAmmoTier";
	public static final String NBT_AMMO_TIER_ALIAS = "tarkovcraft_ammo_tier";
	public static final String NBT_LOADED_TIER = "TarkovLoadedTier";

	private final String id;
	private final String badge;
	private final float damageMultiplier;
	private final int penetrationDelta;
	private final float armorWearMultiplier;
	private final Formatting color;
	private final int argbColor;

	AmmoTier(String id, String badge, float damageMultiplier, int penetrationDelta, float armorWearMultiplier, Formatting color, int argbColor) {
		this.id = id;
		this.badge = badge;
		this.damageMultiplier = damageMultiplier;
		this.penetrationDelta = penetrationDelta;
		this.armorWearMultiplier = armorWearMultiplier;
		this.color = color;
		this.argbColor = argbColor;
	}

	public String getId() {
		return id;
	}

	public String getBadge() {
		return badge;
	}

	public float getDamageMultiplier() {
		return damageMultiplier;
	}

	public int getPenetrationDelta() {
		return penetrationDelta;
	}

	public float getArmorWearMultiplier() {
		return armorWearMultiplier;
	}

	public Formatting getColor() {
		return color;
	}

	public int getArgbColor() {
		return argbColor;
	}

	public String getTranslationKey() {
		return "ammotier.tarkovcraft." + id;
	}

	public static String formatArmorClassTarget(int pen) {
		if (pen >= 55) return "Class 6";
		if (pen >= 45) return "Class 5";
		if (pen >= 35) return "Class 4";
		if (pen >= 25) return "Class 3";
		if (pen >= 15) return "Class 2";
		return "Class 1 / Flesh";
	}

	public static AmmoTier byId(String id) {
		if (id == null || id.isBlank()) return FMJ;
		for (AmmoTier tier : values()) {
			if (tier.id.equalsIgnoreCase(id) || tier.badge.equalsIgnoreCase(id)) {
				return tier;
			}
		}
		return FMJ;
	}

	public static boolean isTaczAmmoStack(ItemStack stack) {
		if (stack == null || stack.isEmpty()) return false;
		Identifier id = Registries.ITEM.getId(stack.getItem());
		if ("tacz".equals(id.getNamespace()) && "ammo".equals(id.getPath())) {
			return true;
		}
		return stack.hasNbt() && stack.getNbt() != null && stack.getNbt().contains("AmmoId");
	}

	public static boolean isTaczGunStack(ItemStack stack) {
		if (stack == null || stack.isEmpty()) return false;
		Identifier id = Registries.ITEM.getId(stack.getItem());
		if ("tacz".equals(id.getNamespace()) && "modern_kinetic_gun".equals(id.getPath())) {
			return true;
		}
		return stack.hasNbt() && stack.getNbt() != null && stack.getNbt().contains("GunId");
	}

	public static String getAmmoId(ItemStack stack) {
		if (stack == null || stack.isEmpty() || !stack.hasNbt() || stack.getNbt() == null) {
			return "";
		}
		return stack.getNbt().getString("AmmoId");
	}

	public static AmmoTier getFromAmmoStack(ItemStack stack) {
		if (stack == null || stack.isEmpty() || !stack.hasNbt() || stack.getNbt() == null) {
			return FMJ;
		}
		NbtCompound nbt = stack.getNbt();
		if (nbt.contains(NBT_AMMO_TIER)) {
			return byId(nbt.getString(NBT_AMMO_TIER));
		}
		if (nbt.contains(NBT_AMMO_TIER_ALIAS)) {
			return byId(nbt.getString(NBT_AMMO_TIER_ALIAS));
		}
		return FMJ;
	}

	public static void setOnAmmoStack(ItemStack stack, AmmoTier tier) {
		if (stack == null || stack.isEmpty()) return;
		if (tier == FMJ) {
			if (stack.hasNbt() && stack.getNbt() != null) {
				stack.getNbt().remove(NBT_AMMO_TIER);
			}
		} else {
			stack.getOrCreateNbt().putString(NBT_AMMO_TIER, tier.id);
		}
	}

	public static AmmoTier getLoadedTierFromGun(ItemStack gunStack) {
		if (gunStack == null || gunStack.isEmpty() || !gunStack.hasNbt() || gunStack.getNbt() == null) {
			return FMJ;
		}
		NbtCompound nbt = gunStack.getNbt();
		if (nbt.contains(NBT_LOADED_TIER)) {
			return byId(nbt.getString(NBT_LOADED_TIER));
		}
		return FMJ;
	}

	public static void stampLoadedTierOnGun(ItemStack gunStack, AmmoTier tier) {
		if (gunStack == null || gunStack.isEmpty()) return;
		gunStack.getOrCreateNbt().putString(NBT_LOADED_TIER, tier.id);
	}

	public static ItemStack createTaczAmmoStack(String ammoId, AmmoTier tier, int count) {
		Identifier taczAmmoItemId = new Identifier("tacz", "ammo");
		Item item = Registries.ITEM.containsId(taczAmmoItemId)
			? Registries.ITEM.get(taczAmmoItemId)
			: Items.IRON_NUGGET;
		ItemStack stack = new ItemStack(item, Math.max(1, count));
		String fullAmmoId = ammoId.contains(":") ? ammoId : ("tacz:" + ammoId);
		stack.getOrCreateNbt().putString("AmmoId", fullAmmoId);
		if (tier != FMJ) {
			stack.getOrCreateNbt().putString(NBT_AMMO_TIER, tier.id);
		}
		return stack;
	}
}

			AmmoTier target = requested != null ? requested : AmmoTier.FMJ;
			if (allowedTiers.contains(target)) {

			}
			if (target == AmmoTier.IGN && allowedTiers.contains(AmmoTier.AP)) {
				return AmmoTier.AP;
			}
			if (target == AmmoTier.HP && allowedTiers.contains(AmmoTier.FMJ)) {
				return AmmoTier.FMJ;
			}
			return allowedTiers.contains(AmmoTier.FMJ) ? AmmoTier.FMJ : allowedTiers.iterator().next();
		}

	public static boolean isShotgunCaliber(String ammoId) {
		if (ammoId == null) return false;
		String l = ammoId.toLowerCase(Locale.ROOT);
		return l.contains("12g") || l.contains("20g") || l.contains("410") || l.contains("gauge") || l.contains("shot");
	}

	public static EnumSet<AmmoTier> resolveAllowedTiers(String ammoId, CaliberGroup group) {
		String l = ammoId == null ? "" : ammoId.toLowerCase(Locale.ROOT);
		if (isShotgunCaliber(l) || group == CaliberGroup.SHOTGUN) {
			return EnumSet.of(AmmoTier.HP, AmmoTier.FMJ, AmmoTier.AP);
		}
		if (group == CaliberGroup.HEAVY_SNIPER || l.contains("50bmg") || l.contains("338")
			|| l.contains("127") || l.contains("12_7") || l.contains("145")) {
			return EnumSet.of(AmmoTier.FMJ, AmmoTier.AP, AmmoTier.IGN);
		}
		if (l.contains("57x28") || l.contains("46x30")) {
			return EnumSet.allOf(AmmoTier.class);
		}
		if (group == CaliberGroup.PISTOL_SMG || group == CaliberGroup.MAGNUM_PISTOL
			|| l.contains("9x18") || l.contains("9x19") || l.contains("380") || l.contains("45acp") || l.contains("9mm")) {
			return EnumSet.of(AmmoTier.HP, AmmoTier.FMJ, AmmoTier.AP);
		}
		return EnumSet.allOf(AmmoTier.class);
	}

	public static boolean isTierAllowed(String ammoId, AmmoTier tier) {
		return getSpecOrDynamic(ammoId).isTierAllowed(tier);

	public static boolean isBallisticAmmoId(String ammoId) {
		if (ammoId == null || !VALID_AMMO_ID.matcher(ammoId).matches()) {
			return false;
		}
		String lower = ammoId.toLowerCase(Locale.ROOT);
		return !lower.contains("40mm") && !lower.contains("rpg_rocket");
	}

	public static List<CaliberSpec> getDefaultCalibers() {
		return List.copyOf(DEFAULT_SPECS.values());
	}

	public static synchronized void discoverRuntimeCalibers() {
		if (runtimeDiscovered) {
			return;
		}
		runtimeDiscovered = true;
		try {
			Class<?> apiClass = Class.forName("com.tacz.guns.api.TimelessAPI");
			Method m = apiClass.getMethod("getAllCommonAmmoIndex");
			Object raw = m.invoke(null);
			if (raw instanceof Set<?> entries) {
				for (Object entryObj : entries) {
					if (entryObj instanceof Map.Entry<?, ?> entry && entry.getKey() != null) {
						String id = entry.getKey().toString();
						registerRuntimeCaliber(id);
					}
				}
			}
		} catch (Throwable ignored) {
			// Zero-hard-dependency: TaCZ is optional at compile/test time
		}
	}

	public static synchronized void registerRuntimeCaliber(String ammoId) {
		if (!isBallisticAmmoId(ammoId) || DEFAULT_SPECS.containsKey(ammoId) || RUNTIME_SPECS.containsKey(ammoId)) {
			return;
		}
		RUNTIME_SPECS.put(ammoId, buildDynamicSpec(ammoId));
	}

	public static synchronized List<CaliberSpec> getAllKnownCalibers() {
		discoverRuntimeCalibers();
		List<CaliberSpec> out = new ArrayList<>(DEFAULT_SPECS.values());
		out.addAll(RUNTIME_SPECS.values());
		return Collections.unmodifiableList(out);
	}

	public static synchronized List<CaliberSpec> getAllCalibersWithExtras(Collection<String> extraAmmoIds) {
		discoverRuntimeCalibers();
		Map<String, CaliberSpec> merged = new LinkedHashMap<>(DEFAULT_SPECS);
		merged.putAll(RUNTIME_SPECS);
		if (extraAmmoIds != null) {
			for (String id : extraAmmoIds) {
				if (isBallisticAmmoId(id) && !merged.containsKey(id)) {
					merged.put(id, buildDynamicSpec(id));
				}
			}
		}
		return List.copyOf(merged.values());
	}

	public static CaliberSpec getSpecOrDynamic(String ammoId) {
		CaliberSpec known = DEFAULT_SPECS.get(ammoId);
		if (known != null) {
			return known;
		}
		synchronized (TaCZCaliberCatalog.class) {
			CaliberSpec rt = RUNTIME_SPECS.get(ammoId);
			if (rt != null) {
				return rt;
			}
		}
		return buildDynamicSpec(ammoId);
	}

	public static CaliberSpec buildDynamicSpec(String ammoId) {
		String clean = ammoId == null ? "unknown" : ammoId;
		int colon = clean.indexOf(':');
		String path = colon >= 0 ? clean.substring(colon + 1) : clean;
		String label = path.replace('_', ' ').toUpperCase(Locale.ROOT);
		int basePen = TaCZBulletHelper.estimateBaseCaliberPenetration(clean);
		return new CaliberSpec(clean, label, CaliberGroup.ASSAULT_RIFLE, basePen);
	}
}

public final class ArenaBossSpawner {
	private static final String BOSS_TAG = "tarkovcraft:arena_boss";

	private ArenaBossSpawner() {
	}

	public static void register() {
		ServerEntityEvents.ENTITY_UNLOAD.register((entity, world) -> {
			if (entity.getCommandTags().contains(BOSS_TAG)
				&& entity.getRemovalReason() != null && entity.getRemovalReason().shouldDestroy()) {
				clearActiveBoss(world, entity.getUuid());
			}
		});
		ServerLivingEntityEvents.AFTER_DEATH.register((entity, source) -> {
			if (!(entity instanceof PillagerEntity boss) || !boss.getCommandTags().contains(BOSS_TAG)
				|| !(boss.getWorld() instanceof ServerWorld world)) return;
			clearActiveBoss(world, boss.getUuid());
			TarkovLootGenerator.BossLoot loot = TarkovLootGenerator.createBossLoot(world.random);
			boss.dropStack(boss.getEquippedStack(EquipmentSlot.CHEST).copy());
			boss.dropStack(boss.getEquippedStack(EquipmentSlot.HEAD).copy());
			boss.dropStack(loot.weapon());
			for (ItemStack trophy : loot.trophies()) boss.dropStack(trophy);
		});
	}

	private static void clearActiveBoss(ServerWorld world, java.util.UUID bossId) {
		for (ServerWorld dimension : world.getServer().getWorlds()) {
			ArenaEventsState state = ArenaEventsState.get(dimension);
			for (ArenaEventsState.Arena arena : state.arenas()) {
				if (bossId.equals(arena.activeBoss)) {
					arena.activeBoss = null;
					state.markDirty();
					return;
				}
			}
		}
	}

	public static boolean spawn(ServerWorld world, ArenaEventsState.Arena arena, List<ServerPlayerEntity> players) {
		if (world.getDifficulty() == Difficulty.PEACEFUL) return false;
		PillagerEntity boss = EntityType.PILLAGER.create(world);
		if (boss == null) return false;
		BlockPos pos = findSpawn(world, arena, boss);
		if (pos == null) return false;

		TarkovLootGenerator.BossLoot loot = TarkovLootGenerator.createBossLoot(world.random);
		boss.setCustomName(Text.translatable("entity.tarkovcraft.arena_boss"));
		boss.setCustomNameVisible(true);
		boss.addCommandTag(BOSS_TAG);
		boss.addCommandTag(MobGearSpawner.TAG_GEAR_ROLLED);
		boss.setPersistent();
		boss.setCanPickUpLoot(false);
		equip(boss, EquipmentSlot.HEAD, loot.helmet());
		equip(boss, EquipmentSlot.CHEST, loot.bodyArmor());
		ItemStack crossbow = new ItemStack(Items.CROSSBOW);
		crossbow.addEnchantment(Enchantments.QUICK_CHARGE, 2);
		equip(boss, EquipmentSlot.MAINHAND, crossbow);

		int armor = armorClass(loot.bodyArmor()) + armorClass(loot.helmet());
		setAttribute(boss, EntityAttributes.GENERIC_ARMOR, armor * 1.5);
		setAttribute(boss, EntityAttributes.GENERIC_ARMOR_TOUGHNESS, 6);
		setAttribute(boss, EntityAttributes.GENERIC_MAX_HEALTH, 80);
		boss.setHealth(boss.getMaxHealth());
		if (!world.spawnEntity(boss)) return false;

		arena.bossSpawned = true;
		arena.activeBoss = boss.getUuid();
		ArenaEventsState.get(world).markDirty();
		Text msg = Text.translatable("message.tarkovcraft.arena_boss_spawned", pos.getX(), pos.getY(), pos.getZ());
		players.forEach(player -> player.sendMessage(msg, false));
		return true;
	}

	private static int armorClass(ItemStack stack) {
		return stack.getItem() instanceof TarkovArmorItem armor ? armor.getArmorClass() : 0;
	}

	private static void equip(PillagerEntity boss, EquipmentSlot slot, ItemStack stack) {
		boss.equipStack(slot, stack);
		boss.setEquipmentDropChance(slot, 0);
	}

	private static void setAttribute(PillagerEntity boss, EntityAttribute attribute, double value) {
		var instance = boss.getAttributeInstance(attribute);
		if (instance != null) instance.setBaseValue(value);
	}

	private static BlockPos findSpawn(ServerWorld world, ArenaEventsState.Arena arena, PillagerEntity boss) {
		BlockPos center = arena.center();
		for (BlockPos column : BlockPos.iterateOutwards(center, 8, 0, 8)) {
			for (int dy = 0; dy <= 8; dy++) {
				BlockPos down = column.down(dy);
				if (tryPosition(world, arena, boss, down)) return down;
				BlockPos up = column.up(dy);
				if (dy != 0 && tryPosition(world, arena, boss, up)) return up;
			}
		}
		return null;
	}

	private static boolean tryPosition(ServerWorld world, ArenaEventsState.Arena arena, PillagerEntity boss, BlockPos pos) {
		if (!ArenaEvents.canPlaceAt(world, arena, pos, null)) return false;
		boss.refreshPositionAndAngles(pos.getX() + 0.5, pos.getY(), pos.getZ() + 0.5, 0, 0);
		return world.isSpaceEmpty(boss);
	}
}

package com.tarkovcraft.arena;

import com.google.gson.Gson;
import com.google.gson.GsonBuilder;
import com.tarkovcraft.TarkovCraftMod;
import net.fabricmc.loader.api.FabricLoader;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.world.ServerWorld;
import net.minecraft.util.math.BlockPos;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.StandardCopyOption;
import java.util.Arrays;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.Set;

final class ArenaConfig {
	private static final Gson GSON = new GsonBuilder().setPrettyPrinting().create();
	private static final Path FILE = FabricLoader.getInstance().getConfigDir().resolve("tarkovcraft_arenas.json");

	private ArenaConfig() { }

	static boolean load(MinecraftServer server) {
		if (!Files.exists(FILE)) return save(server);
		try (var reader = Files.newBufferedReader(FILE)) {
			Config config = GSON.fromJson(reader, Config.class);
			if (config == null || config.dimensions() == null) throw new IOException("Missing dimensions");
			Map<ServerWorld, Map<String, ArenaEventsState.Arena>> configured = new LinkedHashMap<>();
			for (ServerWorld world : server.getWorlds()) {
				Map<String, Definition> definitions = config.dimensions().get(world.getRegistryKey().getValue().toString());
				if (definitions == null) continue;
				var old = ArenaEventsState.get(world).definitions();
				Map<String, ArenaEventsState.Arena> converted = new LinkedHashMap<>();
				definitions.forEach((id, definition) -> {
					var arena = definition.toArena();
					var previous = old.get(id);
					if (previous != null) {
						arena.cratePos = previous.cratePos;
						arena.activeBoss = previous.activeBoss;
						arena.bossSpawned = previous.bossSpawned;
					}
					converted.put(id, arena);
				});
				configured.put(world, converted);
			}
			configured.forEach((world, definitions) -> {
				var state = ArenaEventsState.get(world);
				state.definitions().clear();
				state.definitions().putAll(definitions);
				state.markDirty();
			});
			if (config.lobby() != null) {
				RaidProfileState.get(server).setLobby(config.lobby().dimension(), config.lobby().position());
			}
			return true;
		} catch (IOException | RuntimeException e) {
			TarkovCraftMod.LOGGER.error("Could not load {}", FILE, e);
			return false;
		}
	}

	static boolean save(MinecraftServer server) {
		Map<String, Map<String, Definition>> worlds = new LinkedHashMap<>();
		for (ServerWorld world : server.getWorlds()) {
			Map<String, Definition> arenas = new LinkedHashMap<>();
			ArenaEventsState.get(world).definitions().forEach((id, arena) -> arenas.put(id, Definition.from(arena)));
			worlds.put(world.getRegistryKey().getValue().toString(), arenas);
		}
		var profile = RaidProfileState.get(server);
		Lobby lobby = profile.lobbyPos() == null ? null : new Lobby(profile.lobbyDimension(), profile.lobbyPos());
		Path temporary = FILE.resolveSibling(FILE.getFileName() + ".tmp");
		try {
			Files.createDirectories(FILE.getParent());
			Files.writeString(temporary, GSON.toJson(new Config(worlds, lobby)));
			Files.move(temporary, FILE, StandardCopyOption.REPLACE_EXISTING);
			return true;
		} catch (IOException e) {
			TarkovCraftMod.LOGGER.error("Could not save {}", FILE, e);
			return false;
		}
	}

	private record Config(Map<String, Map<String, Definition>> dimensions, Lobby lobby) { }
	private record Lobby(String dimension, BlockPos position) { }

	private record Definition(String displayName, BlockPos anchor, List<BlockPos> corners, int durationMinutes,
		boolean airdropsEnabled, int maxGroundY, boolean bossEnabled, Set<BlockPos> airdropPoints, Set<BlockPos> pointsOfInterest,
		Set<BlockPos> pmcSpawns, Set<BlockPos> scavSpawns, Map<String, ArenaEventsState.Extract> extracts, Set<BlockPos> lootPoints) {
		static Definition from(ArenaEventsState.Arena arena) {
			return new Definition(arena.displayName, arena.anchor, Arrays.asList(arena.corners.clone()), arena.durationMinutes,
				arena.airdropsEnabled, arena.maxGroundY, arena.bossEnabled, arena.airdropPoints, arena.pointsOfInterest,
				arena.pmcSpawns, arena.scavSpawns, arena.extracts, arena.lootPoints);
		}

		ArenaEventsState.Arena toArena() {
			var arena = new ArenaEventsState.Arena(anchor);
			arena.displayName = displayName == null ? "" : displayName;
			arena.durationMinutes = durationMinutes == 0 ? 15 : Math.max(1, Math.min(60, durationMinutes));
			arena.airdropsEnabled = airdropsEnabled;
			arena.maxGroundY = maxGroundY;
			arena.bossEnabled = bossEnabled;
			if (corners != null) {
				for (int i = 0; i < Math.min(4, corners.size()); i++) {
					if (corners.get(i) != null) arena.setCorner(i, corners.get(i));
				}
			}
			if (airdropPoints != null) arena.airdropPoints.addAll(airdropPoints);
			if (pointsOfInterest != null) arena.pointsOfInterest.addAll(pointsOfInterest);
			if (pmcSpawns != null) arena.pmcSpawns.addAll(pmcSpawns);
			if (scavSpawns != null) arena.scavSpawns.addAll(scavSpawns);
			if (extracts != null) arena.extracts.putAll(extracts);
			if (lootPoints != null) arena.lootPoints.addAll(lootPoints);
			return arena;
		}
	}
}

	private static int create(CommandContext<ServerCommandSource> ctx, String displayName) {
		String id = StringArgumentType.getString(ctx, "id");
		if (!id.matches("[a-z0-9_-]{1,48}") || displayName.isBlank() || displayName.length() > 80) {
			return error(ctx, "Use an ID of 1–48 lowercase letters, digits, underscores or hyphens and a name up to 80 characters.");
		}
		var state = ArenaEventsState.get(ctx.getSource().getWorld());
		if (!state.create(id, position(ctx))) return error(ctx, "Arena already exists in this dimension.");
		state.getArena(id).displayName = displayName;
		return save(ctx, "Arena created. Set its four corners with /tarkov arena set_corner.");
	}

	private static LiteralArgumentBuilder<ServerCommandSource> addPoint(String command,
		java.util.function.Function<ArenaEventsState.Arena, java.util.Set<BlockPos>> points) {

				.executes(ctx -> {
					String value = StringArgumentType.getString(ctx, "enabled");
					if (!value.equals("on") && !value.equals("off")) return error(ctx, "Expected on or off.");
					return update(ctx, arena -> {
						setter.accept(arena, value.equals("on"));
						return command + " → " + value;
					});
				})));
	}

		CommandContext<ServerCommandSource> ctx, java.util.function.Function<ArenaEventsState.Arena, String> edit
	) {
		return update(ctx, edit, "Invalid arena setting.");
	}

	private static int update(
		CommandContext<ServerCommandSource> ctx, java.util.function.Function<ArenaEventsState.Arena, String> edit, String reason
	) {
		ArenaEventsState state = ArenaEventsState.get(ctx.getSource().getWorld());
		ArenaEventsState.Arena arena = state.getArena(StringArgumentType.getString(ctx, "id"));
		if (arena == null) return error(ctx, "Unknown arena in this dimension.");
		if (RaidSession.isActiveArena(ctx.getSource().getServer(), arena)) return error(ctx, "Wait until the raid ends.");
		String message = edit.apply(arena);
		if (message == null) return error(ctx, reason);
		state.markDirty();
		return save(ctx, message);
	}

	private static int save(CommandContext<ServerCommandSource> ctx, String message) {
		return ArenaConfig.save(ctx.getSource().getServer()) ? success(ctx, message)
			: error(ctx, "The world state was updated, but the arena JSON could not be saved. Check the server log.");
	}

	private static BlockPos position(CommandContext<ServerCommandSource> ctx) {
		return BlockPos.ofFloored(ctx.getSource().getPosition());
	}

	private static int success(CommandContext<ServerCommandSource> ctx, String message) {
		ctx.getSource().sendFeedback(() -> Text.literal(message), true);
		return 1;
	}

	private static int error(CommandContext<ServerCommandSource> ctx, String message) {
		ctx.getSource().sendError(Text.literal(message));
		return 0;
	}

package com.tarkovcraft.arena;

import com.tarkovcraft.airdrop.AirdropManager;
import com.tarkovcraft.block.SearchableLootBlockEntity;
import com.tarkovcraft.config.TarkovConfig;
import net.minecraft.server.MinecraftServer;
import net.minecraft.server.network.ServerPlayerEntity;
import net.minecraft.server.world.ServerWorld;
import net.minecraft.util.math.BlockPos;

import java.util.List;

public final class ArenaEvents {
	private ArenaEvents() {
	}

	public static void tick(MinecraftServer server) {
		TarkovConfig config = TarkovConfig.get();
		for (ServerWorld world : server.getWorlds()) {
			ArenaEventsState state = ArenaEventsState.get(world);
			if (state.retiredCrates.removeIf(pos -> HackableCrateEvent.clearRetiredCrate(world, pos))) state.markDirty();
			for (ArenaEventsState.Arena arena : state.arenas()) {
				if (!arena.ready()) continue;
				List<ServerPlayerEntity> players = world.getPlayers(p -> p.isAlive() && !p.isSpectator()
					&& arena.contains(p.getBlockX(), p.getBlockZ()));
				if (players.isEmpty()) continue;

				if (config.airdropEnabled && arena.airdropsEnabled
					&& ++arena.airdropTicks >= config.getAirdropIntervalTicks()) {
					arena.airdropTicks = AirdropManager.spawnForArena(world, arena) != null
						? 0 : Math.max(0, config.getAirdropIntervalTicks() - 20 * 10);
				}
				if (arena.crateTicks == 0) arena.crateTicks = (240 + world.random.nextInt(61)) * 20;
				if (--arena.crateTicks == 0 && !HackableCrateEvent.activate(world, arena, players)) {
					arena.crateTicks = 20 * 10;
				}
				if (arena.bossEnabled && !arena.bossSpawned) {
					if (arena.bossRetryTicks == 0) {
						arena.bossRetryTicks = ArenaBossSpawner.spawn(world, arena, players) ? 0 : 20 * 10;
					} else {
						arena.bossRetryTicks--;
					}
				}
			}
		}
	}

	static boolean canPlaceAt(ServerWorld world, ArenaEventsState.Arena arena, BlockPos pos, SearchableLootBlockEntity oldCrate) {
		if (!arena.contains(pos.getX(), pos.getZ()) || !world.isInBuildLimit(pos)
			|| !world.isChunkLoaded(pos.getX() >> 4, pos.getZ() >> 4)) return false;
		BlockPos floor = pos.down();
		return world.getBlockState(floor).isSolidBlock(world, floor)
			&& (world.getBlockState(pos).isAir() || oldCrate != null && pos.equals(oldCrate.getPos()))
			&& world.getBlockState(pos.up()).isAir();
	}
}

package com.tarkovcraft.arena;

import net.minecraft.nbt.NbtCompound;
import net.minecraft.server.world.ServerWorld;
import net.minecraft.util.math.BlockPos;
import net.minecraft.world.PersistentState;

import java.util.Arrays;
import java.util.Collection;
import java.util.Comparator;
import java.util.LinkedHashMap;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;
import java.util.UUID;

public final class ArenaEventsState extends PersistentState {
	private final Map<String, Arena> arenas = new LinkedHashMap<>();
	final Set<BlockPos> retiredCrates = new LinkedHashSet<>();

	public static ArenaEventsState get(ServerWorld world) {
		return world.getPersistentStateManager().getOrCreate(
			ArenaEventsState::fromNbt, ArenaEventsState::new, "tarkovcraft_arena_events");
	}

	public Arena getArena(String id) {
		return arenas.get(id);
	}

	public Collection<Arena> arenas() {
		return arenas.values();
	}

	Map<String, Arena> definitions() {
		return arenas;
	}

	public boolean create(String id, BlockPos anchor) {
		if (arenas.containsKey(id)) return false;
		arenas.put(id, new Arena(anchor.toImmutable()));
		markDirty();
		return true;
	}

	@Override
	public NbtCompound writeNbt(NbtCompound nbt) {
		NbtCompound definitions = new NbtCompound();
		arenas.forEach((id, arena) -> definitions.put(id, arena.writeNbt()));
		nbt.put("Arenas", definitions);
		nbt.putLongArray("RetiredCrates", retiredCrates.stream().mapToLong(BlockPos::asLong).toArray());
		return nbt;
	}

	static ArenaEventsState fromNbt(NbtCompound nbt) {
		ArenaEventsState state = new ArenaEventsState();
		NbtCompound definitions = nbt.getCompound("Arenas");
		for (String id : definitions.getKeys()) {
			state.arenas.put(id, Arena.readNbt(definitions.getCompound(id)));
		}
		for (long pos : nbt.getLongArray("RetiredCrates")) state.retiredCrates.add(BlockPos.fromLong(pos));
		return state;
	}

	public static final class Arena {
		final BlockPos anchor;
		final BlockPos[] corners = new BlockPos[4];
		private List<BlockPos> polygon = List.of();
		final Set<BlockPos> airdropPoints = new LinkedHashSet<>();
		final Set<BlockPos> pointsOfInterest = new LinkedHashSet<>();
		boolean airdropsEnabled = true;
		boolean bossEnabled;
		int maxGroundY;
		BlockPos cratePos;
		boolean bossSpawned;
		UUID activeBoss;
		int airdropTicks;
		int crateTicks;
		int bossRetryTicks;
		String displayName = "";
		int durationMinutes = 15;
		final Set<BlockPos> pmcSpawns = new LinkedHashSet<>();
		final Set<BlockPos> scavSpawns = new LinkedHashSet<>();
		final Map<String, Extract> extracts = new LinkedHashMap<>();
		final Set<BlockPos> lootPoints = new LinkedHashSet<>();

		Arena(BlockPos anchor) {
			this.anchor = anchor;
			this.maxGroundY = anchor.getY() + 8;
		}

		void setCorner(int index, BlockPos pos) {
			corners[index] = pos.toImmutable();
			rebuildPolygon();
		}

		private void rebuildPolygon() {
			polygon = List.of();
			if (Arrays.stream(corners).anyMatch(p -> p == null)) return;
			double centerX = Arrays.stream(corners).mapToInt(BlockPos::getX).average().orElseThrow();
			double centerZ = Arrays.stream(corners).mapToInt(BlockPos::getZ).average().orElseThrow();
			List<BlockPos> ordered = Arrays.stream(corners)
				.sorted(Comparator.comparingDouble(p -> Math.atan2(p.getZ() - centerZ, p.getX() - centerX)))
				.toList();
			for (int i = 0; i < ordered.size(); i++) {
				BlockPos a = ordered.get(i);
				BlockPos b = ordered.get((i + 1) % ordered.size());
				BlockPos c = ordered.get((i + 2) % ordered.size());
				if (cross(a, b, c.getX(), c.getZ()) == 0) return;
			}
			polygon = ordered;
		}

		private static long cross(BlockPos a, BlockPos b, int x, int z) {
			return ((long) b.getX() - a.getX()) * ((long) z - a.getZ())
				- ((long) b.getZ() - a.getZ()) * ((long) x - a.getX());
		}

		public BlockPos center() {
			if (Arrays.stream(corners).anyMatch(p -> p == null)) return anchor;
			long x = 0;
			long z = 0;
			for (BlockPos corner : corners) {
				x += corner.getX();
				z += corner.getZ();
			}
			return new BlockPos((int) Math.round(x / 4.0), anchor.getY(), (int) Math.round(z / 4.0));
		}

		public boolean ready() {
			return !polygon.isEmpty();
		}

		public boolean contains(int x, int z) {
			return contains((double) x, z);
		}

		public boolean contains(double x, double z) {
			if (!ready()) return false;
			boolean inside = false;
			for (int i = 0; i < polygon.size(); i++) {
				BlockPos a = polygon.get(i);
				BlockPos b = polygon.get((i + 1) % polygon.size());
				double side = (b.getX() - (double) a.getX()) * (z - a.getZ())
					- (b.getZ() - (double) a.getZ()) * (x - a.getX());
				if (Math.abs(side) < 1.0e-8 && between(x, a.getX(), b.getX()) && between(z, a.getZ(), b.getZ())) return true;
				if ((a.getZ() > z) != (b.getZ() > z) && (side > 0) == (b.getZ() > a.getZ())) inside = !inside;
			}
			return inside;
		}

		private static boolean between(double value, int first, int second) {
			return value >= Math.min(first, second) && value <= Math.max(first, second);
		}

		public List<BlockPos> corners() {
			return polygon;
		}

		public Set<BlockPos> airdropPoints() {
			return Set.copyOf(airdropPoints);
		}

		public boolean airdropsEnabled() {
			return airdropsEnabled;
		}

		public int maxGroundY() {
			return maxGroundY;
		}

		NbtCompound writeNbt() {
			NbtCompound tag = new NbtCompound();
			tag.putLong("Anchor", anchor.asLong());
			for (int i = 0; i < corners.length; i++) {
				if (corners[i] != null) tag.putLong("Corner" + i, corners[i].asLong());
			}
			tag.putLongArray("AirdropPoints", airdropPoints.stream().mapToLong(BlockPos::asLong).toArray());
			tag.putLongArray("Poi", pointsOfInterest.stream().mapToLong(BlockPos::asLong).toArray());
			tag.putBoolean("AirdropsEnabled", airdropsEnabled);
			tag.putInt("MaxGroundY", maxGroundY);
			tag.putBoolean("BossEnabled", bossEnabled);
			tag.putString("DisplayName", displayName);
			tag.putInt("DurationMinutes", durationMinutes);
			tag.putLongArray("PmcSpawns", pmcSpawns.stream().mapToLong(BlockPos::asLong).toArray());
			tag.putLongArray("ScavSpawns", scavSpawns.stream().mapToLong(BlockPos::asLong).toArray());
			tag.putLongArray("LootPoints", lootPoints.stream().mapToLong(BlockPos::asLong).toArray());
			NbtCompound exits = new NbtCompound();
			extracts.forEach((name, extract) -> {
				NbtCompound exit = new NbtCompound();
				exit.putLong("Pos", extract.pos().asLong());
				exit.putBoolean("Loud", extract.loud());
				exits.put(name, exit);
			});
			tag.put("Extracts", exits);
			tag.putBoolean("BossSpawned", bossSpawned);
			if (activeBoss != null) tag.putUuid("ActiveBoss", activeBoss);
			if (cratePos != null) tag.putLong("CratePos", cratePos.asLong());
			return tag;
		}

		static Arena readNbt(NbtCompound tag) {
			Arena arena = new Arena(BlockPos.fromLong(tag.getLong("Anchor")));
			for (int i = 0; i < arena.corners.length; i++) {
				if (tag.contains("Corner" + i)) arena.corners[i] = BlockPos.fromLong(tag.getLong("Corner" + i));
			}
			arena.rebuildPolygon();
			for (long point : tag.getLongArray("AirdropPoints")) arena.airdropPoints.add(BlockPos.fromLong(point));
			for (long point : tag.getLongArray("Poi")) arena.pointsOfInterest.add(BlockPos.fromLong(point));
			arena.airdropsEnabled = !tag.contains("AirdropsEnabled") || tag.getBoolean("AirdropsEnabled");
			if (tag.contains("MaxGroundY")) arena.maxGroundY = tag.getInt("MaxGroundY");
			arena.bossEnabled = tag.getBoolean("BossEnabled");
			arena.displayName = tag.getString("DisplayName");
			if (tag.contains("DurationMinutes")) arena.durationMinutes = tag.getInt("DurationMinutes");
			for (long point : tag.getLongArray("PmcSpawns")) arena.pmcSpawns.add(BlockPos.fromLong(point));
			for (long point : tag.getLongArray("ScavSpawns")) arena.scavSpawns.add(BlockPos.fromLong(point));
			for (long point : tag.getLongArray("LootPoints")) arena.lootPoints.add(BlockPos.fromLong(point));
			NbtCompound exits = tag.getCompound("Extracts");
			for (String name : exits.getKeys()) {
				NbtCompound exit = exits.getCompound(name);
				arena.extracts.put(name, new Extract(BlockPos.fromLong(exit.getLong("Pos")), exit.getBoolean("Loud")));
			}
			arena.bossSpawned = tag.getBoolean("BossSpawned");
			if (tag.containsUuid("ActiveBoss")) arena.activeBoss = tag.getUuid("ActiveBoss");
			if (tag.contains("CratePos")) arena.cratePos = BlockPos.fromLong(tag.getLong("CratePos"));
			return arena;
		}
	}

	record Extract(BlockPos pos, boolean loud) { }

final class ArenaResetService {
	private ArenaResetService() { }

	static BlockPos openable(BlockPos pos, BlockState state) {
		if (state.getBlock() instanceof DoorBlock) {
			return state.get(Properties.DOUBLE_BLOCK_HALF) == DoubleBlockHalf.UPPER ? pos.down() : pos.toImmutable();
		}
		return state.getBlock() instanceof TrapdoorBlock || state.getBlock() instanceof FenceGateBlock
			? pos.toImmutable() : null;
	}

	static void reset(ServerWorld world, ArenaEventsState.Arena arena, Set<BlockPos> interactedOpenables) {
		for (BlockPos pos : interactedOpenables) {
			BlockState state = world.getBlockState(pos);
			if (openable(pos, state) != null && state.get(Properties.OPEN)) {
				world.setBlockState(pos, state.with(Properties.OPEN, false), Block.NOTIFY_ALL);
			}
		}
		interactedOpenables.clear();

		var state = ArenaEventsState.get(world);
		if (arena.cratePos != null) {
			if (!HackableCrateEvent.clearRetiredCrate(world, arena.cratePos)) state.retiredCrates.add(arena.cratePos);
			arena.cratePos = null;
		}
		var containers = new java.util.HashSet<String>();
		for (BlockPos pos : arena.lootPoints) {
			if (!arena.contains(pos.getX() + 0.5, pos.getZ() + 0.5)) continue;
			if (world.getBlockEntity(pos) instanceof SearchableLootBlockEntity crate) {
				if (!crate.isEventCrate()) crate.refillLoot(world.random);
			} else {
				var blockState = world.getBlockState(pos);
				if (VanillaContainerBridge.isSupportedBlock(blockState.getBlock())
					&& containers.add(VanillaContainerBridge.resolveCanonicalKey(pos, blockState, new UUID(0, 0)))) {
					VanillaContainerBridge.refillArenaContainer(world, pos);
				}
			}
		}
		var obsolete = new ArrayList<Entity>();
		for (Entity entity : world.iterateEntities()) {
			if ((entity instanceof CorpseEntity || entity instanceof SleeperPlayerEntity || entity instanceof ItemEntity
				|| entity instanceof AirdropEntity || entity instanceof MobEntity)
				&& arena.contains(entity.getX(), entity.getZ())) obsolete.add(entity);
		}
		obsolete.forEach(Entity::discard);
		arena.airdropTicks = 0;
		arena.crateTicks = 0;
		arena.bossRetryTicks = 0;
		arena.bossSpawned = false;
		arena.activeBoss = null;
		state.markDirty();
	}
}

package com.tarkovcraft.arena;

import com.tarkovcraft.block.SearchableLootBlockEntity;
import com.tarkovcraft.loot.TarkovLootPool;
import com.tarkovcraft.registry.ModBlocks;
import net.minecraft.block.Block;
import net.minecraft.server.network.ServerPlayerEntity;
import net.minecraft.server.world.ServerWorld;
import net.minecraft.sound.SoundCategory;
import net.minecraft.sound.SoundEvents;
import net.minecraft.text.Text;
import net.minecraft.util.hit.HitResult;
import net.minecraft.util.math.BlockPos;
import net.minecraft.world.RaycastContext;

import java.util.List;
import java.util.UUID;

public final class HackableCrateEvent {
	private static final int HACK_TICKS = 45 * 20;
	private UUID hacker;
	private int remaining;

	public static boolean activate(ServerWorld world, ArenaEventsState.Arena arena, List<ServerPlayerEntity> players) {
		BlockPos oldPos = arena.cratePos;
		boolean oldChunkLoaded = oldPos != null && world.isChunkLoaded(oldPos.getX() >> 4, oldPos.getZ() >> 4);
		SearchableLootBlockEntity oldCrate = oldChunkLoaded
			&& world.getBlockEntity(oldPos) instanceof SearchableLootBlockEntity crate && crate.isEventCrate() ? crate : null;
		if (oldCrate != null && oldCrate.isHackInProgress()) return false;

		List<BlockPos> points = arena.pointsOfInterest.stream()
			.map(pos -> crateAtPoi(world, arena, pos, oldCrate)).filter(pos -> pos != null).distinct().toList();
		BlockPos pos = points.isEmpty() ? findCentralSite(world, arena, oldCrate) : points.get(world.random.nextInt(points.size()));
		if (pos == null) return false;
		if (pos.equals(oldPos) && oldCrate != null) world.removeBlock(oldPos, false);
		if (!world.setBlockState(pos, ModBlocks.AIRDROP_CRATE.getDefaultState(), Block.NOTIFY_ALL)) return false;
		if (!(world.getBlockEntity(pos) instanceof SearchableLootBlockEntity crate)) return false;
		crate.lockForEvent();
		if (oldCrate != null && !pos.equals(oldPos)) world.removeBlock(oldPos, false);
		ArenaEventsState state = ArenaEventsState.get(world);
		if (oldPos != null && !oldChunkLoaded) state.retiredCrates.add(oldPos);
		arena.cratePos = pos;
		state.markDirty();

		world.playSound(null, pos, SoundEvents.BLOCK_BELL_USE, SoundCategory.BLOCKS, 5.0f, 0.55f);
		Text announcement = Text.translatable("message.tarkovcraft.hack_crate_activated", pos.getX(), pos.getY(), pos.getZ())
			.append(". ").append(Text.translatable("message.tarkovcraft.hack_crate_prompt"));
		players.forEach(player -> player.sendMessage(announcement, false));
		return true;
	}

	static boolean clearRetiredCrate(ServerWorld world, BlockPos pos) {
		if (!world.isChunkLoaded(pos.getX() >> 4, pos.getZ() >> 4)) return false;
		if (world.getBlockEntity(pos) instanceof SearchableLootBlockEntity crate && crate.isEventCrate()) {
			world.removeBlock(pos, false);
		}
		return true;
	}

	public void start(ServerWorld world, BlockPos pos, ServerPlayerEntity player) {
		if (!canHack(world, pos, player)) return;
		if (hacker != null && !hacker.equals(player.getUuid())) {
			player.sendMessage(Text.translatable("message.tarkovcraft.hack_crate_busy"), true);
			return;
		}
		if (hacker == null) {
			hacker = player.getUuid();
			remaining = HACK_TICKS;
			reportProgress(world, pos, player);
		}
	}

	public boolean isInProgress() {
		return hacker != null;
	}

	public void tick(ServerWorld world, BlockPos pos, SearchableLootBlockEntity crate) {
		if (hacker == null) return;
		ServerPlayerEntity player = world.getServer().getPlayerManager().getPlayer(hacker);
		if (player == null || !canHack(world, pos, player)) {
			hacker = null;
			if (player != null) player.sendMessage(Text.translatable("message.tarkovcraft.hack_crate_interrupted"), true);
			return;
		}
		if (--remaining == 0) {
			hacker = null;
			crate.unlockEventCrate(world.random);
			world.playSound(null, pos, SoundEvents.BLOCK_NOTE_BLOCK_PLING.value(), SoundCategory.BLOCKS, 0.9f, 1.25f);
			player.sendMessage(Text.translatable("message.tarkovcraft.hack_crate_complete"), true);
		} else if (remaining % 20 == 0) {
			reportProgress(world, pos, player);
		}
	}

	private void reportProgress(ServerWorld world, BlockPos pos, ServerPlayerEntity player) {
		player.sendMessage(Text.translatable("message.tarkovcraft.hack_crate_progress", remaining / 20), true);
		world.playSound(null, pos, SoundEvents.BLOCK_NOTE_BLOCK_BIT.value(), SoundCategory.BLOCKS, 0.8f, 1.2f);
	}

	private static boolean canHack(ServerWorld world, BlockPos pos, ServerPlayerEntity player) {
		if (player.getServerWorld() != world || !player.isAlive() || player.isSpectator()
			|| player.squaredDistanceTo(pos.toCenterPos()) > 4.5 * 4.5) return false;
		var hit = world.raycast(new RaycastContext(player.getEyePos(), pos.toCenterPos(),
			RaycastContext.ShapeType.COLLIDER, RaycastContext.FluidHandling.NONE, player));
		return hit.getType() == HitResult.Type.BLOCK && hit.getBlockPos().equals(pos);
	}

	private static BlockPos crateAtPoi(ServerWorld world, ArenaEventsState.Arena arena, BlockPos pos, SearchableLootBlockEntity old) {
		if (ArenaEvents.canPlaceAt(world, arena, pos, old)) return pos;
		if (ArenaEvents.canPlaceAt(world, arena, pos.up(), old)) return pos.up();
		return null;
	}

	private static BlockPos findCentralSite(ServerWorld world, ArenaEventsState.Arena arena, SearchableLootBlockEntity old) {
		BlockPos center = arena.center();
		for (int attempt = 0; attempt < 36; attempt++) {
			BlockPos column = attempt == 0 ? center : center.add(world.random.nextInt(25) - 12, 0, world.random.nextInt(25) - 12);
			for (int dy = 0; dy <= 8; dy++) {
				BlockPos down = column.down(dy);
				if (ArenaEvents.canPlaceAt(world, arena, down, old)) return down;
				BlockPos up = column.up(dy);
				if (dy != 0 && ArenaEvents.canPlaceAt(world, arena, up, old)) return up;
			}
		}
		return null;
	}
}

			|| path.contains("reload") || path.contains("search") || path.contains("grass") || path.contains("bush");
	}

		return path.contains("explode") || path.contains("explosion") || path.contains("shoot")
			|| path.contains("gunshot") || path.contains("firework") || path.contains("grenade");
	}

		return stack.getItem() instanceof TarkovArmorItem armor ? armor.getHearingMultiplier() : 1.0f;

		this.containerTitleKey = containerTitleKey;
		this.defaultLootPool = defaultLootPool != null ? defaultLootPool : TarkovLootPool.MILITARY;
	}

	public int getGridWidth() {
		return gridWidth;
	}

	public int getGridHeight() {
		return gridHeight;
	}

	public String getContainerTitleKey() {
		return containerTitleKey;
	}

	public TarkovLootPool getDefaultLootPool() {
		return defaultLootPool;
	}

	/**
	 * Creates a placeable loot crate ItemStack configured for the given TarkovLootPool.
	 * When placed in the world, loot generates lazily on first open so map builders can reuse/copy crates.
	 */
	public static ItemStack createLootCrateStack(TarkovLootPool pool) {
		SearchableLootBlock block = defaultBlockForPool(pool);
		return createLootCrateStack(block, pool, pool.getDefaultGridWidth(), pool.getDefaultGridHeight());
	}

	public static ItemStack createLootCrateStack(
		SearchableLootBlock block,
		TarkovLootPool pool,
		int width,
		int height
	) {
		SearchableLootBlock targetBlock = block != null ? block : defaultBlockForPool(pool);
		int clampedW = Math.max(1, Math.min(10, width > 0 ? width : targetBlock.getGridWidth()));
		int clampedH = Math.max(1, Math.min(10, height > 0 ? height : targetBlock.getGridHeight()));
		ItemStack stack = new ItemStack(targetBlock, 1);
		NbtCompound beTag = new NbtCompound();
		beTag.putString("LootPool", pool.getId());
		beTag.putBoolean("PendingLoot", true);
		beTag.putInt("DefaultW", clampedW);
		beTag.putInt("DefaultH", clampedH);
		BlockItem.setBlockEntityNbt(stack, ModBlocks.LOOT_CONTAINER_ENTITY, beTag);
		stack.setCustomName(
			Text.translatable("item.tarkovcraft.loot_crate_preset", Text.translatable(pool.getTranslationKey()))
				.styled(s -> s.withItalic(false).withColor(Formatting.GOLD))
		);
		return stack;
	}

	public static SearchableLootBlock defaultBlockForPool(TarkovLootPool pool) {
		if (pool == null) {
			return ModBlocks.TECHNICAL_CRATE;
		}
		return switch (pool.getDefaultBlockKey()) {
			case "weapon_crate" -> ModBlocks.WEAPON_CRATE;
			case "medbag" -> ModBlocks.MEDBAG;
			case "airdrop_crate" -> ModBlocks.AIRDROP_CRATE;
			default -> ModBlocks.TECHNICAL_CRATE;
		};
	}

	@Override
	public void appendTooltip(ItemStack stack, @Nullable BlockView world, List<Text> tooltip, TooltipContext options) {
		super.appendTooltip(stack, world, tooltip, options);
		NbtCompound beTag = BlockItem.getBlockEntityNbt(stack);
		TarkovLootPool pool = null;
		int w = gridWidth;
		int h = gridHeight;
		boolean pending = false;
		if (beTag != null) {
			if (beTag.contains("LootPool", NbtElement.STRING_TYPE)) {
				pool = TarkovLootPool.byId(beTag.getString("LootPool"));
			}
			if (beTag.contains("DefaultW", NbtElement.NUMBER_TYPE)) {
				w = Math.max(1, Math.min(10, beTag.getInt("DefaultW")));
			}
			if (beTag.contains("DefaultH", NbtElement.NUMBER_TYPE)) {
				h = Math.max(1, Math.min(10, beTag.getInt("DefaultH")));
			}
			pending = beTag.getBoolean("PendingLoot");
		}
		if (pool != null) {
			tooltip.add(Text.translatable("tooltip.tarkovcraft.crate_pool", Text.translatable(pool.getTranslationKey()))
				.formatted(Formatting.YELLOW));
		}
		tooltip.add(Text.translatable("tooltip.tarkovcraft.crate_size", w, h).formatted(Formatting.GRAY));
		if (pending) {
			tooltip.add(Text.translatable("tooltip.tarkovcraft.crate_pending").formatted(Formatting.DARK_GRAY));
		}
	}

	@Override
	public BlockRenderType getRenderType(BlockState state) {
		return BlockRenderType.MODEL;
	}

	@Nullable
	@Override
	public BlockEntity createBlockEntity(BlockPos pos, BlockState state) {
		return new SearchableLootBlockEntity(ModBlocks.LOOT_CONTAINER_ENTITY, pos, state, gridWidth, gridHeight);
	}

	@Nullable
	@Override
	public <T extends BlockEntity> BlockEntityTicker<T> getTicker(World world, BlockState state, BlockEntityType<T> type) {
		return world.isClient ? null : checkType(type, ModBlocks.LOOT_CONTAINER_ENTITY, SearchableLootBlockEntity::serverTick);
	}

	@Override
	public ActionResult onUse(BlockState state, World world, BlockPos pos, PlayerEntity player, Hand hand, BlockHitResult hit) {
		// Opening loot containers is strictly via [F] interact, NOT right-click
		return ActionResult.PASS;
	}
}

public class SearchableLootBlockEntity extends BlockEntity {

	private int defaultWidth;
	private int defaultHeight;
	private GridContainer grid;
	private ContainerSearchMemory searchMemory = new ContainerSearchMemory();

	private TarkovLootPool lootPool = null;
	private boolean pendingLootGeneration = false;
	private long lastLootEpochMs = 0L;
	private boolean autoRespawn = true;
	private boolean eventCrate;
	@Nullable
	private HackableCrateEvent hack;

	public SearchableLootBlockEntity(BlockPos pos, BlockState state) {
		this(ModBlocks.LOOT_CONTAINER_ENTITY, pos, state,
			state.getBlock() instanceof SearchableLootBlock b ? b.getGridWidth() : 4,

	public SearchableLootBlockEntity(BlockEntityType<?> type, BlockPos pos, BlockState state, int width, int height) {

		this.defaultHeight = Math.max(1, height);
		this.grid = new GridContainer(this.defaultWidth, this.defaultHeight);
	}

	public GridContainer getGrid() {
		return grid;
	}

	public ContainerSearchMemory getSearchMemory() {
		return searchMemory;
	}

	public SearchTarget asSearchTarget() {
		return new SearchTarget(grid.getContainerId(), List.of(grid), searchMemory, this::markDirty);
	}

	public int getSmokeTicksRemaining() {
		return smokeTicksRemaining;
	}

	public void setSmokeTicksRemaining(int smokeTicksRemaining) {
		this.smokeTicksRemaining = Math.max(0, smokeTicksRemaining);
		markDirty();
	}

	@Nullable
	public TarkovLootPool getLootPool() {
		return lootPool;
	}

	public TarkovLootPool getEffectiveLootPool() {
		if (lootPool != null) {
			return lootPool;
		}
		if (getCachedState().getBlock() instanceof SearchableLootBlock lootBlock) {
			return lootBlock.getDefaultLootPool();
		}
		return TarkovLootPool.MILITARY;
	}

	public void setLootPool(@Nullable TarkovLootPool lootPool) {

		this.defaultWidth = clampedW;

	public void populateFromPool(TarkovLootPool pool, Random random) {
		TarkovLootPool effective = pool != null ? pool : getEffectiveLootPool();
		this.lootPool = effective;
		this.pendingLootGeneration = false;
		this.lastLootEpochMs = System.currentTimeMillis();
		this.grid.clear();
		this.searchMemory = new ContainerSearchMemory();
		TarkovLootGenerator.populateLootGrid(this.grid, effective, random);
		markDirty();
	}

	public long getLastLootEpochMs() {
		return lastLootEpochMs;
	}

		this.lastLootEpochMs = Math.max(0L, lastLootEpochMs);

		if (hack == null) return;
		populateFromPool(TarkovLootPool.AIRDROP, random);
		hack = null;
		markDirty();
	}

	public static boolean isRespawnDue(long lastLootEpochMs) {
		long intervalMs = TarkovConfig.get().getLootRespawnMillis();
		if (intervalMs <= 0L || lastLootEpochMs <= 0L) {
			return false;
		}
		return (System.currentTimeMillis() - lastLootEpochMs) >= intervalMs;
	}

	public void refillLoot(Random random) {
		populateFromPool(getEffectiveLootPool(), random);
	}

	public static void serverTick(World world, BlockPos pos, BlockState state, SearchableLootBlockEntity be) {
		if (!(world instanceof ServerWorld sw)) return;
		if (be.hack != null) be.hack.tick(sw, pos, be);
		if (be.smokeTicksRemaining <= 0) return;
		be.smokeTicksRemaining--;
		if (be.smokeTicksRemaining % 3 == 0) {
			double cx = pos.getX() + 0.5;
			double cy = pos.getY() + 1.05;
			double cz = pos.getZ() + 0.5;
			double pillarHeight = (be.smokeTicksRemaining % 12) * 0.9;
			sw.spawnParticles(ParticleTypes.CAMPFIRE_SIGNAL_SMOKE, cx, cy, cz, 2, 0.15, 0.2, 0.15, 0.06);
			sw.spawnParticles(RED_FLARE_DUST, cx, cy + 0.15 + pillarHeight, cz, 4, 0.25, 0.35, 0.25, 0.01);
		}
		if (be.smokeTicksRemaining == 0) {
			be.markDirty();
		}
	}

	public Text getDisplayTitle() {
		if (lootPool != null) {
			return Text.translatable("container.tarkovcraft.loot_crate_typed", Text.translatable(lootPool.getTranslationKey()));
		}
		if (getCachedState().getBlock() instanceof SearchableLootBlock lootBlock) {
			return Text.translatable(lootBlock.getContainerTitleKey());
		}
		return Text.translatable("container.tarkovcraft.loot_crate");
	}

	public void openForPlayer(ServerPlayerEntity player) {
		if (hack != null && world instanceof ServerWorld serverWorld) {
			hack.start(serverWorld, pos, player);
			return;
		}
		if (world != null) {
			if (pendingLootGeneration) {
				populateFromPool(getEffectiveLootPool(), world.getRandom());
			} else if (autoRespawn && isRespawnDue(lastLootEpochMs)) {
				populateFromPool(getEffectiveLootPool(), world.getRandom());
			}
		}
		if (smokeTicksRemaining > 0) {
			smokeTicksRemaining = 0;
			markDirty();
		}
		player.openHandledScreen(new ExtendedScreenHandlerFactory() {
			@Override
			public void writeScreenOpeningData(ServerPlayerEntity player, PacketByteBuf buf) {
				buf.writeByte(TarkovScreenHandler.MODE_LOOT_BLOCK);
				buf.writeBlockPos(pos);
			}

			@Override
			public Text getDisplayName() {
				return getDisplayTitle();
			}

			@Nullable
			@Override
			public ScreenHandler createMenu(int syncId, PlayerInventory inv, PlayerEntity p) {
				return TarkovScreenHandler.forLootBlock(syncId, inv, SearchableLootBlockEntity.this);
			}
		});
	}

	@Override
	protected void writeNbt(NbtCompound nbt) {
		super.writeNbt(nbt);
		nbt.putInt("DefaultW", defaultWidth);
		nbt.putInt("DefaultH", defaultHeight);
		nbt.put("Grid", grid.toNbt());
		nbt.put("SearchMemory", searchMemory.toNbt());
		nbt.putInt("SmokeTicks", smokeTicksRemaining);
		if (lootPool != null) {
			nbt.putString("LootPool", lootPool.getId());
		}
		if (pendingLootGeneration) {
			nbt.putBoolean("PendingLoot", true);
		}
		if (lastLootEpochMs > 0L) {
			nbt.putLong("LastLootEpochMs", lastLootEpochMs);
		}
		nbt.putBoolean("AutoRespawn", autoRespawn);
		if (eventCrate) {
			nbt.putBoolean("EventCrate", true);
			nbt.putBoolean("HackLocked", hack != null);
		}
	}

	@Override
	public void readNbt(NbtCompound nbt) {
		super.readNbt(nbt);
		if (nbt.contains("DefaultW")) {
			this.defaultWidth = Math.max(1, Math.min(10, nbt.getInt("DefaultW")));
		}
		if (nbt.contains("DefaultH")) {
			this.defaultHeight = Math.max(1, Math.min(10, nbt.getInt("DefaultH")));
		}
		if (nbt.contains("Grid", NbtElement.COMPOUND_TYPE)) {
			this.grid = GridContainer.fromNbt(nbt.getCompound("Grid"), defaultWidth, defaultHeight);
		} else {
			this.grid = new GridContainer(defaultWidth, defaultHeight);
		}
		if (nbt.contains("SearchMemory", NbtElement.COMPOUND_TYPE)) {
			this.searchMemory = ContainerSearchMemory.fromNbt(nbt.getCompound("SearchMemory"));
		} else {
			this.searchMemory = new ContainerSearchMemory();
		}
		this.smokeTicksRemaining = Math.max(0, nbt.getInt("SmokeTicks"));
		if (nbt.contains("LootPool", NbtElement.STRING_TYPE)) {
			this.lootPool = TarkovLootPool.byId(nbt.getString("LootPool"));
		} else {
			this.lootPool = null;
		}
		this.pendingLootGeneration = nbt.getBoolean("PendingLoot");
		this.lastLootEpochMs = Math.max(0L, nbt.getLong("LastLootEpochMs"));
		this.autoRespawn = !nbt.contains("AutoRespawn") || nbt.getBoolean("AutoRespawn");
		this.eventCrate = nbt.getBoolean("EventCrate");
		this.hack = eventCrate && nbt.getBoolean("HackLocked") ? new HackableCrateEvent() : null;
	}
}

package com.tarkovcraft.client;

import com.tarkovcraft.ammo.AmmoTier;

import net.minecraft.util.Identifier;
import org.jetbrains.annotations.Nullable;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.UUID;

/**
 * Client-only sender helpers for TarkovCraft C2S packets introduced in the EFT UI refactor.
 * Keeps client networking imports strictly inside com.tarkovcraft.client.*.
 */
@Environment(EnvType.CLIENT)
public final class ClientNetworkSender {
	private ClientNetworkSender() {
	}

	public static void sendQuickEquip(byte srcLoc, int srcIdx, @Nullable UUID srcEntryId) {
		ClientPlayNetworking.send(TarkovNetworking.C2S_QUICK_EQUIP, buildSourceBuf(srcLoc, srcIdx, srcEntryId));
	}

	public static void sendUseItem(byte srcLoc, int srcIdx, @Nullable UUID srcEntryId) {
		ClientPlayNetworking.send(TarkovNetworking.C2S_USE_ITEM, buildSourceBuf(srcLoc, srcIdx, srcEntryId));
	}

	public static void sendSplitStack(byte srcLoc, int srcIdx, @Nullable UUID srcEntryId, int splitAmount) {
		PacketByteBuf buf = buildSourceBuf(srcLoc, srcIdx, srcEntryId);
		buf.writeVarInt(splitAmount);
		ClientPlayNetworking.send(TarkovNetworking.C2S_SPLIT_STACK, buf);
	}

	public static void sendRequestSearch(UUID targetId) {
		PacketByteBuf buf = PacketByteBufs.create();
		buf.writeUuid(targetId);
		ClientPlayNetworking.send(TarkovNetworking.C2S_REQUEST_SEARCH, buf);
	}

	public static void sendHealPart(byte srcLoc, int srcIdx, @Nullable UUID srcEntryId, @Nullable BodyPart part) {
		PacketByteBuf buf = buildSourceBuf(srcLoc, srcIdx, srcEntryId);
		buf.writeByte(part == null ? -1 : part.ordinal());
		ClientPlayNetworking.send(TarkovNetworking.C2S_HEAL_PART, buf);
	}

	public static void sendOpenCase(byte srcLoc, int srcIdx, @Nullable UUID srcEntryId) {
		ClientPlayNetworking.send(TarkovNetworking.C2S_OPEN_CASE, buildSourceBuf(srcLoc, srcIdx, srcEntryId));
	}

	public static void sendCloseCase() {
		PacketByteBuf buf = PacketByteBufs.create();
		buf.writeBoolean(false);
		ClientPlayNetworking.send(TarkovNetworking.C2S_CLOSE_CASE, buf);
	}

	public static void sendCloseCase(UUID containerUuid) {
		PacketByteBuf buf = PacketByteBufs.create();
		buf.writeBoolean(containerUuid != null);
		if (containerUuid != null) {
			buf.writeUuid(containerUuid);
		}
		ClientPlayNetworking.send(TarkovNetworking.C2S_CLOSE_CASE, buf);
	}

	public static void sendWorkbenchCraftAmmo(String ammoId, AmmoTier tier, boolean upgradeFromFmj) {
		PacketByteBuf buf = PacketByteBufs.create();
		buf.writeString(ammoId, 64);
		buf.writeByte(tier.ordinal());
		buf.writeBoolean(upgradeFromFmj);
		ClientPlayNetworking.send(TarkovNetworking.C2S_WORKBENCH_CRAFT_AMMO, buf);
	}

	public static void sendWorkbenchCraftGear(Identifier targetItemId) {
		PacketByteBuf buf = PacketByteBufs.create();
		buf.writeIdentifier(targetItemId);
		ClientPlayNetworking.send(TarkovNetworking.C2S_WORKBENCH_CRAFT_GEAR, buf);
	}

	public static void registerWorkbenchClientReceiver() {
		ClientPlayNetworking.registerGlobalReceiver(TarkovNetworking.S2C_WORKBENCH_SYNC, (client, handler, buf, responseSender) -> {
			int itemCount = buf.readVarInt();
			Map<Identifier, Integer> items = new HashMap<>(itemCount);
			for (int i = 0; i < itemCount; i++) {
				items.put(buf.readIdentifier(), buf.readVarInt());
			}
			int fmjCount = buf.readVarInt();
			Map<String, Integer> fmj = new HashMap<>(fmjCount);
			for (int i = 0; i < fmjCount; i++) {
				fmj.put(buf.readString(64), buf.readVarInt());
			}
			int extraCount = buf.readVarInt();
			List<String> extras = new ArrayList<>(extraCount);
			for (int i = 0; i < extraCount; i++) {
				extras.add(buf.readString(64));
			}
			client.execute(() -> ClientWorkbenchCache.update(items, fmj, extras));
		});
	}

	private static PacketByteBuf buildSourceBuf(byte srcLoc, int srcIdx, @Nullable UUID srcEntryId) {
		PacketByteBuf buf = PacketByteBufs.create();
		buf.writeByte(srcLoc);
		buf.writeVarInt(srcIdx);
		buf.writeBoolean(srcEntryId != null);
		if (srcEntryId != null) {
			buf.writeUuid(srcEntryId);
		}
		return buf;
	}
}

package com.tarkovcraft.client;

import net.fabricmc.api.EnvType;
import net.fabricmc.api.Environment;
import net.minecraft.item.Item;
import net.minecraft.registry.Registries;
import net.minecraft.util.Identifier;

import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

/**
 * Client-side cache of workbench crafting material counts, FMJ ammo stocks, and extra gunpack calibers.
 */
@Environment(EnvType.CLIENT)
public final class ClientWorkbenchCache {
	private static Map<Identifier, Integer> itemCounts = new HashMap<>();
	private static Map<String, Integer> fmjAmmoCounts = new HashMap<>();
	private static List<String> extraCalibers = List.of();
	private static int revision = 0;

	private ClientWorkbenchCache() {
	}

	public static int getRevision() {
		return revision;
	}

	public static void update(
		Map<Identifier, Integer> items,
		Map<String, Integer> fmjCounts,
		List<String> extras
	) {
		itemCounts = new HashMap<>(items);
		fmjAmmoCounts = new HashMap<>(fmjCounts);
		extraCalibers = List.copyOf(extras);
		revision++;
	}

	public static int getItemCount(Item item) {
		if (item == null) return 0;
		return itemCounts.getOrDefault(Registries.ITEM.getId(item), 0);
	}

	public static int getFmjCount(String ammoId) {
		if (ammoId == null) return 0;
		return fmjAmmoCounts.getOrDefault(ammoId, 0);
	}

	public static List<String> getExtraCalibers() {
		return Collections.unmodifiableList(extraCalibers);
	}
}

package com.tarkovcraft.client.compat;

/**
 * Full EMI integration for TarkovCraft:
 * 1. Workbench Gear recipes (Helmets, Armor, Masks, Plates, Rigs, Backpacks)
 * 2. Workbench Tiered Ammo recipes (Scratch & FMJ -> HP/AP/IGN upgrades)
 * 3. EMI-Loot-style City & Crate Loot Tables for all 16 TarkovLootPools
 * 4. Extra EMI sidebar stacks for tiered TaCZ ammo & preconfigured city loot crates
 */
@Environment(EnvType.CLIENT)
public final class TarkovEmiPlugin implements EmiPlugin {

	public static final EmiStack WORKBENCH_ICON = EmiStack.of(ModBlocks.TARKOV_WORKBENCH);
	public static final EmiStack CRATE_ICON = EmiStack.of(ModBlocks.WEAPON_CRATE);

	public static final EmiRecipeCategory WORKBENCH_GEAR_CATEGORY = new EmiRecipeCategory(
		new Identifier(TarkovCraftMod.MOD_ID, "workbench_gear"),
		WORKBENCH_ICON
	);
	public static final EmiRecipeCategory WORKBENCH_AMMO_CATEGORY = new EmiRecipeCategory(
		new Identifier(TarkovCraftMod.MOD_ID, "workbench_ammo"),
		WORKBENCH_ICON
	);
	public static final EmiRecipeCategory LOOT_TABLES_CATEGORY = new EmiRecipeCategory(
		new Identifier(TarkovCraftMod.MOD_ID, "loot_tables"),
		CRATE_ICON
	);

	@Override
	public void register(EmiRegistry registry) {
		registry.addCategory(WORKBENCH_GEAR_CATEGORY);
		registry.addCategory(WORKBENCH_AMMO_CATEGORY);
		registry.addCategory(LOOT_TABLES_CATEGORY);

		registry.addWorkstation(WORKBENCH_GEAR_CATEGORY, WORKBENCH_ICON);
		registry.addWorkstation(WORKBENCH_AMMO_CATEGORY, WORKBENCH_ICON);
		registry.addWorkstation(LOOT_TABLES_CATEGORY, EmiStack.of(ModBlocks.WEAPON_CRATE));
		registry.addWorkstation(LOOT_TABLES_CATEGORY, EmiStack.of(ModBlocks.MEDBAG));
		registry.addWorkstation(LOOT_TABLES_CATEGORY, EmiStack.of(ModBlocks.TECHNICAL_CRATE));
		registry.addWorkstation(LOOT_TABLES_CATEGORY, EmiStack.of(ModBlocks.AIRDROP_CRATE));

		registerGearRecipes(registry);
		registerAmmoRecipes(registry);
		registerLootTableRecipes(registry);
	}

	private static void registerGearRecipes(EmiRegistry registry) {
		for (TarkovWorkbenchRecipes.GearCategory cat : TarkovWorkbenchRecipes.GearCategory.values()) {
			for (TarkovWorkbenchRecipes.GearRecipe recipe : TarkovWorkbenchRecipes.getGearRecipes(cat)) {
				Identifier itemId = Registries.ITEM.getId(recipe.resultItem());
				Identifier recipeId = new Identifier(TarkovCraftMod.MOD_ID, "/workbench_gear/" + itemId.getPath());
				List<EmiIngredient> inputs = new ArrayList<>();
				for (TarkovWorkbenchRecipes.Ingredient ing : recipe.materials()) {
					inputs.add(EmiStack.of(ing.item(), ing.count()));
				}
				EmiStack output = EmiStack.of(recipe.resultItem());
				registry.addRecipe(new WorkbenchEmiRecipe(WORKBENCH_GEAR_CATEGORY, recipeId, inputs, output));
			}
		}
	}

	private static void registerAmmoRecipes(EmiRegistry registry) {
		if (!Registries.ITEM.containsId(new Identifier("tacz", "ammo"))) {
			return;
		}
		for (TaCZCaliberCatalog.CaliberSpec spec : TaCZCaliberCatalog.getAllKnownCalibers()) {
			String safeCaliber = spec.ammoId().replace(':', '_').toLowerCase(Locale.ROOT);
			for (AmmoTier tier : spec.getAllowedTiersList()) {
				ItemStack outStack = AmmoTier.createTaczAmmoStack(spec.ammoId(), tier, spec.getBatchSize());
				if (outStack.isEmpty()) {
					continue;
				}
				// Expose tiered ammo variant in EMI item sidebar
				registry.addEmiStack(EmiStack.of(AmmoTier.createTaczAmmoStack(spec.ammoId(), tier, 1)));

				// Scratch recipe
				TarkovWorkbenchRecipes.AmmoRecipe scratch = TarkovWorkbenchRecipes.getAmmoRecipe(spec.ammoId(), tier, false);
				List<EmiIngredient> scratchInputs = new ArrayList<>();
				for (TarkovWorkbenchRecipes.Ingredient ing : scratch.materials()) {
					scratchInputs.add(EmiStack.of(ing.item(), ing.count()));
				}
				Identifier scratchId = new Identifier(TarkovCraftMod.MOD_ID, "/workbench_ammo/scratch_" + safeCaliber + "_" + tier.getId());
				registry.addRecipe(new WorkbenchEmiRecipe(WORKBENCH_AMMO_CATEGORY, scratchId, scratchInputs, EmiStack.of(outStack)));

				// Upgrade from FMJ recipe
				if (tier != AmmoTier.FMJ) {
					TarkovWorkbenchRecipes.AmmoRecipe upg = TarkovWorkbenchRecipes.getAmmoRecipe(spec.ammoId(), tier, true);
					List<EmiIngredient> upgInputs = new ArrayList<>();
					ItemStack fmjStack = AmmoTier.createTaczAmmoStack(spec.ammoId(), AmmoTier.FMJ, upg.requiredFmjCount());
					if (!fmjStack.isEmpty()) {
						upgInputs.add(EmiStack.of(fmjStack));
					}
					for (TarkovWorkbenchRecipes.Ingredient ing : upg.materials()) {
						upgInputs.add(EmiStack.of(ing.item(), ing.count()));
					}
					Identifier upgId = new Identifier(TarkovCraftMod.MOD_ID, "/workbench_ammo/upgrade_" + safeCaliber + "_" + tier.getId());
					registry.addRecipe(new WorkbenchEmiRecipe(WORKBENCH_AMMO_CATEGORY, upgId, upgInputs, EmiStack.of(outStack)));
				}
			}
		}
	}

	private static void registerLootTableRecipes(EmiRegistry registry) {
		for (TarkovLootPool pool : TarkovLootPool.values()) {
			ItemStack crateStack = SearchableLootBlock.createLootCrateStack(pool);
			if (pool != TarkovLootPool.SCAV_POCKET) {
				registry.addEmiStack(EmiStack.of(crateStack));
			}
			// Sample deterministic representative drops from this pool (32 rolls -> up to 24 unique items)
			Map<String, EmiStack> uniqueDrops = new LinkedHashMap<>();
			Random rng = Random.create(0x7A9C0FL ^ (pool.ordinal() * 7919L));
			for (int sample = 0; sample < 10 && uniqueDrops.size() < 24; sample++) {
				GridContainer probe = new GridContainer(8, 8);
				TarkovLootGenerator.populateLootGrid(probe, pool, rng);
				for (GridEntry entry : probe.getEntries()) {
					ItemStack st = entry.getStack();
					if (st.isEmpty()) {
						continue;
					}
					String dedupKey = Registries.ITEM.getId(st.getItem()).toString()
						+ (st.hasNbt() ? "|" + st.getNbt().getString("GunId") + "|" + st.getNbt().getString("AttachmentId") + "|" + st.getNbt().getString("AmmoId") : "");
					if (!uniqueDrops.containsKey(dedupKey)) {
						ItemStack one = st.copy();
						one.setCount(1);
						uniqueDrops.put(dedupKey, EmiStack.of(one));
						if (uniqueDrops.size() >= 24) {
							break;
						}
					}
				}
			}
			if (!uniqueDrops.isEmpty()) {
				Identifier id = new Identifier(TarkovCraftMod.MOD_ID, "/loot_table/" + pool.getId());
				registry.addRecipe(new LootPoolEmiRecipe(id, pool, EmiStack.of(crateStack), new ArrayList<>(uniqueDrops.values())));
			}
		}
	}

	private static final class WorkbenchEmiRecipe implements EmiRecipe {
		private final EmiRecipeCategory category;
		private final Identifier id;
		private final List<EmiIngredient> inputs;
		private final EmiStack output;

		private WorkbenchEmiRecipe(EmiRecipeCategory category, Identifier id, List<EmiIngredient> inputs, EmiStack output) {
			this.category = category;
			this.id = id;
			this.inputs = inputs;
			this.output = output;
		}

		@Override
		public EmiRecipeCategory getCategory() {
			return category;
		}

		@Override
		public @Nullable Identifier getId() {
			return id;
		}

		@Override
		public List<EmiIngredient> getInputs() {
			return inputs;
		}

		@Override
		public List<EmiStack> getOutputs() {
			return List.of(output);
		}

		@Override
		public int getDisplayWidth() {
			return 144;
		}

		@Override
		public int getDisplayHeight() {
			return 38;
		}

		@Override
		public void addWidgets(WidgetHolder widgets) {
			int maxShown = Math.min(4, inputs.size());
			for (int i = 0; i < maxShown; i++) {
				widgets.addSlot(inputs.get(i), 4 + i * 19, 10);
			}
			widgets.addTexture(EmiTexture.EMPTY_ARROW, 84, 11);
			widgets.addSlot(output, 114, 6).large(true).recipeContext(this);
		}
	}

	private static final class LootPoolEmiRecipe implements EmiRecipe {
		private final Identifier id;
		private final TarkovLootPool pool;
		private final EmiStack crateIcon;
		private final List<EmiStack> outputs;

		private LootPoolEmiRecipe(Identifier id, TarkovLootPool pool, EmiStack crateIcon, List<EmiStack> outputs) {
			this.id = id;
			this.pool = pool;
			this.crateIcon = crateIcon;
			this.outputs = outputs;
		}

		@Override
		public EmiRecipeCategory getCategory() {
			return LOOT_TABLES_CATEGORY;
		}

		@Override
		public @Nullable Identifier getId() {
			return id;
		}

		@Override
		public List<EmiIngredient> getInputs() {
			return List.of(crateIcon);
		}

		@Override
		public List<EmiStack> getOutputs() {
			return outputs;
		}

		@Override
		public int getDisplayWidth() {
			return 152;
		}

		@Override
		public int getDisplayHeight() {
			int rows = Math.max(1, (outputs.size() + 7) / 8);
			return 24 + rows * 18;
		}

		@Override
		public void addWidgets(WidgetHolder widgets) {
			widgets.addSlot(crateIcon, 2, 2);
			widgets.addText(Text.translatable(pool.getTranslationKey()), 24, 6, 0xFFDDDDDD, true);
			for (int i = 0; i < outputs.size(); i++) {
				int col = i % 8;
				int row = i / 8;
				widgets.addSlot(outputs.get(i), 4 + col * 18, 22 + row * 18).recipeContext(this);
			}
		}
	}
}
