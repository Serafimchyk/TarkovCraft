package com.tarkovcraft.item;

/**
 * Zonal armor piece (Helmet, Face Mask, or Body Armor) with Armor Class (0..6),
 * ballistic material, ricochet chance, weight, durability, and explicit HitZone coverage.
 * Zones NOT in coveredZones take full unmitigated damage.
 */
public class TarkovArmorItem extends Item implements TarkovSizedItem {
	private static final List<HitZone> HELMET_VULNERABLE_ZONES = List.of(
		HitZone.HEAD_TOP, HitZone.HEAD_EARS, HitZone.HEAD_EYES, HitZone.HEAD_JAWS
	);
	private static final List<HitZone> BODY_ARMOR_VULNERABLE_ZONES = List.of(
		HitZone.THORAX, HitZone.STOMACH, HitZone.NECK, HitZone.ARMPIT_LEFT, HitZone.ARMPIT_RIGHT
	);

	private final EquipmentSlotType targetSlot;
	private final ItemSize storedItemSize;
	private final int armorClass;
	private final ArmorMaterialType material;
	private final float ricochetChance;
	private final float weightKg;
	private final Set<HitZone> coveredZones;
	private final ItemRarityTier rarityTier;
	private final boolean gasProtection;

	public TarkovArmorItem(
		Settings settings,
		EquipmentSlotType targetSlot,
		ItemSize storedItemSize,
		int armorClass,
		int maxDurability,
		ArmorMaterialType material,
		float ricochetChance,
		float weightKg,
		Set<HitZone> coveredZones
	) {
		this(settings, targetSlot, storedItemSize, armorClass, maxDurability, material, ricochetChance, weightKg,
			coveredZones, ItemRarityTier.UNCOMMON, false);
	}

	public TarkovArmorItem(
		Settings settings,
		EquipmentSlotType targetSlot,
		ItemSize storedItemSize,
		int armorClass,
		int maxDurability,
		ArmorMaterialType material,
		float ricochetChance,
		float weightKg,
		Set<HitZone> coveredZones,
		ItemRarityTier rarityTier
	) {
		this(settings, targetSlot, storedItemSize, armorClass, maxDurability, material, ricochetChance, weightKg,
			coveredZones, rarityTier, false);
	}

	public TarkovArmorItem(
		Settings settings,
		EquipmentSlotType targetSlot,
		ItemSize storedItemSize,
		int armorClass,
		int maxDurability,
		ArmorMaterialType material,
		float ricochetChance,
		float weightKg,
		Set<HitZone> coveredZones,
		ItemRarityTier rarityTier,
		boolean gasProtection
	) {
		super(armorClass > 0
			? settings.maxCount(1).maxDamage(Math.max(1, maxDurability))
			: settings.maxCount(1));
		this.targetSlot = targetSlot;
		this.storedItemSize = storedItemSize;
		this.armorClass = armorClass;
		this.material = material;
		this.ricochetChance = Math.max(0.0f, Math.min(0.65f, ricochetChance));
		this.weightKg = Math.max(0.0f, weightKg);
		Set<HitZone> zones = (coveredZones == null || coveredZones.isEmpty())
			? EnumSet.noneOf(HitZone.class)
			: EnumSet.copyOf(coveredZones);
		this.coveredZones = Collections.unmodifiableSet(zones);
		this.rarityTier = rarityTier != null ? rarityTier : ItemRarityTier.UNCOMMON;
		this.gasProtection = gasProtection;
	}

	public boolean isGasProtection() {
		return gasProtection;
	}

	public EquipmentSlotType getTargetSlot() {
		return targetSlot;
	}

	@Override
	public ItemSize getStoredItemSize() {
		return storedItemSize;
	}

	@Override
	public ItemRarityTier getRarityTier() {
		return rarityTier;
	}

	public int getArmorClass() {
		return armorClass;
	}

	public ArmorMaterialType getMaterial() {
		return material;
	}

	public float getRicochetChance() {
		return ricochetChance;
	}

	@Override
	public float getWeightKg() {
		return weightKg;
	}

	public Set<HitZone> getCoveredZones() {
		return coveredZones;
	}

	public boolean coversZone(HitZone zone) {
		return armorClass > 0 && zone != null && coveredZones.contains(zone);
	}

	public static int getRemainingDurability(ItemStack stack) {
		if (stack == null || stack.isEmpty() || !(stack.getItem() instanceof TarkovArmorItem armor)) {
			return 0;
		}
		if (armor.getArmorClass() <= 0 || !stack.isDamageable()) {
			return 0;
		}
		return Math.max(0, stack.getMaxDamage() - stack.getDamage());
	}

	public static boolean isBroken(ItemStack stack) {
		if (stack == null || stack.isEmpty() || !(stack.getItem() instanceof TarkovArmorItem armor)) {
			return false;
		}
		return armor.getArmorClass() > 0 && getRemainingDurability(stack) <= 0;
	}

	@Override
	public void appendTooltip(ItemStack stack, @Nullable World world, List<Text> tooltip, TooltipContext context) {
		if (armorClass > 0) {
			tooltip.add(Text.translatable("tooltip.tarkovcraft.armor_class", armorClass).formatted(Formatting.AQUA));
			tooltip.add(Text.translatable("tooltip.tarkovcraft.armor_material",
				Text.translatable(material.getTranslationKey())).formatted(Formatting.GRAY));

			int maxDur = Math.max(1, stack.getMaxDamage());
			int curDur = getRemainingDurability(stack);
			if (curDur <= 0) {
				tooltip.add(Text.translatable("tooltip.tarkovcraft.armor_broken", maxDur).formatted(Formatting.RED));
			} else {
				float ratio = curDur / (float) maxDur;
				Formatting durColor = ratio < 0.25f ? Formatting.RED
					: (ratio < 0.50f ? Formatting.YELLOW : Formatting.GRAY);
				tooltip.add(Text.translatable("tooltip.tarkovcraft.armor_durability", curDur, maxDur).formatted(durColor));
			}

			if (ricochetChance > 0.0f) {
				tooltip.add(Text.translatable("tooltip.tarkovcraft.armor_ricochet", Math.round(ricochetChance * 100.0f))
					.formatted(Formatting.DARK_AQUA));
			}

			if (!coveredZones.isEmpty()) {
				String zonesStr = coveredZones.stream()
					.map(z -> Text.translatable(z.getTranslationKey()).getString())
					.collect(Collectors.joining(", "));
				tooltip.add(Text.translatable("tooltip.tarkovcraft.armor_zones", zonesStr).formatted(Formatting.DARK_GREEN));
			}

			List<HitZone> slotZones = switch (targetSlot) {
				case HELMET -> HELMET_VULNERABLE_ZONES;
				case BODY_ARMOR -> BODY_ARMOR_VULNERABLE_ZONES;
				default -> List.of();
			};
			if (!slotZones.isEmpty()) {
				String exposedStr = slotZones.stream()
					.filter(z -> !coveredZones.contains(z))
					.map(z -> Text.translatable(z.getTranslationKey()).getString())
					.collect(Collectors.joining(", "));
				if (!exposedStr.isEmpty()) {
					tooltip.add(Text.translatable("tooltip.tarkovcraft.armor_exposed", exposedStr).formatted(Formatting.RED));
				}
			}
		} else {
			tooltip.add(Text.translatable("tooltip.tarkovcraft.armor_cosmetic").formatted(Formatting.DARK_GRAY));
		}

		tooltip.add(Text.translatable("tooltip.tarkovcraft.armor_weight", String.format(Locale.ROOT, "%.1f", weightKg))
			.formatted(Formatting.DARK_GRAY));
	}
}

/**
 * Ballistic armor plates and armor repair kits that restore durability on equipped TarkovArmorItem pieces.
 */
public class TarkovArmorRepairItem extends Item implements TarkovSizedItem {
	private static final EquipmentSlotType[] REPAIR_PRIORITY = {
		EquipmentSlotType.BODY_ARMOR,
		EquipmentSlotType.HELMET,
		EquipmentSlotType.FACE_MASK
	};

	private final ItemSize storedItemSize;
	private final ItemRarityTier rarityTier;
	private final float weightKg;
	private final int repairAmount;

	public TarkovArmorRepairItem(
		Settings settings,
		int repairAmount,
		ItemSize storedItemSize,
		ItemRarityTier rarityTier,
		float weightKg
	) {
		super(settings.maxCount(1));
		this.repairAmount = Math.max(1, repairAmount);
		this.storedItemSize = storedItemSize != null ? storedItemSize : ItemSize.ONE_BY_ONE;
		this.rarityTier = rarityTier != null ? rarityTier : ItemRarityTier.UNCOMMON;
		this.weightKg = Math.max(0.0f, weightKg);
	}

	public int getRepairAmount() {
		return repairAmount;
	}

	@Override
	public ItemSize getStoredItemSize() {
		return storedItemSize;
	}

	@Override
	public ItemRarityTier getRarityTier() {
		return rarityTier;
	}

	@Override
	public float getWeightKg() {
		return weightKg;
	}

	@Override
	public TypedActionResult<ItemStack> use(World world, PlayerEntity user, Hand hand) {
		ItemStack stack = user.getStackInHand(hand);
		if (world.isClient) {
			return TypedActionResult.pass(stack);
		}
		if (!(user instanceof ServerPlayerEntity sp)) {
			return TypedActionResult.fail(stack);
		}
		if (!applyRepair(sp, stack)) {
			return TypedActionResult.fail(stack);
		}
		if (stack.isEmpty() && !sp.isCreative()) {
			sp.setStackInHand(hand, ItemStack.EMPTY);
		}
		return TypedActionResult.success(sp.getStackInHand(hand), false);
	}

	public static boolean canRepairAny(ServerPlayerEntity player) {
		if (player == null || !(player instanceof PlayerLoadoutHolder holder)) {
			return false;
		}
		return findFirstDamagedArmor(holder.tarkovcraft$getLoadout()) != null;
	}

	@Nullable
	private static ItemStack findFirstDamagedArmor(PlayerLoadout loadout) {
		if (loadout == null) {
			return null;
		}
		for (EquipmentSlotType slot : REPAIR_PRIORITY) {
			ItemStack eq = loadout.getEquipment(slot);
			if (!eq.isEmpty()
				&& eq.getItem() instanceof TarkovArmorItem armor
				&& armor.getArmorClass() > 0
				&& eq.isDamageable()
				&& eq.getDamage() > 0) {
				return eq;
			}
		}
		return null;
	}

	public static boolean applyRepair(ServerPlayerEntity player, ItemStack stack) {
		if (player == null || stack == null || stack.isEmpty() || !(stack.getItem() instanceof TarkovArmorRepairItem repair)) {
			return false;
		}
		if (player.isSpectator() || !player.isAlive() || !(player instanceof PlayerLoadoutHolder holder)) {
			return false;
		}

		PlayerLoadout loadout = holder.tarkovcraft$getLoadout();
		ItemStack targetArmor = findFirstDamagedArmor(loadout);
		if (targetArmor == null) {
			player.sendMessage(Text.translatable("message.tarkovcraft.repair_not_needed"), true);
			return false;
		}

		int curDamage = targetArmor.getDamage();
		int repaired = Math.min(curDamage, repair.repairAmount);
		targetArmor.setDamage(curDamage - repaired);

		if (!player.isCreative()) {
			stack.decrement(1);
		}
		holder.tarkovcraft$syncLoadout();
		player.getWorld().playSound(null, player.getX(), player.getY(), player.getZ(),
			SoundEvents.BLOCK_ANVIL_USE, SoundCategory.PLAYERS, 0.55f, 1.3f);
		player.sendMessage(Text.translatable("message.tarkovcraft.repair_done", repaired), true);
		return true;
	}

	@Override
	public void appendTooltip(ItemStack stack, @Nullable World world, List<Text> tooltip, TooltipContext context) {
		tooltip.add(Text.translatable("tooltip.tarkovcraft.repair_use", repairAmount).formatted(Formatting.AQUA));
		tooltip.add(Text.translatable("tooltip.tarkovcraft.item_size",
			storedItemSize.width(), storedItemSize.height()).formatted(Formatting.GRAY));
		tooltip.add(Text.translatable("tooltip.tarkovcraft.med_weight",
			String.format(Locale.ROOT, "%.2f", weightKg)).formatted(Formatting.DARK_GRAY));
	}
}

/**
 * Openable TarkovCraft storage case (Wallet, Keytool, Docs, SICC, Medcase, THICC, etc.).
 * Persists an internal GridContainer in NBT via ContainerItemHelper.
 */
public class TarkovCaseItem extends Item implements TarkovSizedItem {
	public enum CaseRole {
		GENERAL("general"),
		MEDICAL("medical"),
		INJECTOR("injector"),
		AMMO("ammo"),
		GRENADE("grenade"),
		FOOD("food"),
		WEAPON("weapon"),
		ARMOR_PLATE("armor_plate"),
		MONEY("money");

		private final String translationSuffix;

		CaseRole(String translationSuffix) {
			this.translationSuffix = translationSuffix;
		}

		public String getTranslationKey() {
			return "tooltip.tarkovcraft.case_role." + translationSuffix;
		}

		public static CaseRole fromString(String raw) {
			if (raw == null || raw.isBlank()) {
				return GENERAL;
			}
			for (CaseRole role : values()) {
				if (role.name().equalsIgnoreCase(raw)) {
					return role;
				}
			}
			return GENERAL;
		}
	}

	private final int gridWidth;
	private final int gridHeight;
	private final ItemSize storedItemSize;
	private final ItemRarityTier rarityTier;
	private final float weightKg;
	private final CaseRole caseRole;

	public TarkovCaseItem(
		Settings settings,
		int gridWidth,
		int gridHeight,
		ItemSize storedItemSize,
		ItemRarityTier rarityTier,
		float weightKg
	) {
		this(settings, gridWidth, gridHeight, storedItemSize, rarityTier, weightKg, CaseRole.GENERAL);
	}

	public TarkovCaseItem(
		Settings settings,
		int gridWidth,
		int gridHeight,
		ItemSize storedItemSize,
		ItemRarityTier rarityTier,
		float weightKg,
		CaseRole caseRole
	) {
		super(settings.maxCount(1));
		this.gridWidth = Math.max(1, gridWidth);
		this.gridHeight = Math.max(1, gridHeight);
		this.storedItemSize = storedItemSize != null ? storedItemSize : ItemSize.ONE_BY_ONE;
		this.rarityTier = rarityTier != null ? rarityTier : ItemRarityTier.RARE;
		this.weightKg = Math.max(0.0f, weightKg);
		this.caseRole = caseRole != null ? caseRole : CaseRole.GENERAL;
	}

	public CaseRole getCaseRole() {
		return caseRole;
	}

	public boolean isMedicalCase() {
		return caseRole == CaseRole.MEDICAL || caseRole == CaseRole.INJECTOR;
	}

	/**
	 * Returns true if the given ItemStack is allowed inside this storage case.
	 */
	public boolean canAcceptItem(ItemStack stack) {
		if (stack == null || stack.isEmpty()) {
			return false;
		}
		Item item = stack.getItem();
		return switch (caseRole) {
			case GENERAL -> true;
			case MEDICAL -> item instanceof TarkovMedicalItem || item instanceof TarkovStimulatorItem;
			case INJECTOR -> item instanceof TarkovStimulatorItem;
			case AMMO -> isTaczAmmoOrAttachment(stack);
			case GRENADE -> item instanceof TarkovGrenadeItem;
			case FOOD -> stack.isFood() || item instanceof TarkovFoodItem
				|| item instanceof PotionItem || item instanceof MilkBucketItem;
			case WEAPON -> AmmoTier.isTaczGunStack(stack) || isTaczAmmoOrAttachment(stack)
				|| item instanceof TarkovMeleeItem || item instanceof TarkovGrenadeItem
				|| item instanceof SwordItem || item instanceof BowItem
				|| item instanceof CrossbowItem || item instanceof TridentItem;
			case ARMOR_PLATE -> item instanceof TarkovArmorRepairItem || item instanceof TarkovArmorItem;
			case MONEY -> item == Items.EMERALD || item == Items.GOLD_INGOT || item == Items.GOLD_NUGGET
				|| item == Items.DIAMOND || item == Items.NETHERITE_INGOT || item == Items.NETHERITE_SCRAP
				|| item == Items.AMETHYST_SHARD || item == Items.PAPER;
		};
	}

	private static boolean isTaczAmmoOrAttachment(ItemStack stack) {
		if (AmmoTier.isTaczAmmoStack(stack)) {
			return true;
		}
		Identifier id = Registries.ITEM.getId(stack.getItem());
		if (!"tacz".equals(id.getNamespace())) {
			return false;
		}
		String path = id.getPath();
		return path.contains("ammo") || path.contains("attachment") || path.contains("mag");
	}

	public int getGridWidth() {
		return gridWidth;
	}

	public int getGridHeight() {
		return gridHeight;
	}

	@Override
	public ItemSize getStoredItemSize() {
		return storedItemSize;
	}

	@Override
	public ItemRarityTier getRarityTier() {
		return rarityTier;
	}

	@Override
	public float getWeightKg() {
		return weightKg;
	}

	@Override
	public TypedActionResult<ItemStack> use(World world, PlayerEntity user, Hand hand) {
		ItemStack stack = user.getStackInHand(hand);
		if (world.isClient) {
			return TypedActionResult.pass(stack);
		}
		if (!(user instanceof ServerPlayerEntity sp) || sp.isSpectator()) {
			return TypedActionResult.fail(stack);
		}
		UUID caseUuid = ContainerItemHelper.getOrCreateContainerUuid(stack);
		if (sp instanceof PlayerLoadoutHolder holder && !sp.isCreative()) {
			holder.tarkovcraft$syncLoadout();
		}
		TarkovScreenHandler.openCaseScreen(sp, caseUuid);
		return TypedActionResult.success(stack, false);
	}

	@Override
	public void appendTooltip(ItemStack stack, @Nullable World world, List<Text> tooltip, TooltipContext context) {
		tooltip.add(Text.translatable("tooltip.tarkovcraft.case_grid",
			gridWidth, gridHeight, gridWidth * gridHeight).formatted(Formatting.AQUA));
		tooltip.add(Text.translatable("tooltip.tarkovcraft.case_filter",
			Text.translatable(caseRole.getTranslationKey())).formatted(Formatting.GOLD));
		tooltip.add(Text.translatable("tooltip.tarkovcraft.case_open_hint").formatted(Formatting.YELLOW));
		tooltip.add(Text.translatable("tooltip.tarkovcraft.item_size",
			storedItemSize.width(), storedItemSize.height()).formatted(Formatting.GRAY));
		tooltip.add(Text.translatable("tooltip.tarkovcraft.med_weight",
			String.format(Locale.ROOT, "%.2f", weightKg)).formatted(Formatting.DARK_GRAY));
	}
}

package com.tarkovcraft.item;

/**
 * TarkovCraft food and drink provisions with FoodComponent, grid size, rarity, and weight.
 */
public class TarkovFoodItem extends Item implements TarkovSizedItem {
	private final ItemSize storedItemSize;
	private final ItemRarityTier rarityTier;
	private final float weightKg;
	private final boolean drink;

	public TarkovFoodItem(
		Settings settings,
		int hunger,
		float saturation,
		boolean drink,
		ItemSize storedItemSize,
		ItemRarityTier rarityTier,
		float weightKg
	) {
		super(buildSettings(settings, hunger, saturation, drink));
		this.storedItemSize = storedItemSize != null ? storedItemSize : ItemSize.ONE_BY_ONE;
		this.rarityTier = rarityTier != null ? rarityTier : ItemRarityTier.COMMON;
		this.weightKg = Math.max(0.0f, weightKg);
		this.drink = drink;
	}

	private static Settings buildSettings(Settings settings, int hunger, float saturation, boolean drink) {
		FoodComponent.Builder builder = new FoodComponent.Builder()
			.hunger(Math.max(0, hunger))
			.saturationModifier(Math.max(0.0f, saturation));
		if (drink || hunger <= 0) {
			builder.alwaysEdible();
		}
		return settings.maxCount(16).food(builder.build());
	}

	public boolean isDrink() {
		return drink;
	}

	@Override
	public UseAction getUseAction(ItemStack stack) {
		return drink ? UseAction.DRINK : UseAction.EAT;
	}

	@Override
	public ItemSize getStoredItemSize() {
		return storedItemSize;
	}

	@Override
	public ItemRarityTier getRarityTier() {
		return rarityTier;
	}

	@Override
	public float getWeightKg() {
		return weightKg;
	}

	@Override
	public void appendTooltip(ItemStack stack, @Nullable World world, List<Text> tooltip, TooltipContext context) {
		tooltip.add(Text.translatable("tooltip.tarkovcraft.item_size",
			storedItemSize.width(), storedItemSize.height()).formatted(Formatting.GRAY));
		tooltip.add(Text.translatable("tooltip.tarkovcraft.med_weight",
			String.format(Locale.ROOT, "%.2f", weightKg)).formatted(Formatting.DARK_GRAY));
	}
}

package com.tarkovcraft.item;

import net.minecraft.item.Item;

/**
 * Functional Tarkov item for tactical headsets, storage cases, and grenades.
 */
public class TarkovFunctionalItem extends Item implements TarkovSizedItem {
	private final String category;
	private final ItemSize storedItemSize;
	private final ItemRarityTier rarityTier;
	private final float weightKg;

	public TarkovFunctionalItem(
		Settings settings,
		String category,
		ItemSize storedItemSize,
		ItemRarityTier rarityTier,
		float weightKg
	) {
		super("grenade".equalsIgnoreCase(category) ? settings.maxCount(4) : settings.maxCount(1));
		this.category = category != null ? category : "gear";
		this.storedItemSize = storedItemSize != null ? storedItemSize : ItemSize.ONE_BY_ONE;
		this.rarityTier = rarityTier != null ? rarityTier : ItemRarityTier.UNCOMMON;
		this.weightKg = Math.max(0.0f, weightKg);
	}

	public String getCategory() {
		return category;
	}

	@Override
	public ItemSize getStoredItemSize() {
		return storedItemSize;
	}

	@Override
	public ItemRarityTier getRarityTier() {
		return rarityTier;
	}

	@Override
	public float getWeightKg() {
		return weightKg;
	}

	@Override
	public void appendTooltip(ItemStack stack, @Nullable World world, List<Text> tooltip, TooltipContext context) {
		tooltip.add(Text.translatable("tooltip.tarkovcraft.category", formatCategory(category)).formatted(Formatting.AQUA));
		tooltip.add(Text.translatable("tooltip.tarkovcraft.item_size",
			storedItemSize.width(), storedItemSize.height()).formatted(Formatting.GRAY));
		tooltip.add(Text.translatable("tooltip.tarkovcraft.med_weight",
			String.format(Locale.ROOT, "%.2f", weightKg)).formatted(Formatting.DARK_GRAY));
	}

	private static String formatCategory(String raw) {
		if (raw == null || raw.isBlank()) {
			return "Gear";
		}
		String spaced = raw.replace('_', ' ');
		return Character.toUpperCase(spaced.charAt(0)) + spaced.substring(1);
	}
}

/**
 * Throwable TarkovCraft tactical grenade (Frag, Impact Frag, Flashbang, Smoke, CS Gas).
 */
public class TarkovGrenadeItem extends Item implements TarkovSizedItem {
	public enum Category {
		FRAG,
		IMPACT,
		FLASH,
		SMOKE,
		GAS
	}

	public enum GrenadeType {
		FRAG_F1(40, 0, 7.5f, 300.0f, 22, 0.70f, 0, Category.FRAG),
		FRAG_V40(56, 0, 4.5f, 190.0f, 15, 0.45f, 0, Category.FRAG),
		FRAG_RGN(70, 6, 5.5f, 230.0f, 18, 0.55f, 0, Category.IMPACT),
		FRAG_RGO(70, 6, 7.0f, 280.0f, 24, 0.70f, 0, Category.IMPACT),
		FLASH_M7290(40, 0, 11.0f, 0.0f, 0, 0.0f, 0, Category.FLASH),
		SMOKE_M18(30, 0, 5.5f, 0.0f, 0, 0.0f, 300, Category.SMOKE),
		GAS_CS(35, 0, 5.5f, 0.0f, 0, 0.0f, 240, Category.GAS);

		private final int fuseTicks;
		private final int armingTicks;
		private final float blastRadius;
		private final float maxDamage;
		private final int penetrationPower;
		private final float bleedChance;
		private final int cloudTicks;
		private final Category category;

		GrenadeType(
			int fuseTicks,
			int armingTicks,
			float blastRadius,
			float maxDamage,
			int penetrationPower,
			float bleedChance,
			int cloudTicks,
			Category category
		) {
			this.fuseTicks = fuseTicks;
			this.armingTicks = armingTicks;
			this.blastRadius = blastRadius;
			this.maxDamage = maxDamage;
			this.penetrationPower = penetrationPower;
			this.bleedChance = bleedChance;
			this.cloudTicks = cloudTicks;
			this.category = category;
		}

		public int getFuseTicks() {
			return fuseTicks;
		}

		public int getArmingTicks() {
			return armingTicks;
		}

		public boolean isImpactFuse() {
			return armingTicks > 0;
		}

		public float getBlastRadius() {
			return blastRadius;
		}

		public float getMaxDamage() {
			return maxDamage;
		}

		public int getPenetrationPower() {
			return penetrationPower;
		}

		public float getBleedChance() {
			return bleedChance;
		}

		public int getCloudTicks() {
			return cloudTicks;
		}

		public Category getCategory() {
			return category;
		}

		public static GrenadeType fromString(String raw) {
			if (raw == null || raw.isBlank()) {
				return FRAG_F1;
			}
			for (GrenadeType type : values()) {
				if (type.name().equalsIgnoreCase(raw.trim())) {
					return type;
				}
			}
			return FRAG_F1;
		}
	}

	private final GrenadeType grenadeType;
	private final ItemSize storedItemSize;
	private final ItemRarityTier rarityTier;
	private final float weightKg;

	public TarkovGrenadeItem(
		Settings settings,
		GrenadeType grenadeType,
		ItemSize storedItemSize,
		ItemRarityTier rarityTier,
		float weightKg
	) {
		super(settings.maxCount(4));
		this.grenadeType = grenadeType != null ? grenadeType : GrenadeType.FRAG_F1;
		this.storedItemSize = storedItemSize != null ? storedItemSize : ItemSize.ONE_BY_ONE;
		this.rarityTier = rarityTier != null ? rarityTier : ItemRarityTier.RARE;
		this.weightKg = Math.max(0.0f, weightKg);
	}

	public GrenadeType getGrenadeType() {
		return grenadeType;
	}

	@Override
	public ItemSize getStoredItemSize() {
		return storedItemSize;
	}

	@Override
	public ItemRarityTier getRarityTier() {
		return rarityTier;
	}

	@Override
	public float getWeightKg() {
		return weightKg;
	}

	@Override
	public TypedActionResult<ItemStack> use(World world, PlayerEntity user, Hand hand) {
		ItemStack stack = user.getStackInHand(hand);
		user.getItemCooldownManager().set(this, 20);
		world.playSound(null, user.getX(), user.getY(), user.getZ(),
			SoundEvents.ENTITY_SNOWBALL_THROW, SoundCategory.PLAYERS, 0.6f, 0.75f);

		if (!world.isClient) {
			TarkovGrenadeEntity grenade = new TarkovGrenadeEntity(world, user, stack, grenadeType);
			grenade.setVelocity(user, user.getPitch(), user.getYaw(), 0.0f, 1.35f, 1.0f);
			world.spawnEntity(grenade);

			if (!user.isCreative()) {
				stack.decrement(1);
				if (user instanceof ServerPlayerEntity && user instanceof PlayerLoadoutHolder holder) {
					holder.tarkovcraft$syncLoadout();
				}
			}
		}

		return TypedActionResult.success(stack, world.isClient());
	}

	@Override
	public void appendTooltip(ItemStack stack, @Nullable World world, List<Text> tooltip, TooltipContext context) {
		String typeKey = switch (grenadeType.getCategory()) {
			case FRAG -> "tooltip.tarkovcraft.grenade_type_frag";
			case IMPACT -> "tooltip.tarkovcraft.grenade_type_impact";
			case FLASH -> "tooltip.tarkovcraft.grenade_type_flash";
			case SMOKE -> "tooltip.tarkovcraft.grenade_type_smoke";
			case GAS -> "tooltip.tarkovcraft.grenade_type_gas";
		};
		Formatting typeColor = switch (grenadeType.getCategory()) {
			case FRAG, IMPACT -> Formatting.RED;
			case FLASH -> Formatting.YELLOW;
			case SMOKE -> Formatting.GREEN;
			case GAS -> Formatting.GOLD;
		};
		tooltip.add(Text.translatable(typeKey).formatted(typeColor));

		String fuseSec = formatNumber(grenadeType.getFuseTicks() / 20.0f);
		if (grenadeType.isImpactFuse()) {
			tooltip.add(Text.translatable("tooltip.tarkovcraft.grenade_impact_fuse", fuseSec).formatted(Formatting.GRAY));
		} else {
			tooltip.add(Text.translatable("tooltip.tarkovcraft.grenade_fuse", fuseSec).formatted(Formatting.GRAY));
		}

		tooltip.add(Text.translatable("tooltip.tarkovcraft.grenade_radius", formatNumber(grenadeType.getBlastRadius()))
			.formatted(Formatting.GRAY));

		if (grenadeType.getCloudTicks() > 0) {
			tooltip.add(Text.translatable("tooltip.tarkovcraft.grenade_cloud", grenadeType.getCloudTicks() / 20)
				.formatted(Formatting.AQUA));
		}

		tooltip.add(Text.translatable("tooltip.tarkovcraft.item_size",
			storedItemSize.width(), storedItemSize.height()).formatted(Formatting.GRAY));
		tooltip.add(Text.translatable("tooltip.tarkovcraft.med_weight",
			String.format(Locale.ROOT, "%.2f", weightKg)).formatted(Formatting.DARK_GRAY));
	}

	private static String formatNumber(float value) {
		if (Math.abs(value - Math.round(value)) < 0.05f) {
			return String.valueOf(Math.round(value));
		}
		return String.format(Locale.ROOT, "%.1f", value);
	}
}

package com.tarkovcraft.item;

/**
 * TarkovCraft medical item (bandages, tourniquets, splints, painkillers, medkits, surgical kits).
 * Tracks remaining uses or HP pool via NBT_REMAINING without item stacking.
 */
public class TarkovMedicalItem extends Item implements TarkovSizedItem {
	public static final String NBT_REMAINING = "TarkovMedRemaining";

	private final MedicalType medicalType;
	private final ItemSize storedItemSize;
	private final ItemRarityTier rarityTier;
	private final float weightKg;

	public TarkovMedicalItem(Settings settings, MedicalType medicalType) {
		this(settings, medicalType, medicalType.getGridSize(), medicalType.getRarityTier(), medicalType.getWeightKg());
	}

	public TarkovMedicalItem(
		Settings settings,
		MedicalType medicalType,
		ItemSize storedItemSize,
		ItemRarityTier rarityTier,
		float weightKg
	) {
		super(settings.maxCount(1));
		this.medicalType = medicalType;
		this.storedItemSize = storedItemSize != null ? storedItemSize : medicalType.getGridSize();
		this.rarityTier = rarityTier != null ? rarityTier : medicalType.getRarityTier();
		this.weightKg = weightKg >= 0.0f ? weightKg : medicalType.getWeightKg();
	}

	public MedicalType getMedicalType() {
		return medicalType;
	}

	@Override
	public ItemSize getStoredItemSize() {
		return storedItemSize;
	}

	@Override
	public float getWeightKg() {
		return weightKg;
	}

	@Override
	public ItemRarityTier getRarityTier() {
		return rarityTier;
	}

	public static int getRemaining(ItemStack stack) {
		if (stack == null || stack.isEmpty() || !(stack.getItem() instanceof TarkovMedicalItem med)) {
			return 0;
		}
		int max = med.getMedicalType().getMaxCapacity();
		NbtCompound nbt = stack.getNbt();
		if (nbt == null || !nbt.contains(NBT_REMAINING)) {
			return max;
		}
		return Math.max(0, Math.min(max, nbt.getInt(NBT_REMAINING)));
	}

	public static void setRemaining(ItemStack stack, int value) {
		if (stack == null || stack.isEmpty() || !(stack.getItem() instanceof TarkovMedicalItem med)) {
			return;
		}
		if (value <= 0) {
			stack.setCount(0);
			return;
		}
		int clamped = Math.min(med.getMedicalType().getMaxCapacity(), value);
		stack.getOrCreateNbt().putInt(NBT_REMAINING, clamped);
	}

	@Override
	public boolean isItemBarVisible(ItemStack stack) {
		int max = medicalType.getMaxCapacity();
		return max > 1 && getRemaining(stack) < max;
	}

	@Override
	public int getItemBarStep(ItemStack stack) {
		int max = Math.max(1, medicalType.getMaxCapacity());
		int step = Math.round(13.0f * getRemaining(stack) / (float) max);
		return Math.max(0, Math.min(13, step));
	}

	@Override
	public int getItemBarColor(ItemStack stack) {
		float ratio = getRemaining(stack) / (float) Math.max(1, medicalType.getMaxCapacity());
		if (ratio >= 0.50f) return 0x55CC55;
		if (ratio >= 0.25f) return 0xE0B23B;
		return 0xE53935;
	}

	@Override
	public TypedActionResult<ItemStack> use(World world, PlayerEntity user, Hand hand) {
		ItemStack stack = user.getStackInHand(hand);
		if (world.isClient) {
			return TypedActionResult.pass(stack);
		}
		if (!(user instanceof ServerPlayerEntity sp) || !(sp instanceof PlayerLoadoutHolder holder)
			|| sp.isSpectator() || !sp.isAlive()) {
			return TypedActionResult.fail(stack);
		}
		if (sp.getItemCooldownManager().isCoolingDown(this)) {
			return TypedActionResult.fail(stack);
		}

		BodyPartHealth health = holder.tarkovcraft$getBodyHealth();
		MedicalHelper.ApplyResult res = MedicalHelper.applyMedical(health, stack, null, sp.isCreative());
		if (!res.applied()) {
			sp.sendMessage(Text.translatable("message.tarkovcraft.med_not_needed"), true);
			return TypedActionResult.fail(stack);
		}

		if (res.exhaustionToAdd() > 0.0f) {
			sp.addExhaustion(res.exhaustionToAdd());
		}
		sp.getItemCooldownManager().set(this, medicalType.getCooldownTicks());
		if (stack.isEmpty() && !sp.isCreative()) {
			sp.setStackInHand(hand, ItemStack.EMPTY);
		}
		sp.setHealth(Math.max(1.0f, health.getTotalRatio() * sp.getMaxHealth()));
		holder.tarkovcraft$syncBodyHealth();
		holder.tarkovcraft$syncLoadout();
		MedicalHelper.playMedicalSound(sp, medicalType);
		return TypedActionResult.success(sp.getStackInHand(hand), false);
	}

	@Override
	public void appendTooltip(ItemStack stack, @Nullable World world, List<Text> tooltip, TooltipContext context) {
		int rem = getRemaining(stack);
		int max = medicalType.getMaxCapacity();
		float ratio = rem / (float) Math.max(1, max);
		Formatting resCol = ratio >= 0.50f ? Formatting.GREEN : (ratio >= 0.25f ? Formatting.YELLOW : Formatting.RED);

		if (medicalType.getCapacityMode() == MedicalType.CapacityMode.HP_POOL) {
			tooltip.add(Text.translatable("tooltip.tarkovcraft.med_pool", rem, max).formatted(resCol));
			tooltip.add(Text.translatable("tooltip.tarkovcraft.med_max_heal", medicalType.getMaxHealPerUse())
				.formatted(Formatting.GRAY));
		} else {
			tooltip.add(Text.translatable("tooltip.tarkovcraft.med_uses", rem, max).formatted(resCol));
		}

		if (medicalType.isCuresLightBleed()) {
			if (medicalType.getCapacityMode() == MedicalType.CapacityMode.HP_POOL) {
				tooltip.add(Text.translatable("tooltip.tarkovcraft.med_cures_light_bleed_cost",
					medicalType.getLightBleedCost()).formatted(Formatting.GOLD));
			} else {
				tooltip.add(Text.translatable("tooltip.tarkovcraft.med_cures_light_bleed").formatted(Formatting.GOLD));
			}
		}
		if (medicalType.isCuresHeavyBleed()) {
			if (medicalType.getCapacityMode() == MedicalType.CapacityMode.HP_POOL) {
				tooltip.add(Text.translatable("tooltip.tarkovcraft.med_cures_heavy_bleed_cost",
					medicalType.getHeavyBleedCost()).formatted(Formatting.RED));
			} else {
				tooltip.add(Text.translatable("tooltip.tarkovcraft.med_cures_heavy_bleed").formatted(Formatting.RED));
			}
		}
		if (medicalType.isCuresFracture()) {
			tooltip.add(Text.translatable("tooltip.tarkovcraft.med_cures_fracture").formatted(Formatting.YELLOW));
		}
		if (medicalType.getPainkillerTicks() > 0) {
			tooltip.add(Text.translatable("tooltip.tarkovcraft.med_painkiller", medicalType.getPainkillerTicks() / 20)
				.formatted(Formatting.AQUA));
		}
		if (medicalType.isSurgicalKit()) {
			tooltip.add(Text.translatable("tooltip.tarkovcraft.med_surgery",
				Math.round(medicalType.getSurgeryRetention() * 100.0f)).formatted(Formatting.LIGHT_PURPLE));
		}

		tooltip.add(Text.translatable("tooltip.tarkovcraft.med_weight",
			String.format(Locale.ROOT, "%.2f", weightKg)).formatted(Formatting.DARK_GRAY));
	}
}

package com.tarkovcraft.item;

import com.tarkovcraft.grid.ItemSize;
import com.tarkovcraft.search.ItemRarityTier;

/**
 * Common contract for TarkovCraft items that define grid dimensions, rarity tier, and optional weight.
 */
public interface TarkovSizedItem {
	ItemSize getStoredItemSize();

	ItemRarityTier getRarityTier();

	default float getWeightKg() {
		return -1.0f;
	}
}
