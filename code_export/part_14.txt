	private static synchronized void ensurePools() {
		if (poolsInitialized) {
			return;
		}
		poolsInitialized = true;

		List<Item> all = new ArrayList<>();
		all.add(ModItems.SLING_BACKPACK);
		all.add(ModItems.BERKUT_BACKPACK);
		all.add(ModItems.TRIZIP_BACKPACK);
		all.add(ModItems.SCAV_VEST);
		all.add(ModItems.BANK_ROBBER_RIG);
		all.add(ModItems.TRITON_RIG);
		all.add(ModItems.SSH68_HELMET);
		all.add(ModItems.RATNIK_6B47_HELMET);
		all.add(ModItems.ALTYN_HELMET);
		all.add(ModItems.BALACLAVA_MASK);
		all.add(ModItems.DEATH_SHADOW_MASK);
		all.add(ModItems.PACA_ARMOR);
		all.add(ModItems.PRESS_ARMOR);
		all.add(ModItems.ZABRALO_ARMOR);
		all.add(ModItems.BANDAGE);
		all.add(ModItems.CAT_TOURNIQUET);
		all.add(ModItems.SPLINT);
		all.add(ModItems.ALU_SPLINT);
		all.add(ModItems.ANALGIN);
		all.add(ModItems.AI2_MEDKIT);
		all.add(ModItems.SALEWA_MEDKIT);
		all.add(ModItems.CMS_KIT);
		all.add(ModItems.SURV12_KIT);
		all.addAll(TarkovItemCatalog.getArmorItems());
		all.addAll(TarkovItemCatalog.getRigItems());
		all.addAll(TarkovItemCatalog.getBackpackItems());
		all.addAll(TarkovItemCatalog.getMedicalItems());
		all.addAll(TarkovItemCatalog.getFoodItems());
		all.addAll(TarkovItemCatalog.getTacticalItems());

		for (Item item : all) {
			String path = Registries.ITEM.getId(item).getPath().toLowerCase(Locale.ROOT);
			ItemSize sz = ItemSizeRegistry.getSize(new ItemStack(item));
			boolean is1x1 = sz.width() == 1 && sz.height() == 1;

			if (item instanceof TarkovArmorItem armor) {
				int ac = armor.getArmorClass();
				if (armor.getTargetSlot() == EquipmentSlotType.BODY_ARMOR) {
					if (ac >= 1 && ac <= 3) ARMOR_1_TO_3.add(item);
					if (ac >= 2 && ac <= 5) ARMOR_2_TO_5.add(item);
					if (ac >= 4) ARMOR_4_TO_6.add(item);
				} else if (armor.getTargetSlot() == EquipmentSlotType.HELMET) {
					if (ac >= 1 && ac <= 3) HELMETS_1_TO_3.add(item);
					if (ac >= 2 && ac <= 5) HELMETS_2_TO_5.add(item);
					if (ac >= 4) HELMETS_4_TO_6.add(item);
				} else if (armor.getTargetSlot() == EquipmentSlotType.FACE_MASK) {
					if (ac <= 1) COSMETIC_MASKS.add(item);
					if (ac >= 2) TACTICAL_MASKS.add(item);
				}
			} else if (item instanceof TarkovArmorRepairItem) {
				PLATES_REPAIR.add(item);
			} else if (item instanceof TacticalRigItem rig) {
				ALL_RIGS.add(item);
				int cells = rig.getCompartmentLayouts().stream().mapToInt(s -> s.width() * s.height()).sum();
				if (cells <= 20) LIGHT_RIGS.add(item);
				if (cells >= 16) LARGE_RIGS.add(item);
			} else if (item instanceof BackpackItem bp) {
				ALL_BACKPACKS.add(item);
				int cells = bp.getGridWidth() * bp.getGridHeight();
				if (cells <= 20) LIGHT_BACKPACKS.add(item);
				if (cells >= 12 && cells <= 30) MEDIUM_BACKPACKS.add(item);
				if (cells >= 24) LARGE_BACKPACKS.add(item);
			} else if (item instanceof TarkovMedicalItem med) {
				ALL_MEDS.add(item);
				if (is1x1) ONE_BY_ONE_MEDS.add(item);
				switch (med.getMedicalType()) {
					case BANDAGE, ARMY_BANDAGE, CAT_TOURNIQUET, ESMARCH, SPLINT, ANALGIN, AI2_MEDKIT, CAR_MEDKIT ->
						BASIC_MEDS.add(item);
					case CMS_KIT, SURV12_KIT, GRIZZLY, AFAK, IFAK, SALEWA_MEDKIT ->
						HIGH_TIER_MEDS.add(item);
					default -> {
					}
				}
			} else if (item instanceof TarkovStimulatorItem stim) {
				ALL_STIMS.add(item);
				if (is1x1) ONE_BY_ONE_STIMS.add(item);
				switch (stim.getPreset()) {
					case PROPITAL, ETG_CHANGE, SJ6, MULE, OBDOLBOS_2, PNB, TRIMADOL, ZAGUSTIN, PERFOTORAN ->
						TOP_STIMS.add(item);
					default -> {
					}
				}
			} else if (item instanceof TarkovFoodItem) {
				ALL_FOOD.add(item);
				if (is1x1) SMALL_FOOD.add(item);
			} else if (item instanceof TarkovGrenadeItem) {
				ALL_GRENADES.add(item);
				if (is1x1) ONE_BY_ONE_GRENADES.add(item);
			} else if (item instanceof TarkovCaseItem caze) {
				ALL_CASES.add(item);
				if (caze.isMedicalCase()) MED_CASES.add(item);
				if (sz.width() * sz.height() <= 2) SMALL_CASES.add(item);
				if (caze.getGridWidth() * caze.getGridHeight() >= 25) LARGE_CASES.add(item);
			} else if (item instanceof TarkovMeleeItem) {
				MELEE_WEAPONS.add(item);
			} else if (item instanceof TarkovFunctionalItem fn && "headset".equalsIgnoreCase(fn.getCategory())) {
				HEADSETS.add(item);
			}
		}
	}

	public static ItemStack createTaczGunStack(GunPreset preset, AmmoTier loadedTier, Random random) {
		if (preset == null || !Registries.ITEM.containsId(TACZ_GUN_ITEM_ID)) {
			return ItemStack.EMPTY;
		}
		Item gunItem = Registries.ITEM.get(TACZ_GUN_ITEM_ID);
		ItemStack stack = new ItemStack(gunItem, 1);
		NbtCompound nbt = stack.getOrCreateNbt();
		nbt.putString("GunId", preset.gunId());
		nbt.putString("GunFireMode", preset.fireMode());
		int minAmmo = Math.max(1, preset.magSize() / 2);
		int loadedCount = minAmmo + random.nextInt(Math.max(1, preset.magSize() - minAmmo + 1));
		nbt.putInt("GunCurrentAmmoCount", loadedCount);
		nbt.putBoolean("HasBulletInBarrel", true);
		AmmoTier.stampLoadedTierOnGun(stack, loadedTier != null ? loadedTier : AmmoTier.FMJ);
		return stack;
	}

	public static void populateLootGrid(GridContainer grid, TarkovLootPool pool, Random random) {
		if (grid == null || pool == null || random == null) {
			return;
		}
		ensurePools();

		// Guarantee / chance for real TaCZ firearms + matching tiered ammo in military crates and airdrops
		if (pool == TarkovLootPool.MILITARY && random.nextFloat() < 0.45f) {
			GunPreset gun = MILITARY_GUNS.get(random.nextInt(MILITARY_GUNS.size()));
			AmmoTier tier = random.nextFloat() < 0.35f ? AmmoTier.AP : AmmoTier.FMJ;
			ItemStack gunStack = createTaczGunStack(gun, tier, random);
			if (!gunStack.isEmpty()) {
				grid.tryAutoPlace(gunStack);
				grid.tryAutoPlace(AmmoTier.createTaczAmmoStack(gun.ammoId(), tier, Math.min(30, Math.max(10, gun.magSize()))));
			}
		} else if (pool == TarkovLootPool.AIRDROP) {
			int gunCount = random.nextBoolean() ? 2 : 1;
			for (int g = 0; g < gunCount; g++) {
				GunPreset gun = AIRDROP_GUNS.get(random.nextInt(AIRDROP_GUNS.size()));
				AmmoTier tier = random.nextFloat() < 0.45f ? AmmoTier.IGN : AmmoTier.AP;
				ItemStack gunStack = createTaczGunStack(gun, tier, random);
				if (!gunStack.isEmpty()) {
					grid.tryAutoPlace(gunStack);
					grid.tryAutoPlace(AmmoTier.createTaczAmmoStack(gun.ammoId(), tier, Math.min(40, Math.max(15, gun.magSize()))));
				}
			}
		}

		int baseRolls = switch (pool) {
			case MILITARY, MEDICAL -> 4 + random.nextInt(4);
			case CIVILIAN -> 3 + random.nextInt(4);
			case AIRDROP -> 8 + random.nextInt(5);
			case SCAV_POCKET -> 1 + random.nextInt(3);
		};
		float mult = TarkovConfig.get().lootDensityMultiplier;
		int rolls = Math.max(1, Math.round(baseRolls * mult));

		for (int i = 0; i < rolls; i++) {
			ItemStack candidate = rollItem(pool, random);
			if (!candidate.isEmpty()) {
				grid.tryAutoPlace(candidate);
			}
		}
	}

	public static ItemStack rollItem(TarkovLootPool pool, Random random) {
		ensurePools();
		return switch (pool) {
			case MILITARY -> rollMilitary(random);
			case MEDICAL -> rollMedical(random);
			case CIVILIAN -> rollCivilian(random);
			case AIRDROP -> rollAirdrop(random);
			case SCAV_POCKET -> rollScavPocket(random);
		};
	}

	private static ItemStack rollMilitary(Random random) {
		int p = random.nextInt(100);
		if (p < 30) {
			return rollAmmo(random, 45, 20, 30, 5, 12, 30);
		}
		if (p < 50) {
			return random.nextBoolean() ? pickStack(ARMOR_2_TO_5, random, 1) : pickStack(HELMETS_2_TO_5, random, 1);
		}
		if (p < 62) {
			return pickStack(PLATES_REPAIR, random, 1);
		}
		if (p < 76) {
			return random.nextBoolean() ? pickStack(ALL_RIGS, random, 1) : pickStack(MEDIUM_BACKPACKS, random, 1);
		}
		if (p < 88) {
			return pickStack(ALL_GRENADES, random, 1 + random.nextInt(2));
		}
		int sub = random.nextInt(3);
		if (sub == 0) return pickStack(HEADSETS, random, 1);
		if (sub == 1) return pickStack(TACTICAL_MASKS, random, 1);
		return pickStack(MELEE_WEAPONS, random, 1);
	}

	private static ItemStack rollMedical(Random random) {
		int p = random.nextInt(100);
		if (p < 48) {
			return pickStack(ALL_MEDS, random, 1);
		}
		if (p < 82) {
			return pickStack(ALL_STIMS, random, 1);
		}
		if (p < 96) {
			return pickStack(ALL_FOOD, random, 1 + random.nextInt(2));
		}
		return pickStack(MED_CASES.isEmpty() ? ALL_CASES : MED_CASES, random, 1);
	}

	private static ItemStack rollCivilian(Random random) {
		int p = random.nextInt(100);
		if (p < 35) {
			return pickStack(ALL_FOOD, random, 1 + random.nextInt(3));
		}
		if (p < 57) {
			return pickStack(COSMETIC_MASKS, random, 1);
		}
		if (p < 69) {
			return random.nextBoolean() ? pickStack(LIGHT_RIGS, random, 1) : pickStack(LIGHT_BACKPACKS, random, 1);
		}
		if (p < 84) {
			return pickStack(BASIC_MEDS, random, 1);
		}
		if (p < 96) {
			Item[] mats = {Items.IRON_INGOT, Items.COPPER_INGOT, Items.LEATHER, Items.GUNPOWDER, Items.GOLD_INGOT};
			return new ItemStack(mats[random.nextInt(mats.length)], 2 + random.nextInt(5));
		}
		return pickStack(SMALL_CASES.isEmpty() ? ALL_CASES : SMALL_CASES, random, 1);
	}

	private static ItemStack rollAirdrop(Random random) {
		int p = random.nextInt(100);
		if (p < 24) {
			return random.nextBoolean() ? pickStack(ARMOR_4_TO_6, random, 1) : pickStack(HELMETS_4_TO_6, random, 1);
		}
		if (p < 46) {
			return rollAmmo(random, 0, 0, 55, 45, 20, 40);
		}
		if (p < 60) {
			return pickStack(TOP_STIMS.isEmpty() ? ALL_STIMS : TOP_STIMS, random, 1);
		}
		if (p < 74) {
			return random.nextBoolean() ? pickStack(LARGE_BACKPACKS, random, 1) : pickStack(LARGE_RIGS, random, 1);
		}
		if (p < 84) {
			return pickStack(LARGE_CASES.isEmpty() ? ALL_CASES : LARGE_CASES, random, 1);
		}
		int sub = random.nextInt(3);
		if (sub == 0) return pickStack(ALL_GRENADES, random, 1 + random.nextInt(3));
		if (sub == 1) return pickStack(HIGH_TIER_MEDS.isEmpty() ? ALL_MEDS : HIGH_TIER_MEDS, random, 1);
		return pickStack(PLATES_REPAIR, random, 1);
	}

	private static ItemStack rollScavPocket(Random random) {
		int p = random.nextInt(100);
		if (p < 32) {
			return pickStack(SMALL_FOOD.isEmpty() ? ALL_FOOD : SMALL_FOOD, random, 1 + random.nextInt(2));
		}
		if (p < 56) {
			return pickStack(ONE_BY_ONE_MEDS.isEmpty() ? BASIC_MEDS : ONE_BY_ONE_MEDS, random, 1);
		}
		if (p < 76) {
			return rollAmmo(random, 70, 30, 0, 0, 8, 16);
		}
		if (p < 86) {
			return pickStack(ONE_BY_ONE_STIMS.isEmpty() ? ALL_STIMS : ONE_BY_ONE_STIMS, random, 1);
		}
		if (p < 94) {
			return pickStack(ONE_BY_ONE_GRENADES.isEmpty() ? ALL_GRENADES : ONE_BY_ONE_GRENADES, random, 1);
		}
		Item[] mats = {Items.IRON_INGOT, Items.COPPER_INGOT, Items.GUNPOWDER};
		return new ItemStack(mats[random.nextInt(mats.length)], 1 + random.nextInt(3));
	}

	private static ItemStack rollAmmo(
		Random random,
		int wFmj,
		int wHp,
		int wAp,
		int wIgn,
		int minCount,
		int maxCount
	) {
		List<TaCZCaliberCatalog.CaliberSpec> calibers = TaCZCaliberCatalog.getDefaultCalibers();
		TaCZCaliberCatalog.CaliberSpec spec = calibers.get(random.nextInt(calibers.size()));
		int total = Math.max(1, wFmj + wHp + wAp + wIgn);
		int roll = random.nextInt(total);
		AmmoTier tier;
		if (roll < wFmj) {
			tier = AmmoTier.FMJ;
		} else if (roll < wFmj + wHp) {
			tier = AmmoTier.HP;
		} else if (roll < wFmj + wHp + wAp) {
			tier = AmmoTier.AP;
		} else {
			tier = AmmoTier.IGN;
		}
		int count = minCount + random.nextInt(Math.max(1, maxCount - minCount + 1));
		return AmmoTier.createTaczAmmoStack(spec.ammoId(), tier, count);
	}

	private static ItemStack pickStack(List<Item> pool, Random random, int count) {
		if (pool.isEmpty()) {
			return ItemStack.EMPTY;
		}
		Item item = pool.get(random.nextInt(pool.size()));
		int clampedCount = Math.min(count, item.getMaxCount());
		return new ItemStack(item, Math.max(1, clampedCount));
	}

	public static ItemStack pickRandomLightHelmet(Random random) {
		ensurePools();
		return pickStack(HELMETS_1_TO_3, random, 1);
	}

	public static ItemStack pickRandomLightArmor(Random random) {
		ensurePools();
		return pickStack(ARMOR_1_TO_3, random, 1);
	}

	public static ItemStack pickRandomScavRig(Random random) {
		ensurePools();
		return pickStack(LIGHT_RIGS.isEmpty() ? ALL_RIGS : LIGHT_RIGS, random, 1);
	}

	public static ItemStack pickRandomScavBackpack(Random random) {
		ensurePools();
		return pickStack(LIGHT_BACKPACKS.isEmpty() ? ALL_BACKPACKS : LIGHT_BACKPACKS, random, 1);
	}
}

	@Inject(method = "renderHotbar", at = @At("HEAD"), cancellable = true)
	private void tarkovcraft$renderTarkovHud(float tickDelta, DrawContext context, CallbackInfo ci) {
		if (client.player == null || client.player.isCreative() || client.player.isSpectator()) {
			WorldInteractState.updateTarget(null);
			return;
		}
		ci.cancel();

		if (client.currentScreen instanceof TarkovInventoryScreen || client.currentScreen instanceof TarkovDeathScreen) {
			WorldInteractState.updateTarget(null);
			return;
		}

		PlayerLoadout loadout = ClientLoadoutCache.getLoadout();
		int screenW = client.getWindow().getScaledWidth();
		int screenH = client.getWindow().getScaledHeight();

		// 1. Top-right Tarkov Body-Part Silhouette HUD
		BodyHealthHudHelper.drawCompactTopRight(context, client.textRenderer, client.player, screenW);

		// 2. Center crosshair [F] prompt / scrollable action menu

			renderWorldInteractPrompt(context, loadout, screenW, screenH);
		} else {
			WorldInteractState.updateTarget(null);
		}

		// 3. Tarkov Quick-Use Hotbar (dynamic keybind labels)
		int slotSize = 24;
		int gap = 2;
		int groupGap = 8;
		int totalW = slotSize + groupGap + (3 * slotSize + 2 * gap) + groupGap + (6 * slotSize + 5 * gap);
		int startX = (screenW - totalW) / 2;
		int y = screenH - slotSize - 6;

		int selected = client.player.getInventory().selectedSlot;
		boolean meleeActive = loadout.isMeleeSelected();

		int x = startX;
		String meleeKey = TarkovCraftClient.getShortKeyLabel(TarkovCraftClient.toggleMeleeKey);
		int mBg = meleeActive ? 0xDD3D3628 : 0xBB14171B;
		int mBorder = meleeActive ? 0xFFD4B46A : 0xFF424A52;
		drawHotbarBox(context, x, y, slotSize, mBg, mBorder, meleeKey, loadout.getEquipment(EquipmentSlotType.MELEE));
		x += slotSize + groupGap;

		for (int i = 0; i < 3; i++) {
			boolean isSel = !meleeActive && selected == i;
			int bg = isSel ? 0xDD2E3A2B : 0xBB14171B;
			int border = isSel ? 0xFFD4B46A : 0xFF424A52;
			String keyLabel = TarkovCraftClient.getShortKeyLabel(client.options.hotbarKeys[i]);
			drawHotbarBox(context, x, y, slotSize, bg, border, keyLabel, loadout.getHotbarDisplayStack(i));
			x += slotSize + gap;
		}
		x += groupGap - gap;

		for (int i = 3; i < 9; i++) {
			boolean isSel = !meleeActive && selected == i;
			int bg = isSel ? 0xDD2E3A2B : 0xBB181C20;
			int border = isSel ? 0xFFD4B46A : 0xFF3A4148;
			String keyLabel = TarkovCraftClient.getShortKeyLabel(client.options.hotbarKeys[i]);
			drawHotbarBox(context, x, y, slotSize, bg, border, keyLabel, loadout.getHotbarDisplayStack(i));
			x += slotSize + gap;
		}

	@Unique
	private void renderWorldInteractPrompt(DrawContext context, PlayerLoadout loadout, int screenW, int screenH) {
		if (client.player == null || client.world == null) return;
		String key = TarkovCraftClient.getShortKeyLabel(TarkovCraftClient.interactKey);

		double reach = 4.2;
		Vec3d eye = client.player.getCameraPosVec(1.0f);
		Vec3d look = client.player.getRotationVec(1.0f);
		Vec3d end = eye.add(look.multiply(reach));

		double maxDist = reach;
		if (client.crosshairTarget != null && client.crosshairTarget.getType() == HitResult.Type.BLOCK) {
			maxDist = Math.min(reach, eye.distanceTo(client.crosshairTarget.getPos()));
		}

		Box searchBox = client.player.getBoundingBox().stretch(look.multiply(reach)).expand(1.2);
		EntityHitResult eHit = ProjectileUtil.raycast(
			client.player, eye, end, searchBox,
			e -> (e instanceof CorpseEntity || e instanceof ItemEntity) && !e.isRemoved(),
			maxDist * maxDist
		);
		Entity target = eHit != null ? eHit.getEntity() : TarkovNetworking.findClosestLookedItem(client.player, eye, look, maxDist);

		WorldInteractState.updateTarget(target);

		// EFT Scrollable 2-Action Menu for Backpacks & Rigs on the ground!
		if (WorldInteractState.hasMultiActionMenu() && target instanceof ItemEntity itemEntity) {
			ItemStack st = itemEntity.getStack();
			boolean slotEmpty = (st.getItem() instanceof BackpackItem && loadout.getEquipment(EquipmentSlotType.BACKPACK).isEmpty())
				|| (st.getItem() instanceof TacticalRigItem && loadout.getEquipment(EquipmentSlotType.RIG).isEmpty());

			String itemTitle = st.getName().getString();
			String opt0 = Text.translatable("action.tarkovcraft.search").getString();
			String opt1 = Text.translatable(slotEmpty ? "action.tarkovcraft.equip" : "action.tarkovcraft.take").getString();
			int sel = WorldInteractState.getSelectedAction();

			int boxW = Math.max(120, Math.max(client.textRenderer.getWidth(itemTitle) + 18, 132));
			int bx = screenW / 2 + 12;
			int by = screenH / 2 - 14;

			// Header with item name
			context.fill(bx, by, bx + boxW, by + 13, 0xDD0C0F12);
			context.drawBorder(bx, by, boxW, 43, 0xFF3A424A);
			context.drawTextWithShadow(client.textRenderer, itemTitle, bx + 4, by + 3, 0xFFB8C0C8);

			// Option 0: Search (top, default)
			drawWorldMenuRow(context, bx + 1, by + 13, boxW - 2, 14, key, opt0, sel == WorldInteractState.ACTION_SEARCH, "↑↓");
			// Option 1: Equip / Take (bottom, scroll down)
			drawWorldMenuRow(context, bx + 1, by + 27, boxW - 2, 15, key, opt1, sel == WorldInteractState.ACTION_EQUIP_OR_TAKE, "");
			return;
		}

		// Standard single-line prompt
		Text prompt = null;
		if (target instanceof CorpseEntity corpse) {
			prompt = Text.translatable("prompt.tarkovcraft.search", key, corpse.getOwnerName());
		} else if (target instanceof ItemEntity itemEntity && !itemEntity.getStack().isEmpty()) {
			ItemStack st = itemEntity.getStack();
			boolean willEquip = st.getItem() instanceof TarkovArmorItem a && loadout.getEquipment(a.getTargetSlot()).isEmpty();
			prompt = Text.translatable(willEquip ? "prompt.tarkovcraft.equip" : "prompt.tarkovcraft.take", key, st.getName().getString());
		} else if (client.crosshairTarget instanceof BlockHitResult bhr && bhr.getType() == HitResult.Type.BLOCK) {
			BlockState state = client.world.getBlockState(bhr.getBlockPos());
			if (state.getBlock() instanceof SearchableLootBlock lootBlock) {
				prompt = Text.translatable("prompt.tarkovcraft.search", key, Text.translatable(lootBlock.getContainerTitleKey()).getString());
			} else if (state.getBlock() instanceof TarkovWorkbenchBlock wb) {
				prompt = Text.translatable("prompt.tarkovcraft.workbench", key, wb.getName().getString());
			}
		}

		if (prompt != null) {
			int pw = client.textRenderer.getWidth(prompt) + 14;
			int px = (screenW - pw) / 2;
			int py = screenH / 2 + 18;
			context.fill(px, py, px + pw, py + 16, 0xCC121619);
			context.drawBorder(px, py, pw, 16, 0xFFD4B46A);
			context.drawCenteredTextWithShadow(client.textRenderer, prompt, screenW / 2, py + 4, 0xFFF2F2F2);
		}
	}

	@Unique
	private void drawWorldMenuRow(DrawContext context, int x, int y, int w, int h, String key, String label, boolean selected, String hint) {
		int bg = selected ? 0xE6343D30 : 0xCC15191D;
		int textCol = selected ? 0xFFF4F4F4 : 0xFF8C96A0;
		context.fill(x, y, x + w, y + h, bg);
		if (selected) {
			context.fill(x, y, x + 2, y + h, 0xFFD4B46A);
			context.drawTextWithShadow(client.textRenderer, "[" + key + "] " + label, x + 5, y + 3, 0xFFD4B46A);
		} else {
			context.drawTextWithShadow(client.textRenderer, "    " + label, x + 5, y + 3, textCol);
		}
		if (!hint.isEmpty()) {
			int hw = client.textRenderer.getWidth(hint);
			context.drawTextWithShadow(client.textRenderer, hint, x + w - hw - 4, y + 3, 0xFF7D8791);
		}
	}

	@Unique
	private void drawHotbarBox(DrawContext context, int x, int y, int size, int bg, int border, String label, ItemStack stack) {
		context.fill(x, y, x + size, y + size, bg);
		context.drawBorder(x, y, size, size, border);
		if (!stack.isEmpty()) {
			context.drawItem(stack, x + 4, y + 4);
			context.drawItemInSlot(client.textRenderer, stack, x + 4, y + 4);
		}
		context.getMatrices().push();
		context.getMatrices().scale(0.65f, 0.65f, 1.0f);
		int bx = (int) ((x + 2) / 0.65f);
		int by = (int) ((y + 2) / 0.65f);
		context.drawTextWithShadow(client.textRenderer, label, bx, by, 0xFFD8DEE4);
		context.getMatrices().pop();
	}
}

/**

 * as 3D gear models instead of tiny floating spinning 2D sprites.
 */
@Mixin(ItemEntityRenderer.class)
public abstract class ItemEntityRendererMixin {
	@Unique
	private static final Identifier GEAR_TEX = new Identifier("minecraft", "textures/block/white_wool.png");

	@Inject(method = "render(Lnet/minecraft/entity/ItemEntity;FFLnet/minecraft/client/util/math/MatrixStack;Lnet/minecraft/client/render/VertexConsumerProvider;I)V",
		at = @At("HEAD"), cancellable = true)
	private void tarkovcraft$renderGroundedGear(
		ItemEntity entity, float yaw, float tickDelta,
		MatrixStack matrices, VertexConsumerProvider vertexConsumers, int light,
		CallbackInfo ci
	) {
		ItemStack stack = entity.getStack();
		if (stack.isEmpty()) return;
		Item item = stack.getItem();
		if (!(item instanceof BackpackItem || item instanceof TacticalRigItem || item instanceof TarkovArmorItem)) {
			return;
		}

		ci.cancel();
		matrices.push();
		// Fixed rotation based on entity UUID so it stays still on the ground
		float groundYaw = (entity.getUuid().hashCode() & 0xFF) * (360.0f / 256.0f);
		matrices.multiply(RotationAxis.POSITIVE_Y.rotationDegrees(groundYaw));

		VertexConsumer vc = vertexConsumers.getBuffer(RenderLayer.getEntitySolid(GEAR_TEX));
		if (item instanceof BackpackItem bp) {
			float scale = Math.min(1.25f, 0.7f + (bp.getGridWidth() * bp.getGridHeight()) * 0.02f);
			// Main backpack body lying flat
			drawBox(matrices, vc, -0.22f * scale, 0.0f, -0.28f * scale, 0.22f * scale, 0.20f * scale, 0.28f * scale, light, 82, 96, 70);
			// Top pouch + straps
			drawBox(matrices, vc, -0.17f * scale, 0.20f * scale, -0.18f * scale, 0.17f * scale, 0.25f * scale, 0.18f * scale, light, 60, 72, 52);

			drawBox(matrices, vc, 0.12f * scale, 0.02f, -0.30f * scale, 0.20f * scale, 0.14f * scale, 0.30f * scale, light, 42, 46, 38);

			drawBox(matrices, vc, -0.22f, 0.09f, -0.12f, -0.10f, 0.16f, 0.12f, light, 68, 63, 52);
			drawBox(matrices, vc, -0.06f, 0.09f, -0.12f, 0.06f, 0.16f, 0.12f, light, 68, 63, 52);
			drawBox(matrices, vc, 0.10f, 0.09f, -0.12f, 0.22f, 0.16f, 0.12f, light, 68, 63, 52);

	@Unique
	private static void drawBox(
		MatrixStack matrices, VertexConsumer vc,
		float x1, float y1, float z1, float x2, float y2, float z2,
		int light, int r, int g, int b
	) {
		Matrix4f mat = matrices.peek().getPositionMatrix();
		Matrix3f norm = matrices.peek().getNormalMatrix();
		quad(vc, mat, norm, x1, y2, z2, x2, y2, z2, x2, y2, z1, x1, y2, z1, 0, 1, 0, light, r, g, b);
		quad(vc, mat, norm, x1, y1, z1, x2, y1, z1, x2, y1, z2, x1, y1, z2, 0, -1, 0, light, r, g, b);
		quad(vc, mat, norm, x2, y1, z1, x1, y1, z1, x1, y2, z1, x2, y2, z1, 0, 0, -1, light, r, g, b);
		quad(vc, mat, norm, x1, y1, z2, x2, y1, z2, x2, y2, z2, x1, y2, z2, 0, 0, 1, light, r, g, b);
		quad(vc, mat, norm, x1, y1, z1, x1, y1, z2, x1, y2, z2, x1, y2, z1, -1, 0, 0, light, r, g, b);
		quad(vc, mat, norm, x2, y1, z2, x2, y1, z1, x2, y2, z1, x2, y2, z2, 1, 0, 0, light, r, g, b);
	}

	@Unique
	private static void quad(
		VertexConsumer vc, Matrix4f mat, Matrix3f norm,
		float x1, float y1, float z1, float x2, float y2, float z2,
		float x3, float y3, float z3, float x4, float y4, float z4,
		float nx, float ny, float nz, int light, int r, int g, int b
	) {
		vc.vertex(mat, x1, y1, z1).color(r, g, b, 255).texture(0, 0).overlay(OverlayTexture.DEFAULT_UV).light(light).normal(norm, nx, ny, nz).next();
		vc.vertex(mat, x2, y2, z2).color(r, g, b, 255).texture(1, 0).overlay(OverlayTexture.DEFAULT_UV).light(light).normal(norm, nx, ny, nz).next();
		vc.vertex(mat, x3, y3, z3).color(r, g, b, 255).texture(1, 1).overlay(OverlayTexture.DEFAULT_UV).light(light).normal(norm, nx, ny, nz).next();
		vc.vertex(mat, x4, y4, z4).color(r, g, b, 255).texture(0, 1).overlay(OverlayTexture.DEFAULT_UV).light(light).normal(norm, nx, ny, nz).next();
	}
}

import net.fabricmc.api.EnvType;

import org.spongepowered.asm.mixin.injection.At;
import org.spongepowered.asm.mixin.injection.Inject;
import org.spongepowered.asm.mixin.injection.callback.CallbackInfo;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

/**
 * Allows multiple KeyBindings on the same physical key (e.g. F for TarkovCraft interact + vanilla swapOffhand,
 * or TaCZ shoot/aim alongside vanilla attack/use) without overwriting each other in KEY_TO_BINDINGS.
 */
@Environment(EnvType.CLIENT)
@Mixin(KeyBinding.class)
public abstract class KeyBindingMixin {
	@Shadow
	@Final
	private static Map<String, KeyBinding> KEYS_BY_ID;

	@Shadow
	@Final
	private static Map<InputUtil.Key, KeyBinding> KEY_TO_BINDINGS;

	@Shadow
	private InputUtil.Key boundKey;

	@Shadow
	private int timesPressed;

	@Unique
	private static final Map<InputUtil.Key, List<KeyBinding>> TARKOV_KEY_TO_BINDINGS = new HashMap<>();

	@Unique
	private static void tarkovcraft$registerKey(KeyBinding kb) {
		InputUtil.Key k = ((KeyBindingMixin) (Object) kb).boundKey;
		if (k == null || k.equals(InputUtil.UNKNOWN_KEY)) {
			return;
		}
		List<KeyBinding> list = TARKOV_KEY_TO_BINDINGS.computeIfAbsent(k, ignored -> new ArrayList<>(2));
		if (!list.contains(kb)) {
			list.add(kb);
		}
	}

	@Inject(method = "<init>(Ljava/lang/String;Lnet/minecraft/client/util/InputUtil$Type;ILjava/lang/String;)V", at = @At("TAIL"))
	private void tarkovcraft$onConstruct(String translationKey, InputUtil.Type type, int code, String category, CallbackInfo ci) {
		tarkovcraft$registerKey((KeyBinding) (Object) this);
	}

	@Inject(method = "updateKeysByCode", at = @At("TAIL"))
	private static void tarkovcraft$onUpdateKeysByCode(CallbackInfo ci) {
		TARKOV_KEY_TO_BINDINGS.clear();
		for (KeyBinding kb : KEYS_BY_ID.values()) {
			tarkovcraft$registerKey(kb);
		}
	}

	@Inject(method = "onKeyPressed", at = @At("TAIL"))
	private static void tarkovcraft$onKeyPressed(InputUtil.Key key, CallbackInfo ci) {
		if (key == null || key.equals(InputUtil.UNKNOWN_KEY)) {
			return;
		}
		List<KeyBinding> list = TARKOV_KEY_TO_BINDINGS.get(key);
		if (list == null || list.size() <= 1) {
			return;
		}
		KeyBinding vanillaHandled = KEY_TO_BINDINGS.get(key);
		for (KeyBinding kb : list) {
			if (kb != vanillaHandled) {
				((KeyBindingMixin) (Object) kb).timesPressed++;
			}
		}
	}

	@Inject(method = "setKeyPressed", at = @At("TAIL"))
	private static void tarkovcraft$onSetKeyPressed(InputUtil.Key key, boolean pressed, CallbackInfo ci) {
		if (key == null || key.equals(InputUtil.UNKNOWN_KEY)) {
			return;
		}
		List<KeyBinding> list = TARKOV_KEY_TO_BINDINGS.get(key);
		if (list == null || list.size() <= 1) {
			return;
		}
		KeyBinding vanillaHandled = KEY_TO_BINDINGS.get(key);
		for (KeyBinding kb : list) {
			if (kb != vanillaHandled) {
				kb.setPressed(pressed);
			}
		}
	}
}

package com.tarkovcraft.mixin;

import com.tarkovcraft.config.TarkovConfig;
import com.tarkovcraft.entity.CorpseEntity;
import com.tarkovcraft.entity.MobGearSpawner;
import net.minecraft.entity.LivingEntity;
import net.minecraft.entity.damage.DamageSource;
import net.minecraft.entity.mob.HostileEntity;
import net.minecraft.entity.mob.IllagerEntity;
import net.minecraft.entity.mob.PiglinEntity;
import net.minecraft.entity.mob.SkeletonEntity;
import net.minecraft.entity.mob.ZombieEntity;
import org.spongepowered.asm.mixin.Mixin;
import org.spongepowered.asm.mixin.injection.At;
import org.spongepowered.asm.mixin.injection.Inject;
import org.spongepowered.asm.mixin.injection.callback.CallbackInfo;

/**
 * Spawns a lootable CorpseEntity when humanoid hostile mobs die so players
 * can loot them and strip their unsearched backpacks whole.
 */
@Mixin(LivingEntity.class)
public abstract class LivingEntityDeathMixin {
	@Inject(method = "onDeath", at = @At("TAIL"))
	private void tarkovcraft$spawnMobCorpse(DamageSource damageSource, CallbackInfo ci) {
		LivingEntity self = (LivingEntity) (Object) this;
		if (self.getWorld().isClient) {
			return;
		}
		if (self instanceof ZombieEntity
			|| self instanceof SkeletonEntity
			|| self instanceof IllagerEntity
			|| self instanceof PiglinEntity
			|| (self instanceof HostileEntity && self.getHeight() >= 1.6f)) {
			CorpseEntity corpse = null;
			if (self.getCommandTags().contains(MobGearSpawner.TAG_GEARED_MOB)) {
				corpse = CorpseEntity.fromGearedMob(self.getWorld(), self);
			} else if (self.getWorld().random.nextFloat() < TarkovConfig.get().corpseBaseLootChance) {
				corpse = CorpseEntity.fromRegularMob(self.getWorld(), self);
			}
			if (corpse != null) {
				self.getWorld().spawnEntity(corpse);
			}
		}
	}
}

package com.tarkovcraft.mixin;

import com.tarkovcraft.client.screen.TarkovDeathScreen;
import com.tarkovcraft.network.TarkovNetworking;
import net.minecraft.client.MinecraftClient;
import net.minecraft.client.gui.screen.DeathScreen;
import net.minecraft.client.gui.screen.Screen;
import net.minecraft.client.gui.screen.ingame.InventoryScreen;
import net.minecraft.client.network.ClientPlayerEntity;
import net.minecraft.client.option.GameOptions;
import org.jetbrains.annotations.Nullable;
import org.spongepowered.asm.mixin.Final;
import org.spongepowered.asm.mixin.Mixin;
import org.spongepowered.asm.mixin.Shadow;
import org.spongepowered.asm.mixin.injection.At;
import org.spongepowered.asm.mixin.injection.Inject;
import org.spongepowered.asm.mixin.injection.callback.CallbackInfo;

/**
 * 1. Redirects Survival InventoryScreen to our server-backed TarkovInventoryScreen.
 * 2. Redirects Survival DeathScreen to our Tarkov KIA Death Screen!
 * 3. Kills vanilla swapHandsKey (offhand swap on F) in Survival/Adventure mode.
 */
@Mixin(MinecraftClient.class)
public abstract class MinecraftClientMixin {
	@Shadow
	@Nullable
	public ClientPlayerEntity player;

	@Shadow
	@Final
	public GameOptions options;

	@Shadow
	public abstract void setScreen(@Nullable Screen screen);

	@Inject(method = "setScreen", at = @At("HEAD"), cancellable = true)
	private void tarkovcraft$redirectScreens(@Nullable Screen screen, CallbackInfo ci) {
		if (player != null && !player.isCreative()) {
			if (screen instanceof InventoryScreen) {
				TarkovNetworking.sendOpenOwnInventory();
				ci.cancel();
			} else if (screen instanceof DeathScreen) {
				ci.cancel();
				this.setScreen(new TarkovDeathScreen());
			}
		}
	}

	@Inject(method = "handleInputEvents", at = @At("HEAD"))
	private void tarkovcraft$consumeSwapHandsKey(CallbackInfo ci) {
		if (player != null && !player.isCreative()) {
			while (options.swapHandsKey.wasPressed()) {
				// Intentionally swallowed — offhand swap is disabled in TarkovCraft
			}
		}
	}
}
