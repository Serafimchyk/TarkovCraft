import com.tarkovcraft.ammo.AmmoTier;
import com.tarkovcraft.ammo.TaCZCaliberCatalog;
import com.tarkovcraft.health.TaCZBulletHelper;

import java.util.concurrent.ConcurrentHashMap;

/**
 * Client-only zero-hard-dependency helper that inspects TaCZ firearms via TimelessAPI reflection
 * and extracts caliber, fire mode, ammo count, barrel chamber state, loaded tier, and allowed tiers.
 */
@Environment(EnvType.CLIENT)
public final class TaCZGunInfoHelper {
	private record GunMeta(String ammoId, int baseMagCapacity, int rpm) {
	}

	public record GunSummary(
		String caliberName,
		String ammoId,
		String fireMode,
		int currentAmmo,
		int maxMag,
		boolean bulletInBarrel,
		AmmoTier loadedTier,
		int penetration,
		List<AmmoTier> allowedTiers
	) {
	}

	private static final ConcurrentHashMap<String, GunMeta> GUN_META_CACHE = new ConcurrentHashMap<>();
	private static final GunMeta EMPTY_META = new GunMeta("", 0, 0);

	private TaCZGunInfoHelper() {
	}

	public static GunSummary inspectGun(ItemStack stack) {
		if (stack == null || stack.isEmpty()) {
			return new GunSummary("Unknown", "", "", 0, 0, false, AmmoTier.FMJ, 22, List.of(AmmoTier.values()));
		}
		NbtCompound nbt = stack.getNbt();
		String gunId = nbt != null ? nbt.getString("GunId") : "";
		int curAmmo = nbt != null ? nbt.getInt("GunCurrentAmmoCount") : 0;
		boolean barrel = nbt != null && nbt.getBoolean("HasBulletInBarrel");
		String fireMode = nbt != null ? nbt.getString("GunFireMode").toUpperCase(Locale.ROOT) : "";
		AmmoTier loadedTier = AmmoTier.getLoadedTierFromGun(stack);

		GunMeta meta = resolveGunMeta(gunId);
		String ammoId = meta.ammoId();
		if (ammoId.isBlank() && nbt != null && nbt.contains("AmmoId")) {
			ammoId = nbt.getString("AmmoId");
		}

		String caliberName;
		List<AmmoTier> allowedTiers;
		int pen;
		if (!ammoId.isBlank()) {
			TaCZCaliberCatalog.CaliberSpec spec = TaCZCaliberCatalog.getSpecOrDynamic(ammoId);
			caliberName = spec.displayName();
			allowedTiers = spec.getAllowedTiersList();
			pen = TaCZBulletHelper.computePenetration(ammoId, loadedTier);
		} else {
			caliberName = "Ballistic";
			allowedTiers = List.of(AmmoTier.values());
			pen = TaCZBulletHelper.computePenetration("", loadedTier);
		}

		return new GunSummary(
			caliberName,
			ammoId,
			fireMode,
			Math.max(0, curAmmo),
			Math.max(curAmmo, meta.baseMagCapacity()),
			barrel,
			loadedTier,
			pen,
			allowedTiers
		);
	}

	private static GunMeta resolveGunMeta(String gunId) {
		if (gunId == null || gunId.isBlank()) {
			return EMPTY_META;
		}
		return GUN_META_CACHE.computeIfAbsent(gunId, TaCZGunInfoHelper::reflectGunMeta);
	}

	private static GunMeta reflectGunMeta(String gunId) {
		try {
			Identifier id = Identifier.tryParse(gunId);
			if (id == null) {
				return EMPTY_META;
			}
			Class<?> apiClass = Class.forName("com.tacz.guns.api.TimelessAPI");
			Method getGunIndex = apiClass.getMethod("getCommonGunIndex", Identifier.class);
			Object raw = getGunIndex.invoke(null, id);
			Object gunIndex = (raw instanceof Optional<?> opt) ? opt.orElse(null) : raw;
			if (gunIndex == null) {
				return EMPTY_META;
			}
			Object gunData = gunIndex.getClass().getMethod("getGunData").invoke(gunIndex);
			if (gunData == null) {
				return EMPTY_META;
			}

			String ammoId = "";
			try {
				Object rawAmmo = gunData.getClass().getMethod("getAmmoId").invoke(gunData);
				if (rawAmmo != null) {
					ammoId = rawAmmo.toString();
				}
			} catch (Throwable ignored) {
			}

			int magCap = 0;
			try {
				Object rawAmt = gunData.getClass().getMethod("getAmmoAmount").invoke(gunData);
				if (rawAmt instanceof Number n) {
					magCap = n.intValue();
				}
			} catch (Throwable ignored) {
			}

			int rpm = 0;
			try {
				Method mRpm = gunData.getClass().getMethod("getRoundsPerMinute");
				Object rawRpm = mRpm.invoke(gunData);
				if (rawRpm instanceof Number n) {
					rpm = n.intValue();
				}
			} catch (Throwable ignored) {
			}

			return new GunMeta(ammoId, Math.max(0, magCap), Math.max(0, rpm));
		} catch (Throwable ignored) {
			return EMPTY_META;
		}
	}
}

				return;
			}
			if (AmmoTier.isTaczAmmoStack(stack)) {
				AmmoTier tier = AmmoTier.getFromAmmoStack(stack);
				if (!lines.isEmpty()) {
					lines.set(0, lines.get(0).copy().append(" ")
						.append(Text.literal("[" + tier.getBadge() + "]").formatted(tier.getColor())));

					if (tier == AmmoTier.HP) {
						lines.add(idx++, Text.translatable("tooltip.tarkovcraft.ammo_shotgun_hp").formatted(Formatting.GREEN));
					} else if (tier == AmmoTier.FMJ) {
						lines.add(idx++, Text.translatable("tooltip.tarkovcraft.ammo_shotgun_fmj").formatted(Formatting.GRAY));
					} else if (tier == AmmoTier.AP) {
						lines.add(idx++, Text.translatable("tooltip.tarkovcraft.ammo_shotgun_ap").formatted(Formatting.GOLD));
					}
				}
				if (tier == AmmoTier.FMJ) {
					lines.add(idx, Text.translatable("tooltip.tarkovcraft.ammo_upgrade_hint").formatted(Formatting.DARK_GRAY));
				}
				return;
			}
			if (AmmoTier.isTaczGunStack(stack)) {
				TaCZGunInfoHelper.GunSummary info = TaCZGunInfoHelper.inspectGun(stack);
				int idx = Math.min(1, lines.size());
				lines.add(idx++, Text.translatable("tooltip.tarkovcraft.gun_caliber", info.caliberName())
					.formatted(Formatting.GOLD));
				if (info.maxMag() > 0 || info.currentAmmo() > 0) {
					if (info.bulletInBarrel()) {
						lines.add(idx++, Text.translatable("tooltip.tarkovcraft.gun_ammo_count_chamber",

					} else {
						lines.add(idx++, Text.translatable("tooltip.tarkovcraft.gun_ammo_count",

					}
				}
				if (!info.fireMode().isBlank()) {
					lines.add(idx++, Text.translatable("tooltip.tarkovcraft.gun_fire_mode", info.fireMode())
						.formatted(Formatting.GRAY));
				}
				lines.add(idx++, Text.translatable("tooltip.tarkovcraft.gun_loaded_tier_full",
					info.loadedTier().getBadge(),
					info.penetration(),
					String.format(Locale.ROOT, "%.2f", info.loadedTier().getDamageMultiplier()))
					.formatted(info.loadedTier().getColor()));
				String allowedStr = String.join(", ", info.allowedTiers().stream().map(AmmoTier::getBadge).toList());
				lines.add(idx, Text.translatable("tooltip.tarkovcraft.gun_allowed_tiers", allowedStr)
					.formatted(Formatting.DARK_GRAY));
				return;
			}
			if (!stack.isFood()) {
				return;
			}
			FoodComponent fc = stack.getItem().getFoodComponent();
			if (fc == null) {
				return;
			}
			int energy = fc.getHunger() * 5;
			int sat = Math.round(fc.getHunger() * fc.getSaturationModifier() * 10.0f);
			int insertIdx = Math.min(1, lines.size());
			lines.add(insertIdx, Text.translatable("tooltip.tarkovcraft.food_stats", energy, sat)
				.formatted(Formatting.GREEN));
		});

		));

				: com.tarkovcraft.audio.HearingProfile.forLoadout(ClientLoadoutCache.getLoadout()));
			while (ammoRadialKey.wasPressed()) {
				if (client.player != null && client.currentScreen == null && client.player.isAlive()
						&& !client.player.isSpectator() && ClientPlayNetworking.canSend(AmmoSelectionNetworking.SELECT)) {
					ItemStack gun = client.player.getMainHandStack();
					String ammoId = AmmoReloadTracker.getGunAmmoId(gun);
					if (gun.getNbt() != null && TaCZCaliberCatalog.isBallisticAmmoId(ammoId)) {
						String gunId = gun.getNbt().getString("GunId");
						if (!gunId.isEmpty() && gunId.length() <= AmmoSelectionNetworking.MAX_GUN_ID_LENGTH) {
							client.setScreen(new AmmoRadialScreen(gunId, ammoId, AmmoReloadTracker.countAvailableAmmo(client.player, ammoId)));
						}
					}
				}
			}
			while (toggleMeleeKey.wasPressed()) {
				if (client.player != null && !client.player.isCreative()) {
					TarkovNetworking.sendToggleMelee();
				}
			}
			while (interactKey.wasPressed()) {
				if (client.player != null && client.currentScreen == null) {
					TarkovNetworking.sendInteract((byte) WorldInteractState.getSelectedAction());
				}
			}
			while (toggleHitboxesKey.wasPressed()) {
				boolean on = HitZoneDebugRenderer.toggle();
				if (client.player != null) {
					client.player.sendMessage(Text.translatable(
						on ? "message.tarkovcraft.hitboxes_on" : "message.tarkovcraft.hitboxes_off"
					), true);
				}
			}
		});
	}

	public static String getShortKeyLabel(KeyBinding key) {
		if (key == null || key.isUnbound()) {
			return "-";
		}
		String raw = key.getBoundKeyLocalizedText().getString().trim();
		String lower = raw.toLowerCase();
		if (lower.contains("кнопка ") || lower.contains("button ")) {
			String digits = raw.replaceAll("[^0-9]", "");
			if (!digits.isEmpty()) {
				return "M" + digits;
			}
		}
		if (lower.equals("left button") || lower.equals("лкм")) return "LMB";
		if (lower.equals("right button") || lower.equals("пкм")) return "RMB";
		if (lower.equals("middle button") || lower.equals("скм")) return "MMB";
		if (raw.length() > 4) {
			return raw.substring(0, 3).toUpperCase();
		}
		return raw.toUpperCase();
	}
}

package com.tarkovcraft.client;

import com.tarkovcraft.item.BackpackItem;
import com.tarkovcraft.item.TacticalRigItem;
import com.tarkovcraft.item.TarkovCaseItem;

import org.jetbrains.annotations.Nullable;

/**
 * Tracks the currently targeted world interactable and the scroll-wheel selected action
 * for multi-action items on the ground (Backpacks & Tactical Rigs):
 *   0 = Search (default, top)
 *   1 = Equip / Take (scroll down, bottom)
 */
public final class WorldInteractState {
	public static final int ACTION_SEARCH = 0;
	public static final int ACTION_EQUIP_OR_TAKE = 1;

	private static int lastTargetEntityId = -1;
	private static boolean multiActionActive = false;
	private static int selectedAction = ACTION_SEARCH;

	private WorldInteractState() {
	}

	public static void updateTarget(@Nullable Entity entity) {
		if (entity instanceof ItemEntity ie
			&& !ie.getStack().isEmpty()
			&& (ie.getStack().getItem() instanceof BackpackItem
			|| ie.getStack().getItem() instanceof TacticalRigItem
			|| ie.getStack().getItem() instanceof TarkovCaseItem)) {
			if (ie.getId() != lastTargetEntityId) {
				lastTargetEntityId = ie.getId();
				selectedAction = ACTION_SEARCH; // Default top option: Search
			}
			multiActionActive = true;
		} else {
			lastTargetEntityId = -1;
			multiActionActive = false;
			selectedAction = ACTION_SEARCH;
		}
	}

	public static boolean hasMultiActionMenu() {
		return multiActionActive;
	}

	public static int getSelectedAction() {
		return multiActionActive ? selectedAction : ACTION_SEARCH;
	}

	public static void scrollAction(int delta) {
		if (!multiActionActive) return;
		selectedAction = Math.max(ACTION_SEARCH, Math.min(ACTION_EQUIP_OR_TAKE, selectedAction + delta));
	}
}

	private static int healFull(ServerCommandSource source) {
		ServerPlayerEntity player = source.getPlayer();
		if (player == null || !(player instanceof PlayerLoadoutHolder holder)) return 0;
		holder.tarkovcraft$getBodyHealth().resetFull();
		player.setHealth(player.getMaxHealth());
		holder.tarkovcraft$syncBodyHealth();
		source.sendFeedback(() -> Text.literal("[TarkovCraft] All body parts restored to 435/435 HP."), false);
		return 1;
	}

	private static int simulateHit(ServerCommandSource source, String zoneStr, float damage, int penetration) {
		ServerPlayerEntity player = source.getPlayer();
		if (player == null || !(player instanceof PlayerLoadoutHolder holder)) return 0;

		HitZone zone = HitZone.byId(zoneStr);
		BodyPartHealth health = holder.tarkovcraft$getBodyHealth();
		PlayerLoadout loadout = holder.tarkovcraft$getLoadout();

		boolean killed = health.applyHit(
			loadout, zone, damage, penetration, 1.0f, false, true,
			player.getRandom().nextFloat(), player.getRandom().nextFloat(), player.getRandom().nextFloat(),
			"Scav Sniper", "AK-74M", "5.45x39 PS", 32.5f
		);
		holder.tarkovcraft$syncBodyHealth();
		holder.tarkovcraft$syncLoadout();

		if (killed) {
			player.setHealth(0.0f);
			player.onDeath(player.getDamageSources().generic());
		} else {
			player.setHealth(Math.max(1.0f, health.getTotalRatio() * player.getMaxHealth()));
			source.sendFeedback(() -> Text.literal(
				"[TarkovCraft] Hit " + zone.getId() + " (" + damage + " raw, pen=" + penetration + ") -> "
					+ zone.getBodyPart().getId() + "=" + Math.round(health.getHp(zone.getBodyPart()))
					+ ", thorax=" + Math.round(health.getHp(com.tarkovcraft.health.BodyPart.THORAX))
			), false);
		}
		return 1;
	}

	private static SearchableLootBlock resolveCrateBlock(String blockName, TarkovLootPool pool) {
		if (blockName == null || blockName.isBlank() || "default".equalsIgnoreCase(blockName)) {
			return SearchableLootBlock.defaultBlockForPool(pool);
		}
		return switch (blockName.toLowerCase(java.util.Locale.ROOT)) {
			case "weapon_crate" -> ModBlocks.WEAPON_CRATE;
			case "medbag" -> ModBlocks.MEDBAG;
			case "airdrop_crate" -> ModBlocks.AIRDROP_CRATE;
			default -> ModBlocks.TECHNICAL_CRATE;
		};
	}

	private static int giveLootCrate(ServerCommandSource source, String poolRaw, String blockName, int width, int height) {
		ServerPlayerEntity player = source.getPlayer();
		if (player == null) {
			return 0;
		}
		TarkovLootPool pool = TarkovLootPool.byId(poolRaw);
		if (pool == null) {
			source.sendError(Text.literal("[TarkovCraft] Unknown loot pool: " + poolRaw + ". Use /tarkov crate list"));
			return 0;
		}
		SearchableLootBlock block = resolveCrateBlock(blockName, pool);
		int w = width > 0 ? width : pool.getDefaultGridWidth();
		int h = height > 0 ? height : pool.getDefaultGridHeight();
		ItemStack crateStack = SearchableLootBlock.createLootCrateStack(block, pool, w, h);

		boolean inserted = player.getInventory().insertStack(crateStack);
		if (!inserted && !crateStack.isEmpty()) {
			if (player instanceof PlayerLoadoutHolder holder) {
				holder.tarkovcraft$getLoadout().tryInsertItem(crateStack, player.getUuid());
				holder.tarkovcraft$syncLoadout();
			}
			if (!crateStack.isEmpty()) {
				player.dropItem(crateStack, false);
			}
		}
		source.sendFeedback(() -> Text.literal("[TarkovCraft] Given loot crate: ")
			.append(Text.translatable(pool.getTranslationKey()).formatted(Formatting.GOLD))
			.append(Text.literal(" (" + w + "x" + h + ")").formatted(Formatting.GRAY)), false);
		return 1;
	}

	private static int spawnLootCrate(ServerCommandSource source, String poolRaw, String blockName) {
		ServerPlayerEntity player = source.getPlayer();
		if (player == null) {
			return 0;
		}
		TarkovLootPool pool = TarkovLootPool.byId(poolRaw);
		if (pool == null) {
			source.sendError(Text.literal("[TarkovCraft] Unknown loot pool: " + poolRaw + ". Use /tarkov crate list"));
			return 0;
		}
		SearchableLootBlock block = resolveCrateBlock(blockName, pool);
		BlockPos pos = player.getBlockPos().offset(player.getHorizontalFacing(), 2);
		player.getWorld().setBlockState(pos, block.getDefaultState());
		BlockEntity be = player.getWorld().getBlockEntity(pos);
		if (be instanceof SearchableLootBlockEntity lootBe) {
			lootBe.resizeGrid(pool.getDefaultGridWidth(), pool.getDefaultGridHeight());
			lootBe.populateFromPool(pool, player.getWorld().getRandom());
		}
		source.sendFeedback(() -> Text.literal("[TarkovCraft] Spawned ")
			.append(Text.translatable(pool.getTranslationKey()).formatted(Formatting.GOLD))
			.append(Text.literal(" crate at " + pos.toShortString())), false);
		return 1;
	}

	private static int setLookedCratePool(ServerCommandSource source, String poolRaw) {
		ServerPlayerEntity player = source.getPlayer();
		if (player == null) {
			return 0;
		}
		TarkovLootPool pool = TarkovLootPool.byId(poolRaw);
		if (pool == null) {
			source.sendError(Text.literal("[TarkovCraft] Unknown loot pool: " + poolRaw + ". Use /tarkov crate list"));
			return 0;
		}
		HitResult hit = player.raycast(8.0, 0.0f, false);
		if (hit.getType() != HitResult.Type.BLOCK) {
			source.sendError(Text.literal("[TarkovCraft] Look directly at a TarkovCraft loot crate (within 8 blocks)."));
			return 0;
		}
		BlockPos pos = ((BlockHitResult) hit).getBlockPos();
		BlockEntity be = player.getWorld().getBlockEntity(pos);
		if (!(be instanceof SearchableLootBlockEntity lootBe)) {
			source.sendError(Text.literal("[TarkovCraft] Target block at " + pos.toShortString() + " is not a TarkovCraft loot crate."));
			return 0;
		}
		lootBe.populateFromPool(pool, player.getWorld().getRandom());
		source.sendFeedback(() -> Text.literal("[TarkovCraft] Crate at " + pos.toShortString() + " set to ")
			.append(Text.translatable(pool.getTranslationKey()).formatted(Formatting.GOLD))
			.append(Text.literal(" and refilled.")), false);
		return 1;
	}

	private static int refillCratesInRadius(ServerCommandSource source, int radius) {
		BlockPos center = BlockPos.ofFloored(source.getPosition());
		int count = 0;
		int minChunkX = (center.getX() - radius) >> 4;
		int maxChunkX = (center.getX() + radius) >> 4;
		int minChunkZ = (center.getZ() - radius) >> 4;
		int maxChunkZ = (center.getZ() + radius) >> 4;
		int rSq = radius * radius;

		for (int cx = minChunkX; cx <= maxChunkX; cx++) {
			for (int cz = minChunkZ; cz <= maxChunkZ; cz++) {
				if (!source.getWorld().isChunkLoaded(cx, cz)) {
					continue;
				}
				var chunk = source.getWorld().getChunk(cx, cz);
				for (BlockEntity be : chunk.getBlockEntities().values()) {
					if (be instanceof SearchableLootBlockEntity lootBe
						&& be.getPos().getSquaredDistance(center) <= rSq) {
						lootBe.refillLoot(source.getWorld().getRandom());
						count++;
					}
				}
			}
		}
		int finalCount = count;
		source.sendFeedback(() -> Text.literal("[TarkovCraft] Refilled " + finalCount + " loot crate(s) within " + radius + " blocks."), true);
		return finalCount;
	}

	private static int assignCityZone(ServerCommandSource source, String poolRaw, int radius) {
		TarkovLootPool pool = TarkovLootPool.byId(poolRaw);
		if (pool == null) {
			source.sendError(Text.literal("[TarkovCraft] Unknown loot pool: " + poolRaw + ". Use /tarkov crate list"));
			return 0;
		}
		var world = source.getWorld();
		BlockPos center = BlockPos.ofFloored(source.getPosition());
		int rSq = radius * radius;
		int minChunkX = (center.getX() - radius) >> 4;
		int maxChunkX = (center.getX() + radius) >> 4;
		int minChunkZ = (center.getZ() - radius) >> 4;
		int maxChunkZ = (center.getZ() + radius) >> 4;

		int modCount = 0;
		int vanillaCount = 0;
		java.util.Set<String> handledVanilla = new java.util.HashSet<>();

		for (int cx = minChunkX; cx <= maxChunkX; cx++) {
			for (int cz = minChunkZ; cz <= maxChunkZ; cz++) {
				if (!world.isChunkLoaded(cx, cz)) {
					continue;
				}
				var chunk = world.getChunk(cx, cz);
				for (BlockEntity be : chunk.getBlockEntities().values()) {
					BlockPos p = be.getPos();
					if (p.getSquaredDistance(center) > rSq) {
						continue;
					}
					if (be instanceof SearchableLootBlockEntity lootBe) {
						lootBe.populateFromPool(pool, world.getRandom());
						modCount++;
					} else {
						var state = world.getBlockState(p);
						if (VanillaContainerBridge.isSupportedBlock(state.getBlock())
							&& !(state.getBlock() instanceof net.minecraft.block.EnderChestBlock)) {
							String key = VanillaContainerBridge.resolveCanonicalKey(p, state, java.util.UUID.randomUUID());
							if (handledVanilla.add(key) && VanillaContainerBridge.assignZoneAndRefill(world, p, pool)) {
								vanillaCount++;
							}
						}
					}
				}
			}
		}

		int fMod = modCount;
		int fVan = vanillaCount;
		source.sendFeedback(() -> Text.literal("[TarkovCraft] Zone ")
			.append(Text.translatable(pool.getTranslationKey()).formatted(Formatting.GOLD))
			.append(Text.literal(" applied within " + radius + "m: " + fVan + " vanilla containers + " + fMod + " Tarkov crates.")), true);
		return fMod + fVan;
	}

	private static int scatterCityCrates(ServerCommandSource source, int radius, String poolRaw) {
		TarkovLootPool fixedPool = "auto".equalsIgnoreCase(poolRaw) ? null : TarkovLootPool.byId(poolRaw);
		if (!"auto".equalsIgnoreCase(poolRaw) && fixedPool == null) {
			source.sendError(Text.literal("[TarkovCraft] Unknown loot pool: " + poolRaw + " (use 'auto' or /tarkov crate list)"));
			return 0;
		}
		var world = source.getWorld();
		var rng = world.getRandom();
		BlockPos center = BlockPos.ofFloored(source.getPosition());
		int minY = Math.max(world.getBottomY() + 2, center.getY() - 24);
		int maxY = Math.min(world.getTopY() - 4, center.getY() + 24);

		java.util.List<BlockPos> placed = new java.util.ArrayList<>();
		int maxCrates = Math.min(80, Math.max(8, (radius * radius) / 220));
		int attempts = maxCrates * 35;

		for (int a = 0; a < attempts && placed.size() < maxCrates; a++) {
			int dx = rng.nextInt(radius * 2 + 1) - radius;
			int dz = rng.nextInt(radius * 2 + 1) - radius;
			if (dx * dx + dz * dz > radius * radius) {
				continue;
			}
			int x = center.getX() + dx;
			int z = center.getZ() + dz;
			if (!world.isChunkLoaded(x >> 4, z >> 4)) {
				continue;
			}
			int y = minY + rng.nextInt(Math.max(1, maxY - minY + 1));
			BlockPos pos = new BlockPos(x, y, z);

			// Must be empty space on a solid floor, inside a building (roof overhead, not open sky), away from other crates
			if (!world.getBlockState(pos).isAir() || !world.getBlockState(pos.up()).isAir()) {
				continue;
			}
			BlockPos floorPos = pos.down();
			var floorState = world.getBlockState(floorPos);
			if (!floorState.isSolidBlock(world, floorPos) || floorState.getBlock() instanceof SearchableLootBlock) {
				continue;
			}
			if (world.isSkyVisible(pos)) {
				continue;
			}
			// Require at least 1 nearby wall within 2 blocks so crates sit naturally near walls/corners
			boolean nearWall = false;
			for (net.minecraft.util.math.Direction dir : net.minecraft.util.math.Direction.Type.HORIZONTAL) {
				if (world.getBlockState(pos.offset(dir)).isSolidBlock(world, pos.offset(dir))) {
					nearWall = true;
					break;
				}
			}
			if (!nearWall) {
				continue;
			}
			boolean tooClose = false;
			for (BlockPos prev : placed) {
				if (prev.getSquaredDistance(pos) < 12 * 12) {
					tooClose = true;
					break;
				}
			}
			if (tooClose) {
				continue;
			}

			TarkovLootPool pool = fixedPool != null ? fixedPool : CityZoneClassifier.classify(world, pos, ModBlocks.TECHNICAL_CRATE);
			SearchableLootBlock block = SearchableLootBlock.defaultBlockForPool(pool);
			world.setBlockState(pos, block.getDefaultState());
			BlockEntity be = world.getBlockEntity(pos);
			if (be instanceof SearchableLootBlockEntity lootBe) {
				lootBe.resizeGrid(pool.getDefaultGridWidth(), pool.getDefaultGridHeight());
				lootBe.setLootPool(pool);
				lootBe.setPendingLootGeneration(true);
				lootBe.setAutoRespawn(true);
				placed.add(pos);
			}
		}

		int finalPlaced = placed.size();
		source.sendFeedback(() -> Text.literal("[TarkovCraft] Scattered " + finalPlaced
			+ " indoor loot crate(s) within " + radius + "m (auto-respawn: " + TarkovConfig.get().lootRespawnMinutes + "m)."), true);
		return finalPlaced;
	}

		source.sendFeedback(() -> Text.translatable("message.tarkovcraft.config_reloaded"), true);

	}

	private static int locateStructure(ServerCommandSource source, String typeRaw) {
		TarkovStructureFeature.StructureKind filter = null;
		if (typeRaw != null && !"any".equalsIgnoreCase(typeRaw)) {
			try {
				filter = TarkovStructureFeature.StructureKind.valueOf(typeRaw.toUpperCase(java.util.Locale.ROOT));
			} catch (IllegalArgumentException e) {
				source.sendError(Text.literal("[TarkovCraft] Unknown structure: " + typeRaw
					+ " (use: any, roadblock, med_tent, military_camp, forest_stash)"));
				return 0;
			}
		}

		BlockPos origin = BlockPos.ofFloored(source.getPosition());
		TarkovStructureRegistry registry = TarkovStructureRegistry.get(source.getWorld());
		TarkovStructureRegistry.FoundStructure found = registry.findNearest(filter, origin);
		if (found == null) {
			source.sendError(Text.literal("[TarkovCraft] No generated structures recorded yet for '" + typeRaw
				+ "'. Explore new chunks in Overworld to generate them."));
			return 0;
		}

		BlockPos pos = found.pos();
		int dist = (int) Math.round(Math.sqrt(pos.getSquaredDistance(origin)));
		String tpCmd = "/tp @s " + pos.getX() + " " + (pos.getY() + 1) + " " + pos.getZ();
		Text coords = Text.literal("[" + pos.getX() + ", " + pos.getY() + ", " + pos.getZ() + "]")
			.styled(style -> style
				.withColor(Formatting.GREEN)
				.withClickEvent(new ClickEvent(ClickEvent.Action.SUGGEST_COMMAND, tpCmd))
				.withHoverEvent(new HoverEvent(HoverEvent.Action.SHOW_TEXT, Text.literal("Click to teleport: " + tpCmd)))
			);
		source.sendFeedback(() -> Text.literal("[TarkovCraft] Nearest "
			+ found.kind().name().toLowerCase(java.util.Locale.ROOT) + " is at ")
			.append(coords)
			.append(Text.literal(" (" + dist + " blocks away)")), false);
		return 1;
	}

package com.tarkovcraft.compat.voicechat;

import com.tarkovcraft.audio.HearingProfile;
import net.fabricmc.api.EnvType;
import net.fabricmc.loader.api.FabricLoader;
import net.fabricmc.loader.api.LanguageAdapter;
import net.fabricmc.loader.api.LanguageAdapterException;
import net.fabricmc.loader.api.ModContainer;
import net.minecraft.server.network.ServerPlayerEntity;
import org.slf4j.LoggerFactory;

import java.lang.reflect.Method;
import java.lang.reflect.Proxy;
import java.util.Map;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;
import java.util.function.Consumer;

/** Creates the Simple Voice Chat entrypoint only when its API is available. */
public final class VoicechatAdapter implements LanguageAdapter {
	private static final String API = "de.maxhenkel.voicechat.api.";
	private static final String SIMPLE_VOICE_CHAT = "voicechat";
	private static final String VOICE_PHYSICS = "vc-audio-distance";
	private static final Map<UUID, HearingProfile> SERVER_PROFILES = new ConcurrentHashMap<>();
	private static final Map<UUID, Float> FILTER_STATE = new ConcurrentHashMap<>();
	private static volatile HearingProfile clientProfile = HearingProfile.DEFAULT;

	public static void updateServerProfile(UUID player, HearingProfile profile) {
		if (FabricLoader.getInstance().isModLoaded(SIMPLE_VOICE_CHAT)
			&& !FabricLoader.getInstance().isModLoaded(VOICE_PHYSICS)) SERVER_PROFILES.put(player, profile);
	}

	public static void removePlayer(UUID player) {
		SERVER_PROFILES.remove(player);
	}

	public static void updateClientProfile(HearingProfile profile) {
		if (!profile.muffled() && clientProfile.muffled()) FILTER_STATE.clear();
		clientProfile = profile;
	}

	@Override
	public <T> T create(ModContainer mod, String value, Class<T> type) throws LanguageAdapterException {
		if (!FabricLoader.getInstance().isModLoaded(SIMPLE_VOICE_CHAT) || !type.getName().equals(API + "VoicechatPlugin")) {
			throw new LanguageAdapterException("Simple Voice Chat API is unavailable");
		}
		Object proxy = Proxy.newProxyInstance(type.getClassLoader(), new Class<?>[]{type},
			(instance, method, arguments) -> switch (method.getName()) {
				case "getPluginId", "toString" -> "tarkovcraft";
				case "hashCode" -> System.identityHashCode(instance);
				case "equals" -> instance == arguments[0];
				case "registerEvents" -> {
					try {
						registerEvents(arguments[0]);
					} catch (ReflectiveOperationException exception) {
						LoggerFactory.getLogger("TarkovCraft Voice").warn("Could not register the Simple Voice Chat integration", exception);
					}
					yield null;
				}
				default -> null;
			});
		return type.cast(proxy);
	}

	private static void registerEvents(Object registration) throws ReflectiveOperationException {
		Method register = api("events.EventRegistration").getMethod("registerEvent", Class.class, Consumer.class, int.class);
		// Voice Physics owns proximity routing when installed. Both plugins can still process client PCM.
		if (!FabricLoader.getInstance().isModLoaded(VOICE_PHYSICS)) {
			registerRangeEvents(register, registration);
		}
		if (FabricLoader.getInstance().getEnvironmentType() == EnvType.CLIENT) {
			registerClientEvents(register, registration);
		}
	}

	private static void registerRangeEvents(Method register, Object registration) throws ReflectiveOperationException {
		Class<?> distanceEvent = api("events.VoiceDistanceEvent");
		Method getDistance = distanceEvent.getMethod("getDistance");
		Method setDistance = distanceEvent.getMethod("setDistance", float.class);
		Class<?> soundEvent = api("events.EntitySoundPacketEvent");
		Method source = soundEvent.getMethod("getSource");
		Method receiver = soundEvent.getMethod("getReceiverConnection");
		Method packet = soundEvent.getMethod("getPacket");
		Method voice = soundEvent.getMethod("getVoicechat");
		Method cancel = soundEvent.getMethod("cancel");
		Class<?> connection = api("VoicechatConnection");
		Method apiPlayer = connection.getMethod("getPlayer");
		Method nativePlayer = api("Player").getMethod("getPlayer");
		Class<?> entityPacket = api("packets.EntitySoundPacket");
		Method packetDistance = entityPacket.getMethod("getDistance");
		Method builderMethod = entityPacket.getMethod("entitySoundPacketBuilder");
		Class<?> builderType = api("packets.EntitySoundPacket$Builder");
		Method builderDistance = builderType.getMethod("distance", float.class);
		Method build = builderType.getMethod("build");
		Method send = api("VoicechatServerApi").getMethod("sendEntitySoundPacketTo", connection, entityPacket);
		register.invoke(registration, distanceEvent, guarded(event ->
			setDistance.invoke(event, (float) getDistance.invoke(event) * HearingProfile.ACTIVE_RANGE)), -1000);
		register.invoke(registration, soundEvent, guarded(event -> {
			if (!"proximity".equals(source.invoke(event))) return;
			Object target = receiver.invoke(event);
			if (target == null) return;
			Object player = nativePlayer.invoke(apiPlayer.invoke(target));
			HearingProfile profile = player instanceof ServerPlayerEntity serverPlayer
				? SERVER_PROFILES.getOrDefault(serverPlayer.getUuid(), HearingProfile.DEFAULT)
				: HearingProfile.DEFAULT;
			if (profile.range() == HearingProfile.ACTIVE_RANGE) return;
			Object original = packet.invoke(event);
			float range = (float) packetDistance.invoke(original) / HearingProfile.ACTIVE_RANGE * profile.range();
			Object builder = builderMethod.invoke(original);
			builderDistance.invoke(builder, range);
			Object adjusted = build.invoke(builder);
			if (Boolean.TRUE.equals(cancel.invoke(event))) send.invoke(voice.invoke(event), target, adjusted);
		}), -1000);
	}

	private static void registerClientEvents(Method register, Object registration) throws ReflectiveOperationException {
		Class<?> pcm = api("events.ClientReceiveSoundEvent");
		Method read = pcm.getMethod("getRawAudio");
		Method write = pcm.getMethod("setRawAudio", short[].class);
		Method getId = pcm.getMethod("getId");
		Consumer<Object> process = guarded(event -> {
			HearingProfile profile = clientProfile;
			if (profile == HearingProfile.DEFAULT) return;
			short[] input = (short[]) read.invoke(event);
			if (input == null || input.length == 0) return;
			UUID id = (UUID) getId.invoke(event);
			short[] output = new short[input.length];
			float previous = id == null ? 0.0f : FILTER_STATE.getOrDefault(id, 0.0f);
			for (int i = 0; i < output.length; i++) {
				float sample = input[i];
				if (profile.muffled()) {
					previous += 0.16f * (sample - previous);
					sample = previous * profile.ambientGain();
				} else {
					float gained = Math.abs(sample * profile.ambientGain());
					float knee = 0.72f * Short.MAX_VALUE;
					sample = Math.copySign(gained <= knee ? gained : knee + (gained - knee) * 0.25f, sample);
				}
				output[i] = (short) Math.max(Short.MIN_VALUE, Math.min(Short.MAX_VALUE, Math.round(sample)));
			}
			if (profile.muffled() && id != null) FILTER_STATE.put(id, previous);
			write.invoke(event, (Object) output);
		});
		for (String subtype : new String[]{"EntitySound", "LocationalSound", "StaticSound"}) {
			register.invoke(registration, api("events.ClientReceiveSoundEvent$" + subtype), process, -1000);
		}
		Class<?> icon = api("events.NameTagIconRenderEvent");
		Method cancel = icon.getMethod("cancel");
		register.invoke(registration, icon, guarded(event -> cancel.invoke(event)), 1000);
		register.invoke(registration, api("events.ClientVoicechatConnectionEvent"),
			(Consumer<Object>) event -> FILTER_STATE.clear(), 0);
	}

	private static Class<?> api(String name) throws ClassNotFoundException {
		return Class.forName(API + name, false, VoicechatAdapter.class.getClassLoader());
	}

	@FunctionalInterface
	private interface EventAction {
		void run(Object event) throws ReflectiveOperationException;
	}

	private static Consumer<Object> guarded(EventAction action) {
		return new Consumer<>() {
			private boolean failed;

			@Override
			public void accept(Object event) {
				if (failed) return;
				try {
					action.run(event);
				} catch (ReflectiveOperationException exception) {
					failed = true;
					LoggerFactory.getLogger("TarkovCraft Voice").warn("Disabled an incompatible Simple Voice Chat event handler", exception);
				}
			}
		};
	}
}

public final class TarkovConfig {
	private static final Logger LOGGER = LoggerFactory.getLogger("TarkovCraft/Config");
	private static final Gson GSON = new GsonBuilder().setPrettyPrinting().disableHtmlEscaping().create();
	private static volatile TarkovConfig instance = new TarkovConfig();

		if (config != null) {
			config.validateAndClamp();
			instance = config;
		}
	}

	public static synchronized TarkovConfig loadOrCreate() {
		try {
			Path path = FabricLoader.getInstance().getConfigDir().resolve("tarkovcraft.json");
			return loadFromPath(path);
		} catch (Throwable t) {
			LOGGER.warn("Could not resolve Fabric config dir ({}); using in-memory defaults.", t.getMessage());
			TarkovConfig fallback = new TarkovConfig();
			fallback.validateAndClamp();
			instance = fallback;
			return fallback;
		}
	}

	public static synchronized void save() {
		try {
			Path path = FabricLoader.getInstance().getConfigDir().resolve("tarkovcraft.json");
			saveToPath(path, instance);
		} catch (Throwable t) {
			LOGGER.warn("Failed to save config/tarkovcraft.json: {}", t.getMessage());
		}
	}

	public static synchronized TarkovConfig loadFromPath(Path path) {
		TarkovConfig loaded = null;
		if (Files.exists(path)) {
			try (Reader reader = Files.newBufferedReader(path, StandardCharsets.UTF_8)) {
				loaded = GSON.fromJson(reader, TarkovConfig.class);
			} catch (Exception e) {
				LOGGER.warn("Failed to parse config {}: {}; restoring defaults.", path, e.getMessage());
			}
		}
		if (loaded == null) {
			loaded = new TarkovConfig();
		}
		loaded.validateAndClamp();
		saveToPath(path, loaded);
		instance = loaded;
		return loaded;
	}

	public static synchronized void saveToPath(Path path, TarkovConfig cfg) {
		try {
			Path parent = path.getParent();
			if (parent != null) {
				Files.createDirectories(parent);
			}
			try (Writer writer = Files.newBufferedWriter(path, StandardCharsets.UTF_8)) {
				GSON.toJson(cfg, writer);
			}
		} catch (Exception e) {
			LOGGER.warn("Failed to write config {}: {}", path, e.getMessage());
		}
	}

	public void validateAndClamp() {
		airdropIntervalMinutes = clampInt(airdropIntervalMinutes, 1, 1440);
		airdropMinRadius = clampInt(airdropMinRadius, 8, 256);
		airdropMaxRadius = clampInt(airdropMaxRadius, airdropMinRadius, 512);
		airdropSmokeSeconds = clampInt(airdropSmokeSeconds, 0, 1800);
		lootDensityMultiplier = clampFloat(lootDensityMultiplier, 0.1f, 5.0f, 1.0f);
		mobGearSpawnChance = clampFloat(mobGearSpawnChance, 0.0f, 1.0f, 0.08f);
		mobBackpackChance = clampFloat(mobBackpackChance, 0.0f, 1.0f, 0.35f);
		corpseBaseLootChance = clampFloat(corpseBaseLootChance, 0.0f, 1.0f, 0.25f);
		corpseLifetimeSeconds = clampInt(corpseLifetimeSeconds, 30, 7200);
		containerRevealSeconds = clampFloat(containerRevealSeconds, 0.1f, 10.0f, 1.0f);
		searchTimeMultiplier = clampFloat(searchTimeMultiplier, 0.1f, 5.0f, 1.0f);
		grenadeDamageMultiplier = clampFloat(grenadeDamageMultiplier, 0.1f, 5.0f, 1.0f);
		lootRespawnMinutes = clampInt(lootRespawnMinutes, 0, 10080);
		vanillaContainerLootChance = clampFloat(vanillaContainerLootChance, 0.05f, 1.0f, 0.72f);
	}

	public long getLootRespawnMillis() {
		return lootRespawnMinutes <= 0 ? 0L : (long) lootRespawnMinutes * 60_000L;
	}

	public int getAirdropIntervalTicks() {
		return airdropIntervalMinutes * 60 * 20;
	}

	public int getContainerRevealTicks() {
		return Math.max(2, Math.round(containerRevealSeconds * 20.0f));
	}

	public int getCorpseLifetimeTicks() {
		return corpseLifetimeSeconds * 20;
	}

	private static int clampInt(int value, int min, int max) {
		return Math.max(min, Math.min(max, value));
	}

	private static float clampFloat(float value, float min, float max, float fallback) {
		if (Float.isNaN(value) || Float.isInfinite(value)) {
			return fallback;
		}
		return Math.max(min, Math.min(max, value));
	}

/**
 * Bridge between vanilla storage containers (Chest, Trapped Chest, Barrel, Shulker Box, Ender Chest)
 * and TarkovCraft's Tetris grid & §6 search pipeline.
 */
public final class VanillaContainerBridge {
	public static final int SINGLE_WIDTH = 6;
	public static final int SINGLE_HEIGHT = 5;
	public static final int DOUBLE_WIDTH = 9;
	public static final int DOUBLE_HEIGHT = 7;
	public static final int MAX_AUTO_HEIGHT = 60;

	private VanillaContainerBridge() {
	}

	/**
	 * Returns true for the 5 supported vanilla storage block hierarchies
	 * (ChestBlock covers both normal and TrappedChestBlock).
	 */
	public static boolean isSupportedBlock(Block block) {
		return block instanceof ChestBlock
			|| block instanceof BarrelBlock
			|| block instanceof ShulkerBoxBlock
			|| block instanceof EnderChestBlock;
	}

	public static String canonicalBlockKey(BlockPos pos) {
		return "block:" + pos.getX() + "," + pos.getY() + "," + pos.getZ();
	}

	public static String canonicalDoubleKey(BlockPos a, BlockPos b) {
		boolean aLess = a.getX() != b.getX()
			? a.getX() < b.getX()
			: (a.getY() != b.getY() ? a.getY() < b.getY() : a.getZ() <= b.getZ());
		return canonicalBlockKey(aLess ? a : b);
	}

	public static String canonicalEnderKey(UUID playerId) {
		return "ender:" + playerId;
	}

	public static String resolveCanonicalKey(BlockPos pos, BlockState state, UUID playerId) {
		Block block = state.getBlock();
		if (block instanceof EnderChestBlock) {
			return canonicalEnderKey(playerId);
		}
		if (block instanceof ChestBlock
			&& state.contains(ChestBlock.CHEST_TYPE)
			&& state.get(ChestBlock.CHEST_TYPE) != ChestType.SINGLE) {
			BlockPos otherPos = pos.offset(ChestBlock.getFacing(state));
			return canonicalDoubleKey(pos, otherPos);
		}
		return canonicalBlockKey(pos);
	}

	public static void removeForBrokenBlock(ServerWorld world, BlockPos pos, BlockState state) {
		Block block = state.getBlock();
		if (block instanceof EnderChestBlock || !isSupportedBlock(block)) {
			return;
		}
		VanillaContainerState vcState = VanillaContainerState.get(world);
		vcState.removeEntry(canonicalBlockKey(pos));
		if (block instanceof ChestBlock
			&& state.contains(ChestBlock.CHEST_TYPE)
			&& state.get(ChestBlock.CHEST_TYPE) != ChestType.SINGLE) {
			BlockPos otherPos = pos.offset(ChestBlock.getFacing(state));
			vcState.removeEntry(canonicalDoubleKey(pos, otherPos));
		}
	}

	public static boolean isContainerBlocked(World world, BlockPos pos, BlockState state) {
		Block block = state.getBlock();
		if (block instanceof ChestBlock) {
			if (ChestBlock.isChestBlocked(world, pos)) {
				return true;
			}
			if (state.contains(ChestBlock.CHEST_TYPE) && state.get(ChestBlock.CHEST_TYPE) != ChestType.SINGLE) {
				BlockPos otherPos = pos.offset(ChestBlock.getFacing(state));
				return ChestBlock.isChestBlocked(world, otherPos);
			}
			return false;
		}
		if (block instanceof EnderChestBlock) {
			BlockPos above = pos.up();
			return world.getBlockState(above).isSolidBlock(world, above);
		}
		if (block instanceof ShulkerBoxBlock) {
			BlockEntity be = world.getBlockEntity(pos);
			if (be instanceof ShulkerBoxBlockEntity shulkerBe
				&& shulkerBe.getAnimationStage() == ShulkerBoxBlockEntity.AnimationStage.CLOSED) {
				Box box = ShulkerEntity.calculateBoundingBox(state.get(ShulkerBoxBlock.FACING), 0.0f, 0.5f)
					.offset(pos)
					.contract(1.0E-6);
				return !world.isSpaceEmpty(box);
			}
			return false;
		}
		return false;
	}

	/**
	 * Attempts to open a vanilla container at {@code pos} inside TarkovScreenHandler.
	 * Returns true if {@code pos} is a supported vanilla container (even if blocked).
	 */
	public static boolean tryOpenForPlayer(ServerPlayerEntity player, BlockPos pos) {
		if (!(player.getWorld() instanceof ServerWorld world)) {
			return false;
		}
		BlockState state = world.getBlockState(pos);
		Block block = state.getBlock();
		if (!isSupportedBlock(block)) {
			return false;
		}

		if (isContainerBlocked(world, pos, state)) {
			player.sendMessage(Text.translatable("message.tarkovcraft.container_blocked"), true);
			return true;
		}

		EnderChestBlockEntity enderChestBe = null;
		Inventory liveInv;
		Text title = block.getName();

		if (block instanceof EnderChestBlock) {
			BlockEntity be = world.getBlockEntity(pos);
			if (!(be instanceof EnderChestBlockEntity ecBe)) {
				return false;
			}
			enderChestBe = ecBe;
			liveInv = player.getEnderChestInventory();
		} else if (block instanceof ChestBlock chestBlock) {
			BlockEntity be1 = world.getBlockEntity(pos);
			if (be1 instanceof LootableContainerBlockEntity loot1) {
				loot1.checkLootInteraction(player);
			}
			if (state.contains(ChestBlock.CHEST_TYPE) && state.get(ChestBlock.CHEST_TYPE) != ChestType.SINGLE) {
				BlockPos otherPos = pos.offset(ChestBlock.getFacing(state));
				BlockEntity be2 = world.getBlockEntity(otherPos);
				if (be2 instanceof LootableContainerBlockEntity loot2) {
					loot2.checkLootInteraction(player);
				}
			}
			liveInv = ChestBlock.getInventory(chestBlock, state, world, pos, true);
			if (liveInv instanceof Nameable named) {
				title = named.getDisplayName();
			} else if (be1 instanceof Nameable named && named.hasCustomName()) {
				title = named.getDisplayName();
			}
		} else {
			BlockEntity be = world.getBlockEntity(pos);
			if (be instanceof LootableContainerBlockEntity lootBe) {
				lootBe.checkLootInteraction(player);
			}
			liveInv = be instanceof Inventory inv ? inv : null;
			if (be instanceof Nameable named && named.hasCustomName()) {
				title = named.getDisplayName();
			}
		}

		if (liveInv == null) {
			return false;
		}

		boolean isDouble = liveInv.size() > 27;
		int baseW = isDouble ? DOUBLE_WIDTH : SINGLE_WIDTH;
		int baseH = isDouble ? DOUBLE_HEIGHT : SINGLE_HEIGHT;

		String key = resolveCanonicalKey(pos, state, player.getUuid());
		VanillaContainerState vcState = VanillaContainerState.get(world);
		VanillaContainerState.StoredVanillaContainer stored = vcState.getOrCreateEntry(key, baseW, baseH);
		boolean isShulker = block instanceof ShulkerBoxBlock;

		boolean changed = false;
		if (!(block instanceof EnderChestBlock) && !stored.isPlayerPlaced() && TarkovConfig.get().cityAutoLootVanillaContainers) {
			if (stored.getLastLootEpochMs() == 0L) {
				if (!liveInv.isEmpty()) {
					// Already contains player-placed items before auto-loot ever touched it -> protect it
					stored.setPlayerPlaced(true);
					changed = true;
				} else {
					TarkovLootPool pool = stored.getAssignedPool() != null
						? stored.getAssignedPool()
						: CityZoneClassifier.classify(world, pos, block);
					stored.setAssignedPool(pool);
					populateAndSyncVanillaLoot(stored, liveInv, baseW, baseH, pool, isShulker, isDouble, world.getRandom());
					changed = true;
				}
			} else if (SearchableLootBlockEntity.isRespawnDue(stored.getLastLootEpochMs())) {
				TarkovLootPool pool = stored.getAssignedPool() != null
					? stored.getAssignedPool()
					: CityZoneClassifier.classify(world, pos, block);
				stored.setAssignedPool(pool);
				populateAndSyncVanillaLoot(stored, liveInv, baseW, baseH, pool, isShulker, isDouble, world.getRandom());
				changed = true;
			}
		}

		if (reconcile(stored, liveInv, baseW, baseH)) {
			changed = true;
		}
		if (stored.getGrid().isEmpty()) {
			stored.getSearchMemory().markGridRevealed(player.getUuid());
			changed = true;
		}
		if (changed) {
			vcState.markDirty();
		}

		if (!(block instanceof EnderChestBlock) && !stored.isPlayerPlaced() && stored.getAssignedPool() != null) {
			title = title.copy()
				.append(Text.literal(" ("))
				.append(Text.translatable(stored.getAssignedPool().getTranslationKey()))
				.append(Text.literal(")"));
		}

		TarkovScreenHandler.openVanillaContainer(
			player, pos, title, liveInv, stored, vcState, isShulker, enderChestBe
		);
		return true;
	}

	private static void populateAndSyncVanillaLoot(
		VanillaContainerState.StoredVanillaContainer stored,
		Inventory liveInv,
		int baseW,
		int baseH,
		TarkovLootPool pool,
		boolean isShulker,
		boolean isDouble,
		net.minecraft.util.math.random.Random random
	) {
		GridContainer freshGrid = new GridContainer(baseW, baseH);
		float chance = isShulker ? 1.0f : TarkovConfig.get().vanillaContainerLootChance;
		if (random.nextFloat() <= chance) {
			TarkovLootGenerator.populateVanillaContainerGrid(freshGrid, pool, isShulker, isDouble, random);
		}
		stored.setGrid(freshGrid);
		stored.setSearchMemory(new ContainerSearchMemory());
		stored.setLastLootEpochMs(System.currentTimeMillis());
		syncGridToInventory(freshGrid, liveInv);
	}

	/**
	 * Marks a vanilla container position as player-placed so city auto-loot & hourly respawn never overwrite it.
	 */
	public static void markPlayerPlaced(ServerWorld world, BlockPos pos) {
		BlockState state = world.getBlockState(pos);
		Block block = state.getBlock();
		if (!isSupportedBlock(block) || block instanceof EnderChestBlock) {
			return;
		}
		String key = resolveCanonicalKey(pos, state, UUID.randomUUID());
		VanillaContainerState vcState = VanillaContainerState.get(world);
		VanillaContainerState.StoredVanillaContainer stored = vcState.getOrCreateEntry(key, SINGLE_WIDTH, SINGLE_HEIGHT);
		stored.setPlayerPlaced(true);
		vcState.markDirty();
	}

	/**
	 * Assigns a specific city loot pool to a vanilla container and forces an immediate refill.
	 */
	public static boolean assignZoneAndRefill(ServerWorld world, BlockPos pos, TarkovLootPool pool) {
		BlockState state = world.getBlockState(pos);
		Block block = state.getBlock();
		if (!isSupportedBlock(block) || block instanceof EnderChestBlock) {
			return false;
		}
		Inventory liveInv;
		if (block instanceof ChestBlock chestBlock) {
			liveInv = ChestBlock.getInventory(chestBlock, state, world, pos, true);
		} else {
			BlockEntity be = world.getBlockEntity(pos);
			liveInv = be instanceof Inventory inv ? inv : null;
		}
		if (liveInv == null) {
			return false;
		}
		boolean isDouble = liveInv.size() > 27;
		boolean isShulker = block instanceof ShulkerBoxBlock;
		int baseW = isDouble ? DOUBLE_WIDTH : SINGLE_WIDTH;
		int baseH = isDouble ? DOUBLE_HEIGHT : SINGLE_HEIGHT;
		String key = resolveCanonicalKey(pos, state, UUID.randomUUID());
		VanillaContainerState vcState = VanillaContainerState.get(world);
		VanillaContainerState.StoredVanillaContainer stored = vcState.getOrCreateEntry(key, baseW, baseH);
		stored.setPlayerPlaced(false);
		stored.setAssignedPool(pool);
		populateAndSyncVanillaLoot(stored, liveInv, baseW, baseH, pool, isShulker, isDouble, world.getRandom());
		vcState.markDirty();
		return true;
	}

	public static boolean refillArenaContainer(ServerWorld world, BlockPos pos) {
		BlockState state = world.getBlockState(pos);
		Block block = state.getBlock();
		if (!isSupportedBlock(block) || block instanceof EnderChestBlock) return false;
		String key = resolveCanonicalKey(pos, state, new UUID(0L, 0L));
		var stored = VanillaContainerState.get(world).getEntry(key);
		if (stored != null && stored.isPlayerPlaced()) return false;
		TarkovLootPool pool = stored != null ? stored.getAssignedPool() : null;
		return assignZoneAndRefill(world, pos, pool != null ? pool : CityZoneClassifier.classify(world, pos, block));
	}

	/**
	 * Reconciles persisted Tetris GridContainer against the live vanilla Inventory.
	 * - Fast-path: 100% multiset match -> preserves exact coordinates, rotations, UUIDs, and search memory.
	 * - Delta-path: keeps matching entries untouched, removes missing entries, and places newly
	 *   arrived items (e.g. from hoppers) with auto-grow height so zero items are ever lost.
	 *
	 * @return true if stored grid was modified
	 */
	public static boolean reconcile(
		VanillaContainerState.StoredVanillaContainer stored,
		Inventory liveInv,
		int baseW,
		int baseH
	) {
		List<ItemStack> liveStacks = new ArrayList<>();
		for (int i = 0; i < liveInv.size(); i++) {
			ItemStack st = liveInv.getStack(i);
			if (!st.isEmpty()) {
				liveStacks.add(st.copy());
			}
		}

		GridContainer grid = stored.getGrid();
		List<GridEntry> currentEntries = grid.getEntries();

		boolean[] liveMatched = new boolean[liveStacks.size()];
		List<GridEntry> keptEntries = new ArrayList<>();
		List<GridEntry> removedEntries = new ArrayList<>();

		for (GridEntry entry : currentEntries) {
			int matchIdx = findMatchingUnused(entry.getStack(), liveStacks, liveMatched);
			if (matchIdx >= 0) {
				liveMatched[matchIdx] = true;
				keptEntries.add(entry);
			} else {
				removedEntries.add(entry);
			}
		}

		List<ItemStack> addedStacks = new ArrayList<>();
		for (int i = 0; i < liveStacks.size(); i++) {
			if (!liveMatched[i]) {
				addedStacks.add(liveStacks.get(i));
			}
		}

		// Fast-path: 100% match -> zero changes
		if (removedEntries.isEmpty() && addedStacks.isEmpty() && grid.getWidth() >= baseW) {
			return false;
		}

		for (GridEntry removed : removedEntries) {
			grid.remove(removed.getEntryId());
		}

		// Ensure width >= baseW and try placing additions in the existing grid first
		boolean needsRebuild = grid.getWidth() < baseW || grid.getHeight() < baseH;
		if (!needsRebuild) {
			List<GridEntry> newlyPlaced = new ArrayList<>();
			for (ItemStack added : addedStacks) {
				GridEntry placed = grid.tryPlaceInFreeCell(added);
				if (placed == null) {
					needsRebuild = true;
					break;
				}
				newlyPlaced.add(placed);
			}
			if (!needsRebuild) {
				return true;
			}
			// Roll back partial placements before rebuilding with taller height
			for (GridEntry np : newlyPlaced) {
				grid.remove(np.getEntryId());
			}
		}

		GridContainer grown = rebuildWithAutoGrow(
			Math.max(baseW, grid.getWidth()),
			Math.max(baseH, grid.getHeight()),
			grid.getContainerId(),
			keptEntries,
			addedStacks
		);
		stored.setGrid(grown);
		return true;
	}

	private static GridContainer rebuildWithAutoGrow(
		int width,
		int startHeight,
		UUID containerId,
		List<GridEntry> keptEntries,
		List<ItemStack> addedStacks
	) {
		for (int h = startHeight; h <= MAX_AUTO_HEIGHT; h += 3) {
			GridContainer candidate = new GridContainer(width, h, containerId);
			boolean allPlaced = true;

			for (GridEntry kept : keptEntries) {
				if (!placePreservingId(candidate, kept)) {
					allPlaced = false;
					break;
				}
			}
			if (allPlaced) {
				for (ItemStack added : addedStacks) {
					if (candidate.tryPlaceInFreeCell(added) == null) {
						allPlaced = false;
						break;
					}
				}
			}
			if (allPlaced) {
				return candidate;
			}
		}

		// Fallback (should never happen with MAX_AUTO_HEIGHT = 60, guarantees zero silent drops)
		GridContainer fallback = new GridContainer(width, MAX_AUTO_HEIGHT, containerId);
		for (GridEntry kept : keptEntries) {
			placePreservingId(fallback, kept);
		}
		for (ItemStack added : addedStacks) {
			fallback.tryPlaceInFreeCell(added);
		}
		return fallback;
	}

	private static boolean placePreservingId(GridContainer target, GridEntry old) {
		if (target.placeWithId(old.getEntryId(), old.getStack(), old.getX(), old.getY(), old.isRotated()) != null) {
			return true;
		}
		ItemSize baseSize = ItemSizeRegistry.getSize(old.getStack());
		boolean[] orientations = baseSize.width() == baseSize.height()
			? new boolean[]{false}
			: new boolean[]{false, true};
		for (boolean rot : orientations) {
			for (int y = 0; y < target.getHeight(); y++) {
				for (int x = 0; x < target.getWidth(); x++) {
					if (target.placeWithId(old.getEntryId(), old.getStack(), x, y, rot) != null) {
						return true;
					}
				}
			}
		}
		return false;
	}

	private static int findMatchingUnused(ItemStack target, List<ItemStack> liveStacks, boolean[] used) {
		for (int i = 0; i < liveStacks.size(); i++) {
			if (!used[i] && stacksEqual(target, liveStacks.get(i))) {
				return i;
			}
		}
		return -1;
	}

	private static boolean stacksEqual(ItemStack a, ItemStack b) {
		return a.getCount() == b.getCount() && ItemStack.canCombine(a, b);
	}

	/**
	 * Immediately mirrors every GridEntry in {@code grid} back into {@code inv} (write-through).
	 */
	public static void syncGridToInventory(GridContainer grid, Inventory inv) {
		int slot = 0;
		for (GridEntry entry : grid.getEntries()) {
			if (slot >= inv.size()) {
				break;
			}
			inv.setStack(slot++, entry.getStack().copy());
		}
		while (slot < inv.size()) {
			inv.setStack(slot++, ItemStack.EMPTY);
		}
		inv.markDirty();
	}
}

package com.tarkovcraft.container;

import com.tarkovcraft.grid.GridContainer;
import com.tarkovcraft.loot.TarkovLootPool;
import com.tarkovcraft.search.ContainerSearchMemory;
import net.minecraft.nbt.NbtCompound;
import net.minecraft.nbt.NbtElement;
import net.minecraft.nbt.NbtList;
import net.minecraft.server.world.ServerWorld;
import net.minecraft.world.PersistentState;
import org.jetbrains.annotations.Nullable;

import java.util.LinkedHashMap;
import java.util.Map;

/**
 * Per-world persistent storage mapping canonical vanilla container keys
 * (single/double chest, barrel, shulker, player ender chest) to their
 * TarkovCraft Tetris GridContainer layout and per-player ContainerSearchMemory.
 */
public final class VanillaContainerState extends PersistentState {
	public static final String DATA_NAME = "tarkovcraft_vanilla_containers";

	private final Map<String, StoredVanillaContainer> containers = new LinkedHashMap<>();

	public static final class StoredVanillaContainer {
		private GridContainer grid;
		private ContainerSearchMemory searchMemory;
		@Nullable
		private TarkovLootPool assignedPool;
		private long lastLootEpochMs;
		private boolean playerPlaced;

		public StoredVanillaContainer(GridContainer grid, ContainerSearchMemory searchMemory) {
			this.grid = grid;
			this.searchMemory = searchMemory;
		}

		public GridContainer getGrid() {
			return grid;
		}

		public void setGrid(GridContainer grid) {
			this.grid = grid;
		}

		public ContainerSearchMemory getSearchMemory() {
			return searchMemory;
		}

		public void setSearchMemory(ContainerSearchMemory searchMemory) {
			this.searchMemory = searchMemory;
		}

		@Nullable
		public TarkovLootPool getAssignedPool() {
			return assignedPool;
		}

		public void setAssignedPool(@Nullable TarkovLootPool assignedPool) {
			this.assignedPool = assignedPool;
		}

		public long getLastLootEpochMs() {
			return lastLootEpochMs;
		}

		public void setLastLootEpochMs(long lastLootEpochMs) {
			this.lastLootEpochMs = Math.max(0L, lastLootEpochMs);
		}

		public boolean isPlayerPlaced() {
			return playerPlaced;
		}

		public void setPlayerPlaced(boolean playerPlaced) {
			this.playerPlaced = playerPlaced;
		}

		public NbtCompound toNbt() {
			NbtCompound tag = new NbtCompound();
			tag.put("Grid", grid.toNbt());
			tag.put("SearchMemory", searchMemory.toNbt());
			if (assignedPool != null) {
				tag.putString("AssignedPool", assignedPool.getId());
			}
			if (lastLootEpochMs > 0L) {
				tag.putLong("LastLootEpochMs", lastLootEpochMs);
			}
			if (playerPlaced) {
				tag.putBoolean("PlayerPlaced", true);
			}
			return tag;
		}

		public static StoredVanillaContainer fromNbt(NbtCompound tag, int defaultWidth, int defaultHeight) {
			GridContainer grid = tag.contains("Grid", NbtElement.COMPOUND_TYPE)
				? GridContainer.fromNbt(tag.getCompound("Grid"), defaultWidth, defaultHeight)
				: new GridContainer(defaultWidth, defaultHeight);
			ContainerSearchMemory mem = tag.contains("SearchMemory", NbtElement.COMPOUND_TYPE)
				? ContainerSearchMemory.fromNbt(tag.getCompound("SearchMemory"))
				: new ContainerSearchMemory();
			StoredVanillaContainer stored = new StoredVanillaContainer(grid, mem);
			if (tag.contains("AssignedPool", NbtElement.STRING_TYPE)) {
				stored.assignedPool = TarkovLootPool.byId(tag.getString("AssignedPool"));
			}
			stored.lastLootEpochMs = Math.max(0L, tag.getLong("LastLootEpochMs"));
			stored.playerPlaced = tag.getBoolean("PlayerPlaced");
			return stored;
		}
	}

	public static VanillaContainerState get(ServerWorld world) {
		return world.getPersistentStateManager().getOrCreate(
			VanillaContainerState::fromNbt,
			VanillaContainerState::new,
			DATA_NAME
		);
	}

	@Nullable
	public StoredVanillaContainer getEntry(String key) {
		return containers.get(key);
	}

	public StoredVanillaContainer getOrCreateEntry(String key, int baseWidth, int baseHeight) {
		StoredVanillaContainer existing = containers.get(key);
		if (existing != null) {
			return existing;
		}
		StoredVanillaContainer created = new StoredVanillaContainer(
			new GridContainer(baseWidth, baseHeight),
			new ContainerSearchMemory()
		);
		containers.put(key, created);
		markDirty();
		return created;
	}

	public void putEntry(String key, StoredVanillaContainer entry) {
		containers.put(key, entry);
		markDirty();
	}

	public void removeEntry(String key) {
		if (containers.remove(key) != null) {
			markDirty();
		}
	}

	public int size() {
		return containers.size();
	}

	@Override
	public NbtCompound writeNbt(NbtCompound nbt) {
		NbtList list = new NbtList();
		for (Map.Entry<String, StoredVanillaContainer> entry : containers.entrySet()) {
			NbtCompound row = entry.getValue().toNbt();
			row.putString("Key", entry.getKey());
			list.add(row);
		}
		nbt.put("Containers", list);
		return nbt;
	}

	public static VanillaContainerState fromNbt(NbtCompound nbt) {
		VanillaContainerState state = new VanillaContainerState();
		if (nbt == null || !nbt.contains("Containers", NbtElement.LIST_TYPE)) {
			return state;
		}
		NbtList list = nbt.getList("Containers", NbtElement.COMPOUND_TYPE);
		for (int i = 0; i < list.size(); i++) {
			NbtCompound row = list.getCompound(i);
			String key = row.getString("Key");
			if (key == null || key.isEmpty()) {
				continue;
			}
			StoredVanillaContainer stored = StoredVanillaContainer.fromNbt(row, 6, 5);
			state.containers.put(key, stored);
		}
		return state;
	}
}

		BlockState belowState = sw.getBlockState(below);
		boolean hitGround = this.isOnGround()
			|| (!belowState.isAir() && !belowState.isReplaceable())
			|| !sw.getFluidState(getBlockPos()).isEmpty()
			|| this.ticksInAir > MAX_AIR_TICKS;

		if (landingPoint != null) {
			if (getY() <= landingPoint.getY() + 0.08) {
				BlockState support = sw.getBlockState(landingPoint.down());
				if (sw.getBlockState(landingPoint).isAir()
					&& sw.getTopY(Heightmap.Type.MOTION_BLOCKING, landingPoint.getX(), landingPoint.getZ()) == landingPoint.getY()
					&& support.isSolidBlock(sw, landingPoint.down())
					&& !support.isIn(BlockTags.LEAVES) && !support.isIn(BlockTags.LOGS)) {
					placeCrate(sw, landingPoint);
				}
				discard();
			} else if (hitGround) {
				discard();
			}
		} else if (hitGround) {
			land(sw);
		}
	}

	private void land(ServerWorld sw) {
		BlockPos placePos = getBlockPos();
		if (this.ticksInAir > MAX_AIR_TICKS && sw.getBlockState(placePos.down()).isReplaceable()) {
			placePos = sw.getTopPosition(Heightmap.Type.MOTION_BLOCKING_NO_LEAVES, placePos);
		}
		for (int i = 0; i < 4; i++) {
			BlockState current = sw.getBlockState(placePos);
			if (current.isAir() || current.isReplaceable()) {
				break;
			}
			placePos = placePos.up();
		}

		if (sw.getBlockState(placePos).isReplaceable()) placeCrate(sw, placePos);
		discard();
	}

	private static void placeCrate(ServerWorld sw, BlockPos placePos) {
		if (!sw.setBlockState(placePos, ModBlocks.AIRDROP_CRATE.getDefaultState(), Block.NOTIFY_ALL)) return;
		if (sw.getBlockEntity(placePos) instanceof SearchableLootBlockEntity be) {
			be.populateFromPool(TarkovLootPool.AIRDROP, sw.random);
			be.setSmokeTicksRemaining(TarkovConfig.get().airdropSmokeSeconds * 20);
		}
		sw.playSound(null, placePos, SoundEvents.BLOCK_ANVIL_LAND, SoundCategory.BLOCKS, 1.0f, 0.85f);
		sw.spawnParticles(ParticleTypes.CLOUD, placePos.getX() + 0.5, placePos.getY() + 0.4, placePos.getZ() + 0.5,
			16, 0.6, 0.15, 0.6, 0.04);
	}
}

/**

 * NO fake generic container: holds the deceased's REAL equipment, REAL 6x(1x1) pockets,

		World world,

	public static PlayerLoadout transferPlayerInventoryToCorpse(
		PlayerLoadout sourceLoadout,
		GridContainer targetPocketsGrid,
		List<ItemStack> extras,
		List<ItemStack> overflow
	) {
		PlayerLoadout copy = sourceLoadout.copyLoot();
		for (int i = 0; i < PlayerLoadout.POCKET_COUNT; i++) {
			ItemStack p = copy.getPocket(i);
			if (!p.isEmpty()) {
				if (targetPocketsGrid.place(p.copy(), i, 0, false) == null) {
					overflow.add(p.copy());
				}
				copy.setPocket(i, ItemStack.EMPTY);
			}
		}

		// Store any offHand or vanilla main-inventory extras into corpse pockets -> empty equipment -> rig -> backpack -> overflow
		if (extras != null) {
			for (ItemStack extra : extras) {
				if (extra == null || extra.isEmpty()) {
					continue;
				}
				ItemStack rem = extra.copy();
				if (!targetPocketsGrid.tryAutoPlace(rem) && !rem.isEmpty()) {
					tryPlaceExtraIntoCorpseLoadout(copy, rem);
				}
				if (!rem.isEmpty()) {
					overflow.add(rem);
				}
			}
		}

		markContainerUnsearched(copy.getEquipment(EquipmentSlotType.BACKPACK));
		markContainerUnsearched(copy.getEquipment(EquipmentSlotType.RIG));
		return copy;
	}

	private static void tryPlaceExtraIntoCorpseLoadout(PlayerLoadout corpseLoadout, ItemStack rem) {
		if (rem.isEmpty()) {
			return;
		}
		// 1. Empty matching equipment slot (armor, backpack, rig, melee, weapons)
		for (EquipmentSlotType slot : EquipmentSlotType.values()) {
			if (corpseLoadout.getEquipment(slot).isEmpty() && slot.accepts(rem)) {
				corpseLoadout.setEquipment(slot, rem.copy());
				rem.setCount(0);
				return;
			}
		}
		// 2. Equipped Rig compartments
		ItemStack rig = corpseLoadout.getEquipment(EquipmentSlotType.RIG);
		if (!rig.isEmpty() && rig.getItem() instanceof TacticalRigItem) {
			List<GridContainer> grids = ContainerItemHelper.readRigGrids(rig);
			boolean changed = false;
			for (GridContainer g : grids) {
				int before = rem.getCount();
				boolean full = g.tryAutoPlace(rem);
				if (full || rem.getCount() < before) {
					changed = true;
					if (rem.isEmpty()) break;
				}
			}
			if (changed) {
				ContainerItemHelper.writeRigGrids(rig, grids);
				if (rem.isEmpty()) return;
			}
		}
		// 3. Equipped Backpack grid
		ItemStack bp = corpseLoadout.getEquipment(EquipmentSlotType.BACKPACK);
		if (!bp.isEmpty() && bp.getItem() instanceof BackpackItem) {
			GridContainer bpGrid = ContainerItemHelper.readBackpackGrid(bp);
			int before = rem.getCount();
			boolean full = bpGrid.tryAutoPlace(rem);
			if (full || rem.getCount() < before) {
				ContainerItemHelper.writeBackpackGrid(bp, bpGrid);
			}
		}
	}

	public static CorpseEntity fromMob(World world, LivingEntity mob) {

	}

	public static CorpseEntity fromGearedMob(World world, LivingEntity mob) {
		CorpseEntity corpse = new CorpseEntity(ModEntities.CORPSE, world);
		corpse.setPosition(mob.getX(), mob.getY(), mob.getZ());
		corpse.setYaw(mob.getYaw());
		corpse.setOwnerName(mob.getDisplayName().getString());
		Random random = world.getRandom();

		ItemStack headStack = mob.getEquippedStack(EquipmentSlot.HEAD);
		if (!headStack.isEmpty() && headStack.getItem() instanceof TarkovArmorItem armor
			&& armor.getTargetSlot() == EquipmentSlotType.HELMET) {
			corpse.loadout.setEquipment(EquipmentSlotType.HELMET, headStack.copy());
		} else if (random.nextFloat() < 0.45f) {
			ItemStack rolledHelmet = TarkovLootGenerator.pickRandomLightHelmet(random);
			if (!rolledHelmet.isEmpty()) {
				corpse.loadout.setEquipment(EquipmentSlotType.HELMET, rolledHelmet);
			}
		}

		ItemStack chestStack = mob.getEquippedStack(EquipmentSlot.CHEST);
		if (!chestStack.isEmpty() && chestStack.getItem() instanceof TarkovArmorItem armor
			&& armor.getTargetSlot() == EquipmentSlotType.BODY_ARMOR) {
			corpse.loadout.setEquipment(EquipmentSlotType.BODY_ARMOR, chestStack.copy());
		} else if (random.nextFloat() < 0.5f) {
			ItemStack rolledArmor = TarkovLootGenerator.pickRandomLightArmor(random);
			if (!rolledArmor.isEmpty()) {
				corpse.loadout.setEquipment(EquipmentSlotType.BODY_ARMOR, rolledArmor);
			}
		}

		if (mob.getCommandTags().contains(MobGearSpawner.TAG_GEARED_RIG) || random.nextFloat() < 0.65f) {
			ItemStack rig = TarkovLootGenerator.pickRandomScavRig(random);
			if (!rig.isEmpty()) {
				List<GridContainer> rigGrids = ContainerItemHelper.readRigGrids(rig);
				for (GridContainer g : rigGrids) {
					TarkovLootGenerator.populateLootGrid(g, TarkovLootPool.SCAV_POCKET, random);
				}
				ContainerItemHelper.writeRigGrids(rig, rigGrids);
				markContainerUnsearched(rig);
				corpse.loadout.setEquipment(EquipmentSlotType.RIG, rig);
			}
		}

		if (mob.getCommandTags().contains(MobGearSpawner.TAG_GEARED_BACKPACK)) {
			ItemStack bp = TarkovLootGenerator.pickRandomScavBackpack(random);
			if (!bp.isEmpty()) {
				GridContainer bpGrid = ContainerItemHelper.readBackpackGrid(bp);
				TarkovLootGenerator.populateLootGrid(bpGrid, TarkovLootPool.CIVILIAN, random);
				ContainerItemHelper.writeBackpackGrid(bp, bpGrid);
				markContainerUnsearched(bp);
				corpse.loadout.setEquipment(EquipmentSlotType.BACKPACK, bp);
			}
		}

		TarkovLootGenerator.populateLootGrid(corpse.pocketsGrid, TarkovLootPool.SCAV_POCKET, random);
		corpse.syncVisualFlags();
		return corpse;
	}

	public void populateScavLoot(Random random) {
		ItemStack helmet = TarkovLootGenerator.pickRandomLightHelmet(random);
		if (!helmet.isEmpty() && random.nextFloat() < 0.65f) {
			loadout.setEquipment(EquipmentSlotType.HELMET, helmet);
		}
		ItemStack armor = TarkovLootGenerator.pickRandomLightArmor(random);
		if (!armor.isEmpty() && random.nextFloat() < 0.6f) {
			loadout.setEquipment(EquipmentSlotType.BODY_ARMOR, armor);
		}

		ItemStack rig = TarkovLootGenerator.pickRandomScavRig(random);
		if (!rig.isEmpty()) {
			List<GridContainer> rigGrids = ContainerItemHelper.readRigGrids(rig);
			for (GridContainer g : rigGrids) {
				TarkovLootGenerator.populateLootGrid(g, TarkovLootPool.SCAV_POCKET, random);
			}
			ContainerItemHelper.writeRigGrids(rig, rigGrids);
			markContainerUnsearched(rig);
			loadout.setEquipment(EquipmentSlotType.RIG, rig);
		}

		ItemStack bp = TarkovLootGenerator.pickRandomScavBackpack(random);
		if (!bp.isEmpty()) {
			GridContainer bpGrid = ContainerItemHelper.readBackpackGrid(bp);
			TarkovLootGenerator.populateLootGrid(bpGrid, TarkovLootPool.CIVILIAN, random);
			ContainerItemHelper.writeBackpackGrid(bp, bpGrid);
			markContainerUnsearched(bp);
			loadout.setEquipment(EquipmentSlotType.BACKPACK, bp);
		}

		TarkovLootGenerator.populateLootGrid(pocketsGrid, TarkovLootPool.SCAV_POCKET, random);
		syncVisualFlags();
	}

	private static void markContainerUnsearched(ItemStack stack) {
		if (!stack.isEmpty() && ContainerItemHelper.isContainerItem(stack)) {
			ContainerItemHelper.setRequiresSearch(stack, true);
			ContainerItemHelper.saveSearchMemory(stack, new ContainerSearchMemory());
		}
	}

		this.dataTracker.set(HAS_HELMET, !loadout.getEquipment(EquipmentSlotType.HELMET).isEmpty());
	}

	/**
	 * Ordered list of all searchable containers currently on this corpse:
	 * 1. Pockets (6x1)
	 * 2. Tactical Rig (all compartments, searched as one unit)
	 * 3. Backpack (its full WxH grid)
	 */
	public List<SearchTarget> buildCorpseSearchTargets() {
		List<SearchTarget> targets = new ArrayList<>();
		targets.add(new SearchTarget(pocketsGrid.getContainerId(), List.of(pocketsGrid), pocketsSearchMemory, () -> {}));

		ItemStack rig = loadout.getEquipment(EquipmentSlotType.RIG);
		if (!rig.isEmpty() && rig.getItem() instanceof TacticalRigItem && ContainerItemHelper.requiresSearch(rig)) {
			UUID rigId = ContainerItemHelper.getOrCreateContainerUuid(rig);
			List<GridContainer> rigGrids = ContainerItemHelper.readRigGrids(rig);
			ContainerSearchMemory rigMem = ContainerItemHelper.getSearchMemory(rig);
			targets.add(new SearchTarget(rigId, rigGrids, rigMem, () -> {
				ContainerItemHelper.saveSearchMemory(rig, rigMem);
			}));
		}

		ItemStack bp = loadout.getEquipment(EquipmentSlotType.BACKPACK);
		if (!bp.isEmpty() && bp.getItem() instanceof BackpackItem && ContainerItemHelper.requiresSearch(bp)) {
			UUID bpId = ContainerItemHelper.getOrCreateContainerUuid(bp);
			GridContainer bpGrid = ContainerItemHelper.readBackpackGrid(bp);
			ContainerSearchMemory bpMem = ContainerItemHelper.getSearchMemory(bp);
			targets.add(new SearchTarget(bpId, List.of(bpGrid), bpMem, () -> {
				ContainerItemHelper.saveSearchMemory(bp, bpMem);
			}));
		}

		return targets;
	}

	public boolean isCompletelyEmpty() {
		return loadout.isEmpty() && pocketsGrid.isEmpty();
	}

	@Override
	public boolean canHit() {
		return !this.isRemoved();
	}

	@Override
	public boolean damage(DamageSource source, float amount) {
		// Allow Creative-mode players to immediately punch/remove corpses!
		if (!this.getWorld().isClient && source.getAttacker() instanceof PlayerEntity p && p.isCreative()) {
			this.discard();
			return true;
		}
		return false;
	}

	@Override
	public void tick() {
		super.tick();
		if (!this.getWorld().isClient) {
			ageTicks++;
			if (isCompletelyEmpty()) {
				emptyTicks++;
				if (emptyTicks >= EMPTY_DESPAWN_TICKS) {
					this.discard();
					return;
				}
			} else {
				emptyTicks = 0;
			}
			if (ageTicks >= TarkovConfig.get().getCorpseLifetimeTicks()) {
				this.discard();
			}
		}
	}

	@Override
	public ActionResult interact(PlayerEntity player, Hand hand) {
		// Looting corpses is strictly via [F] interact, NOT right-click
		return ActionResult.PASS;
	}

	public void openForPlayer(ServerPlayerEntity player) {
		if (player.getUuid().equals(ownerUuid) && RaidSession.isScav(player)) {
			player.sendMessage(Text.translatable("message.tarkovcraft.raid.own_corpse"), true);
			return;
		}
		player.openHandledScreen(new ExtendedScreenHandlerFactory() {
			@Override
			public void writeScreenOpeningData(ServerPlayerEntity player, PacketByteBuf buf) {
				buf.writeByte(TarkovScreenHandler.MODE_CORPSE);
				buf.writeInt(CorpseEntity.this.getId());
			}

			@Override
			public Text getDisplayName() {
				return Text.translatable("entity.tarkovcraft.corpse_title", getOwnerName());
			}

			@Nullable
			@Override
			public ScreenHandler createMenu(int syncId, PlayerInventory inv, PlayerEntity p) {
				return TarkovScreenHandler.forCorpse(syncId, inv, CorpseEntity.this);
			}
		});
		if (player.currentScreenHandler instanceof TarkovScreenHandler tsh) {
			tsh.syncToClient(player);
		}
	}

	@Override
	protected void readCustomDataFromNbt(NbtCompound nbt) {
		if (nbt.contains("OwnerName", NbtElement.STRING_TYPE)) {
			setOwnerName(nbt.getString("OwnerName"));
		}
		ownerUuid = nbt.containsUuid("OwnerUuid") ? nbt.getUuid("OwnerUuid") : null;
		if (nbt.contains("Loadout", NbtElement.COMPOUND_TYPE)) {
			loadout.readNbt(nbt.getCompound("Loadout"));
		}
		if (nbt.contains("PocketsGrid", NbtElement.COMPOUND_TYPE)) {
			pocketsGrid = GridContainer.fromNbt(nbt.getCompound("PocketsGrid"), 6, 1);
		}
		if (nbt.contains("PocketsSearchMemory", NbtElement.COMPOUND_TYPE)) {
			pocketsSearchMemory = ContainerSearchMemory.fromNbt(nbt.getCompound("PocketsSearchMemory"));
		}
		ageTicks = nbt.getInt("AgeTicks");
		syncVisualFlags();
	}

	@Override
	protected void writeCustomDataToNbt(NbtCompound nbt) {
		nbt.putString("OwnerName", getOwnerName());
		if (ownerUuid != null) nbt.putUuid("OwnerUuid", ownerUuid);
		nbt.put("Loadout", loadout.writeLootNbt());
		nbt.put("PocketsGrid", pocketsGrid.toNbt());
		nbt.put("PocketsSearchMemory", pocketsSearchMemory.toNbt());
		nbt.putInt("AgeTicks", ageTicks);
	}
}

package com.tarkovcraft.entity;

import com.tarkovcraft.config.TarkovConfig;

import net.minecraft.util.math.random.Random;

/**
 * Assigns Tarkov Scav/Raider gear tags and visible armor pieces to eligible Overworld/hostile mobs on spawn.
 */
public final class MobGearSpawner {
	public static final String TAG_GEAR_ROLLED = "tarkovcraft:gear_rolled";
	public static final String TAG_GEARED_MOB = "tarkovcraft:geared_mob";
	public static final String TAG_GEARED_RIG = "tarkovcraft:geared_rig";
	public static final String TAG_GEARED_BACKPACK = "tarkovcraft:geared_backpack";

	private MobGearSpawner() {
	}

	public static void register() {
		ServerEntityEvents.ENTITY_LOAD.register(MobGearSpawner::onEntityLoad);
	}

	public static boolean isEligibleMob(Entity entity) {
		if (!(entity instanceof MobEntity mob) || mob.isBaby()) {
			return false;
		}
		return mob instanceof ZombieEntity
			|| mob instanceof HuskEntity
			|| mob instanceof SkeletonEntity
			|| mob instanceof StrayEntity
			|| mob instanceof PillagerEntity
			|| mob instanceof VindicatorEntity;
	}

	private static void onEntityLoad(Entity entity, ServerWorld world) {
		if (!isEligibleMob(entity)) {
			return;
		}
		MobEntity mob = (MobEntity) entity;
		if (mob.getCommandTags().contains(TAG_GEAR_ROLLED)) {
			return;
		}
		mob.addCommandTag(TAG_GEAR_ROLLED);

		TarkovConfig cfg = TarkovConfig.get();
		Random random = world.random;
		if (random.nextFloat() >= cfg.mobGearSpawnChance) {
			return;
		}

		mob.addCommandTag(TAG_GEARED_MOB);
		if (mob.getEquippedStack(EquipmentSlot.HEAD).isEmpty() && random.nextFloat() < 0.65f) {
			ItemStack helmet = TarkovLootGenerator.pickRandomLightHelmet(random);
			if (!helmet.isEmpty()) {
				mob.equipStack(EquipmentSlot.HEAD, helmet);
				mob.setEquipmentDropChance(EquipmentSlot.HEAD, 0.0f);
			}
		}
		if (mob.getEquippedStack(EquipmentSlot.CHEST).isEmpty() && random.nextFloat() < 0.75f) {
			ItemStack armor = TarkovLootGenerator.pickRandomLightArmor(random);
			if (!armor.isEmpty()) {
				mob.equipStack(EquipmentSlot.CHEST, armor);
				mob.setEquipmentDropChance(EquipmentSlot.CHEST, 0.0f);
			}
		}
		if (random.nextFloat() < 0.65f) {
			mob.addCommandTag(TAG_GEARED_RIG);
		}
		if (random.nextFloat() < cfg.mobBackpackChance) {
			mob.addCommandTag(TAG_GEARED_BACKPACK);
		}
	}

	@Override
	public void tick() {
		super.tick();
		if (!(getWorld() instanceof ServerWorld) || !isAlive()) return;
		BodyPartHealth.TickResult result = bodyHealth.tickEffects(false);
		if (result.stateChanged()) updateVanillaHealth();
		if (bodyHealth.isDead()) {
			onDeath(getDamageSources().generic());
			return;
		}
		if (age % 10 == 0) tickHostileTargets();
	}

	private void tickHostileTargets() {
		for (HostileEntity hostile : getWorld().getEntitiesByClass(HostileEntity.class,
			getBoundingBox().expand(48.0), mob -> mob.isAlive() && !mob.isAiDisabled())) {
			LivingEntity target = hostile.getTarget();
			if (target != null && target.isAlive()
				&& (!(target instanceof SleeperPlayerEntity) || hostile.squaredDistanceTo(target) <= hostile.squaredDistanceTo(this))) {
				continue;
			}
			double followRange = hostile.getAttributeValue(EntityAttributes.GENERIC_FOLLOW_RANGE);
			if (hostile.squaredDistanceTo(this) > followRange * followRange
				|| !hostile.canTarget(this) || !hostile.canSee(this)) continue;

			hostile.setTarget(this);
			var brain = hostile.getBrain();
			if (brain.hasMemoryModule(MemoryModuleType.ATTACK_TARGET)
				&& brain.getOptionalRegisteredMemory(MemoryModuleType.ATTACK_TARGET).filter(LivingEntity::isAlive).isEmpty()) {
				brain.remember(MemoryModuleType.ATTACK_TARGET, this, 20);
			}
		}
	}

	@Override
	protected void applyDamage(DamageSource source, float amount) {
		if (getWorld().isClient || amount <= 0.0f) return;
		String sourceName = source.getName();
		if ("outOfWorld".equals(sourceName) || "genericKill".equals(sourceName)) {
			bodyHealth.applyHit(loadout, HitZone.THORAX, 9999.0f, 100, 0.0f, true, false, 1.0f, sourceName, "", "", 0.0f);
			updateVanillaHealth();
			return;
		}

		StatusEffectInstance resistance = getStatusEffect(StatusEffects.RESISTANCE);
		if (resistance != null && !source.isIn(DamageTypeTags.BYPASSES_RESISTANCE)) {
			amount *= Math.max(0, 25 - (resistance.getAmplifier() + 1) * 5) / 25.0f;
			if (amount <= 0.001f) return;
		}
		if (source.isIn(DamageTypeTags.IS_FALL)) {
			bodyHealth.applyFallDamage(amount, loadout.calculateWeightKg(), random.nextFloat());
			updateVanillaHealth();
			return;
		}

		TaCZBulletHelper.BulletInfo bullet = TaCZBulletHelper.inspectDamageSource(source);
		boolean bypassArmor = source.isIn(DamageTypeTags.IS_DROWNING)
			|| source.isIn(DamageTypeTags.IS_FIRE)
			|| source.isIn(DamageTypeTags.IS_FREEZING)
			|| !bullet.isTaczBullet() && source.isIn(DamageTypeTags.BYPASSES_ARMOR);
		boolean canRicochet = !bypassArmor && (bullet.isTaczBullet() || source.isIn(DamageTypeTags.IS_PROJECTILE));
		float damage = amount * DAMAGE_SCALE * bullet.damageMultiplier();
		if (getAbsorptionAmount() > 0 && !source.isIn(DamageTypeTags.BYPASSES_INVULNERABILITY)) {
			float absorbed = Math.min(damage, getAbsorptionAmount() * DAMAGE_SCALE);
			setAbsorptionAmount(Math.max(0, getAbsorptionAmount() - absorbed / DAMAGE_SCALE));
			damage -= absorbed;
			if (damage <= 0.001f) return;
		}
		Entity attacker = source.getAttacker();
		String killer = attacker == null ? sourceName : attacker.getDisplayName().getString();
		bodyHealth.applyHit(loadout, HitZoneResolver.resolveFromDamage(this, source),
			damage, bullet.penetrationPower(), bullet.armorWearMultiplier(), bypassArmor, canRicochet,
			random.nextFloat(), random.nextFloat(), random.nextFloat(), killer, bullet.weaponName(),
			bullet.ammoName(), attacker == null ? 0.0f : distanceTo(attacker));
		updateVanillaHealth();
	}

	private void updateVanillaHealth() {
		setHealth(bodyHealth.isDead() ? 0.0f : Math.max(1.0f, bodyHealth.getTotalRatio() * getMaxHealth()));
	}

	@Override
	public void onDeath(DamageSource source) {
		if (dead) return;
		super.onDeath(source);
		if (!(getWorld() instanceof ServerWorld world) || ownerUuid == null) return;

		List<ItemStack> extras = new ArrayList<>(vanillaInventory.size());
		for (int i = 0; i < vanillaInventory.size(); i++) {
			extras.add(ItemStack.fromNbt(vanillaInventory.getCompound(i)));
		}
		List<ItemStack> overflow = new ArrayList<>();
		CorpseEntity corpse = CorpseEntity.fromSnapshot(world, getPos(), getYaw(),
			ownerUuid, getName().getString(), loadout, extras, overflow);
		if (world.spawnEntity(corpse)) {
			overflow.forEach(this::dropStack);
		}
		RaidSession.onSleeperDeath(this, source);
	}

	@Override
	protected boolean shouldDropLoot() {
		return false;
	}

	@Override
	protected SoundEvent getHurtSound(DamageSource source) {
		return SoundEvents.ENTITY_PLAYER_HURT;
	}

	@Override
	protected SoundEvent getDeathSound() {
		return SoundEvents.ENTITY_PLAYER_DEATH;
	}

	@Override
	protected float getActiveEyeHeight(EntityPose pose, EntityDimensions dimensions) {
		return Math.min(1.62f, dimensions.height);
	}

	@Override
	public Iterable<ItemStack> getArmorItems() {
		return ARMOR_SLOTS.stream().map(this::getEquippedStack).toList();
	}

	@Override
	public ItemStack getEquippedStack(EquipmentSlot slot) {
		return equipment.getOrDefault(slot, ItemStack.EMPTY);
	}

	@Override
	public void equipStack(EquipmentSlot slot, ItemStack stack) {
		equipment.put(slot, stack);
	}

	@Override
	public Arm getMainArm() {
		return dataTracker.get(LEFT_HANDED) ? Arm.LEFT : Arm.RIGHT;
	}
}

/**
 * Thrown TarkovCraft tactical grenade with bounce physics, impact/timed fuses,
 * block-safe fragmentation with zonal BodyPartHealth damage, flashbang LOS checks,
 * smoke screen (M18), and tear gas cloud (CS Model 8230).
 */
public class TarkovGrenadeEntity extends ThrownItemEntity {
	private static final HitZone[] FRAG_ZONES = {
		HitZone.THORAX,
		HitZone.STOMACH,
		HitZone.LEFT_ARM,
		HitZone.RIGHT_ARM,
		HitZone.LEFT_LEG,
		HitZone.RIGHT_LEG,
		HitZone.HEAD_JAWS
	};

	private static final DustParticleEffect SMOKE_GREEN_DUST =
		new DustParticleEffect(new Vector3f(0.28f, 0.72f, 0.34f), 4.0f);
	private static final DustParticleEffect SMOKE_WHITE_DUST =
		new DustParticleEffect(new Vector3f(0.78f, 0.82f, 0.78f), 4.0f);
	private static final DustParticleEffect GAS_CS_DUST =
		new DustParticleEffect(new Vector3f(0.68f, 0.78f, 0.22f), 2.8f);

	private TarkovGrenadeItem.GrenadeType grenadeType = TarkovGrenadeItem.GrenadeType.FRAG_F1;
	private int fuseRemaining = 40;
	private int cloudRemaining = 0;
	private boolean detonated = false;
	private boolean emitting = false;
	private boolean resting = false;

	public TarkovGrenadeEntity(EntityType<? extends ThrownItemEntity> type, World world) {
		super(type, world);
	}

	public TarkovGrenadeEntity(
		World world,
		LivingEntity owner,
		ItemStack sourceStack,
		TarkovGrenadeItem.GrenadeType grenadeType
	) {
		super(ModEntities.GRENADE, owner, world);
		this.grenadeType = grenadeType != null ? grenadeType : TarkovGrenadeItem.GrenadeType.FRAG_F1;
		this.fuseRemaining = this.grenadeType.getFuseTicks();
		if (sourceStack != null && !sourceStack.isEmpty()) {
			this.setItem(sourceStack.copyWithCount(1));
		}
	}

	@Override
	protected Item getDefaultItem() {
		return Items.SNOWBALL;
	}

	@Override
	protected float getGravity() {
		return 0.045f;
	}

	@Override
	protected void onBlockHit(BlockHitResult hit) {
		if (emitting) {
			setVelocity(Vec3d.ZERO);
			return;
		}
		if (!getWorld().isClient && !detonated && grenadeType.isImpactFuse() && age >= grenadeType.getArmingTicks()) {
			detonate();
			return;
		}

		Vec3d vel = getVelocity();
		Direction.Axis axis = hit.getSide().getAxis();
		double vx = axis == Direction.Axis.X ? -vel.x * 0.55 : vel.x * 0.72;
		double vy = axis == Direction.Axis.Y ? -vel.y * 0.42 : vel.y * 0.72;
		double vz = axis == Direction.Axis.Z ? -vel.z * 0.55 : vel.z * 0.72;
		Vec3d nextVel = new Vec3d(vx, vy, vz);

		if (hit.getSide() == Direction.UP && nextVel.lengthSquared() < 0.015) {
			resting = true;
			setNoGravity(true);
			setVelocity(Vec3d.ZERO);
		} else {
			setVelocity(nextVel);
		}
		getWorld().playSound(null, getX(), getY(), getZ(),
			SoundEvents.BLOCK_METAL_HIT, SoundCategory.PLAYERS, 0.45f, 1.25f);
	}

	@Override
	protected void onEntityHit(EntityHitResult hit) {
		if (emitting) {
			return;
		}
		if (!getWorld().isClient && !detonated && grenadeType.isImpactFuse() && age >= grenadeType.getArmingTicks()) {
			detonate();
			return;
		}
		Entity target = hit.getEntity();
		if (!getWorld().isClient) {
			target.damage(getDamageSources().thrown(this, getOwner()), 1.0f);
		}
		Vec3d vel = getVelocity();
		setVelocity(vel.x * -0.35, Math.abs(vel.y) * 0.35, vel.z * -0.35);
		getWorld().playSound(null, getX(), getY(), getZ(),
			SoundEvents.BLOCK_METAL_HIT, SoundCategory.PLAYERS, 0.45f, 1.25f);
	}

	@Override
	public void tick() {
		if (resting && getWorld().getBlockState(getBlockPos().down()).isAir()) {
			resting = false;
			setNoGravity(false);
		}
		if (resting || emitting) {
			setVelocity(Vec3d.ZERO);
		}
		super.tick();

		if (getWorld().isClient) {
			if (!detonated && age % 2 == 0) {
				getWorld().addParticle(ParticleTypes.SMOKE, getX(), getY() + 0.15, getZ(), 0.0, 0.02, 0.0);
			}
			return;
		}

		if (!detonated) {
			fuseRemaining--;
			if (fuseRemaining <= 0) {
				detonate();
			}
		} else if (emitting) {
			cloudRemaining--;
			tickCloud((ServerWorld) getWorld());
			if (cloudRemaining <= 0) {
				discard();
			}
		}
	}

	private void detonate() {
		if (detonated || !(getWorld() instanceof ServerWorld sw)) {
			return;
		}
		detonated = true;

		switch (grenadeType.getCategory()) {
			case FRAG, IMPACT -> detonateFrag(sw);
			case FLASH -> detonateFlash(sw);
			case SMOKE -> startSmokeCloud(sw);
			case GAS -> startGasCloud(sw);
		}
	}

	private void detonateFrag(ServerWorld sw) {
		sw.playSound(null, getX(), getY(), getZ(),
			SoundEvents.ENTITY_GENERIC_EXPLODE, SoundCategory.PLAYERS, 3.0f, 0.9f);
		sw.spawnParticles(ParticleTypes.EXPLOSION_EMITTER, getX(), getY() + 0.25, getZ(), 1, 0.1, 0.1, 0.1, 0.0);
		sw.spawnParticles(ParticleTypes.EXPLOSION, getX(), getY() + 0.25, getZ(), 12, 0.8, 0.5, 0.8, 0.05);
		sw.spawnParticles(ParticleTypes.CAMPFIRE_COSY_SMOKE, getX(), getY() + 0.25, getZ(), 18, 1.2, 0.6, 1.2, 0.03);
		sw.spawnParticles(ParticleTypes.LAVA, getX(), getY() + 0.2, getZ(), 14, 0.8, 0.5, 0.8, 0.12);

		float radius = grenadeType.getBlastRadius();
		Vec3d origin = getPos().add(0.0, 0.2, 0.0);
		DamageSource source = getDamageSources().explosion(this, getOwner());
		String killerName = getOwner() != null ? getOwner().getName().getString() : "Grenade";
		String weaponName = getStack().isEmpty() ? grenadeType.name() : getStack().getName().getString();

		List<LivingEntity> targets = sw.getEntitiesByClass(LivingEntity.class, getBoundingBox().expand(radius),
			e -> e.isAlive() && !e.isSpectator());
		for (LivingEntity target : targets) {
			double dist = origin.distanceTo(target.getPos().add(0.0, target.getHeight() * 0.5, 0.0));
			if (dist > radius) {
				continue;
			}
			float exposure = computeExposure(sw, origin, target);
			if (exposure <= 0.0f) {
				continue;
			}

			float falloff = Math.max(0.0f, 1.0f - (float) (dist / radius));
			float effectiveDmg = grenadeType.getMaxDamage()
				* (0.25f + 0.75f * falloff * falloff)
				* exposure
				* com.tarkovcraft.config.TarkovConfig.get().grenadeDamageMultiplier;

			if (target instanceof ServerPlayerEntity sp && !sp.isCreative() && !sp.isSpectator()
				&& sp instanceof PlayerLoadoutHolder holder) {
				BodyPartHealth health = holder.tarkovcraft$getBodyHealth();
				float perZone = effectiveDmg / 4.0f;
				for (int i = 0; i < 4; i++) {
					HitZone zone = (i == 0) ? HitZone.THORAX : FRAG_ZONES[random.nextInt(FRAG_ZONES.length)];
					health.applyHit(
						holder.tarkovcraft$getLoadout(),
						zone,
						perZone,
						grenadeType.getPenetrationPower(),
						1.0f, false, false, 1.0f,
						random.nextFloat(),
						random.nextFloat(),
						killerName,
						weaponName,
						"Shrapnel",
						(float) dist
					);
				}

				if (random.nextFloat() < grenadeType.getBleedChance() * falloff * exposure) {
					HitZone bleedZone = FRAG_ZONES[random.nextInt(FRAG_ZONES.length)];
					health.addInjury(bleedZone.getBodyPart(),
						falloff > 0.6f ? InjuryType.HEAVY_BLEED : InjuryType.LIGHT_BLEED);
				}

				holder.tarkovcraft$syncBodyHealth();
				holder.tarkovcraft$syncLoadout();
				if (health.isDead()) {
					sp.setHealth(0.0f);
					sp.onDeath(source);
				} else {
					sp.setHealth(Math.max(1.0f, health.getTotalRatio() * sp.getMaxHealth()));
				}
			} else if (!(target instanceof PlayerEntity)) {
				target.damage(source, Math.max(1.0f, effectiveDmg / 9.0f));
			}

			Vec3d push = target.getPos().subtract(origin);
			if (push.lengthSquared() > 1.0E-4) {
				Vec3d knock = push.normalize().multiply(0.55 * falloff * exposure);
				target.addVelocity(knock.x, Math.max(0.15 * falloff, knock.y), knock.z);
				target.velocityModified = true;
			}
		}

		discard();
	}

	private void detonateFlash(ServerWorld sw) {
		sw.playSound(null, getX(), getY(), getZ(),
			SoundEvents.ENTITY_FIREWORK_ROCKET_LARGE_BLAST, SoundCategory.PLAYERS, 3.0f, 1.4f);
		sw.playSound(null, getX(), getY(), getZ(),
			SoundEvents.ENTITY_GENERIC_EXPLODE, SoundCategory.PLAYERS, 1.2f, 1.8f);
		sw.spawnParticles(ParticleTypes.FLASH, getX(), getY() + 0.35, getZ(), 4, 0.2, 0.2, 0.2, 0.0);
		sw.spawnParticles(ParticleTypes.EXPLOSION, getX(), getY() + 0.35, getZ(), 3, 0.3, 0.2, 0.3, 0.02);

		float radius = grenadeType.getBlastRadius();
		Vec3d origin = getPos().add(0.0, 0.25, 0.0);
		List<LivingEntity> targets = sw.getEntitiesByClass(LivingEntity.class, getBoundingBox().expand(radius),
			e -> e.isAlive() && !e.isSpectator());
		for (LivingEntity target : targets) {
			double dist = origin.distanceTo(target.getEyePos());
			if (dist > radius) {
				continue;
			}
			float exposure = computeExposure(sw, origin, target);
			if (exposure <= 0.0f) {
				continue;
			}
			float falloff = Math.max(0.0f, 1.0f - (float) (dist / radius));
			Vec3d toFlash = origin.subtract(target.getEyePos());
			double facingMult = 0.5;
			if (toFlash.lengthSquared() > 1.0E-4) {
				double dot = target.getRotationVec(1.0f).dotProduct(toFlash.normalize());
				facingMult = dot > 0.25 ? 1.0 : 0.5;
			}

			int blindTicks = Math.max(40, (int) (220 * falloff * facingMult));
			target.addStatusEffect(new StatusEffectInstance(StatusEffects.BLINDNESS, blindTicks, 0));
			target.addStatusEffect(new StatusEffectInstance(StatusEffects.DARKNESS, 80, 0));
			target.addStatusEffect(new StatusEffectInstance(StatusEffects.NAUSEA, 160, 0));
			target.addStatusEffect(new StatusEffectInstance(StatusEffects.SLOWNESS, 100, 1));
		}

		discard();
	}

	private void startSmokeCloud(ServerWorld sw) {
		emitting = true;
		resting = true;
		setNoGravity(true);
		setVelocity(Vec3d.ZERO);
		cloudRemaining = grenadeType.getCloudTicks();
		sw.playSound(null, getX(), getY(), getZ(),
			SoundEvents.BLOCK_FIRE_EXTINGUISH, SoundCategory.PLAYERS, 1.2f, 0.8f);
	}

	private void startGasCloud(ServerWorld sw) {
		emitting = true;
		resting = true;
		setNoGravity(true);
		setVelocity(Vec3d.ZERO);
		cloudRemaining = grenadeType.getCloudTicks();
		sw.playSound(null, getX(), getY(), getZ(),
			SoundEvents.BLOCK_FIRE_EXTINGUISH, SoundCategory.PLAYERS, 1.2f, 1.0f);
	}

	private void tickCloud(ServerWorld sw) {
		float radius = grenadeType.getBlastRadius();
		double spread = radius * 0.65;

		if (grenadeType.getCategory() == TarkovGrenadeItem.Category.SMOKE) {
			double smokeSpread = Math.max(2.6, radius * 0.72);
			sw.spawnParticles(ParticleTypes.CAMPFIRE_SIGNAL_SMOKE, getX(), getY() + 0.45, getZ(), 10, smokeSpread, 0.8, smokeSpread, 0.012);
			sw.spawnParticles(ParticleTypes.CAMPFIRE_SIGNAL_SMOKE, getX(), getY() + 1.85, getZ(), 8, smokeSpread * 0.85, 0.7, smokeSpread * 0.85, 0.01);
			sw.spawnParticles(ParticleTypes.CLOUD, getX(), getY() + 1.2, getZ(), 14, smokeSpread * 0.9, 1.15, smokeSpread * 0.9, 0.008);
			if (cloudRemaining % 2 == 0) {
				sw.spawnParticles(SMOKE_GREEN_DUST, getX(), getY() + 1.0, getZ(), 20, smokeSpread, 1.2, smokeSpread, 0.01);
				sw.spawnParticles(SMOKE_WHITE_DUST, getX(), getY() + 1.4, getZ(), 18, smokeSpread, 1.2, smokeSpread, 0.01);
				sw.spawnParticles(ParticleTypes.LARGE_SMOKE, getX(), getY() + 0.8, getZ(), 12, smokeSpread, 1.0, smokeSpread, 0.015);
			}
			if (cloudRemaining % 8 == 0) {
				Box box = getBoundingBox().expand(radius, 3.8, radius);
				List<LivingEntity> targets = sw.getEntitiesByClass(LivingEntity.class, box,
					e -> e.isAlive() && !e.isSpectator());
				Vec3d origin = getPos();
				for (LivingEntity target : targets) {
					double dx = target.getX() - origin.x;
					double dz = target.getZ() - origin.z;
					double dy = target.getY() - origin.y;
					if (dx * dx + dz * dz <= radius * radius && dy >= -1.0 && dy <= 4.0) {
						target.addStatusEffect(new StatusEffectInstance(StatusEffects.BLINDNESS, 45, 0, false, false, true));
						target.addStatusEffect(new StatusEffectInstance(StatusEffects.DARKNESS, 55, 0, false, false, true));
					}
				}
			}
			if (cloudRemaining % 30 == 0) {
				sw.playSound(null, getX(), getY(), getZ(),
					SoundEvents.BLOCK_FIRE_EXTINGUISH, SoundCategory.PLAYERS, 0.4f, 0.9f);
			}
			return;
		}

		if (grenadeType.getCategory() == TarkovGrenadeItem.Category.GAS) {
			if (cloudRemaining % 2 == 0) {
				sw.spawnParticles(GAS_CS_DUST, getX(), getY() + 0.8, getZ(), 12, spread, 0.9, spread, 0.01);
				sw.spawnParticles(ParticleTypes.CAMPFIRE_COSY_SMOKE, getX(), getY() + 0.5, getZ(), 5, spread, 0.8, spread, 0.01);
				sw.spawnParticles(ParticleTypes.SNEEZE, getX(), getY() + 0.8, getZ(), 4, spread * 0.8, 0.7, spread * 0.8, 0.02);
			}
			if (cloudRemaining % 10 == 0) {
				Box box = getBoundingBox().expand(radius);
				List<LivingEntity> targets = sw.getEntitiesByClass(LivingEntity.class, box,
					e -> e.isAlive() && !e.isSpectator());
				Vec3d origin = getPos();
				for (LivingEntity target : targets) {
					if (target.getPos().distanceTo(origin) > radius) {
						continue;
					}
					if (!hasGasMaskProtection(target)) {
						target.addStatusEffect(new StatusEffectInstance(StatusEffects.NAUSEA, 100, 0));
						target.addStatusEffect(new StatusEffectInstance(StatusEffects.BLINDNESS, 60, 0));
						target.addStatusEffect(new StatusEffectInstance(StatusEffects.POISON, 60, 0));
						target.addStatusEffect(new StatusEffectInstance(StatusEffects.SLOWNESS, 60, 0));
					}
				}
			}
		}
	}

	private float computeExposure(ServerWorld sw, Vec3d origin, LivingEntity target) {
		Vec3d lowPoint = target.getPos().add(0.0, 0.5, 0.0);
		Vec3d eyePoint = target.getEyePos();
		boolean lowClear = sw.raycast(new RaycastContext(
			origin, lowPoint, RaycastContext.ShapeType.COLLIDER, RaycastContext.FluidHandling.NONE, this
		)).getType() == HitResult.Type.MISS;
		boolean eyeClear = sw.raycast(new RaycastContext(
			origin, eyePoint, RaycastContext.ShapeType.COLLIDER, RaycastContext.FluidHandling.NONE, this
		)).getType() == HitResult.Type.MISS;

		if (lowClear && eyeClear) return 1.0f;
		if (lowClear || eyeClear) return 0.5f;
		return 0.0f;
	}

	public static boolean hasGasMaskProtection(LivingEntity entity) {
		if (!(entity instanceof PlayerLoadoutHolder holder)) {
			return false;
		}
		return hasGasMaskProtection(holder.tarkovcraft$getLoadout());
	}

	public static boolean hasGasMaskProtection(PlayerLoadout loadout) {
		if (loadout == null) {
			return false;
		}
		ItemStack mask = loadout.getEquipment(EquipmentSlotType.FACE_MASK);
		return !mask.isEmpty()
			&& mask.getItem() instanceof com.tarkovcraft.item.TarkovArmorItem armor
			&& armor.isGasProtection();
	}

	@Override
	public void writeCustomDataToNbt(NbtCompound nbt) {
		super.writeCustomDataToNbt(nbt);
		nbt.putString("GrenadeType", grenadeType.name());
		nbt.putInt("FuseRemaining", fuseRemaining);
		nbt.putInt("CloudRemaining", cloudRemaining);
		nbt.putBoolean("Detonated", detonated);
		nbt.putBoolean("Emitting", emitting);
		nbt.putBoolean("Resting", resting);
	}

	@Override
	public void readCustomDataFromNbt(NbtCompound nbt) {
		super.readCustomDataFromNbt(nbt);
		if (nbt.contains("GrenadeType")) {
			grenadeType = TarkovGrenadeItem.GrenadeType.fromString(nbt.getString("GrenadeType"));
		}
		if (nbt.contains("FuseRemaining")) {
			fuseRemaining = nbt.getInt("FuseRemaining");
		}
		cloudRemaining = nbt.getInt("CloudRemaining");
		detonated = nbt.getBoolean("Detonated");
		emitting = nbt.getBoolean("Emitting");
		resting = nbt.getBoolean("Resting");
		if (resting || emitting) {
			setNoGravity(true);
		}
	}
