package com.tarkovcraft.client.screen;

import net.fabricmc.api.EnvType;
import net.fabricmc.api.Environment;
import net.minecraft.client.font.TextRenderer;
import net.minecraft.client.gui.DrawContext;
import net.minecraft.item.ItemStack;
import net.minecraft.text.Text;

import java.util.function.IntConsumer;

/**
 * Compact modal for choosing an exact split count (1 .. count - 1) before sending C2S_SPLIT_STACK.
 */
@Environment(EnvType.CLIENT)
public final class TarkovSplitModal {
	private static final int WIN_W = 164;
	private static final int WIN_H = 84;

	private final int maxSplit;
	private final String itemName;
	private final IntConsumer onConfirm;

	private int x;
	private int y;
	private int splitAmount;
	private boolean draggingSlider = false;

	public TarkovSplitModal(ItemStack stack, int anchorX, int anchorY, int screenW, int screenH, IntConsumer onConfirm) {
		this.maxSplit = Math.max(1, stack.getCount() - 1);
		this.itemName = stack.getName().getString();
		this.splitAmount = Math.max(1, stack.getCount() / 2);
		this.onConfirm = onConfirm;
		this.x = Math.max(6, Math.min(screenW - WIN_W - 6, anchorX - WIN_W / 2));
		this.y = Math.max(6, Math.min(screenH - WIN_H - 6, anchorY - WIN_H / 2));
	}

	public void render(DrawContext ctx, TextRenderer tr, int mouseX, int mouseY) {
		ctx.getMatrices().push();
		ctx.getMatrices().translate(0.0f, 0.0f, TarkovUiTheme.Z_MODAL);

		ctx.fill(x, y, x + WIN_W, y + WIN_H, TarkovUiTheme.BG_MODAL);
		ctx.drawBorder(x, y, WIN_W, WIN_H, TarkovUiTheme.BORDER_POPUP);

		ctx.fill(x + 1, y + 1, x + WIN_W - 1, y + 16, TarkovUiTheme.BG_MODAL_HEADER);
		String title = Text.translatable("ui.tarkovcraft.split_title").getString() + " // " + itemName;
		TarkovUiTheme.drawFlatText(ctx, tr, tr.trimToWidth(title, Math.round((WIN_W - 10) / TarkovUiTheme.TEXT_SCALE_SMALL)),
			x + 5, y + 5, TarkovUiTheme.ACCENT_SAND, TarkovUiTheme.TEXT_SCALE_SMALL);

		int total = maxSplit + 1;
		int remain = total - splitAmount;
		TarkovUiTheme.drawCenteredFlatText(ctx, tr, splitAmount + " / " + remain,
			x + WIN_W / 2, y + 22, TarkovUiTheme.TEXT_PRIMARY, TarkovUiTheme.TEXT_SCALE_SMALL);

		// Slider track
		int trackX = x + 10;
		int trackY = y + 36;
		int trackW = WIN_W - 20;
		ctx.fill(trackX, trackY, trackX + trackW, trackY + 6, 0xFF1A1E23);
		ctx.drawBorder(trackX, trackY, trackW, 6, TarkovUiTheme.BORDER_SLOT);

		float ratio = maxSplit <= 1 ? 1.0f : (splitAmount - 1) / (float) (maxSplit - 1);
		int fillW = Math.max(4, Math.round(ratio * (trackW - 2)));
		ctx.fill(trackX + 1, trackY + 1, trackX + 1 + fillW, trackY + 5, TarkovUiTheme.ACCENT_SAND);

		int knobX = trackX + Math.round(ratio * (trackW - 6));
		ctx.fill(knobX, trackY - 2, knobX + 6, trackY + 8, TarkovUiTheme.TEXT_PRIMARY);

		// Buttons: 50% | OK | Cancel
		drawButton(ctx, tr, x + 8, y + 56, 36, 18, "50%", mouseX, mouseY, false);
		drawButton(ctx, tr, x + 50, y + 56, 52, 18, Text.translatable("ui.tarkovcraft.confirm").getString(), mouseX, mouseY, true);
		drawButton(ctx, tr, x + 108, y + 56, 48, 18, Text.translatable("ui.tarkovcraft.cancel").getString(), mouseX, mouseY, false);

		ctx.getMatrices().pop();
	}

	private void drawButton(DrawContext ctx, TextRenderer tr, int bx, int by, int bw, int bh, String label, int mx, int my, boolean accent) {
		boolean hov = mx >= bx && mx < bx + bw && my >= by && my < by + bh;
		int bg = hov ? TarkovUiTheme.HOVER_MENU_BG : (accent ? 0xFF283226 : 0xFF1A1E24);
		int textCol = hov ? TarkovUiTheme.HOVER_MENU_TEXT : (accent ? TarkovUiTheme.ACCENT_SAND : TarkovUiTheme.TEXT_PRIMARY);
		ctx.fill(bx, by, bx + bw, by + bh, bg);
		ctx.drawBorder(bx, by, bw, bh, hov ? TarkovUiTheme.BORDER_HOVER : TarkovUiTheme.BORDER_SLOT);
		TarkovUiTheme.drawCenteredFlatText(ctx, tr, label, bx + bw / 2, by + 6, textCol, TarkovUiTheme.TEXT_SCALE_SMALL);
	}

	public boolean contains(double mx, double my) {
		return mx >= x && mx < x + WIN_W && my >= y && my < y + WIN_H;
	}

	/**
	 * Returns true if the modal should close after handling the click.
	 */
	public boolean handleMouseClicked(double mx, double my, int button) {
		if (button != 0) {
			return false;
		}
		if (!contains(mx, my)) {
			return true;
		}

		// Slider hit
		int trackX = x + 10;
		int trackY = y + 32;
		int trackW = WIN_W - 20;
		if (mx >= trackX && mx <= trackX + trackW && my >= trackY && my <= trackY + 14) {
			draggingSlider = true;
			updateSliderFromMouse(mx);
			return false;
		}

		// 50% button
		if (mx >= x + 8 && mx < x + 44 && my >= y + 56 && my < y + 74) {
			splitAmount = Math.max(1, Math.min(maxSplit, (maxSplit + 1) / 2));
			return false;
		}

		// OK button
		if (mx >= x + 50 && mx < x + 102 && my >= y + 56 && my < y + 74) {
			confirm();
			return true;
		}

		// Cancel button
		if (mx >= x + 108 && mx < x + 156 && my >= y + 56 && my < y + 74) {
			return true;
		}

		return false;
	}

	public boolean handleMouseDragged(double mx) {
		if (!draggingSlider) {
			return false;
		}
		updateSliderFromMouse(mx);
		return true;
	}

	public void handleMouseReleased() {
		draggingSlider = false;
	}

	public void confirm() {
		onConfirm.accept(Math.max(1, Math.min(maxSplit, splitAmount)));
	}

	private void updateSliderFromMouse(double mx) {
		int trackX = x + 10;
		int trackW = WIN_W - 20;
		float ratio = (float) Math.max(0.0, Math.min(1.0, (mx - trackX) / trackW));
		splitAmount = 1 + Math.round(ratio * Math.max(0, maxSplit - 1));
		splitAmount = Math.max(1, Math.min(maxSplit, splitAmount));
	}
}

@Environment(EnvType.CLIENT)
public final class TarkovUiTheme {

	public static final int BG_CELL_SILHOUETTE = 0xDD090B0D;
	public static final int BG_DIM_DRAG_SOURCE = 0x9907080A;
	public static final int BG_TOOLTIP = 0xE60B0C0E;
	public static final int BG_CONTEXT_MENU = 0xF0111316;
	public static final int BG_MODAL = 0xEE0E1013;
	public static final int BG_MODAL_HEADER = 0xF515181C;

	// Borders & Accents
	public static final int BORDER_PANEL = 0xFF2D3035;
	public static final int BORDER_SLOT = 0xFF32363C;
	public static final int BORDER_CELL = 0xFF272B30;

	private TarkovUiTheme() {
	}

	public static void drawFlatText(DrawContext ctx, TextRenderer tr, String text, int x, int y, int color, float scale) {
		ctx.getMatrices().push();
		ctx.getMatrices().scale(scale, scale, 1.0f);
		ctx.drawText(tr, text, Math.round(x / scale), Math.round(y / scale), color, false);
		ctx.getMatrices().pop();
	}

	public static void drawCenteredFlatText(DrawContext ctx, TextRenderer tr, String text, int cx, int y, int color, float scale) {
		int w = Math.round(tr.getWidth(text) * scale);
		drawFlatText(ctx, tr, text, cx - w / 2, y, color, scale);
	}

	public static void drawRightAlignedFlatText(DrawContext ctx, TextRenderer tr, String text, int rightX, int y, int color, float scale) {
		int w = Math.round(tr.getWidth(text) * scale);
		drawFlatText(ctx, tr, text, rightX - w, y, color, scale);
	}

	public static void drawCellCornerBadge(
		DrawContext ctx, TextRenderer tr, String text,
		int cellX, int cellY, int cellW, int cellH, int color
	) {
		if (text == null || text.isEmpty()) {
			return;
		}
		int rawW = Math.max(1, tr.getWidth(text));
		int maxW = Math.max(8, cellW - 4);
		float scale = Math.min(0.62f, maxW / (float) rawW);
		int drawnW = Math.round(rawW * scale);
		int drawnH = Math.max(4, Math.round(7.0f * scale));

		int tx = cellX + cellW - 2 - drawnW;
		int ty = cellY + cellH - 2 - drawnH;

		ctx.getMatrices().push();
		ctx.getMatrices().translate(0.0f, 0.0f, 165.0f);
		ctx.fill(tx - 1, ty - 1, tx + drawnW + 1, ty + drawnH + 1, 0xA608090C);
		ctx.getMatrices().translate(tx, ty, 1.0f);
		ctx.getMatrices().scale(scale, scale, 1.0f);
		ctx.drawText(tr, text, 0, 0, color, false);
		ctx.getMatrices().pop();
	}

	public static String trimWithEllipsis(TextRenderer tr, String text, int maxPx, float scale) {
		int maxRaw = Math.max(6, Math.round(maxPx / scale));
		if (tr.getWidth(text) <= maxRaw) {
			return text;
		}
		int dotsW = tr.getWidth("...");
		if (maxRaw <= dotsW) {
			return tr.trimToWidth(text, maxRaw);
		}
		return tr.trimToWidth(text, maxRaw - dotsW) + "...";
	}

	public static void drawScaledItem(DrawContext ctx, ItemStack stack, int x, int y, int w, int h) {
		drawScaledItem(ctx, stack, x, y, w, h, false);
	}

	public static void drawScaledItem(DrawContext ctx, ItemStack stack, int x, int y, int w, int h, boolean rotated) {
		if (stack == null || stack.isEmpty() || w <= 0 || h <= 0) {
			return;
		}
		Identifier id = Registries.ITEM.getId(stack.getItem());
		int minDim = Math.min(w, h);
		int maxDim = Math.max(w, h);
		float targetPx;
		if ("tarkovcraft".equals(id.getNamespace())) {
			ItemSize itemSize = ItemSizeRegistry.getSize(stack);
			boolean rectangularItem = itemSize.width() != itemSize.height();
			if (rectangularItem && maxDim > minDim * 1.25f) {
				targetPx = Math.min(maxDim * 0.86f, minDim * 2.0f);
			} else {
				targetPx = Math.min(38.0f, minDim * 0.80f);
			}
		} else {
			targetPx = Math.min(38.0f, minDim * 0.82f);
		}
		float scale = targetPx / 16.0f;
		ctx.enableScissor(x, y, x + w, y + h);
		try {
			ctx.getMatrices().push();
			ctx.getMatrices().translate(x + w * 0.5f, y + h * 0.5f, 0.0f);
			if (rotated) {
				ctx.getMatrices().multiply(RotationAxis.POSITIVE_Z.rotationDegrees(90.0f));
			}
			ctx.getMatrices().scale(scale, scale, 1.0f);
			ctx.drawItem(stack, -8, -8);
			ctx.getMatrices().pop();
		} finally {
			ctx.disableScissor();
		}
	}

	public static int itemCategoryTint(ItemStack stack) {
		if (stack == null || stack.isEmpty()) {
			return BG_CELL_OCCUPIED_DEFAULT;
		}
		Item item = stack.getItem();
		if (item instanceof TarkovArmorItem) {
			return 0xCC18242C;
		}
		if (item instanceof TarkovMedicalItem || item instanceof TarkovStimulatorItem) {
			return 0xCC182824;
		}
		if (item instanceof BackpackItem || item instanceof TacticalRigItem || item instanceof TarkovCaseItem) {
			return 0xCC1E241C;
		}
		Identifier id = Registries.ITEM.getId(item);
		boolean isWeapon = ("tacz".equals(id.getNamespace()) && id.getPath().contains("gun"))
			|| item instanceof SwordItem
			|| item instanceof AxeItem
			|| item instanceof TarkovGrenadeItem;
		if (isWeapon || AmmoTier.isTaczAmmoStack(stack)) {
			return 0xCC26221A;
		}
		if (TarkovScreenHandler.isConsumableItem(stack)) {
			return 0xCC182620;
		}
		return BG_CELL_OCCUPIED_DEFAULT;
	}

	public static void drawFlatTooltip(
		DrawContext ctx,
		TextRenderer tr,
		List<Text> lines,
		int mouseX,
		int mouseY,
		int screenW,
		int screenH
	) {
		if (lines == null || lines.isEmpty()) {
			return;
		}
		int pad = 5;
		int lineH = 10;
		int maxW = 0;
		for (Text line : lines) {
			maxW = Math.max(maxW, tr.getWidth(line));
		}
		int boxW = maxW + pad * 2;
		int boxH = lines.size() * lineH + pad * 2 - 2;

		int bx = mouseX + 10;
		int by = mouseY - 6;
		if (bx + boxW > screenW - 4) {
			bx = Math.max(4, mouseX - boxW - 8);
		}
		if (by + boxH > screenH - 4) {
			by = Math.max(4, screenH - boxH - 4);
		}
		if (by < 4) {
			by = 4;
		}

		ctx.getMatrices().push();
		ctx.getMatrices().translate(0.0f, 0.0f, Z_POPUP);
		ctx.fill(bx, by, bx + boxW, by + boxH, BG_TOOLTIP);
		ctx.drawBorder(bx, by, boxW, boxH, BORDER_TOOLTIP);

		int ty = by + pad;
		for (int i = 0; i < lines.size(); i++) {
			Text line = lines.get(i);
			int col = i == 0 ? TEXT_PRIMARY : TEXT_SECONDARY;
			ctx.drawText(tr, line, bx + pad, ty, col, false);
			ty += lineH;
		}
		ctx.getMatrices().pop();
	}
}
